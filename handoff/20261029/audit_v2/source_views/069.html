  63 MotorMapping::MotorMapping(const std::string& path) {
  64   const YAML::Node root = YAML::LoadFile(path);
  65   const YAML::Node models = root["models"];
  66   const YAML::Node motors = root["motors"];
  67   if (!models || !models.IsMap()) throw std::runtime_error("mapping must contain a models map");
  68   if (!motors || !motors.IsSequence() || motors.size() == 0U || motors.size() > 64U)
  69     throw std::runtime_error("mapping must contain 1..64 motors");
  70 
  71   struct Model {
  72     uint8_t type{};
  73     MotorLimits limits;
  74   };
  75   std::unordered_map<std::string, Model> parsed_models;
  76   for (const auto& item : models)
  77   {
  78     const std::string name = item.first.as<std::string>();
  79     parsed_models.emplace(
  80         name, Model{motorTypeForFamily(item.second["family"].as<std::string>()), loadLimits(item.second, name)});
  81   }
  82 
  83   std::unordered_set<uint64_t> physical_routes;
  84   entries_.reserve(motors.size());
  85   for (const auto& motor : motors)
  86   {
  87     MotorMappingEntry entry;
  88     entry.joint_name = motor["joint_name"].as<std::string>();
  89     entry.enabled = motor["enabled"].as<bool>(false);
  90     const std::string model_name = motor["model"].as<std::string>();
  91     const auto model = parsed_models.find(model_name);
  92     if (model == parsed_models.end()) throw std::runtime_error("unknown model for " + entry.joint_name);
  93     entry.route.motor_type = model->second.type;
  94     entry.route.bus_index = motor["bus_index"].as<uint8_t>();
  95     entry.route.motor_id = motor["motor_id"].as<uint32_t>();
  96     entry.zero_offset = requiredFinite(motor, "zero_offset", entry.joint_name);
  97     entry.motor_direction = requiredFinite(motor, "motor_direction", entry.joint_name);
  98     entry.reduction = requiredFinite(motor, "reduction", entry.joint_name);
  99     entry.joint_lower = requiredFinite(motor, "joint_lower", entry.joint_name);
 100     entry.joint_upper = requiredFinite(motor, "joint_upper", entry.joint_name);
 101     entry.limits = model->second.limits;
 102 
 103     if (entry.joint_name.empty() || entry.joint_name.size() > 64U || std::abs(entry.motor_direction) != 1.0 ||
 104         entry.reduction <= 0.0 || entry.joint_lower > entry.joint_upper)
 105       throw std::runtime_error("invalid mapping for " + entry.joint_name);
 106     if (!physical_routes.emplace(physicalRouteKey(entry.route)).second)
 107       throw std::runtime_error("duplicate motor route for " + entry.joint_name);
 108     if (!joint_indices_.emplace(entry.joint_name, entries_.size()).second)
 109       throw std::runtime_error("duplicate joint name: " + entry.joint_name);
 110     if (entry.enabled) ++enabled_size_;
 111     entries_.push_back(std::move(entry));
 112   }

 131 luvbot_hardware_interfaces::msg::ActuatorMotorCommand MotorMapping::commandFor(const MotorMappingEntry& mapping,
 132                                                                                double q, double dq, double tau,
 133                                                                                double kp, double kd,
 134                                                                                double wrap_offset) const {
 135   if (!mapping.enabled) throw std::runtime_error("motor is disabled for " + mapping.joint_name);
 136   if (!std::isfinite(q) || !std::isfinite(dq) || !std::isfinite(tau) || !std::isfinite(kp) || !std::isfinite(kd) ||
 137       !std::isfinite(wrap_offset))
 138     throw std::runtime_error("non-finite command for " + mapping.joint_name);
 139   const double direction = mapping.motor_direction * mapping.limits.rotation_direction;
 140   const double reduction_squared = mapping.reduction * mapping.reduction;
 141   const double joint_q = std::clamp(q, mapping.joint_lower, mapping.joint_upper);
 142 
 143   luvbot_hardware_interfaces::msg::ActuatorMotorCommand command;
 144   command.route = mapping.route;
 145   command.q =
 146       static_cast<float>(std::clamp((joint_q - wrap_offset - mapping.zero_offset) * direction * mapping.reduction,
 147                                     mapping.limits.p_min, mapping.limits.p_max));
 148   command.dq =
 149       static_cast<float>(std::clamp(dq * direction * mapping.reduction, mapping.limits.v_min, mapping.limits.v_max));
 150   command.tau =
 151       static_cast<float>(std::clamp(tau * direction / mapping.reduction, mapping.limits.t_min, mapping.limits.t_max));
 152   command.kp = static_cast<float>(std::clamp(kp / reduction_squared, mapping.limits.kp_min, mapping.limits.kp_max));
 153   command.kd = static_cast<float>(std::clamp(kd / reduction_squared, mapping.limits.kd_min, mapping.limits.kd_max));
 154   return command;
 155 }

 162 double MotorMapping::initialWrapOffset(const MotorMappingEntry& mapping, double motor_q, double tolerance) const {
 163   if (!std::isfinite(motor_q) || !std::isfinite(tolerance) || tolerance < 0.0)
 164     throw std::runtime_error("invalid initial feedback for " + mapping.joint_name);
 165   const double raw = rawJointPosition(mapping, motor_q);
 166   const double full_turn = 2.0 * std::acos(-1.0);
 167   if (!std::isfinite(raw) || motor_q < mapping.limits.p_min || motor_q > mapping.limits.p_max)
 168     throw std::runtime_error("initial motor position outside protocol limits for " + mapping.joint_name);
 169   const double lower = mapping.joint_lower - tolerance;
 170   const double upper = mapping.joint_upper + tolerance;
 171   if (raw >= lower && raw <= upper) return 0.0;
 172   const double first_turn = std::ceil((lower - raw) / full_turn);
 173   const double last_turn = std::floor((upper - raw) / full_turn);
 174   if (first_turn != last_turn)
 175     throw std::runtime_error("no unique feedback turn within joint limits for " + mapping.joint_name);
 176   const double offset = first_turn * full_turn;
 177   if (!std::isfinite(offset) || raw + offset < lower || raw + offset > upper)
 178     throw std::runtime_error("invalid feedback turn for " + mapping.joint_name);
 179   return offset;
 180 }

 182 JointFeedback MotorMapping::feedbackFor(const MotorMappingEntry& mapping,
 183                                         const luvbot_hardware_interfaces::msg::MotorState& state,
 184                                         double wrap_offset) const {
 185   const double direction = mapping.motor_direction * mapping.limits.rotation_direction;
 186   JointFeedback feedback;
 187   feedback.q = rawJointPosition(mapping, state.q) + wrap_offset;
 188   feedback.dq = state.dq * direction / mapping.reduction;
 189   feedback.tau = state.tau * direction * mapping.reduction;
 190   feedback.error = state.error_code;
 191   feedback.operating_mode = state.operating_mode;
 192   feedback.valid = state.valid && std::isfinite(state.q) && state.q >= mapping.limits.p_min &&
 193                    state.q <= mapping.limits.p_max && std::isfinite(feedback.q) && std::isfinite(feedback.dq) &&
 194                    std::isfinite(feedback.tau) && feedback.q >= mapping.joint_lower - kJointPositionTolerance &&
 195                    feedback.q <= mapping.joint_upper + kJointPositionTolerance;
 196   return feedback;
 197 }

 199 JointFeedback MotorMapping::updateFeedback(const MotorMappingEntry& mapping,
 200                                            const luvbot_hardware_interfaces::msg::MotorState& state,
 201                                            FeedbackWrapState& wrap) const {
 202   auto feedback = feedbackFor(mapping, state, wrap.offset);
 203   if (state.error_code == 0U)
 204     wrap.error_samples = 0U;
 205   else if (state.error_code != wrap.error_code)
 206     wrap.error_samples = 1U;
 207   else
 208     wrap.error_samples = std::min(wrap.error_samples + 1U, kMotorErrorFaultSamples);
 209   wrap.error_code = state.error_code;
 210   const bool persistent_error = wrap.error_samples >= kMotorErrorFaultSamples;
 211   if (wrap.fault || !state.valid || persistent_error || !std::isfinite(state.q) || !std::isfinite(state.dq) ||
 212       !std::isfinite(state.tau) || state.q < mapping.limits.p_min || state.q > mapping.limits.p_max)
 213   {
 214     wrap.stable_samples = 0U;
 215     wrap.fault = wrap.fault || wrap.locked || persistent_error;
 216     feedback.valid = false;
 217     return feedback;
 218   }
 219 
 220   if (state.error_code != 0U)
 221   {
 222     // A transient error must neither confirm/rebase a turn nor hide invalid active feedback.
 223     if (wrap.locked && !feedback.valid)
 224     {
 225       wrap.stable_samples = 0U;
 226       wrap.fault = true;
 227     }
 228     else
 229     {
 230       if (!wrap.ready()) wrap.stable_samples = 0U;
 231       feedback.error_pending = true;
 232       feedback.initializing = !wrap.ready();
 233     }
 234     feedback.valid = false;
 235     return feedback;
 236   }
 237 
 238   // Only startup/damping may change turns; powered control must never silently rebase a target.
 239   if (!wrap.locked && (wrap.stable_samples == 0U || !feedback.valid))
 240   {
 241     try
 242     {
 243       const double offset = initialWrapOffset(mapping, state.q);
 244       if (offset != wrap.offset) wrap.stable_samples = 0U;
 245       wrap.offset = offset;
 246       feedback = feedbackFor(mapping, state, wrap.offset);
 247     } catch (const std::exception&)
 248     {
 249       wrap.stable_samples = 0U;
 250       feedback.valid = false;
 251       feedback.initializing = true;
 252       return feedback;
 253     }
 254   }
 255   if (!feedback.valid)
 256   {
 257     wrap.stable_samples = 0U;
 258     wrap.fault = wrap.locked;
 259     feedback.initializing = !wrap.locked;
 260     return feedback;
 261   }
 262   wrap.stable_samples = std::min(wrap.stable_samples + 1U, 3U);
 263   feedback.valid = wrap.ready();
 264   feedback.initializing = !feedback.valid;
 265   return feedback;
 266 }

 199 JointFeedback MotorMapping::updateFeedback(const MotorMappingEntry& mapping,
 200                                            const luvbot_hardware_interfaces::msg::MotorState& state,
 201                                            FeedbackWrapState& wrap) const {
 202   auto feedback = feedbackFor(mapping, state, wrap.offset);
 203   if (state.error_code == 0U)
 204     wrap.error_samples = 0U;
 205   else if (state.error_code != wrap.error_code)
 206     wrap.error_samples = 1U;
 207   else
 208     wrap.error_samples = std::min(wrap.error_samples + 1U, kMotorErrorFaultSamples);
 209   wrap.error_code = state.error_code;
 210   const bool persistent_error = wrap.error_samples >= kMotorErrorFaultSamples;
 211   if (wrap.fault || !state.valid || persistent_error || !std::isfinite(state.q) || !std::isfinite(state.dq) ||
 212       !std::isfinite(state.tau) || state.q < mapping.limits.p_min || state.q > mapping.limits.p_max)
 213   {
 214     wrap.stable_samples = 0U;
 215     wrap.fault = wrap.fault || wrap.locked || persistent_error;
 216     feedback.valid = false;
 217     return feedback;
 218   }
 219 
 220   if (state.error_code != 0U)
 221   {
 222     // A transient error must neither confirm/rebase a turn nor hide invalid active feedback.
 223     if (wrap.locked && !feedback.valid)
 224     {
 225       wrap.stable_samples = 0U;
 226       wrap.fault = true;
 227     }
 228     else
 229     {
 230       if (!wrap.ready()) wrap.stable_samples = 0U;
 231       feedback.error_pending = true;
 232       feedback.initializing = !wrap.ready();
 233     }
 234     feedback.valid = false;
 235     return feedback;
 236   }
 237 
 238   // Only startup/damping may change turns; powered control must never silently rebase a target.
 239   if (!wrap.locked && (wrap.stable_samples == 0U || !feedback.valid))
 240   {
 241     try
 242     {
 243       const double offset = initialWrapOffset(mapping, state.q);
 244       if (offset != wrap.offset) wrap.stable_samples = 0U;
 245       wrap.offset = offset;
 246       feedback = feedbackFor(mapping, state, wrap.offset);
 247     } catch (const std::exception&)
 248     {
 249       wrap.stable_samples = 0U;
 250       feedback.valid = false;
 251       feedback.initializing = true;
 252       return feedback;
 253     }
 254   }
 255   if (!feedback.valid)
 256   {
 257     wrap.stable_samples = 0U;
 258     wrap.fault = wrap.locked;
 259     feedback.initializing = !wrap.locked;
 260     return feedback;
 261   }
 262   wrap.stable_samples = std::min(wrap.stable_samples + 1U, 3U);
 263   feedback.valid = wrap.ready();
 264   feedback.initializing = !feedback.valid;
 265   return feedback;
 266 }
