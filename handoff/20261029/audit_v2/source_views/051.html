  44 def static_shape(item, kind, onnx):
  45     if item.type.tensor_type.elem_type != onnx.TensorProto.FLOAT:
  46         raise ValueError(f"{kind} '{item.name}' must be float32")
  47     dims = []
  48     for dim in item.type.tensor_type.shape.dim:

  55 def check_model(path, manifest_path=None):
  56     import onnx
  57 
  58     if manifest_path is not None:
  59         graph = check_graph(path, manifest_path)
  60         return [(x["name"], tuple(x["shape"])) for x in graph["inputs"]], (graph["outputs"][0]["name"], tuple(graph["outputs"][0]["shape"]))
  61 
  62     model = onnx.load(str(path))
  63     onnx.checker.check_model(model)
  64     initializers = {item.name for item in model.graph.initializer}
  65     inputs = [item for item in model.graph.input if item.name not in initializers]
  66     outputs = list(model.graph.output)
  67     names = {item.name for item in inputs}
  68     if len(inputs) != 2 or names not in ({"policy", "command_short"}, {"policy", "command"}):
  69         raise ValueError(f"expected policy and command_short/command inputs, got {[item.name for item in inputs]}")
  70     specs = []
  71     for item in inputs:
  72         shape = static_shape(item, "input", onnx)
  73         width = EXPECTED_INPUTS["policy" if item.name == "policy" else "command_short"]
  74         if shape not in ((width,), (1, width)):
  75             raise ValueError(f"input '{item.name}' must contain {width} floats, got {shape}")
  76         specs.append((item.name, shape))
  77     if len(outputs) != 1 or outputs[0].name != EXPECTED_OUTPUT[0]:
  78         raise ValueError(f"expected only output '{EXPECTED_OUTPUT[0]}', got {[item.name for item in outputs]}")

  80     if output_shape not in ((21,), (1, 21)):
  81         raise ValueError(f"output 'action' must contain 21 floats, got {output_shape}")
  82     return specs, (outputs[0].name, output_shape)

  85 def replace_mish(model, onnx):
  86     rewritten = []
  87     count = 0
  88     for node in model.graph.node:
  89         if node.op_type != "Mish" or node.domain not in ("", "ai.onnx"):
  90             rewritten.append(node)
  91             continue
  92         if len(node.input) != 1 or len(node.output) != 1:
  93             raise ValueError(f"Mish node '{node.name}' must have one input and one output")
  94         count += 1
  95         prefix = node.name or f"mish_{count}"
  96         softplus = f"{node.output[0]}__softplus"
  97         tanh = f"{node.output[0]}__tanh"
  98         rewritten.extend(
  99             [
 100                 onnx.helper.make_node("Softplus", node.input, [softplus], name=f"{prefix}__softplus"),
 101                 onnx.helper.make_node("Tanh", [softplus], [tanh], name=f"{prefix}__tanh"),
 102                 onnx.helper.make_node("Mul", [node.input[0], tanh], node.output, name=f"{prefix}__mul"),
 103             ]
 104         )
 105     if count:
 106         del model.graph.node[:]
 107         model.graph.node.extend(rewritten)
 108     return count

 142 def cosine_similarity(reference, candidate):
 143     import numpy as np
 144 
 145     denominator = float(np.linalg.norm(reference) * np.linalg.norm(candidate))
 146     if denominator == 0.0:
 147         return 1.0 if np.array_equal(reference, candidate) else 0.0
 148     return float(np.dot(reference, candidate) / denominator)

 151 def verify_dynamic(rknn, onnx_path, args):
 152     import numpy as np
 153     import onnxruntime as ort
 154     graph = check_graph(onnx_path, args.manifest)
 155     root = args.manifest.resolve().parent
 156     data_path = args.verification_data or asset_path(root, graph["verification_data"])
 157     observations = {}
 158     with np.load(data_path, allow_pickle=False) as data:
 159         for item in graph["inputs"]:
 160             if item["role"] != "observation":
 161                 continue
 162             value = data[item["name"]]
 163             if value.dtype != np.float32 or list(value.shape[1:]) != item["shape"] or not np.isfinite(value).all():
 164                 raise ValueError(f"Invalid recorded input {item['name']}")
 165             observations[item["name"]] = value.copy()
 166     counts = {len(x) for x in observations.values()}
 167     if len(counts) != 1 or min(counts) < 20:
 168         raise ValueError("Dynamic verification requires at least 20 aligned observation samples")
 169     session = ort.InferenceSession(str(onnx_path), providers=["CPUExecutionProvider"])
 170     ret = rknn.init_runtime(**runtime_options(args))
 171     if ret != 0:
 172         raise RuntimeError(f"rknn.init_runtime failed, ret={ret}")
 173     reports, first_feed, first_output = [], None, None
 174     for action in graph["actions"]:
 175         adapter_path = asset_path(root, action["adapter_manifest"])
 176         parameters = load_adapter(adapter_path, graph, action["action_id"])
 177         maximum, worst, minimum_cosine = 0.0, 0.0, 1.0
 178         for i in range(min(counts)):
 179             feed = graph_feeds(graph, {k: v[i] for k, v in observations.items()}, parameters)
 180             expected = np.asarray(session.run(["action"], feed)[0], np.float32).reshape(-1)
 181             started = time.perf_counter()
 182             actual_outputs = rknn.inference(inputs=[feed[x["name"]] for x in graph["inputs"]])
 183             worst = max(worst, time.perf_counter() - started)
 184             if actual_outputs is None or len(actual_outputs) != 1:
 185                 raise RuntimeError("Dynamic RKNN must return one output")
 186             actual = np.asarray(actual_outputs[0], np.float32).reshape(-1)
 187             if actual.shape != expected.shape or not np.isfinite(actual).all():
 188                 raise RuntimeError("Invalid dynamic RKNN output")
 189             error = float(np.max(np.abs(actual - expected)))
 190             cosine = cosine_similarity(expected, actual) if np.linalg.norm(expected) > 1e-8 else 1.0
 191             maximum, minimum_cosine = max(maximum, error), min(minimum_cosine, cosine)
 192             if error > args.max_abs_error or cosine < args.cosine_threshold:
 193                 raise RuntimeError(f"Dynamic RKNN verification failed: {action['action_id']} sample={i} error={error} cosine={cosine}")
 194             if first_feed is None:
 195                 first_feed, first_output = feed, actual.copy()
 196         reports.append({"action_id": action["action_id"], "samples": min(counts),
 197                         "adapter_manifest_sha256": sha256(adapter_path), "max_abs": maximum,
 198                         "minimum_cosine": minimum_cosine, "worst_ms": worst * 1000.0})
 199         print(f"dynamic verification: {reports[-1]}")
 200     if first_feed is None:
 201         raise ValueError("Graph manifest has no selected adapters")
 202     repeated = rknn.inference(inputs=[first_feed[x["name"]] for x in graph["inputs"]])
 203     if repeated is None or not np.allclose(np.asarray(repeated[0]).reshape(-1), first_output, atol=args.max_abs_error, rtol=0):
 204         raise RuntimeError("A-to-B-to-A RKNN verification failed")
 205     return {"actions": reports, "a_to_b_to_a": "passed" if len(reports) > 1 else "one_adapter_only"}

 208 def verify(rknn, onnx_path, input_specs, output_spec, args):
 209     if args.manifest:
 210         return verify_dynamic(rknn, onnx_path, args)
 211     import numpy as np
 212     import onnxruntime as ort
 213 
 214     session = ort.InferenceSession(str(onnx_path), providers=["CPUExecutionProvider"])
 215     ret = rknn.init_runtime(**runtime_options(args))
 216     if ret != 0:
 217         raise RuntimeError(f"rknn.init_runtime failed, ret={ret}")
 218 
 219     rng = np.random.default_rng(0)
 220     recorded = None
 221     if args.verification_data:
 222         with np.load(args.verification_data, allow_pickle=False) as data:
 223             recorded = {}
 224             for name, shape in input_specs:
 225                 key = name
 226                 if name != "policy" and name not in data:
 227                     key = "command" if name == "command_short" else "command_short"
 228                 value = data[key]
 229                 if value.dtype != np.float32 or value.ndim != 2 or value.shape[1] != shape[-1] or \
 230                         not np.isfinite(value).all():
 231                     raise ValueError(f"recorded {name} must be finite float32 [N,{shape[-1]}]")
 232                 recorded[name] = value.copy()
 233         counts = {len(value) for value in recorded.values()}
 234         if len(counts) != 1 or min(counts) < 20:
 235             raise ValueError("recorded policy and command need the same number of frames, at least 20")
 236         sample_count = min(counts)
 237     else:
 238         sample_count = args.samples
 239         print("Random inputs check conversion only; replay recorded observations before hardware use")
 240     minimum_cosine = 1.0
 241     maximum_error = 0.0
 242     error_sum = np.zeros(EXPECTED_OUTPUT[1], dtype=np.float64)
 243     joint_maximum = np.zeros(EXPECTED_OUTPUT[1], dtype=np.float64)
 244     cumulative_error = np.zeros(EXPECTED_OUTPUT[1], dtype=np.float64)
 245     worst_latency = 0.0
 246     for sample in range(sample_count):
 247         feed = {name: recorded[name][sample].reshape(shape) if recorded is not None else
 248                 rng.uniform(-1.0, 1.0, shape).astype(np.float32) for name, shape in input_specs}
 249         expected = np.asarray(session.run([output_spec[0]], feed)[0], dtype=np.float32).reshape(-1)
 250         if expected.size != EXPECTED_OUTPUT[1] or not np.isfinite(expected).all():
 251             raise RuntimeError("ONNX action must contain 21 finite floats")
 252         started = time.perf_counter()
 253         actual_outputs = rknn.inference(inputs=[feed[name] for name, _ in input_specs])
 254         latency = time.perf_counter() - started
 255         worst_latency = max(worst_latency, latency)
 256         if actual_outputs is None or len(actual_outputs) != 1:
 257             raise RuntimeError("RKNN must return exactly one output")
 258         actual = np.asarray(actual_outputs[0], dtype=np.float32).reshape(-1)
 259         if actual.size != EXPECTED_OUTPUT[1] or not np.isfinite(actual).all():
 260             raise RuntimeError(f"RKNN action must contain 21 finite floats, got {actual.shape}")
 261         error = np.abs(expected - actual)
 262         max_error = float(np.max(error))
 263         error_sum += error
 264         joint_maximum = np.maximum(joint_maximum, error)
 265         cumulative_error += actual - expected
 266         cosine = cosine_similarity(expected, actual)
 267         minimum_cosine = min(minimum_cosine, cosine)
 268         maximum_error = max(maximum_error, max_error)
 269         print(f"  sample={sample:02d} cosine={cosine:.6f} max_abs={max_error:.6g} latency_ms={latency * 1000:.3f}")
 270         if cosine < args.cosine_threshold or max_error > args.max_abs_error:
 271             raise RuntimeError(f"verification failed at sample {sample}")
 272     print(f"verification passed: minimum cosine={minimum_cosine:.6f}, maximum abs error={maximum_error:.6g}")
 273     print(f"mean abs error={error_sum.sum() / (sample_count * 21):.6g}, worst inference ms={worst_latency * 1000:.3f}")
 274     print(f"per-joint max abs error: {joint_maximum.tolist()}")
 275     print(f"per-joint mean abs error: {(error_sum / sample_count).tolist()}")
 276     print(f"per-joint signed cumulative error: {cumulative_error.tolist()}")

 279 def main():
 280     args = parse_args()
 281     onnx_path = args.onnx.resolve()
 282     output_path = (args.output or onnx_path.with_suffix(".rknn")).resolve()
 283     if not onnx_path.is_file() or onnx_path.suffix.lower() != ".onnx":
 284         raise ValueError(f"ONNX model not found: {onnx_path}")
 285     if output_path.suffix.lower() != ".rknn":
 286         raise ValueError("output filename must end with .rknn")
 287     if args.samples < 20:
 288         raise ValueError("--samples must be at least 20")
 289     if not math.isfinite(args.cosine_threshold) or not -1.0 <= args.cosine_threshold <= 1.0 or \
 290             not math.isfinite(args.max_abs_error) or args.max_abs_error <= 0:
 291         raise ValueError("verification thresholds must be finite: cosine in [-1,1], max abs error > 0")
 292 
 293     runtime_options(args)
 294     input_specs, output_spec = check_model(onnx_path, args.manifest)
 295     output_specs = [output_spec]
 296     print("inputs: " + ", ".join(f"{name}{shape}" for name, shape in input_specs))
 297     print("outputs: " + ", ".join(f"{name}{shape}" for name, shape in output_specs))
 298     if args.manifest:
 299         graph = load_graph_manifest(args.manifest)
 300         root = args.manifest.resolve().parent
 301         actions = graph.get("actions", [])
 302         if not actions or len({x["action_id"] for x in actions}) != len(actions):
 303             raise ValueError("Dynamic graph requires unique, non-empty action entries")
 304         for action in actions:
 305             load_adapter(asset_path(root, action["adapter_manifest"]), graph, action["action_id"])
 306         print(f"shared {graph['kind']}: actions={len(actions)}, rank={graph['rank']}, alpha={graph['alpha']}; "
 307               "adapter tensors remain runtime inputs (no per-action policy compilation)")
 308     print(f"build: config={rknn_build_config()} do_quantization=False; output={output_path}")
 309     if args.dry_run:
 310         print(f"DRY-RUN: no RKNN build/runtime; after build, validation={'pending formal C++ runtime' if args.build_only else 'Python verification requested'}")
 311         return 0
 312 
 313     with rknn_compatible_model(onnx_path) as rknn_onnx_path:
 314         from rknn.api import RKNN
 315 
 316         rknn = RKNN(verbose=True)
 317         try:
 318             ret = rknn.config(**rknn_build_config())
 319             if ret != 0:
 320                 raise RuntimeError(f"rknn.config failed, ret={ret}")
 321             ret = rknn.load_onnx(model=str(rknn_onnx_path))
 322             if ret != 0:
 323                 raise RuntimeError(f"rknn.load_onnx failed, ret={ret}")
 324             ret = rknn.build(do_quantization=False)
 325             if ret != 0:
 326                 raise RuntimeError(f"rknn.build failed, ret={ret}")
 327             output_path.parent.mkdir(parents=True, exist_ok=True)
 328             with tempfile.TemporaryDirectory(prefix="wbc_rknn_", dir=output_path.parent) as directory:
 329                 candidate = Path(directory) / output_path.name
 330                 ret = rknn.export_rknn(str(candidate))
 331                 if ret != 0:
 332                     raise RuntimeError(f"rknn.export_rknn failed, ret={ret}")
 333                 verification = None if args.build_only else verify(rknn, onnx_path, input_specs, output_spec, args)
 334                 candidate.replace(output_path)
 335                 conversion_record(onnx_path, output_path, args, verification)
 336             print(f"exported: {output_path}; validation={'pending C++ deployment' if args.build_only else 'Python verification completed'}")
 337         finally:
 338             rknn.release()

 125 def cosine_similarity(reference, candidate):
 126     import numpy as np
 127 
 128     denominator = float(np.linalg.norm(reference) * np.linalg.norm(candidate))
 129     if denominator == 0.0:
 130         return 1.0 if np.array_equal(reference, candidate) else 0.0
 131     return float(np.dot(reference, candidate) / denominator)

 134 def verify(rknn, onnx_path, input_specs, output_spec, args):
 135     import numpy as np
 136     import onnxruntime as ort
 137 
 138     session = ort.InferenceSession(str(onnx_path), providers=["CPUExecutionProvider"])
 139     runtime_options = {"target": args.target} if args.target else {}
 140     ret = rknn.init_runtime(**runtime_options)
 141     if ret != 0:
 142         raise RuntimeError(f"rknn.init_runtime failed, ret={ret}")
 143 
 144     rng = np.random.default_rng(0)
 145     recorded = None
 146     if args.verification_data:
 147         with np.load(args.verification_data, allow_pickle=False) as data:
 148             recorded = {}
 149             for name, shape in input_specs:
 150                 key = name
 151                 if name != "policy" and name not in data:
 152                     key = "command" if name == "command_short" else "command_short"
 153                 value = data[key]
 154                 if value.dtype != np.float32 or value.ndim != 2 or value.shape[1] != shape[-1] or \
 155                         not np.isfinite(value).all():
 156                     raise ValueError(f"recorded {name} must be finite float32 [N,{shape[-1]}]")
 157                 recorded[name] = value.copy()
 158         counts = {len(value) for value in recorded.values()}
 159         if len(counts) != 1 or min(counts) < 20:
 160             raise ValueError("recorded policy and command need the same number of frames, at least 20")
 161         sample_count = min(counts)
 162     else:
 163         sample_count = args.samples
 164         print("Random inputs check conversion only; replay recorded observations before hardware use")
 165     minimum_cosine = 1.0
 166     maximum_error = 0.0
 167     error_sum = np.zeros(EXPECTED_OUTPUT[1], dtype=np.float64)
 168     joint_maximum = np.zeros(EXPECTED_OUTPUT[1], dtype=np.float64)
 169     cumulative_error = np.zeros(EXPECTED_OUTPUT[1], dtype=np.float64)
 170     worst_latency = 0.0
 171     for sample in range(sample_count):
 172         feed = {name: recorded[name][sample].reshape(shape) if recorded is not None else
 173                 rng.uniform(-1.0, 1.0, shape).astype(np.float32) for name, shape in input_specs}
 174         expected = np.asarray(session.run([output_spec[0]], feed)[0], dtype=np.float32).reshape(-1)
 175         if expected.size != EXPECTED_OUTPUT[1] or not np.isfinite(expected).all():
 176             raise RuntimeError("ONNX action must contain 21 finite floats")
 177         started = time.perf_counter()
 178         actual_outputs = rknn.inference(inputs=[feed[name] for name, _ in input_specs])
 179         latency = time.perf_counter() - started
 180         worst_latency = max(worst_latency, latency)
 181         if actual_outputs is None or len(actual_outputs) != 1:
 182             raise RuntimeError("RKNN must return exactly one output")
 183         actual = np.asarray(actual_outputs[0], dtype=np.float32).reshape(-1)
 184         if actual.size != EXPECTED_OUTPUT[1] or not np.isfinite(actual).all():
 185             raise RuntimeError(f"RKNN action must contain 21 finite floats, got {actual.shape}")
 186         error = np.abs(expected - actual)
 187         max_error = float(np.max(error))
 188         error_sum += error
 189         joint_maximum = np.maximum(joint_maximum, error)
 190         cumulative_error += actual - expected
 191         cosine = cosine_similarity(expected, actual)
 192         minimum_cosine = min(minimum_cosine, cosine)
 193         maximum_error = max(maximum_error, max_error)
 194         print(f"  sample={sample:02d} cosine={cosine:.6f} max_abs={max_error:.6g} latency_ms={latency * 1000:.3f}")
 195         if cosine < args.cosine_threshold or max_error > args.max_abs_error:
 196             raise RuntimeError(f"verification failed at sample {sample}")
 197     print(f"verification passed: minimum cosine={minimum_cosine:.6f}, maximum abs error={maximum_error:.6g}")
 198     print(f"mean abs error={error_sum.sum() / (sample_count * 21):.6g}, worst inference ms={worst_latency * 1000:.3f}")
 199     print(f"per-joint max abs error: {joint_maximum.tolist()}")
 200     print(f"per-joint mean abs error: {(error_sum / sample_count).tolist()}")
 201     print(f"per-joint signed cumulative error: {cumulative_error.tolist()}")

 204 def main():
 205     args = parse_args()
 206     onnx_path = args.onnx.resolve()
 207     output_path = (args.output or onnx_path.with_suffix(".rknn")).resolve()
 208     if not onnx_path.is_file() or onnx_path.suffix.lower() != ".onnx":
 209         raise ValueError(f"ONNX model not found: {onnx_path}")
 210     if output_path.suffix.lower() != ".rknn":
 211         raise ValueError("output filename must end with .rknn")
 212     if args.samples < 20:
 213         raise ValueError("--samples must be at least 20")
 214     if not math.isfinite(args.cosine_threshold) or not -1.0 <= args.cosine_threshold <= 1.0 or \
 215             not math.isfinite(args.max_abs_error) or args.max_abs_error <= 0:
 216         raise ValueError("verification thresholds must be finite: cosine in [-1,1], max abs error > 0")
 217 
 218     input_specs, output_spec = check_model(onnx_path)
 219     output_specs = [output_spec]
 220     print("inputs: " + ", ".join(f"{name}{shape}" for name, shape in input_specs))
 221     print("outputs: " + ", ".join(f"{name}{shape}" for name, shape in output_specs))
 222 
 223     with rknn_compatible_model(onnx_path) as rknn_onnx_path:
 224         from rknn.api import RKNN
 225 
 226         rknn = RKNN(verbose=True)
 227         try:
 228             ret = rknn.config(target_platform="rk3588")
 229             if ret != 0:
 230                 raise RuntimeError(f"rknn.config failed, ret={ret}")
 231             ret = rknn.load_onnx(model=str(rknn_onnx_path))
 232             if ret != 0:
 233                 raise RuntimeError(f"rknn.load_onnx failed, ret={ret}")
 234             ret = rknn.build(do_quantization=False)
 235             if ret != 0:
 236                 raise RuntimeError(f"rknn.build failed, ret={ret}")
 237             output_path.parent.mkdir(parents=True, exist_ok=True)
 238             with tempfile.TemporaryDirectory(prefix="wbc_rknn_", dir=output_path.parent) as directory:
 239                 candidate = Path(directory) / output_path.name
 240                 ret = rknn.export_rknn(str(candidate))
 241                 if ret != 0:
 242                     raise RuntimeError(f"rknn.export_rknn failed, ret={ret}")
 243                 verify(rknn, onnx_path, input_specs, output_spec, args)
 244                 candidate.replace(output_path)
 245             print(f"exported: {output_path}")
 246         finally:
 247             rknn.release()
 248     return 0
