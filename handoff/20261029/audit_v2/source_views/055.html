  22 def joint_values(mapping, names, label):
  23     values = []
  24     for name in names:
  25         matches = [float(value) for pattern, value in mapping.items() if re.fullmatch(pattern, name)]
  26         if len(matches) != 1 or not np.isfinite(matches[0]):
  27             raise ValueError(f"{label}: expected one finite value for {name}")
  28         values.append(matches[0])
  29     return values

  32 def prepare(source, output):
  33     name = source.name
  34     if not re.fullmatch(r"[A-Za-z0-9_-]{1,32}", name):
  35         raise ValueError("source directory name must be a motion id of 1-32 letters, digits, '_' or '-'")
  36     manifest_path = source / f"{name}.export.json"
  37     manifest = json.loads(manifest_path.read_text(encoding="utf-8"))
  38     if manifest["robot"] != "l0_v5_wbc_pd" or manifest["motion"].split("/")[-1] != name:
  39         raise ValueError("expected a matching l0_v5_wbc_pd motion export")
  40     files = {
  41         "checkpoint": source / f"{name}_bestpoint.pt",
  42         "onnx": source / f"{name}.onnx",
  43         "yaml": source / f"{name}.yaml",
  44         "reference_motion": source / f"{name}_reference_motion.onnx",
  45     }
  46     for key, path in files.items():
  47         if hashlib.sha256(path.read_bytes()).hexdigest() != manifest[f"{key}_sha256"]:
  48             raise ValueError(f"source checksum mismatch: {path}")
  49     inputs, _ = check_model(files["onnx"])
  50     command_name = next(key for key, _ in inputs if key != "policy")
  51     config = yaml.safe_load(files["yaml"].read_text(encoding="utf-8"))
  52     names = config["policy_joint_names"]
  53     if len(names) != 21 or len(set(names)) != 21 or names != config["joint_names_simulation"]:
  54         raise ValueError("expected 21 unique joints in the simulation/policy order")
  55     policy_obs = config["observation"]["policy"]
  56     if list(policy_obs) != ["root_ang_vel_history", "projected_gravity_history", "joint_pos_history",
  57                             "joint_vel_history", "prev_actions"]:
  58         raise ValueError("unexpected policy observation order")
  59     if any(policy_obs[key]["history_steps"] != HISTORY_STEPS for key in list(policy_obs)[:4]) or \
  60             policy_obs["prev_actions"]["steps"] != 3:
  61         raise ValueError("expected history offsets 0/1/2/3/4/8/16 and 3 previous actions")
  62     command = config["observation"]["command_short"]
  63     if list(command) != ["ref_root_pos_future_local", "ref_root_ori_future_b", "ref_joint_pos_future"] or \
  64             any(value["future_steps"] != FUTURE_STEPS for value in command.values()) or \
  65             config["motion"]["future_steps"] != FUTURE_STEPS or \
  66             command["ref_root_pos_future_local"]["body_names"] != "base_link":
  67         raise ValueError("unexpected command signals, order, root body or future offsets")
  68 
  69     _, _, meta = check_reference_model(files["reference_motion"])
  70     if meta["joint_names"].split(",") != names:
  71         raise ValueError("reference joint order does not match policy")
  72     fps, count = float(meta["fps"]), int(meta["num_frames"])
  73     if float(config.get("policy_frequency", fps)) != fps:
  74         raise ValueError("reference and policy frequency mismatch")
  75     root_index = meta["body_names"].split(",").index("base_link")
  76 
  77     robot = {
  78         "policy_type": "rl_wbc", "policy_name": "policy", "policy_frequency": 50,
  79         "lowlevel_frequency": 500, "num_actions": 21,
  80         "joint_order_mj": names, "joint_order_isaac": names.copy(),
  81         "default_joint_angles": joint_values(config["default_joint_pos"], names, "default_joint_pos"),
  82         "kps": joint_values(config["joint_kp"], names, "joint_kp"),
  83         "kds": joint_values(config["joint_kd"], names, "joint_kd"),
  84         "action_scale": [float(value) for value in config["action_scale"]],
  85         # Same default as the handoff's build_monolithic_robot_policy.py.
  86         "clip_actions": float(config.get("clip_actions", 1000000.0)),
  87     }
  88     if len(robot["action_scale"]) != 21 or not np.isfinite(robot["action_scale"]).all() or \
  89             min(robot["kps"] + robot["kds"] + robot["action_scale"]) < 0 or \
  90             not np.isfinite(robot["clip_actions"]) or robot["clip_actions"] <= 0:
  91         raise ValueError("invalid gains, action scales or action clipping")
  92     deploy = {"RobotCfg": robot, "WbcCfg": {
  93         "motion_id": name,
  94         "reference_num_frames": count, "reference_root_body_index": root_index,
  95         "control_dt": 0.02, "start_hold_sec": 1.0, "history_steps": HISTORY_STEPS,
  96         "future_steps": FUTURE_STEPS, "previous_action_steps": 3,
  97         "policy_input_name": "policy", "command_input_name": command_name, "output_name": "action",
  98     }}
  99     if output.exists() and any(output.iterdir()):
 100         raise ValueError("output directory must be empty; do not mix an old RKNN with a new motion")
 101     output.mkdir(parents=True, exist_ok=True)
 102     (output / "rl_config.yaml").write_text(yaml.safe_dump(deploy, sort_keys=False), encoding="utf-8")
 103     for path, filename in [(files["onnx"], "policy.onnx"), (files["yaml"], "source_policy.yaml"),
 104                            (files["reference_motion"], "reference_motion.onnx")]:
 105         shutil.copyfile(path, output / filename)
 106     print(f"Packaged {name}: {count} frames at {fps:g} Hz, {count / fps:.2f}s + 1s start hold")
 107     print(f"Convert {output / 'policy.onnx'} and {output / 'reference_motion.onnx'} to RKNN before deployment")

 110 def install_lora(source, output):
 111     """Validate the converted source; stage only RKNN runtime assets for colcon."""
 112     from wbc_lora_assets import ID_PATTERN, asset_path, check_graph, load_adapter, read_json, sha256
 113 
 114     if output.exists() or output.is_symlink():
 115         raise ValueError("install-lora requires a new output directory; use a versioned bundle name for updates")
 116     if output.is_relative_to(source) or source.is_relative_to(output):
 117         raise ValueError("installation output and source delivery must not be nested")
 118     manifest_path = source / "base_manifest.json"
 119     manifest = read_json(manifest_path)
 120     graph = check_graph(asset_path(source, manifest["model"]["onnx"]), manifest_path)
 121     if graph["kind"] != "policy":
 122         raise ValueError("install-lora requires the full shared policy, not a probe")
 123     files = {manifest_path: sha256(manifest_path)}
 124 
 125     def include(relative, expected=None):
 126         path = asset_path(source, relative)
 127         if path.suffix.lower() not in (".json", ".onnx", ".rknn", ".yaml", ".bin"):
 128             raise ValueError(f"Not a deployment asset (checkpoints are forbidden): {relative}")
 129         if not path.is_file():
 130             raise ValueError(f"Missing deployment asset: {path}")
 131         actual = sha256(path)
 132         if expected is not None and actual != expected:
 133             raise ValueError(f"Deployment asset checksum mismatch: {path}")
 134         files[path] = actual
 135         return path
 136 
 137     def model(onnx, rknn, record_name, expected, graph_record=False):
 138         if Path(onnx).suffix != ".onnx" or Path(rknn).suffix != ".rknn":
 139             raise ValueError("Expected ONNX source and RKNN runtime model paths")
 140         include(onnx, expected)
 141         if record_name != rknn + ".json":
 142             raise ValueError("RKNN conversion record must be beside its model")
 143         record = read_json(include(record_name))
 144         if record.get("format") != "wbc_rknn_conversion_v1" or record.get("source_sha256") != expected or \
 145                 record.get("precision") != "float16" or record.get("do_quantization") is not False:
 146             raise ValueError(f"Invalid conversion record: {record_name}")
 147         if graph_record and record.get("graph_manifest_sha256") != files[manifest_path]:
 148             raise ValueError("Shared RKNN was converted with another graph manifest")
 149         include(rknn, record["rknn_sha256"])
 150 
 151     shared = graph["model"]
 152     model(shared["onnx"], shared["rknn"], shared["rknn_record"], shared["sha256"], True)
 153     seen = set()
 154     for action in graph["actions"]:
 155         action_id = action["action_id"]
 156         if not ID_PATTERN.fullmatch(action_id) or action_id in seen:
 157             raise ValueError("Invalid or duplicate action ID")
 158         seen.add(action_id)
 159         adapter = include(action["adapter_manifest"])
 160         load_adapter(adapter, graph, action_id)
 161         info = read_json(adapter)
 162         weights = asset_path(adapter.parent, info["weights"]["path"])
 163         include(str(weights.relative_to(source)), info["weights"]["sha256"])
 164         config = include(action["config"], action["config_sha256"])
 165         if yaml.safe_load(config.read_text(encoding="utf-8"))["WbcCfg"]["motion_id"] != action_id:
 166             raise ValueError(f"Action/config identity mismatch: {action_id}")
 167         check_reference_model(asset_path(source, action["reference_onnx"]))
 168         model(action["reference_onnx"], action["reference_rknn"], action["reference_rknn_record"],
 169               action["reference_onnx_sha256"])
 170     if not seen:
 171         raise ValueError("No actions in deployment bundle")
 172     # C++ verifies RKNN provenance against the hashes retained in the manifest;
 173     # it does not open ONNX. Keep source models in model_sources for rebuilding.
 174     files = {path: digest for path, digest in files.items() if path.suffix != ".onnx"}
 175     output.parent.mkdir(parents=True, exist_ok=True)
 176     with tempfile.TemporaryDirectory(prefix=".wbc_lora_stage_", dir=output.parent) as temporary:
 177         staged = Path(temporary) / "bundle"
 178         staged.mkdir()
 179         for path, digest in files.items():
 180             destination = staged / path.relative_to(source)
 181             destination.parent.mkdir(parents=True, exist_ok=True)
 182             shutil.copyfile(path, destination)
 183             if sha256(destination) != digest:
 184                 raise ValueError(f"Source changed during installation: {path}")
 185         (staged / "INSTALLATION.json").write_text(json.dumps({
 186             "format": "wbc_lora_installation_v1", "validation_status": "pending_formal_runtime",
 187             "asset_profile": "rknn_runtime_only",
 188             "files": {str(path.relative_to(source)): digest for path, digest in files.items()},
 189         }, indent=2) + "\n", encoding="utf-8")
 190         if output.exists():
 191             raise ValueError("Installation destination appeared during staging")
 192         staged.rename(output)
 193     print(f"Installed {len(seen)} actions / one shared WBC Base at {output}")
 194     print("Asset checks passed; colcon build and formal C++ runtime validation remain required")
 195

 197 if __name__ == "__main__":
 198     parser = argparse.ArgumentParser(description=__doc__)
 199     parser.add_argument("source", type=Path, help="complete exported motion directory")
 200     parser.add_argument("output", type=Path, help="empty deployment policy directory")
 201     parser.add_argument("--mode", choices=("static", "lora", "install-lora"), default="static")
 202     selection = parser.add_mutually_exclusive_group()
 203     selection.add_argument("--actions", nargs="+", help="selected LoRA action IDs")
 204     selection.add_argument("--all-actions", action="store_true", help="export every action in emotion_index.json")
 205     parser.add_argument("--base-actions", nargs="+", default=[], help="subset of --actions using complete Base parameters")
 206     parser.add_argument("--samples", type=int, default=33, help="LoRA CPU verification samples")
 207     parser.add_argument("--fp16", action="store_true", help="export and verify FP16 Base/reference storage in LoRA mode")
 208     args = parser.parse_args()
 209     if args.mode == "lora":
 210         if not args.actions and not args.all_actions:
 211             parser.error("--mode lora requires --actions or --all-actions")
 212         from export_wbc_lora import export_delivery
 213         if len(set(args.base_actions)) != len(args.base_actions) or (args.actions and not set(args.base_actions) <= set(args.actions)):
 214             parser.error("--base-actions must be a unique subset of --actions")
 215         export_delivery(args.source, args.output, args.actions, args.samples, args.base_actions, args.fp16)
 216     else:
 217         if args.actions or args.all_actions or args.base_actions or args.fp16:
 218             parser.error("--actions/--all-actions/--base-actions/--fp16 require --mode lora")
 219         operation = install_lora if args.mode == "install-lora" else prepare
 220         operation(args.source.resolve(), args.output.resolve())
