   1 #!/usr/bin/env python3
   2 """Stage an FP16 WBC delivery for the existing RK3588 build-only converters.
   3 
   4 Never changes the delivery, loads checkpoints, runs inference or compiles RKNN.
   5 The shared policy and adapters are copied byte-for-byte. Reference table/Cast
   6 paths are materialized as FP32 tables with six direct Gather outputs, preserving
   7 the delivered FP16 values and the original frame-index calculation.
   8 """

  28 def load_embedded_model(path):
  29     import onnx
  30 
  31     model = onnx.load(str(path), load_external_data=False)
  32     if any(t.data_location == onnx.TensorProto.EXTERNAL for t in model.graph.initializer):
  33         raise ValueError(f"External ONNX tensor data is not supported: {path}")
  34     if any(a.type in (onnx.AttributeProto.GRAPH, onnx.AttributeProto.GRAPHS)
  35            for n in model.graph.node for a in n.attribute):
  36         raise ValueError(f"Subgraphs are not supported by this preprocessing contract: {path}")
  37     onnx.checker.check_model(model, full_check=True)
  38     return model

  41 def check_fp16_policy(path, graph):
  42     """Check the fixed Base matrices without evaluating the policy."""
  43     import onnx
  44     from onnx import numpy_helper
  45 
  46     model = load_embedded_model(path)
  47     shapes = {p["name"]: p["shape"] for p in graph["parameters"]}
  48     expected = sorted((shapes[f"lora_{i}_a"][1], shapes[f"lora_{i}_b"][0]) for i in range(7))
  49     matrices = []
  50     for tensor in model.graph.initializer:
  51         if tensor.data_type not in (onnx.TensorProto.FLOAT, onnx.TensorProto.FLOAT16):
  52             continue
  53         values = numpy_helper.to_array(tensor)
  54         if not np.isfinite(values).all():
  55             raise ValueError(f"Non-finite policy initializer: {tensor.name}")
  56         if len(tensor.dims) == 2:
  57             if tensor.data_type != onnx.TensorProto.FLOAT16:
  58                 raise ValueError(f"Expected an FP16 Base matrix: {tensor.name}; finish training-side export first")
  59             matrices.append(tuple(tensor.dims))
  60     if sorted(matrices) != expected:
  61         raise ValueError("Expected the seven native FP16 Base matrix initializers; optimized/fused graphs are unsupported")

  64 def normalize_reference(path):
  65     """Only rewrite constant-table -> Gather -> Cast/Identity output paths."""
  66     model = load_embedded_model(path)
  67     _, _, meta = check_reference_model(path)
  68     return normalize_reference_graph(model, meta), meta

  71 def normalize_reference_graph(model, meta):
  72     import onnx
  73     from onnx import helper, numpy_helper
  74 
  75     count = int(meta["num_frames"])
  76     nodes = list(model.graph.node)
  77     producers = {name: (i, node) for i, node in enumerate(nodes) for name in node.output if name}
  78     tensors = {t.name: t for t in model.graph.initializer}
  79     allowed_types = {onnx.TensorProto.FLOAT: np.float32, onnx.TensorProto.FLOAT16: np.float16}
  80 
  81     def unwrap(value):
  82         casts, visited = [], set()
  83         while value in producers:
  84             index, node = producers[value]
  85             if node.op_type not in ("Identity", "Cast"):
  86                 break
  87             if index in visited or node.domain not in ("", "ai.onnx") or \
  88                     len(node.input) != 1 or len(node.output) != 1:
  89                 raise ValueError(f"Unsupported or cyclic reference wrapper: {value}")
  90             visited.add(index)
  91             attrs = {a.name: helper.get_attribute_value(a) for a in node.attribute}
  92             if node.op_type == "Identity":
  93                 if attrs:
  94                     raise ValueError("Unexpected reference Identity attributes")
  95             else:
  96                 if set(attrs) != {"to"} or attrs["to"] not in allowed_types:
  97                     raise ValueError("Reference table/output Cast must only convert FP16/FP32")
  98                 casts.append(allowed_types[attrs["to"]])
  99             value = node.input[0]
 100         return value, list(reversed(casts))
 101 
 102     tables, gather_indices, frame_index = {}, set(), None
 103     for name, shape in REFERENCE_OUTPUTS.items():
 104         value, output_casts = unwrap(name)
 105         if value not in producers:
 106             raise ValueError(f"{name}: expected a Gather producer")
 107         index, node = producers[value]
 108         attrs = {a.name: helper.get_attribute_value(a) for a in node.attribute}
 109         if node.domain not in ("", "ai.onnx") or node.op_type != "Gather" or \
 110                 len(node.input) != 2 or len(node.output) != 1 or \
 111                 set(attrs) - {"axis"} or attrs.get("axis", 0) != 0:
 112             raise ValueError(f"{name}: expected one axis-0 table Gather")
 113         gather_indices.add(index)
 114         if frame_index is None:
 115             frame_index = node.input[1]
 116         if frame_index != node.input[1]:
 117             raise ValueError("All reference outputs must share the same frame-index tensor")
 118         table_name, input_casts = unwrap(node.input[0])
 119         if table_name not in tensors or tensors[table_name].data_type != onnx.TensorProto.FLOAT16:
 120             raise ValueError(f"{name}: expected an embedded FP16 reference table")
 121         table = numpy_helper.to_array(tensors[table_name])
 122         if table.shape != (count,) + shape[1:] or not np.isfinite(table).all():
 123             raise ValueError(f"{name}: invalid reference table shape or value")
 124         # Float Cast commutes with Gather. Materialize only the declared data
 125         # casts; leave all computation of the frame index unchanged.
 126         for dtype in input_casts + output_casts:
 127             table = table.astype(dtype)
 128         if table.dtype != np.float32:
 129             raise ValueError(f"{name}: the table/output path does not produce the required FP32 interface")
 130         tables[name] = table
 131     if len(gather_indices) != 6 or sum(n.op_type == "Gather" for n in nodes) != 6:
 132         raise ValueError("Expected six distinct native reference table Gather branches")
 133 
 134     # Keep the exact original subgraph feeding the common frame index.
 135     retained_nodes, retained_values, pending = set(), set(), [frame_index]
 136     graph_inputs = {v.name for v in model.graph.input}
 137     while pending:
 138         value = pending.pop()
 139         if not value or value in retained_values:
 140             continue
 141         retained_values.add(value)
 142         if value in producers:
 143             index, node = producers[value]
 144             if index in gather_indices:
 145                 raise ValueError("Reference frame index must not depend on a trajectory Gather")
 146             retained_nodes.add(index)
 147             pending.extend(node.input)
 148         elif value not in tensors and value not in graph_inputs:
 149             raise ValueError(f"Unresolved reference frame-index input: {value}")
 150 
 151     prefix = "wbc_preprocess_reference_"
 152     names = set(tensors) | set(producers) | graph_inputs
 153     if any(name.startswith(prefix) for name in names):
 154         raise ValueError("Reference already uses preprocessing names; supply the original FP16 delivery")
 155     retained = [node for i, node in enumerate(nodes) if i in retained_nodes]
 156     initializers = [t for t in model.graph.initializer if t.name in retained_values]
 157     for name in REFERENCE_OUTPUTS:
 158         table_name = prefix + name + "_table"
 159         initializers.append(numpy_helper.from_array(tables[name], table_name))
 160         retained.append(helper.make_node("Gather", [table_name, frame_index], [name],
 161                                          axis=0, name=prefix + name + "_gather"))
 162     del model.graph.node[:]
 163     model.graph.node.extend(retained)
 164     del model.graph.initializer[:]
 165     model.graph.initializer.extend(initializers)
 166     del model.graph.value_info[:]
 167     onnx.checker.check_model(model, full_check=True)
 168     # Exercise the existing converter's graph rewrite, without RKNN or inference.
 169     packed = copy.deepcopy(model)
 170     pack_reference_gathers(packed)
 171     onnx.checker.check_model(packed, full_check=True)
 172     return model

 175 def prepare(source, output, dry_run=False):
 176     import onnx
 177 
 178     source, output = source.resolve(), output.resolve()
 179     if not source.is_dir() or output.exists() or output.is_symlink():
 180         raise ValueError("Source must be a directory and output must not exist")
 181     if output.is_relative_to(source) or source.is_relative_to(output):
 182         raise ValueError("Source and output must be separate, non-nested directories")
 183     manifest_path = source / "base_manifest.json"
 184     manifest_hash = sha256(manifest_path)
 185     graph = read_json(manifest_path)
 186     model_path = asset_path(source, graph["model"]["onnx"])
 187     graph = check_graph(model_path, manifest_path)
 188     if graph["kind"] != "policy":
 189         raise ValueError("Preprocessing requires a full shared policy bundle, not a probe")
 190     check_fp16_policy(model_path, graph)
 191     actions = graph.get("actions", [])
 192     ids = [a["action_id"] for a in actions]
 193     if not ids or len(ids) != len(set(ids)) or any(not ID_PATTERN.fullmatch(x) for x in ids):
 194         raise ValueError("Expected unique, valid action IDs")
 195 
 196     input_hashes = {"base_manifest.json": manifest_hash}
 197     output_hashes, reserved, references = {}, set(), []
 198 
 199     def reserve(relative):
 200         path = Path(relative)
 201         if not relative or path.is_absolute() or ".." in path.parts or path == Path("."):
 202             raise ValueError(f"Invalid bundle path: {relative}")
 203         if any(path == old or path in old.parents or old in path.parents for old in reserved):
 204             raise ValueError(f"Overlapping bundle destinations: {relative}")
 205         reserved.add(path)
 206 
 207     reserve("base_manifest.json")
 208     reserve("PREPROCESSING.json")
 209     reserve("ASSETS.sha256")
 210     for model in [graph["model"]] + [
 211             {"rknn": a["reference_rknn"], "rknn_record": a["reference_rknn_record"]} for a in actions]:
 212         if not model["rknn"].endswith(".rknn") or model["rknn_record"] != model["rknn"] + ".json":
 213             raise ValueError("Expected adjacent RKNN model/conversion record paths")
 214         reserve(model["rknn"])
 215         reserve(model["rknn_record"])
 216 
 217     with ExitStack() as stack:
 218         stage = None
 219         if not dry_run:
 220             output.parent.mkdir(parents=True, exist_ok=True)
 221             temporary = stack.enter_context(tempfile.TemporaryDirectory(prefix=".wbc_preprocess_", dir=output.parent))
 222             stage = Path(temporary) / "bundle"
 223             stage.mkdir()
 224 
 225         def include(relative, expected=None, copy_file=True):
 226             reserve(relative)
 227             path = asset_path(source, relative)
 228             digest = sha256(path)
 229             if expected is not None and digest != expected:
 230                 raise ValueError(f"Delivery checksum mismatch: {relative}; do not regenerate stale hashes on the board")
 231             input_hashes[relative] = digest
 232             if copy_file:
 233                 output_hashes[relative] = digest
 234                 if stage:
 235                     import shutil
 236                     destination = stage / relative
 237                     destination.parent.mkdir(parents=True, exist_ok=True)
 238                     shutil.copyfile(path, destination)
 239                     if sha256(destination) != digest:
 240                         raise ValueError(f"Delivery changed while copying: {relative}")
 241             return path
 242 
 243         include(graph["model"]["onnx"], graph["model"]["sha256"])
 244         for action in actions:
 245             action_id = action["action_id"]
 246             adapter_path = include(action["adapter_manifest"])
 247             load_adapter(adapter_path, graph, action_id)
 248             adapter = read_json(adapter_path)
 249             asset_path(adapter_path.parent, adapter["weights"]["path"])
 250             weights_relative = str(Path(action["adapter_manifest"]).parent / adapter["weights"]["path"])
 251             include(weights_relative, adapter["weights"]["sha256"])
 252             config_path = include(action["config"], action["config_sha256"])
 253             reference = include(action["reference_onnx"], action["reference_onnx_sha256"], copy_file=False)
 254             normalized, meta = normalize_reference(reference)
 255             config = yaml.safe_load(config_path.read_text(encoding="utf-8"))
 256             robot, wbc = config["RobotCfg"], config["WbcCfg"]
 257             joints = meta["joint_names"].split(",")
 258             if wbc["motion_id"] != action_id or robot["policy_type"] != "rl_wbc" or \
 259                     robot["joint_order_mj"] != joints or robot["joint_order_isaac"] != joints or \
 260                     robot["num_actions"] != 21 or robot["policy_frequency"] != 50 or \
 261                     wbc["reference_num_frames"] != int(meta["num_frames"]) or \
 262                     wbc["reference_root_body_index"] != meta["body_names"].split(",").index("base_link"):
 263                 raise ValueError(f"Action/config/reference contract mismatch: {action_id}")
 264             record = {"action_id": action_id, "reference": action["reference_onnx"],
 265                       "source_sha256": action["reference_onnx_sha256"],
 266                       "source_table_dtype": "float16", "prepared_table_dtype": "float32",
 267                       "frame_index_subgraph": "preserved", "num_frames": int(meta["num_frames"])}
 268             if stage:
 269                 destination = stage / action["reference_onnx"]
 270                 destination.parent.mkdir(parents=True, exist_ok=True)
 271                 onnx.save_model(normalized, str(destination), save_as_external_data=False)
 272                 check_reference_model(destination)
 273                 prepared_hash = sha256(destination)
 274                 output_hashes[action["reference_onnx"]] = prepared_hash
 275                 action["reference_onnx_sha256"] = prepared_hash
 276                 record["prepared_sha256"] = prepared_hash
 277             references.append(record)
 278 
 279         # Diagnostics belong to the training archive, not the build-only bundle.
 280         graph["probes"] = []
 281         for key in ("verification_data", "verification_expected", "verification_data_sha256",
 282                     "verification_expected_sha256"):
 283             graph.pop(key, None)
 284         for relative, digest in input_hashes.items():
 285             if sha256(asset_path(source, relative)) != digest:
 286                 raise ValueError(f"Delivery changed during preprocessing: {relative}")
 287         report = {"format": "wbc_lora_preprocess_v1", "source_manifest_sha256": manifest_hash,
 288                   "source_directory": str(source), "prepared_directory": str(output),
 289                   "preprocessor_sha256": sha256(Path(__file__).resolve()),
 290                   "onnx_version": importlib.metadata.version("onnx"),
 291                   "numpy_version": np.__version__, "numerical_verification": "not_run",
 292                   "source_files": input_hashes, "references": references, "dry_run": dry_run}
 293         if stage:
 294             (stage / "base_manifest.json").write_text(json.dumps(graph, indent=2, ensure_ascii=False) + "\n", encoding="utf-8")
 295             check_graph(stage / graph["model"]["onnx"], stage / "base_manifest.json")
 296             output_hashes["base_manifest.json"] = sha256(stage / "base_manifest.json")
 297             report["prepared_files"] = output_hashes
 298             (stage / "PREPROCESSING.json").write_text(json.dumps(report, indent=2, ensure_ascii=False) + "\n", encoding="utf-8")
 299             checksums = [f"{sha256(path)}  {path.relative_to(stage).as_posix()}\n"
 300                          for path in sorted(stage.rglob("*")) if path.is_file()]
 301             (stage / "ASSETS.sha256").write_text("".join(checksums), encoding="utf-8")
 302             if output.exists() or output.is_symlink():
 303                 raise ValueError("Output appeared during preprocessing; refusing to replace it")
 304             stage.rename(output)
 305     print(f"{'DRY-RUN checked' if dry_run else 'Prepared'} {len(actions)} actions: {output}")
 306     print("Source delivery unchanged; no checkpoint loading, inference or RKNN compilation")
 307     return report

 310 def main():
 311     parser = argparse.ArgumentParser(description=__doc__)
 312     parser.add_argument("source", type=Path, help="complete training FP16 delivery directory (read-only)")
 313     parser.add_argument("output", type=Path, help="new prepared bundle directory outside source")
 314     parser.add_argument("--dry-run", action="store_true", help="check the entire delivery and reference rewrite without writing files")
 315     args = parser.parse_args()
 316     if args.output.is_symlink():
 317         parser.error("output must not be a symlink")
 318     prepare(args.source, args.output, args.dry_run)
 319     return 0
