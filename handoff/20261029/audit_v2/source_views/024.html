 114 void LeggedController::plannerCommandCallback(const geometry_msgs::msg::TwistStamped::ConstSharedPtr msg) {
 115   const int64_t stamp_ns = rclcpp::Time(msg->header.stamp).nanoseconds();
 116   const rclcpp::Time current = nh_->now();
 117   const double age = (current - rclcpp::Time(msg->header.stamp)).seconds();
 118   const std::array<double, 6> values{{msg->twist.linear.x, msg->twist.linear.y, msg->twist.linear.z,
 119                                       msg->twist.angular.x, msg->twist.angular.y, msg->twist.angular.z}};
 120   const bool finite = std::all_of(values.begin(), values.end(), [](double value) { return std::isfinite(value); });
 121   const bool planar = msg->twist.linear.z == 0.0 && msg->twist.angular.x == 0.0 && msg->twist.angular.y == 0.0;
 122 
 123   std::lock_guard<std::mutex> lock(plannerCommandMutex_);
 124   plannerCommandSeen_ = true;
 125   if (plannerCommandClockNs_ > 0 && current.nanoseconds() < plannerCommandClockNs_) plannerCommandStampNs_ = 0;
 126   plannerCommandClockNs_ = current.nanoseconds();
 127   if (!finite || !planar || msg->header.frame_id != kPlannerCommandFrame || stamp_ns <= 0 || age < 0.0 || age > 0.5 ||
 128       (plannerCommandStampNs_ > 0 && stamp_ns < plannerCommandStampNs_))
 129   {
 130     plannerCommandValid_ = false;
 131     return;
 132   }
 133 
 134   plannerCommand_.linear.x = msg->twist.linear.x;
 135   plannerCommand_.linear.y = msg->twist.linear.y;
 136   plannerCommand_.angular.z = msg->twist.angular.z;
 137   plannerCommandStampNs_ = stamp_ns;
 138   plannerCommandReceiveTime_ = std::chrono::steady_clock::now();
 139   plannerCommandValid_ = true;
 140 }

 142 std::array<double, 3> LeggedController::readCurrentHeadRpyRad() {
 143   std::array<double, 3> head{{0.0, 0.0, 0.0}};
 144   if (!robot_hw_) return head;
 145   const auto states = robot_hw_->getHyridJointMotorState();
 146   for (std::size_t i = 0; i < jointNames_.size() && i < states.size(); ++i)
 147   {
 148     const auto& name = jointNames_[i];
 149     if (name.find("neck_yaw") != std::string::npos || name.find("head_yaw") != std::string::npos)
 150       head[0] = states[i].q;
 151     else if (name.find("neck_pitch") != std::string::npos || name.find("head_pitch") != std::string::npos)
 152       head[1] = states[i].q;
 153     else if (name.find("neck_roll") != std::string::npos || name.find("head_roll") != std::string::npos)
 154       head[2] = states[i].q;
 155   }
 156   return head;
 157 }

 168   const double wall_sec = nh_->now().seconds();
 169   faceFollowCtrl_.processTarget(

 173 void LeggedController::imitationMotionDoneCallback(const std_msgs::msg::Bool::ConstSharedPtr msg) {
 174   if (!msg->data) return;
 175 
 176   if (sequence_engine_enabled_)
 177   {
 178     const ActiveSequencePhaseSnapshot snapshot = sequenceRuntime_.activePhaseSnapshot(last_controller_wall_sec_);
 179     if (snapshot.active && snapshot.phase.mode == ActionSequenceMode::kRlImitation)
 180     {
 181       const double t_wall = last_controller_wall_sec_;
 182       bool next_phase_is_imitation = false;
 183       sequenceRuntime_.completeImitationPhaseByExternalEvent(t_wall, next_phase_is_imitation);
 184       return;
 185     }
 186   }
 187 
 188   if (policyMux_.source() != PolicyCmdSource::kImitation) return;
 189 
 190   if (policyMux_.singleNodePublishesImitationCommand())
 191   {
 192     return;
 193   }
 194 
 195   returnToRlWithoutGainRamp(nullptr);
 196 }
