   9 std::string quoted(const std::string& value) {
  10   static const char hex[] = "0123456789abcdef";
  11   std::string out = "\"";
  12   for (unsigned char c : value) {
  13     if (c == '"' || c == '\\') { out += '\\'; out += static_cast<char>(c); }
  14     else if (c < 32) { out += "\\u00"; out += hex[c >> 4]; out += hex[c & 15]; }
  15     else out += static_cast<char>(c);
  16   }
  17   return out + '"';
  18 }

  25 void RLWbc::initializeJsonReference() {
  26   node_->get_parameter("wbc_json_base_action", json_base_action_);
  27   if (json_base_action_.empty()) return;
  28   if (!lora_assets_ || !motions_.count(json_base_action_) ||
  29       lora_assets_->action(json_base_action_).parameter_source != "base")
  30     throw std::runtime_error("wbc_json_base_action must select a preloaded complete Base parameter pack");
  31   std::string mapping_file;
  32   if (!node_->get_parameter("mapping_file", mapping_file) || mapping_file.empty())
  33     throw std::runtime_error("JSON WBC requires the robot mapping_file");
  34   json_joint_limits_ = wbcJsonJointLimits(mapping_file, joint_names_);
  35   // Cache through the exact ordinary WBC reference reader before its control
  36   // thread starts. JSON admission never queries a live reference context.
  37   json_base_reference_ = inferReferenceFrame(*motions_.at(json_base_action_), 0);
  38   RCLCPP_INFO(node_->get_logger(), "Text JSON reference uses WBC Base action=%s frame=0 root_height=%.6f; no measured height gate",
  39               json_base_action_.c_str(), json_base_reference_.root_position[2]);
  40   json_status_pub_ = node_->create_publisher<std_msgs::msg::String>("/text_motion/wbc_status", 20);
  41   json_callback_group_ = node_->create_callback_group(rclcpp::CallbackGroupType::MutuallyExclusive);
  42   json_cancel_group_ = node_->create_callback_group(rclcpp::CallbackGroupType::MutuallyExclusive);
  43   rclcpp::SubscriptionOptions options;
  44   options.callback_group = json_callback_group_;
  45   json_sub_ = node_->create_subscription<std_msgs::msg::String>("/text_motion/json",
  46       rclcpp::QoS(1).reliable().durability_volatile(),
  47       std::bind(&RLWbc::jsonReferenceCallback, this, std::placeholders::_1), options);
  48   options.callback_group = json_cancel_group_;
  49   json_cancel_sub_ = node_->create_subscription<std_msgs::msg::String>("/text_motion/cancel", 10,
  50       std::bind(&RLWbc::jsonCancelCallback, this, std::placeholders::_1), options);
  51   RCLCPP_INFO(node_->get_logger(), "JSON WBC enabled: shared Base %s; complete Base parameters=%s; /text_motion/json",
  52               lora_assets_->baseHash().c_str(), json_base_action_.c_str());
  53 }

  62 void RLWbc::jsonReferenceCallback(const std_msgs::msg::String::ConstSharedPtr msg) {
  63   std::string id;
  64   try {
  65     if (msg->data.size() > 4U * 1024U * 1024U) throw std::runtime_error("JSON exceeds 4 MiB");
  66     const auto header = YAML::Load(msg->data);
  67     id = header["request_id"].as<std::string>("");
  68     if (header["root_reference_source"].as<std::string>("snapshot") == "wbc_base_action" &&
  69         header["wbc_base_action"].as<std::string>("") != json_base_action_)
  70       throw std::runtime_error("JSON root reference Base action mismatch");
  71     {
  72       std::lock_guard<std::mutex> lock(mutex_);
  73       if (std::find(json_cancelled_ids_.begin(), json_cancelled_ids_.end(), id) != json_cancelled_ids_.end()) {
  74         publishJsonStatus(id, "cancelled", "cancelled before JSON admission");
  75         return;
  76       }
  77       if (std::find(json_recent_ids_.begin(), json_recent_ids_.end(), id) != json_recent_ids_.end()) {
  78         publishJsonStatus(id, "duplicate", "JSON request already handled");
  79         return;
  80       }
  81     }
  82     std::unique_lock<std::mutex> loading_lock(loading_mutex_, std::try_to_lock);
  83     if (!loading_lock.owns_lock()) throw std::runtime_error("another WBC preparation is loading");
  84     auto reference = std::make_shared<WbcJsonReference>(msg->data, joint_names_, json_joint_limits_, lora_assets_->baseHash(),
  85         &json_base_reference_.root_position, &json_base_reference_.root_orientation);
  86     auto motion = std::make_unique<Motion>();
  87     motion->robot_cfg_path_ = lora_assets_->action(json_base_action_).config;
  88     loadConfig(*motion, motion->robot_cfg_path_);
  89     motion->motion_id_ = reference->id();
  90     motion->json_reference_ = reference;
  91     motion->reference_num_frames_ = static_cast<int>(reference->size());
  92     motion->reference_root_body_index_ = 0;
  93     initializeModel(*motion);  // Reuse policy ABI checks; no reference RKNN is created.
  94     {
  95       std::lock_guard<std::mutex> lock(mutex_);
  96       if (active_ || prepared_ || preparing_) throw std::runtime_error("WBC is preparing/prepared/active");
  97       if (std::find(json_cancelled_ids_.begin(), json_cancelled_ids_.end(), id) != json_cancelled_ids_.end()) {
  98         publishJsonStatus(id, "cancelled", "cancelled before JSON admission");
  99         return;
 100       }
 101       if (std::find(json_recent_ids_.begin(), json_recent_ids_.end(), id) != json_recent_ids_.end()) {
 102         publishJsonStatus(id, "duplicate", "JSON request already handled");
 103         return;
 104       }
 105       for (const auto& entry : motions_) {
 106         const auto& previous = entry.second;
 107         if (previous->json_reference_ && !previous->json_consumed_ && !previous->json_reference_->expired())
 108           throw std::runtime_error("a complete JSON is already waiting for its WBC command");
 109       }
 110       // Retire prior dynamic references only while idle and under the same
 111       // mutex as runLoop/prepare. Ordinary LoRA motions remain preloaded.
 112       for (auto it = motions_.begin(); it != motions_.end();) {
 113         if (it->second->json_reference_) {
 114           if (current_ == it->second.get()) current_ = nullptr;
 115           it = motions_.erase(it);
 116         } else ++it;
 117       }
 118       if (reference->expired()) throw std::runtime_error("JSON expired before registration");
 119       remember(json_recent_ids_, id);
 120       motions_.emplace(id, std::move(motion));
 121     }
 122     publishJsonStatus(id, "ready", "complete JSON registered; Base parameters selected; no actuator command sent");
 123   } catch (const std::exception& error) {
 124     publishJsonStatus(id, "rejected", error.what());
 125   }
 126 }

 128 void RLWbc::jsonCancelCallback(const std_msgs::msg::String::ConstSharedPtr msg) {
 129   const auto& id = msg->data;
 130   if (id.size() > 32 || id.compare(0, 2, "T_") != 0) return;
 131   std::lock_guard<std::mutex> lock(mutex_);
 132   remember(json_cancelled_ids_, id);
 133   const auto motion = motions_.find(id);
 134   if (motion != motions_.end()) motion->second->json_consumed_ = true;
 135   const bool executing = current_ && current_->motion_id_ == id && (active_ || prepared_ || preparing_);
 136   publishJsonStatus(id, executing ? "cancelling" : "cancelled",
 137       executing ? "controller owns WBC to AMP handoff" : "cancelled before WBC activation");
 138 }
