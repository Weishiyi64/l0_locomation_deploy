  67 def load_limits(mapping_file):
  68     mapping = yaml.safe_load(Path(mapping_file).read_text(encoding="utf-8"))
  69     limits = []
  70     for name in JOINTS:
  71         matches = [motor for motor in mapping["motors"] if motor["joint_name"] == name]
  72         if len(matches) != 1 or matches[0].get("enabled") is not True:
  73             raise ValueError("Missing/duplicate/disabled joint in mapping: " + name)
  74         motor = matches[0]
  75         lo, hi = float(motor["joint_lower"]), float(motor["joint_upper"])
  76         if not np.isfinite([lo, hi]).all() or lo >= hi:
  77             raise ValueError("Invalid joint limits: " + name)
  78         limits.append((lo, hi))
  79     return np.asarray(limits, dtype=np.float64)

  82 def check_pose(pose, limits):
  83     if np.asarray(pose).dtype.kind not in "fiu":
  84         raise ValueError("Pose values must be numeric, excluding booleans")
  85     result = np.asarray(pose, dtype=np.float32)
  86     if result.shape != (9,) or not np.isfinite(result).all():
  87         raise ValueError("Pose must contain nine finite radians in robot joint order")
  88     if np.any(result < limits[:, 0] - 1e-6) or np.any(result > limits[:, 1] + 1e-6):
  89         raise ValueError("Pose exceeds the current robot mapping limits")
  90     return np.clip(result, limits[:, 0], limits[:, 1]).astype(np.float32)

  93 def mirror_arms(value):
  94     """Involution between delivered viewer-side columns and anatomical joints.
  95 
  96     Swap arms and invert shoulder/elbow roll; pitch and head axes are unchanged.
  97     Apply once to BOTH model inputs and outputs, never to already adapted JSON.
  98     """
  99     value = np.asarray(value, dtype=np.float32)
 100     result = value.copy()
 101     sign = np.asarray([1, -1, -1], dtype=np.float32)
 102     result[..., :3] = value[..., 3:6] * sign
 103     result[..., 3:6] = value[..., :3] * sign
 104     return result

 107 def configure_generation_limits(pipeline, limits):
 108     """Use mapping limits in every original composer operation, per pipeline.
 109 
 110     The frozen composer keeps its limits in module globals. Give this pipeline
 111     a private module instance so another pipeline's mapping cannot change them.
 112     Source files, model tensors and the original module remain untouched.
 113     """
 114     limits = np.asarray(limits, dtype=np.float64)
 115     if limits.shape != (9, 2) or not np.isfinite(limits).all() or np.any(limits[:, 0] >= limits[:, 1]):
 116         raise ValueError("Generation needs nine finite ordered robot joint limits")
 117     if np.array_equal(getattr(pipeline, "_robot_generation_limits", None), limits):
 118         return
 119     lower, upper = mirror_arms(limits[:, 0]), mirror_arms(limits[:, 1])
 120     model_limits = np.stack([np.minimum(lower, upper), np.maximum(lower, upper)], axis=1)
 121     spec = importlib.util.spec_from_file_location("_l0_robot_motion", BASELINE / "onnx/deploy_rknn/motion.py")
 122     module = importlib.util.module_from_spec(spec)
 123     spec.loader.exec_module(module)
 124     # The source composer uses head/left/right parts; trajectory columns use
 125     # right/left/head. mirror_arms already handles anatomical sides and signs.
 126     module.LIMITS = np.stack([model_limits[6:9], model_limits[3:6], model_limits[:3]])
 127     module.FULL_LIMITS = model_limits
 128     module.LIMITS.setflags(write=False)
 129     module.FULL_LIMITS.setflags(write=False)
 130     previous = pipeline.composer
 131     pipeline.composer = module.MotionComposer(previous.metadata, previous.atoms, previous.lengths, previous.a_pose)
 132     pipeline._robot_generation_limits = limits.copy()

 135 def validate_request(request):
 136     allowed = {"command", "request_id", "mode", "text", "token_ids", "speed", "seed",
 137                "return_home", "return_frames", "end_pose", "limit_policy"}
 138     if not isinstance(request, dict) or set(request) - allowed:
 139         raise ValueError("Unknown request fields")
 140     if not valid_id(request.get("request_id")) or request.get("command", "generate") != "generate":
 141         raise ValueError("Invalid generate request/id")
 142     mode = request.get("mode", "text")
 143     if mode not in ("text", "web"):
 144         raise ValueError("mode must be text or web")
 145     if mode == "text":
 146         text = request.get("text", "")
 147         if not isinstance(text, str) or not text.strip() or len(text) > 128:
 148             raise ValueError("text must contain 1..128 characters; whitespace is preserved")
 149     else:
 150         tokens = request.get("token_ids")
 151         if not isinstance(tokens, list) or len(tokens) != 3 or any(type(t) is not int for t in tokens):
 152             raise ValueError("web mode requires three integer token IDs: head/left/right")
 153     speed = request.get("speed", 1.2)
 154     if type(speed) not in (int, float) or not np.isfinite(speed) or not 0.1 <= speed <= 4.0:
 155         raise ValueError("Robot playback accepts speed in [0.1, 4.0]")
 156     if type(request.get("return_home", True)) is not bool:
 157         raise ValueError("return_home must be boolean")
 158     frames = request.get("return_frames", -1)
 159     if type(frames) is not int or frames not in (-1, 30, 60, 90):
 160         raise ValueError("return_frames must be -1, 30, 60 or 90")
 161     seed = request.get("seed")
 162     if seed is not None and (type(seed) is not int or not 0 <= seed < 2**32):
 163         raise ValueError("seed must be an integer in [0, 2**32)")
 164     if request.get("limit_policy", "reject") not in ("reject", "clip"):
 165         raise ValueError("limit_policy must be reject or clip")

 168 def generate(pipeline, request, initial, limits):
 169     validate_request(request)
 170     configure_generation_limits(pipeline, limits)
 171     initial = check_pose(initial, limits)
 172     end = check_pose(request.get("end_pose", initial), limits)
 173     output = pipeline.generate(
 174         mode=request.get("mode", "text"), text=request.get("text", ""), token_ids=request.get("token_ids"),
 175         init_pose=mirror_arms(initial), end_pose=mirror_arms(end),
 176         return_home=request.get("return_home", True), return_frames=request.get("return_frames", -1),
 177         speed=request.get("speed", 1.2), seed=request.get("seed"))
 178     tokens = np.asarray(output["token_ids"])
 179     if tokens.shape != (3,) or tokens.dtype.kind not in "iu":
 180         raise ValueError("Invalid model token IDs")
 181     # Token order is head/left/right. Own all three axes of each selected part,
 182     # including constant axes; variation thresholds do not express ownership.
 183     mask = np.repeat(tokens[[1, 2, 0]] != 0, 3)
 184     if not mask.any():
 185         return None
 186     motion = mirror_arms(output["motion"])
 187     count = np.asarray(output["frame_count"])
 188     if motion.ndim != 2 or motion.shape[1] != 9 or not 2 <= len(motion) <= 5000 or \
 189             count.shape != (1,) or count.dtype.kind not in "iu" or int(count[0]) != len(motion) or \
 190             not np.isfinite(motion).all():
 191         raise ValueError("Invalid complete model trajectory")
 192     if np.max(np.abs(motion[0, mask] - initial[mask])) > 0.005:
 193         raise ValueError("Generated first frame differs from the captured start pose")
 194     if request.get("return_home", True) and np.max(np.abs(motion[-1, mask] - end[mask])) > 0.005:
 195         raise ValueError("Generated active parts do not reach the requested end pose")
 196     clipped = np.clip(motion, limits[:, 0], limits[:, 1])
 197     adjustments = np.abs(clipped - motion)
 198     if np.any(adjustments > 1e-6) and request.get("limit_policy", "reject") == "reject":
 199         names = [name for j, name in enumerate(JOINTS) if np.any(adjustments[:, j] > 1e-6)]
 200         raise ValueError("Model exceeds robot limits: " + ", ".join(names) + "; explicit limit_policy=clip is required")
 201     return {
 202         "format": "l0_text_motion_v2", "robot": "l0_v5", "angle_unit": "rad", "fps": 50,
 203         "coordinate_contract": "robot_anatomical_v1", "request_id": request["request_id"],
 204         "source_zip_sha256": SOURCE_SHA256, "joint_names": JOINTS, "active_joints": mask.tolist(),
 205         "positions": clipped.tolist(), "frame_count": len(motion),
 206         "token_ids": tokens.tolist(), "material_ids": np.asarray(output["material_ids"]).tolist(),
 207         "backend": pipeline.backend, "inference_ms": output["inference_ms"],
 208         "generation_request": request, "start_pose": initial.tolist(), "end_pose": end.tolist(),
 209         "model_diagnostics": {
 210             "generation_joint_limits_rad": {name: list(map(float, bound)) for name, bound in zip(JOINTS, limits)},
 211             **{key: np.asarray(output[key]).tolist()
 212             for key in ("part_transition_frames", "action_frame_count_before_speed", "frame_count_before_speed",
 213                         "return_home_start_frame", "selector")}},
 214         "limit_policy": request.get("limit_policy", "reject"),
 215         "limit_adjustments": {name: {"count": int(np.count_nonzero(adjustments[:, j] > 1e-6)),
 216                                       "max_rad": float(adjustments[:, j].max())}
 217                               for j, name in enumerate(JOINTS)},
 218         # Renewed immediately before sending; this default also bounds offline replay.
 219         "expires_at_ms": int(time.time() * 1000) + 2000,
 220     }
