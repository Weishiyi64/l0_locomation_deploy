  12 def yaw_xyzw(q):
  13     q = np.asarray(q, dtype=float)
  14     if q.shape != (4,) or not np.isfinite(q).all() or np.linalg.norm(q) < 1e-8:
  15         raise ValueError('Expected a finite nonzero xyzw quaternion')
  16     x, y, z, w = q / np.linalg.norm(q)
  17     return math.atan2(2 * (w * z + x * y), 1 - 2 * (y * y + z * z))

  20 def task_packet(values):
  21     task = np.asarray(values, dtype=float).copy()
  22     if task.shape != (5,) or not np.isfinite(task).all():
  23         raise ValueError('Expected exactly five finite values')
  24     nav, face = np.any(task[:3] != 0), np.any(task[3:] != 0)
  25     if nav and face:
  26         raise ValueError('Nav and Face fields must not be mixed')
  27     if nav:
  28         task[2] = np.clip(task[2], 0.3, 1.2)
  29     if face:
  30         task[3] = wrap(task[3])
  31         task[4] = np.clip(abs(task[4]), 1.57, 3.14)
  32     return task, 'Nav' if nav else 'Face' if face else 'Hold'

  35 class GoalTask:
  36     """World goal with latched position arrival and heading hysteresis.
  37 
  38     ONNX never owns these locks. The upstream scheduler accepts new Nav goals.
  39     """
  40 
  41     def __init__(self, x, y, heading=None, speed=0.5, turn_speed=2.0):
  42         values = [x, y, speed, turn_speed] + ([] if heading is None else [heading])
  43         if not np.isfinite(values).all():
  44             raise ValueError('Non-finite goal')
  45         self.x, self.y, self.heading = x, y, heading
  46         self.speed, self.turn_speed = speed, turn_speed
  47         self.position_locked = self.heading_locked = False
  48 
  49     def keep_position_for(self, other):
  50         """Compare new goal with the completed goal, never with robot position.
  51 
  52         Keep the anchor unchanged for nearby requests so small updates cannot
  53         accumulate into an unbounded shift without starting a new Nav task.
  54         """
  55         return self.position_locked and math.hypot(other.x - self.x, other.y - self.y) <= 0.5
  56 
  57     def update(self, x, y, yaw):
  58         dx, dy = self.x - x, self.y - y
  59         distance = math.hypot(dx, dy)
  60         error = 0.0 if self.heading is None else wrap(self.heading - yaw)
  61         self.position_locked = self.position_locked or distance <= 0.2
  62         self.heading_locked = (
  63             self.heading is not None
  64             and self.position_locked
  65             and ((self.heading_locked and abs(error) <= math.pi / 4) or abs(error) <= math.pi / 12)
  66         )
  67         if not self.position_locked:
  68             c, s = math.cos(yaw), math.sin(yaw)
  69             return task_packet([c * dx + s * dy, -s * dx + c * dy, self.speed, 0.0, 0.0])
  70         if self.heading is not None and not self.heading_locked:
  71             return task_packet([0.0, 0.0, 0.0, error, self.turn_speed])
  72         return task_packet([0.0] * 5)
  73

  75 def absolute_goal(values):
  76     """Decode [world x, world y, speed, absolute yaw, turn-speed magnitude]."""
  77     packet = np.asarray(values, dtype=float)
  78     if packet.shape != (5,) or not np.isfinite(packet).all():
  79         raise ValueError('Goal requires five finite values: [x, y, speed, absolute_yaw, turn_speed]')
  80     x, y, speed, heading, turn_speed = packet
  81     if not float(np.float32(0.3)) <= speed <= float(np.float32(1.2)) or not float(
  82         np.float32(1.57)
  83     ) <= turn_speed <= float(np.float32(3.14)):
  84         raise ValueError('Goal speed must be 0.3..1.2 m/s and turn speed 1.57..3.14 rad/s')
  85     return GoalTask(x, y, wrap(heading), speed, turn_speed)
