  71   declare_if_absent("imitation_return_blend_sec", 0.35);
  72   declare_if_absent("wbc_enabled", false);
  73   declare_if_absent("text_motion_enabled", false);
  74   declare_if_absent("wbc_transition_sec", 0.35);
  75   declare_if_absent("wbc_command_timeout_sec", 0.5);
  76   declare_if_absent("arm_swing.enabled", true);
  77   declare_if_absent("arm_swing.stop_vx_threshold", 0.05);
  78   declare_if_absent("arm_swing.high_vx_threshold", 0.8);
  79   declare_if_absent("arm_swing.vx_hysteresis", 0.03);
  80   declare_if_absent("arm_swing.blend_sec", 0.35);
  81   declare_if_absent("safety.orientation.roll_limit_deg", 30.0);
  82   declare_if_absent("safety.orientation.pitch_limit_deg", 30.0);
  83   declare_if_absent("safety.orientation.consecutive_violation_count", 3);
  84   declare_if_absent("emergency.damping_kd", 5.0);
  85 
  86   std::vector<std::string> joint_names;
  87   std::string robot_cfg_path;
  88   if (!controller_nh->get_parameter("robot_cfg_path", robot_cfg_path) || robot_cfg_path.empty())
  89   {
  90     RCLCPP_ERROR(controller_nh->get_logger(), "Failed to get robot_cfg_path params");
  91     return false;
  92   }
  93 
  94   YAML::Node root = YAML::LoadFile(robot_cfg_path);
  95   YAML::Node rc = root["RobotCfg"];
  96   requireNode(rc, "RobotCfg");
  97 
  98   bool single_node_publishes_imitation_command = false;
  99   {
 100     const std::string policy_type_cfg = rc["policy_type"].as<std::string>("");
 101     single_node_publishes_imitation_command = (policy_type_cfg == "rl_imitation");
 102     RCLCPP_INFO(controller_nh->get_logger(),
 103                 "Policy mux from robot_cfg: policy_type='%s' -> use control/policy/command/imitation on walk=%s",
 104                 policy_type_cfg.c_str(), single_node_publishes_imitation_command ? "yes" : "no");
 105   }
 106 
 107   joint_names = asStringList(rc["joint_order_mj"], "RobotCfg.joint_order_mj");
 108 
 109   if (joint_names.empty())
 110   {
 111     RCLCPP_ERROR(controller_nh->get_logger(), "Failed to get joint names from URDF.");
 112     return false;
 113   }
 114 
 115   std::string robot_type;
 116   controller_nh->get_parameter("robot_type", robot_type);
 117   robot_type = trimCopy(robot_type);
 118   if (robot_type.empty()) robot_type = deriveRobotTypeFromConfigPath(robot_cfg_path);
 119 
 120   std::string policy_root_path;
 121   controller_nh->get_parameter("policy_root_path", policy_root_path);
 122   policy_root_path = trimCopy(policy_root_path);
 123   if (policy_root_path.empty()) policy_root_path = derivePolicyRootPath(robot_cfg_path);
 124 
 125   RCLCPP_INFO(controller_nh->get_logger(), "LeggedController::init, namespace: %s", controller_nh->get_namespace());
 126 
 127   controller_nh->get_parameter("isSim", isSim_);
 128   controller_nh->get_parameter("rviz", enableRviz_);
 129   controller_nh->get_parameter("radar_control_enabled", radarControlEnabled_);
 130   controller_nh->get_parameter("face_follow_enabled", faceFollowEnabled_);
 131 
 132   bool head_script_enabled = false;
 133   controller_nh->get_parameter("head_script_enabled", head_script_enabled);
 134   RCLCPP_INFO(controller_nh->get_logger(), "head_script_enabled: %d", head_script_enabled);
 135   controller_nh->get_parameter("joystick_imitation_enabled", joystick_imitation_enabled_);
 136   controller_nh->get_parameter("joystick_sequence_enabled", joystick_sequence_enabled_);
 137   controller_nh->get_parameter("joystick_back_emergency_sequence_enabled", joystick_back_emergency_sequence_enabled_);
 138   controller_nh->get_parameter("imitation_policy_switch_timeout_sec", imitation_policy_switch_timeout_sec_);
 139   controller_nh->get_parameter("imitation_return_blend_sec", imitation_return_blend_sec_);
 140   controller_nh->get_parameter("wbc_enabled", wbc_enabled_);
 141   controller_nh->get_parameter("text_motion_enabled", text_motion_enabled_);
 142   controller_nh->get_parameter("wbc_transition_sec", wbc_transition_sec_);
 143   controller_nh->get_parameter("wbc_command_timeout_sec", wbc_command_timeout_sec_);
 144   controller_nh->get_parameter("emergency.damping_kd", emergency_damping_kd_);
 145   imitation_return_blend_sec_ = std::max(imitation_return_blend_sec_, 0.0);
 146   if (!std::isfinite(wbc_transition_sec_) || wbc_transition_sec_ < 0.0)
 147   {
 148     RCLCPP_ERROR(controller_nh->get_logger(), "wbc_transition_sec must be finite and non-negative");
 149     return false;
 150   }
 151   if (!std::isfinite(wbc_command_timeout_sec_) || wbc_command_timeout_sec_ <= 0.0)
 152   {
 153     RCLCPP_ERROR(controller_nh->get_logger(), "wbc_command_timeout_sec must be finite and positive");
 154     return false;
 155   }
 156   if (emergency_damping_kd_ < 0.0)
 157   {
 158     RCLCPP_WARN(controller_nh->get_logger(), "emergency.damping_kd %.3f < 0, clamping to 0", emergency_damping_kd_);
 159     emergency_damping_kd_ = 0.0;
 160   }
 161   RCLCPP_INFO(controller_nh->get_logger(), "joystick_imitation_enabled: %d", joystick_imitation_enabled_);
 162   RCLCPP_INFO(controller_nh->get_logger(), "joystick_sequence_enabled: %d", joystick_sequence_enabled_);
 163   RCLCPP_INFO(controller_nh->get_logger(), "joystick_back_emergency_sequence_enabled: %d",
 164               joystick_back_emergency_sequence_enabled_);
 165   RCLCPP_INFO(controller_nh->get_logger(), "imitation_return_blend_sec: %.3f", imitation_return_blend_sec_);
 166   RCLCPP_INFO(controller_nh->get_logger(), "wbc_enabled: %d transition=%.3fs timeout=%.3fs", wbc_enabled_,
 167               wbc_transition_sec_, wbc_command_timeout_sec_);
 168   RCLCPP_INFO(controller_nh->get_logger(), "emergency.damping_kd: %.3f", emergency_damping_kd_);
 169 
 170   ArmSwingController::Params arm_swing_params;
 171   controller_nh->get_parameter("arm_swing.enabled", arm_swing_params.enabled);
 172   controller_nh->get_parameter("arm_swing.stop_vx_threshold", arm_swing_params.stop_vx_threshold);
 173   controller_nh->get_parameter("arm_swing.high_vx_threshold", arm_swing_params.high_vx_threshold);
 174   controller_nh->get_parameter("arm_swing.vx_hysteresis", arm_swing_params.vx_hysteresis);
 175   controller_nh->get_parameter("arm_swing.blend_sec", arm_swing_params.blend_sec);
 176 
 177   std::string head_script_config_path;
 178   controller_nh->get_parameter("head_script_config", head_script_config_path);
 179   controller_nh->get_parameter("joystick_imitation_config_path", joystick_imitation_config_path_);
 180   controller_nh->get_parameter("joystick_sequence_config_path", joystick_sequence_config_path_);
 181   controller_nh->get_parameter("joystick_back_emergency_sequence_path", joystick_back_emergency_sequence_path_);
 182   joystick_imitation_config_path_ = trimCopy(joystick_imitation_config_path_);
 183   joystick_sequence_config_path_ = trimCopy(joystick_sequence_config_path_);
 184   joystick_back_emergency_sequence_path_ = trimCopy(joystick_back_emergency_sequence_path_);
 185   sequence_engine_enabled_ = joystick_sequence_enabled_ || joystick_back_emergency_sequence_enabled_;
 186 
 187   bool ok = true;
 188   policy_frequency = rc["policy_frequency"].as<int>(0);
 189   if (policy_frequency == 0)
 190   {
 191     RCLCPP_ERROR(controller_nh->get_logger(), "Failed to get parameters from /RobotCfg/policy_frequency");
 192     return false;
 193   }
 194   policyMux_.init(controller_nh, joint_names);
 195   JointGainProfile rl_walking_gains;
 196   if (!loadWalkingGainProfile(rc, "RobotCfg", joint_names.size(), rl_walking_gains))
 197   {
 198     return false;
 199   }
 200   policyMux_.configureStartup(single_node_publishes_imitation_command, rl_walking_gains,
 201                               imitation_policy_switch_timeout_sec_, policy_root_path, robot_type);
 202 
 203   RCLCPP_INFO(controller_nh->get_logger(), "Loaded RL walking gains from: %s", robot_cfg_path.c_str());
 204   RCLCPP_INFO(controller_nh->get_logger(), "Dynamic imitation gains will be loaded when a sequence switches policy");
 205 
 206   try
 207   {
 208     legged_controllers_share_ = ament_index_cpp::get_package_share_directory("legged_controllers");
 209   } catch (const std::exception& e)
 210   { RCLCPP_WARN(controller_nh->get_logger(), "Failed to resolve legged_controllers share dir: %s", e.what()); }
 211   sequenceRuntime_.init(controller_nh->get_logger(), legged_controllers_share_);
 212 
 213   if (!ok)
 214   {
 215     RCLCPP_ERROR(controller_nh->get_logger(), "Failed to get parameters from RobotCfg.");
 216     return false;
 217   }
 218 
 219   {
 220     double viz_freq = 0.0;
 221     controller_nh->get_parameter_or("viz_frequency", viz_freq, static_cast<double>(policy_frequency));
 222 
 223     if (viz_freq <= 0.0) viz_freq = static_cast<double>(policy_frequency);
 224     vizPeriod_ = rclcpp::Duration::from_seconds(1.0 / viz_freq);
 225 
 226     double joy_feedback_hz = 2.0;
 227     controller_nh->get_parameter_or("joy_feedback_hz", joy_feedback_hz, 2.0);
 228 
 229     if (joy_feedback_hz <= 0.0) joy_feedback_hz = 2.0;
 230     joyFeedbackPeriod_ = rclcpp::Duration::from_seconds(1.0 / joy_feedback_hz);
 231     lastJoyFeedbackTick_ = rclcpp::Time(0, 0, RCL_ROS_TIME);
 232 
 233     joyFeedbackState_ = false;
 234   }
 235 
 236   dataInit(joint_names);
 237   if (text_motion_enabled_)
 238   {
 239     std::string mapping_file, error;
 240     if (robot_type != "l0_v5" || !controller_nh->get_parameter("mapping_file", mapping_file) ||
 241         !textMotionCtrl_.configure(joint_names, mapping_file, error))
 242     {
 243       RCLCPP_ERROR(controller_nh->get_logger(), "Text motion requires L0 v5 and its valid mapping_file: %s", error.c_str());
 244       return false;
 245     }
 246   }
 247   armSwingCtrl_.configure(arm_swing_params, controller_nh->get_logger());
 248 
 249   headCtrl_.setScriptEnabled(head_script_enabled);
 250   headCtrl_.setTopicEnabled(faceFollowEnabled_);
 251   headCtrl_.setTopicBlendSec(0.1);
 252   headCtrl_.setFallbackBlendSec(0.2);
 253   RCLCPP_INFO(controller_nh->get_logger(), "face_follow_enabled: %d topic=/algorithm/em/follow_face",
 254               faceFollowEnabled_);
 255   if (head_script_enabled)
 256   {
 257     RCLCPP_INFO(controller_nh->get_logger(), "Head script enabled: timeline will drive head");
 258     if (!headCtrl_.loadConfig(head_script_config_path))
 259     {
 260       RCLCPP_ERROR(controller_nh->get_logger(), "Head script config failed to load: %s",
 261                    head_script_config_path.c_str());
 262       return false;
 263     }
 264     RCLCPP_INFO(controller_nh->get_logger(), "Head script loaded from config: %s (%zu segments, duration %.1fs)",
 265                 head_script_config_path.c_str(), headCtrl_.numSegments(), headCtrl_.duration());
 266   }
 267   else
 268   {
 269     RCLCPP_INFO(controller_nh->get_logger(), "Head script disabled by parameter");
 270   }
 271 
 272   sequenceRuntime_.setEnabled(sequence_engine_enabled_);
 273   if (joystick_back_emergency_sequence_enabled_)
 274   {
 275     if (joystick_back_emergency_sequence_path_.empty())
 276     {
 277       RCLCPP_ERROR(controller_nh->get_logger(),
 278                    "Back emergency sequence enabled but joystick_back_emergency_sequence_path is empty");
 279       return false;
 280     }
 281     if (!loadActionSequenceForPlayback(joystick_back_emergency_sequence_path_, "[BackEmergency] Init"))
 282     {
 283       return false;
 284     }
 285     if (!sequenceRuntime_.usesFullCsv() || sequenceRuntime_.usesRlImitation())
 286     {
 287       RCLCPP_ERROR(controller_nh->get_logger(),
 288                    "Back emergency sequence must use full_csv and must not use rl_imitation: %s",
 289                    joystick_back_emergency_sequence_path_.c_str());
 290       return false;
 291     }
 292     RCLCPP_INFO(controller_nh->get_logger(), "Back emergency sequence enabled: %s",
 293                 joystick_back_emergency_sequence_path_.c_str());
 294   }
 295 
 296   if (joystick_imitation_enabled_)
 297   {
 298     if (joystick_imitation_config_path_.empty())
 299     {
 300       RCLCPP_ERROR(controller_nh->get_logger(),
 301                    "Joystick imitation enabled but joystick_imitation_config_path is empty");
 302       return false;
 303     }
 304 
 305     JoystickImitationActions::LoadOptions options;
 306     options.package_share_dir = legged_controllers_share_;
 307     options.policy_root_path = policy_root_path;
 308     options.robot_type = robot_type;
 309     options.validate_policy_assets = true;
 310     if (!joystickImitationActions_.load(joystick_imitation_config_path_, options, controller_nh->get_logger()))
 311     {
 312       return false;
 313     }
 314     if (joystickImitationActions_.empty())
 315     {
 316       joystick_imitation_enabled_ = false;
 317       RCLCPP_WARN(controller_nh->get_logger(),
 318                   "Joystick imitation disabled: no usable actions loaded from %s for robot_type=%s (skipped=%zu)",
 319                   joystick_imitation_config_path_.c_str(), robot_type.c_str(),
 320                   joystickImitationActions_.skippedBindingCount());
 321     }
 322     else if (joystickImitationActions_.skippedBindingCount() > 0)
 323     {
 324       RCLCPP_WARN(
 325           controller_nh->get_logger(), "Joystick imitation skipped %zu unavailable action(s) from %s for robot_type=%s",
 326           joystickImitationActions_.skippedBindingCount(), joystick_imitation_config_path_.c_str(), robot_type.c_str());
 327     }
 328     if (joystick_imitation_enabled_)
 329     {
 330       RCLCPP_INFO(
 331           controller_nh->get_logger(), "Joystick imitation enabled: config=%s robot_type=%s enabled_actions=%zu",
 332           joystick_imitation_config_path_.c_str(), robot_type.c_str(), joystickImitationActions_.bindingCount());
 333     }
 334   }
 335 
 336   if (joystick_sequence_enabled_)
 337   {
 338     if (joystick_sequence_config_path_.empty())
 339     {
 340       RCLCPP_ERROR(controller_nh->get_logger(), "Joystick sequence enabled but joystick_sequence_config_path is empty");
 341       return false;
 342     }
 343 
 344     JoystickSequenceActions::LoadOptions options;
 345     options.package_share_dir = legged_controllers_share_;
 346     options.validate_sequence_files = true;
 347     if (!joystickSequenceActions_.load(joystick_sequence_config_path_, options, controller_nh->get_logger()))
 348     {
 349       return false;
 350     }
 351     RCLCPP_INFO(controller_nh->get_logger(), "Joystick sequence enabled: config=%s enabled_actions=%zu",
 352                 joystick_sequence_config_path_.c_str(), joystickSequenceActions_.bindingCount());
 353   }
 354 
 355   velLimiter_ = std::make_unique<VelocityLimiter>(controller_nh);
 356 
 357   {
 358     const SafetyChecker::Params defaults;
 359     SafetyChecker::Params sp;
 360     double roll_limit_deg = radiansToDegrees(defaults.roll_limit);
 361     double pitch_limit_deg = radiansToDegrees(defaults.pitch_limit);
 362     controller_nh->get_parameter("safety.orientation.roll_limit_deg", roll_limit_deg);
 363     controller_nh->get_parameter("safety.orientation.pitch_limit_deg", pitch_limit_deg);
 364     controller_nh->get_parameter("safety.orientation.consecutive_violation_count", sp.consecutive_violation_count);
 365     sp.roll_limit = degreesToRadians(roll_limit_deg);
 366     sp.pitch_limit = degreesToRadians(pitch_limit_deg);
 367 
 368     if (sp.roll_limit <= 0.0)
 369     {
 370       RCLCPP_WARN(controller_nh->get_logger(), "Invalid roll_limit_deg; using default %.1f deg",
 371                   radiansToDegrees(defaults.roll_limit));
 372       sp.roll_limit = defaults.roll_limit;
 373     }
 374     if (sp.pitch_limit <= 0.0)
 375     {
 376       RCLCPP_WARN(controller_nh->get_logger(), "Invalid pitch_limit_deg; using default %.1f deg",
 377                   radiansToDegrees(defaults.pitch_limit));
 378       sp.pitch_limit = defaults.pitch_limit;
 379     }
 380     if (sp.consecutive_violation_count <= 0)
 381     {
 382       RCLCPP_WARN(controller_nh->get_logger(), "Invalid consecutive_violation_count; using default %d",
 383                   defaults.consecutive_violation_count);
 384       sp.consecutive_violation_count = defaults.consecutive_violation_count;
 385     }
 386 
 387     safetyChecker_ = std::make_unique<SafetyChecker>(sp);
 388     RCLCPP_INFO(controller_nh->get_logger(),
 389                 "Orientation safety always active on real robot: roll_limit=%.1f deg, pitch_limit=%.1f deg, "
 390                 "consecutive_violation_count=%d",
 391                 radiansToDegrees(sp.roll_limit), radiansToDegrees(sp.pitch_limit), sp.consecutive_violation_count);
 392   }
 393 
 394   fsm_ = std::make_unique<StateMachine>(controller_nh->get_logger());
 395 
 396   ModeSubscribe(controller_nh);
 397   ModePublish(controller_nh);
 398 
 399   return true;
 400 }

 402 void LeggedController::dataInit(const std::vector<std::string>& joint_names) {
 403   policy_observation_.joints.name = joint_names;
 404   policy_observation_.joints.position.resize(joint_names.size());
 405   policy_observation_.joints.velocity.resize(joint_names.size());
 406   policy_observation_.joints.effort.resize(joint_names.size());
 407   policyMux_.resizeCommands(joint_names.size());
 408 
 409   rvizVisualPos_.name = joint_names;
 410   jointNames_ = joint_names;
 411   rvizVisualPos_.position.resize(joint_names.size());
 412   rvizVisualPos_.velocity.resize(joint_names.size());
 413   rvizVisualPos_.effort.resize(joint_names.size());
 414 
 415   headCtrl_.initJoints(joint_names, nh_->get_logger());
 416   armSwingCtrl_.initJoints(joint_names, nh_->get_logger());
 417   sequenceRuntime_.initJoints(joint_names);
 418 
 419   const JointGroups groups = classifyJointGroups(joint_names);
 420   is_leg_joint_ = groups.leg;
 421   is_arm_joint_ = groups.arm;
 422   is_head_joint_ = groups.head;
 423 
 424   joyFeedbackArray_.array.resize(2);
 425   joyFeedbackArray_.array[0].type = sensor_msgs::msg::JoyFeedback::TYPE_RUMBLE;
 426   joyFeedbackArray_.array[1].type = sensor_msgs::msg::JoyFeedback::TYPE_RUMBLE;
 427   joyFeedbackArray_.array[0].id = 0;
 428   joyFeedbackArray_.array[1].id = 1;
 429 
 430   policy_observation_.behavior_up_down = 0.0f;
 431   policy_observation_.behavior_left_right = 0.0f;
 432 }
