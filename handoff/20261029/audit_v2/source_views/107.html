  11 def _configure_environment(context):
  12     mode = context.launch_configurations['hpm_mode']
  13     if mode not in ('real', 'isolated'):
  14         raise RuntimeError('hpm_mode must be real or isolated')
  15     if mode == 'isolated':
  16         domain = int(context.launch_configurations['isolated_ros_domain'])
  17         if not 1 <= domain <= 232:
  18             raise RuntimeError('isolated_ros_domain must be between 1 and 232')
  19         return [SetEnvironmentVariable('ROS_DOMAIN_ID', str(domain)),
  20                 SetEnvironmentVariable('ROS_LOCALHOST_ONLY', '1'),
  21                 LogInfo(msg=f'HPM isolated endpoints; ROS_DOMAIN_ID={domain}, ROS_LOCALHOST_ONLY=1')]
  22     return []
