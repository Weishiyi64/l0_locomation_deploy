   1 #!/usr/bin/env python3
   2 """Generate emotion_index.json from lora/<emotion>/<motion> without loading .pt.
   3 
   4 Requires Base checkpoint/ONNX/YAML and each action's lorackpt_N.pt and
   5 motion/reference_motion.onnx. Matching checkpoint_N.pt is optional.
   6 Writes a new index only.
   7 """

   9 import argparse
  10 import hashlib
  11 import json
  12 import os
  13 import re
  14 import sys

  19 ACTION_ID = re.compile(r"[A-Za-z0-9_-]{1,32}")

  22 def checked_file(root, path):
  23     if not path.is_file() or not path.resolve().is_relative_to(root):
  24         raise ValueError(f"Missing file or path outside training root: {path}")
  25     return path
  26

  28 def sha256(path):
  29     digest = hashlib.sha256()
  30     with path.open("rb") as stream:
  31         for chunk in iter(lambda: stream.read(1024 * 1024), b""):
  32             digest.update(chunk)
  33     return digest.hexdigest()
  34 
  35 
  36 def saved_steps(directory, prefix):
  37     result = {}
  38     pattern = re.compile(re.escape(prefix) + r"_([0-9]+)\.pt")
  39     for path in sorted(directory.glob(prefix + "_*.pt")):
  40         match = pattern.fullmatch(path.name)
  41         if not match:
  42             continue
  43         step = int(match.group(1))
  44         if step in result:
  45             raise ValueError(f"Ambiguous filenames for {prefix} step {step}: {directory}")
  46         result[step] = path
  47     return result
  48 
  49 
  50 def generate(root, actions=None, step=None):
  51     root = root.resolve()
  52     if not root.is_dir():
  53         raise ValueError(f"Training root does not exist: {root}")
  54     for name in ("base/base_ckpt.pt", "base/policy.onnx", "base/policy.yaml"):
  55         checked_file(root, root / name)
  56     if step is not None and step < 0:
  57         raise ValueError("--step must be nonnegative")
  58     if actions is not None and (not actions or len(actions) != len(set(actions)) or
  59                                 any(not ACTION_ID.fullmatch(x) for x in actions)):
  60         raise ValueError("--actions must contain unique valid action IDs")
  61     selected = set(actions) if actions is not None else None
  62     entries, seen = [], set()
  63     for directory in sorted((root / "lora").glob("*/*")):
  64         if not directory.is_dir() or (selected is not None and directory.name not in selected):
  65             continue
  66         action = directory.name
  67         if not directory.resolve().is_relative_to(root):
  68             raise ValueError(f"Action directory escapes training root: {directory}")
  69         if not ACTION_ID.fullmatch(action) or action in seen:
  70             raise ValueError(f"Invalid or duplicate action ID: {action}")
  71         seen.add(action)
  72         adapters, checkpoints = saved_steps(directory, "lorackpt"), saved_steps(directory, "checkpoint")
  73         if not adapters:
  74             raise ValueError(f"{action}: missing lorackpt_N.pt")
  75         chosen = max(adapters) if step is None else step
  76         if chosen not in adapters:
  77             raise ValueError(f"{action}: lorackpt file is required for step {chosen}")
  78         adapter = checked_file(root, adapters[chosen])
  79         reference = checked_file(root, directory / "motion/reference_motion.onnx")
  80         entry = {
  81             "emotion": directory.parent.name, "motion": action,
  82             "key": str(directory.relative_to(root / "lora")), "checkpoint_step": chosen,
  83             "adapter": str(adapter.relative_to(root)),
  84             "motion_onnx": str(reference.relative_to(root)),
  85             "adapter_sha256": sha256(adapter),
  86             "motion_onnx_sha256": sha256(reference),
  87         }
  88         if chosen in checkpoints:
  89             checkpoint = checked_file(root, checkpoints[chosen])
  90             entry.update(checkpoint=str(checkpoint.relative_to(root)), checkpoint_sha256=sha256(checkpoint))
  91         motion_npz = directory / "motion" / (action + ".npz")
  92         if motion_npz.exists() or motion_npz.is_symlink():
  93             checked_file(root, motion_npz)
  94             entry["motion_npz"] = str(motion_npz.relative_to(root))
  95         entries.append(entry)
  96     if selected is not None and selected - seen:
  97         raise ValueError("Actions not found: " + ", ".join(sorted(selected - seen)))
  98     if not entries:
  99         raise ValueError("No action directories found under lora/<emotion>/<motion>")
 100     return entries

 103 def main():
 104     parser = argparse.ArgumentParser(description=__doc__)
 105     parser.add_argument("source", type=Path, help="training root containing base/ and lora/")
 106     parser.add_argument("--output", type=Path, default=Path("emotion_index.json"),
 107                         help="new JSON file; relative paths are relative to source (default: emotion_index.json)")
 108     parser.add_argument("--actions", nargs="+", help="only scan these action IDs (default: all actions)")
 109     parser.add_argument("--step", type=int, help="use this checkpoint filename step for every selected action")
 110     parser.add_argument("--dry-run", action="store_true", help="print the generated JSON without writing or replacing any file")
 111     args = parser.parse_args()
 112     root = args.source.resolve()
 113     output = args.output if args.output.is_absolute() else root / args.output
 114     if not args.dry_run and (output.exists() or output.is_symlink()):
 115         parser.error(f"output already exists; choose a new --output path: {output}")
 116     entries = generate(root, args.actions, args.step)
 117     content = json.dumps(entries, indent=2, ensure_ascii=False) + "\n"
 118     if args.dry_run:
 119         sys.stdout.write(content)
 120     else:
 121         output.parent.mkdir(parents=True, exist_ok=True)
 122         with tempfile.TemporaryDirectory(prefix=".emotion_index_", dir=output.parent) as temporary:
 123             candidate = Path(temporary) / "index.json"
 124             candidate.write_text(content, encoding="utf-8")
 125             # Atomic publication, including refusal to overwrite a file/symlink
 126             # that appeared after the initial existence check.
 127             os.link(candidate, output)
 128         print(f"Wrote {len(entries)} actions: {output}")
 129     return 0
