  49 def _load_yaml_file(path: str, description: str) -> Dict[str, Any]:
  50     if yaml is None:
  51         raise RuntimeError(f'PyYAML is required to parse {description}; install python3-yaml.')
  52 
  53     with open(path, 'r', encoding='utf-8') as f:
  54         root = yaml.safe_load(f) or {}
  55     if not isinstance(root, dict):
  56         raise ValueError(f'{description} {path} must contain a YAML map.')
  57     return root

  77 def _parse_sequence_yaml(path: str, package_share_legged_controllers: str) -> Tuple[bool, str]:
  78     """Validate sequence YAML and return (has_rl_imitation_phase, imitation_policy)."""
  79     root = _load_yaml_file(path, 'action sequence')
  80 
  81     sequence = root.get('sequence') or {}
  82     if not isinstance(sequence, dict):
  83         raise ValueError(f'Action sequence {path} must contain sequence map.')
  84     phases = sequence.get('phases') or []
  85     if not isinstance(phases, list) or not phases:
  86         raise ValueError(f'Action sequence {path} must contain non-empty sequence.phases list.')
  87 
  88     has_rl_imitation = False
  89     sequence_dir = os.path.dirname(path)
  90     for index, phase in enumerate(phases):
  91         if not isinstance(phase, dict):
  92             raise ValueError(f'Action sequence {path} phase[{index}] must be a map.')
  93         mode = str(phase.get('mode', '')).strip()
  94         phase_name = str(phase.get('name') or f'phase_{index}')
  95         if mode not in VALID_PHASE_MODES:
  96             raise ValueError(
  97                 f"Action sequence {path} phase '{phase_name}' has invalid mode {mode!r}; "
  98                 f'expected one of {sorted(VALID_PHASE_MODES)}.'
  99             )
 100 
 101         if mode == 'rl_imitation':
 102             has_rl_imitation = True
 103             if phase.get('csv'):
 104                 raise ValueError(f"Action sequence {path} phase '{phase_name}' mode rl_imitation must not set csv.")
 105             if phase.get('velocity_timeline'):
 106                 raise ValueError(
 107                     f"Action sequence {path} phase '{phase_name}' mode rl_imitation does not support velocity_timeline."
 108                 )
 109             continue
 110 
 111         raw_csv = str(phase.get('csv') or '').strip()
 112         if not raw_csv:
 113             raise ValueError(f"Action sequence {path} phase '{phase_name}' mode {mode} requires csv.")
 114         resolved_csv = _resolve_csv_path(raw_csv, sequence_dir, package_share_legged_controllers)
 115         if not os.path.isfile(resolved_csv):
 116             raise ValueError(
 117                 f"Action sequence {path} phase '{phase_name}' csv is missing: "
 118                 f'{raw_csv!r} (resolved: {resolved_csv!r})'
 119             )
 120     raw_imitation = sequence.get('imitation_policy')
 121     if raw_imitation is None:
 122         raw_imitation = sequence.get('imitation_policy_name')
 123     imitation_policy = ''
 124     if raw_imitation is not None:
 125         imitation_policy = str(raw_imitation).strip()
 126 
 127     return has_rl_imitation, imitation_policy

 134 def _validate_imitation_policy_assets(policy_name: str, robot_type: str, policy_root_path: str) -> None:
 135     policy_dir = os.path.normpath(os.path.join(policy_root_path, robot_type, policy_name))
 136     robot_cfg_path = os.path.join(policy_dir, 'rl_config.yaml')
 137     if not os.path.isfile(robot_cfg_path):
 138         raise ValueError(
 139             f"Imitation policy '{policy_name}' is missing rl_config.yaml for robot_type '{robot_type}': "
 140             f'{robot_cfg_path}'
 141         )
 142 
 143     root = _load_yaml_file(robot_cfg_path, 'imitation policy rl_config')
 144     rc = root.get('RobotCfg') or {}
 145     if not isinstance(rc, dict):
 146         raise ValueError(f"Imitation policy '{policy_name}' rl_config.yaml must contain RobotCfg map.")
 147 
 148     policy_type = str(rc.get('policy_type') or '').strip()
 149     if policy_type and policy_type != 'rl_imitation':
 150         raise ValueError(
 151             f"Imitation policy '{policy_name}' RobotCfg.policy_type must be 'rl_imitation', got {policy_type!r}."
 152         )
 153 
 154     policy_model = str(rc.get('policy_name') or '').strip()
 155     motion_model = str(rc.get('motion_name') or '').strip()
 156     if not policy_model:
 157         raise ValueError(f"Imitation policy '{policy_name}' RobotCfg.policy_name is missing.")
 158     if not motion_model:
 159         raise ValueError(f"Imitation policy '{policy_name}' RobotCfg.motion_name is missing.")
 160     if not _model_asset_exists(policy_dir, policy_model):
 161         raise ValueError(
 162             f"Imitation policy '{policy_name}' policy model asset is missing: "
 163             f'{os.path.join(policy_dir, policy_model)}.(onnx|trt)'
 164         )
 165     if not _model_asset_exists(policy_dir, motion_model):
 166         raise ValueError(
 167             f"Imitation policy '{policy_name}' motion model asset is missing: "
 168             f'{os.path.join(policy_dir, motion_model)}.(onnx|trt)'
 169         )

 176 def _button_index(raw_name: str) -> int:
 177     name = _normalize_combo_name(raw_name)
 178     mapping = {

 180         'b': 1,
 181         'x': 3,
 182         'y': 4,
 183         'lb': 6,
 184         'l1': 6,
 185         'rb': 7,

 190         'l2button': 8,

 193         'r2': 9,

 195         'back': 10,
 196         'select': 10,

 199         'lsbutton': 13,

 201         'rs': 14,
 202         'rsbutton': 14,

 208 def _dpad_axis(raw_name: str) -> int:
 209     name = _normalize_combo_name(raw_name)
 210     if name in {'dpadlr', 'dpadx', 'dirlr', 'hatx'}:
 211         return 6
 212     if name in {'dpadud', 'dpady', 'dirud', 'haty'}:
 213         return 7
 214     return -1

 217 def _axis_sign(raw_direction: str, axis_index: int) -> int:
 218     direction = _normalize_combo_name(raw_direction)
 219     if direction in {'positive', 'pos', '+'}:
 220         return 1
 221     if direction in {'negative', 'neg', '-'}:
 222         return -1
 223     if direction == 'up' and axis_index == 7:
 224         return 1
 225     if direction == 'down' and axis_index == 7:
 226         return -1
 227     if direction == 'left' and axis_index == 6:
 228         return 1
 229     if direction == 'right' and axis_index == 6:
 230         return -1
 231     return 0

 234 def _dpad_axis_for_direction(raw_direction: str) -> str:
 235     direction = _normalize_combo_name(raw_direction)
 236     if direction in {'up', 'down'}:
 237         return 'DPad_ud'
 238     if direction in {'left', 'right'}:
 239         return 'DPad_lr'
 240     return ''

 243 def _as_string_list(value: Any) -> List[str]:
 244     if value is None:
 245         return []
 246     if isinstance(value, str):
 247         return [value.strip()]
 248     if isinstance(value, list):
 249         out: List[str] = []
 250         for item in value:
 251             if not isinstance(item, str):
 252                 raise ValueError('combo button entries must be strings')
 253             out.append(item.strip())
 254         return out
 255     raise ValueError('combo buttons must be a string or list')

 258 def _canonical_joystick_combo(combo: Any, action_name: str, kind: str = 'Joystick imitation action') -> str:
 259     if not isinstance(combo, dict):
 260         raise ValueError(f"{kind} '{action_name}' needs combo map.")
 261 
 262     buttons: List[int] = []
 263     axes: List[Tuple[int, int]] = []
 264 
 265     for button_name in _as_string_list(combo.get('buttons')):
 266         index = _button_index(button_name)
 267         if index < 0:
 268             raise ValueError(f"{kind} '{action_name}' has unknown button {button_name!r}.")
 269         buttons.append(index)
 270 
 271     if combo.get('dpad') is not None:
 272         direction = str(combo.get('dpad')).strip()
 273         axis_name = _dpad_axis_for_direction(direction)
 274         axis_index = _dpad_axis(axis_name)
 275         sign = _axis_sign(direction, axis_index)
 276         if axis_index < 0 or sign == 0:
 277             raise ValueError(f"{kind} '{action_name}' has invalid dpad direction {direction!r}.")
 278         axes.append((axis_index, sign))
 279 
 280     raw_axes = combo.get('axes') or []
 281     if not isinstance(raw_axes, list):
 282         raise ValueError(f"{kind} '{action_name}' combo.axes must be a list.")
 283     for axis in raw_axes:
 284         if not isinstance(axis, dict):
 285             raise ValueError(f"{kind} '{action_name}' combo.axes entries must be maps.")
 286         axis_name = str(axis.get('name') or axis.get('axis') or '').strip()
 287         axis_index = _dpad_axis(axis_name)
 288         if axis_index < 0:
 289             raise ValueError(f"{kind} '{action_name}' has unknown D-pad axis {axis_name!r}.")
 290         direction = str(axis.get('direction') or '').strip()
 291         sign = _axis_sign(direction, axis_index)
 292         if sign == 0:
 293             raise ValueError(f"{kind} '{action_name}' axis {axis_name!r} has invalid direction {direction!r}.")
 294         axes.append((axis_index, sign))
 295 
 296     if not buttons and not axes:
 297         raise ValueError(f"{kind} '{action_name}' combo has no requirements.")
 298 
 299     parts = [f'b{index}' for index in sorted(buttons)]
 300     parts.extend(f'a{index}{"+" if sign > 0 else "-"}' for index, sign in sorted(axes))
 301     return '|'.join(parts)

 304 def _iter_joystick_imitation_policy_entries(config_path: str) -> Iterable[str]:
 305     root = _load_yaml_file(config_path, 'joystick imitation action map')
 306     cfg = root.get('joystick_imitation_actions') or root
 307     if not isinstance(cfg, dict):
 308         raise ValueError(f'Joystick imitation map {config_path} must contain a YAML map.')
 309     actions = cfg.get('actions') or []
 310     if not isinstance(actions, list):
 311         raise ValueError(f'Joystick imitation map {config_path} must contain joystick_imitation_actions.actions list.')
 312 
 313     seen_names: Set[str] = set()
 314     seen_combos: Set[str] = set()
 315     for index, action in enumerate(actions):
 316         if not isinstance(action, dict):
 317             raise ValueError(f'Joystick imitation map {config_path} action[{index}] must be a map.')
 318         name = str(action.get('name') or '').strip()
 319         if not name:
 320             raise ValueError(f'Joystick imitation map {config_path} action[{index}] needs non-empty name.')
 321         if name in seen_names:
 322             raise ValueError(f"Joystick imitation map {config_path} has duplicate action name '{name}'.")
 323         seen_names.add(name)
 324 
 325         canonical_combo = _canonical_joystick_combo(action.get('combo'), name)
 326         if canonical_combo in seen_combos:
 327             raise ValueError(f"Joystick imitation map {config_path} has duplicate combo for action '{name}'.")
 328         seen_combos.add(canonical_combo)
 329 
 330         if not bool(action.get('enabled', False)):
 331             continue
 332         policy = str(action.get('policy') or '').strip()
 333         if not policy:
 334             raise ValueError(f"Joystick imitation action '{name}' is enabled but policy is empty.")
 335         yield policy

 356         raise ValueError(
 357             f"Joystick sequence action '{action_name}' sequence {resolved_sequence} must contain "
 358             'non-empty sequence.phases list.'
 359         )
 360 
 361     sequence_dir = os.path.dirname(resolved_sequence)
 362     for index, phase in enumerate(phases):
 363         if not isinstance(phase, dict):
 364             raise ValueError(f"Joystick sequence action '{action_name}' phase[{index}] must be a map.")
 365         phase_name = str(phase.get('name') or f'phase_{index}')
 366         mode = str(phase.get('mode') or '').strip()
 367         if mode != 'rl_with_csv':
 368             raise ValueError(
 369                 f"Joystick sequence action '{action_name}' phase '{phase_name}' must use mode 'rl_with_csv', "

 374             raise ValueError(
 375                 f"Joystick sequence action '{action_name}' phase '{phase_name}' mode rl_with_csv requires csv."
 376             )
 377         resolved_csv = _resolve_csv_path(raw_csv, sequence_dir, package_share_legged_controllers)
 378         if not os.path.isfile(resolved_csv):
 379             raise ValueError(
 380                 f"Joystick sequence action '{action_name}' phase '{phase_name}' csv is missing: "
 381                 f'{raw_csv!r} (resolved: {resolved_csv!r})'

 438 def collect_action_sequence_manifest(
 439     *,
 440     package_share_legged_controllers: str,
 441     joystick_imitation_enabled: bool = False,
 442     joystick_imitation_config_path: str = '',
 443     joystick_sequence_enabled: bool = False,
 444     joystick_sequence_config_path: str = '',
 445     robot_type: str = '',
 446     policy_root_path: str = '',
 447 ) -> ActionSequenceManifest:
 448     """Validate enabled action maps and collect policies that should be preloaded for dynamic imitation."""
 449     policies: Set[str] = set()
 450     sequence_paths: List[str] = []
 451     rl_imitation_sequences: List[str] = []
 452 
 453     if joystick_imitation_enabled:
 454         resolved_map = _resolve_package_path(joystick_imitation_config_path, package_share_legged_controllers)
 455         if not resolved_map or not os.path.isfile(resolved_map):
 456             raise ValueError(f'joystick_imitation_enabled but map is missing: {joystick_imitation_config_path!r}')
 457         for policy in _iter_joystick_imitation_policy_entries(resolved_map):
 458             policies.add(policy)
 459 
 460     if joystick_sequence_enabled:
 461         resolved_map = _resolve_package_path(joystick_sequence_config_path, package_share_legged_controllers)
 462         if not resolved_map or not os.path.isfile(resolved_map):
 463             raise ValueError(f'joystick_sequence_enabled but map is missing: {joystick_sequence_config_path!r}')
 464         sequence_paths.extend(_iter_joystick_sequence_paths(resolved_map, package_share_legged_controllers))
 465 
 466     for raw_sequence_path in sequence_paths:
 467         resolved_sequence = resolve_action_sequence_yaml_path(raw_sequence_path, package_share_legged_controllers)
 468         if not resolved_sequence or not os.path.isfile(resolved_sequence):
 469             raise ValueError(
 470                 f'action sequence referenced by joystick map is missing: '
 471                 f'{raw_sequence_path!r} (resolved: {resolved_sequence!r})'
 472             )
 473         has_imitation, policy = _parse_sequence_yaml(resolved_sequence, package_share_legged_controllers)
 474         if has_imitation:
 475             if not policy:
 476                 raise ValueError(
 477                     f'Action sequence {resolved_sequence} has rl_imitation phase(s) but '
 478                     'sequence.imitation_policy is missing.'
 479                 )
 480             policies.add(policy)
 481             rl_imitation_sequences.append(resolved_sequence)
 482 
 483     sorted_policies = sorted(policies)
 484     skipped_policies: List[str] = []
 485     if sorted_policies and robot_type and policy_root_path:
 486         available_policies: List[str] = []
 487         for policy in sorted_policies:
 488             try:
 489                 _validate_imitation_policy_assets(policy, robot_type, policy_root_path)
 490             except (RuntimeError, ValueError, OSError) as exc:
 491                 skipped_policies.append(f"{policy}: {exc}")
 492                 continue
 493             available_policies.append(policy)
 494         sorted_policies = available_policies
 495 
 496     return ActionSequenceManifest(
 497         sequence_paths=sorted(set(sequence_paths)),
 498         imitation_policies=sorted_policies,
 499         preload_imitation_policies=sorted_policies,
 500         rl_imitation_sequences=sorted(set(rl_imitation_sequences)),
 501         skipped_imitation_policies=skipped_policies,
 502     )

  50     if yaml is None:
  51         raise RuntimeError(f'PyYAML is required to parse {description}; install python3-yaml.')
  52 
  53     with open(path, 'r', encoding='utf-8') as f:
  54         root = yaml.safe_load(f) or {}
  55     if not isinstance(root, dict):
  56         raise ValueError(f'{description} {path} must contain a YAML map.')
  57     return root

  60 def _resolve_csv_path(raw_path: str, sequence_dir: str, package_share_legged_controllers: str) -> str:
  61     if not raw_path:
  62         return raw_path
  63     if os.path.isabs(raw_path):
  64         return os.path.normpath(raw_path)
  65 
  66     relative_to_sequence = os.path.normpath(os.path.join(sequence_dir, raw_path))
  67     if os.path.isfile(relative_to_sequence):
  68         return relative_to_sequence
  69 
  70     relative_to_share = os.path.normpath(os.path.join(package_share_legged_controllers, raw_path))
  71     if os.path.isfile(relative_to_share):
  72         return relative_to_share
  73 
  74     return relative_to_sequence

  77 def _parse_sequence_yaml(path: str, package_share_legged_controllers: str) -> Tuple[bool, str]:
  78     """Validate sequence YAML and return (has_rl_imitation_phase, imitation_policy)."""
  79     root = _load_yaml_file(path, 'action sequence')
  80 
  81     sequence = root.get('sequence') or {}
  82     if not isinstance(sequence, dict):
  83         raise ValueError(f'Action sequence {path} must contain sequence map.')
  84     phases = sequence.get('phases') or []
  85     if not isinstance(phases, list) or not phases:
  86         raise ValueError(f'Action sequence {path} must contain non-empty sequence.phases list.')
  87 
  88     has_rl_imitation = False
  89     sequence_dir = os.path.dirname(path)
  90     for index, phase in enumerate(phases):
  91         if not isinstance(phase, dict):
  92             raise ValueError(f'Action sequence {path} phase[{index}] must be a map.')
  93         mode = str(phase.get('mode', '')).strip()
  94         phase_name = str(phase.get('name') or f'phase_{index}')
  95         if mode not in VALID_PHASE_MODES:
  96             raise ValueError(
  97                 f"Action sequence {path} phase '{phase_name}' has invalid mode {mode!r}; "
  98                 f'expected one of {sorted(VALID_PHASE_MODES)}.'
  99             )
 100 
 101         if mode == 'rl_imitation':
 102             has_rl_imitation = True
 103             if phase.get('csv'):
 104                 raise ValueError(f"Action sequence {path} phase '{phase_name}' mode rl_imitation must not set csv.")
 105             if phase.get('velocity_timeline'):
 106                 raise ValueError(
 107                     f"Action sequence {path} phase '{phase_name}' mode rl_imitation does not support velocity_timeline."
 108                 )
 109             continue
 110 
 111         raw_csv = str(phase.get('csv') or '').strip()
 112         if not raw_csv:
 113             raise ValueError(f"Action sequence {path} phase '{phase_name}' mode {mode} requires csv.")
 114         resolved_csv = _resolve_csv_path(raw_csv, sequence_dir, package_share_legged_controllers)
 115         if not os.path.isfile(resolved_csv):
 116             raise ValueError(
 117                 f"Action sequence {path} phase '{phase_name}' csv is missing: "
 118                 f'{raw_csv!r} (resolved: {resolved_csv!r})'
 119             )
 120     raw_imitation = sequence.get('imitation_policy')
 121     if raw_imitation is None:
 122         raw_imitation = sequence.get('imitation_policy_name')
 123     imitation_policy = ''
 124     if raw_imitation is not None:
 125         imitation_policy = str(raw_imitation).strip()
 126 
 127     return has_rl_imitation, imitation_policy

 134 def _validate_imitation_policy_assets(policy_name: str, robot_type: str, policy_root_path: str) -> None:
 135     policy_dir = os.path.normpath(os.path.join(policy_root_path, robot_type, policy_name))
 136     robot_cfg_path = os.path.join(policy_dir, 'rl_config.yaml')
 137     if not os.path.isfile(robot_cfg_path):
 138         raise ValueError(
 139             f"Imitation policy '{policy_name}' is missing rl_config.yaml for robot_type '{robot_type}': "
 140             f'{robot_cfg_path}'
 141         )
 142 
 143     root = _load_yaml_file(robot_cfg_path, 'imitation policy rl_config')
 144     rc = root.get('RobotCfg') or {}
 145     if not isinstance(rc, dict):
 146         raise ValueError(f"Imitation policy '{policy_name}' rl_config.yaml must contain RobotCfg map.")
 147 
 148     policy_type = str(rc.get('policy_type') or '').strip()
 149     if policy_type and policy_type != 'rl_imitation':
 150         raise ValueError(
 151             f"Imitation policy '{policy_name}' RobotCfg.policy_type must be 'rl_imitation', got {policy_type!r}."
 152         )
 153 
 154     policy_model = str(rc.get('policy_name') or '').strip()
 155     motion_model = str(rc.get('motion_name') or '').strip()
 156     if not policy_model:
 157         raise ValueError(f"Imitation policy '{policy_name}' RobotCfg.policy_name is missing.")
 158     if not motion_model:
 159         raise ValueError(f"Imitation policy '{policy_name}' RobotCfg.motion_name is missing.")
 160     if not _model_asset_exists(policy_dir, policy_model):
 161         raise ValueError(
 162             f"Imitation policy '{policy_name}' policy model asset is missing: "
 163             f'{os.path.join(policy_dir, policy_model)}.(onnx|trt)'
 164         )
 165     if not _model_asset_exists(policy_dir, motion_model):
 166         raise ValueError(
 167             f"Imitation policy '{policy_name}' motion model asset is missing: "
 168             f'{os.path.join(policy_dir, motion_model)}.(onnx|trt)'
 169         )

 172 def _iter_keyboard_sequence_paths(config_path: str) -> Iterable[str]:

 174         raise RuntimeError('PyYAML is required to parse keyboard action maps; install python3-yaml.')

 176         root: Dict[str, Any] = yaml.safe_load(f) or {}
 177

 179     actions = cfg.get('actions') or []
 180     if not isinstance(actions, list):
 181         raise ValueError(f'Keyboard action map {config_path} must contain keyboard_actions.actions list.')

 185         raw = action.get('sequence') or action.get('sequence_path') or ''
 186         raw = str(raw).strip()
 187         if raw:
 188             yield raw

 196     name = _normalize_combo_name(raw_name)
 197     mapping = {
 198         'a': 0,
 199         'b': 1,
 200         'x': 3,
 201         'y': 4,
 202         'lb': 6,
 203         'l1': 6,
 204         'rb': 7,
 205         'r1': 7,

 207         'ltbutton': 8,
 208         'l2': 8,
 209         'l2button': 8,
 210         'rt': 9,
 211         'rtbutton': 9,
 212         'r2': 9,
 213         'r2button': 9,
 214         'back': 10,
 215         'select': 10,

 218         'lsbutton': 13,
 219         'l3': 13,
 220         'rs': 14,
 221         'rsbutton': 14,
 222         'r3': 14,

 227 def _dpad_axis(raw_name: str) -> int:
 228     name = _normalize_combo_name(raw_name)
 229     if name in {'dpadlr', 'dpadx', 'dirlr', 'hatx'}:
 230         return 6
 231     if name in {'dpadud', 'dpady', 'dirud', 'haty'}:
 232         return 7
 233     return -1

 236 def _axis_sign(raw_direction: str, axis_index: int) -> int:
 237     direction = _normalize_combo_name(raw_direction)
 238     if direction in {'positive', 'pos', '+'}:
 239         return 1
 240     if direction in {'negative', 'neg', '-'}:
 241         return -1
 242     if direction == 'up' and axis_index == 7:
 243         return 1
 244     if direction == 'down' and axis_index == 7:
 245         return -1
 246     if direction == 'left' and axis_index == 6:
 247         return 1
 248     if direction == 'right' and axis_index == 6:
 249         return -1
 250     return 0

 253 def _dpad_axis_for_direction(raw_direction: str) -> str:
 254     direction = _normalize_combo_name(raw_direction)
 255     if direction in {'up', 'down'}:
 256         return 'DPad_ud'
 257     if direction in {'left', 'right'}:
 258         return 'DPad_lr'
 259     return ''

 262 def _as_string_list(value: Any) -> List[str]:
 263     if value is None:
 264         return []
 265     if isinstance(value, str):
 266         return [value.strip()]
 267     if isinstance(value, list):
 268         out: List[str] = []
 269         for item in value:
 270             if not isinstance(item, str):
 271                 raise ValueError('combo button entries must be strings')
 272             out.append(item.strip())
 273         return out
 274     raise ValueError('combo buttons must be a string or list')

 277 def _canonical_joystick_combo(combo: Any, action_name: str, kind: str = 'Joystick imitation action') -> str:
 278     if not isinstance(combo, dict):
 279         raise ValueError(f"{kind} '{action_name}' needs combo map.")
 280 
 281     buttons: List[int] = []
 282     axes: List[Tuple[int, int]] = []
 283 
 284     for button_name in _as_string_list(combo.get('buttons')):
 285         index = _button_index(button_name)
 286         if index < 0:
 287             raise ValueError(f"{kind} '{action_name}' has unknown button {button_name!r}.")
 288         buttons.append(index)
 289 
 290     if combo.get('dpad') is not None:
 291         direction = str(combo.get('dpad')).strip()
 292         axis_name = _dpad_axis_for_direction(direction)
 293         axis_index = _dpad_axis(axis_name)
 294         sign = _axis_sign(direction, axis_index)
 295         if axis_index < 0 or sign == 0:
 296             raise ValueError(f"{kind} '{action_name}' has invalid dpad direction {direction!r}.")
 297         axes.append((axis_index, sign))
 298 
 299     raw_axes = combo.get('axes') or []
 300     if not isinstance(raw_axes, list):
 301         raise ValueError(f"{kind} '{action_name}' combo.axes must be a list.")
 302     for axis in raw_axes:
 303         if not isinstance(axis, dict):
 304             raise ValueError(f"{kind} '{action_name}' combo.axes entries must be maps.")
 305         axis_name = str(axis.get('name') or axis.get('axis') or '').strip()
 306         axis_index = _dpad_axis(axis_name)
 307         if axis_index < 0:
 308             raise ValueError(f"{kind} '{action_name}' has unknown D-pad axis {axis_name!r}.")
 309         direction = str(axis.get('direction') or '').strip()
 310         sign = _axis_sign(direction, axis_index)
 311         if sign == 0:
 312             raise ValueError(f"{kind} '{action_name}' axis {axis_name!r} has invalid direction {direction!r}.")
 313         axes.append((axis_index, sign))
 314 
 315     if not buttons and not axes:
 316         raise ValueError(f"{kind} '{action_name}' combo has no requirements.")
 317 
 318     parts = [f'b{index}' for index in sorted(buttons)]
 319     parts.extend(f'a{index}{"+" if sign > 0 else "-"}' for index, sign in sorted(axes))
 320     return '|'.join(parts)

 323 def _iter_joystick_imitation_policy_entries(config_path: str) -> Iterable[str]:
 324     root = _load_yaml_file(config_path, 'joystick imitation action map')
 325     cfg = root.get('joystick_imitation_actions') or root
 326     if not isinstance(cfg, dict):
 327         raise ValueError(f'Joystick imitation map {config_path} must contain a YAML map.')
 328     actions = cfg.get('actions') or []
 329     if not isinstance(actions, list):
 330         raise ValueError(f'Joystick imitation map {config_path} must contain joystick_imitation_actions.actions list.')
 331 
 332     seen_names: Set[str] = set()
 333     seen_combos: Set[str] = set()
 334     for index, action in enumerate(actions):
 335         if not isinstance(action, dict):
 336             raise ValueError(f'Joystick imitation map {config_path} action[{index}] must be a map.')
 337         name = str(action.get('name') or '').strip()
 338         if not name:
 339             raise ValueError(f'Joystick imitation map {config_path} action[{index}] needs non-empty name.')
 340         if name in seen_names:
 341             raise ValueError(f"Joystick imitation map {config_path} has duplicate action name '{name}'.")
 342         seen_names.add(name)
 343 
 344         canonical_combo = _canonical_joystick_combo(action.get('combo'), name)
 345         if canonical_combo in seen_combos:
 346             raise ValueError(f"Joystick imitation map {config_path} has duplicate combo for action '{name}'.")
 347         seen_combos.add(canonical_combo)
 348 
 349         if not bool(action.get('enabled', False)):
 350             continue
 351         policy = str(action.get('policy') or '').strip()
 352         if not policy:
 353             raise ValueError(f"Joystick imitation action '{name}' is enabled but policy is empty.")
 354         yield policy

 461 def collect_action_sequence_manifest(
 462     *,
 463     keyboard_action_enabled: bool,
 464     keyboard_action_config_path: str,
 465     package_share_legged_controllers: str,
 466     joystick_imitation_enabled: bool = False,
 467     joystick_imitation_config_path: str = '',
 468     joystick_sequence_enabled: bool = False,
 469     joystick_sequence_config_path: str = '',
 470     robot_type: str = '',
 471     policy_root_path: str = '',
 472 ) -> ActionSequenceManifest:
 473     """Validate enabled action maps and collect policies that should be preloaded for dynamic imitation."""
 474     policies: Set[str] = set()
 475     sequence_paths: List[str] = []
 476     rl_imitation_sequences: List[str] = []
 477 
 478     if keyboard_action_enabled:
 479         resolved_map = _resolve_package_path(keyboard_action_config_path, package_share_legged_controllers)
 480         if not resolved_map or not os.path.isfile(resolved_map):
 481             raise ValueError(f'keyboard_action_enabled but map is missing: {keyboard_action_config_path!r}')
 482         sequence_paths.extend(_iter_keyboard_sequence_paths(resolved_map))
 483 
 484     if joystick_imitation_enabled:
 485         resolved_map = _resolve_package_path(joystick_imitation_config_path, package_share_legged_controllers)
 486         if not resolved_map or not os.path.isfile(resolved_map):
 487             raise ValueError(f'joystick_imitation_enabled but map is missing: {joystick_imitation_config_path!r}')
 488         for policy in _iter_joystick_imitation_policy_entries(resolved_map):
 489             policies.add(policy)
 490 
 491     if joystick_sequence_enabled:
 492         resolved_map = _resolve_package_path(joystick_sequence_config_path, package_share_legged_controllers)
 493         if not resolved_map or not os.path.isfile(resolved_map):
 494             raise ValueError(f'joystick_sequence_enabled but map is missing: {joystick_sequence_config_path!r}')
 495         sequence_paths.extend(_iter_joystick_sequence_paths(resolved_map, package_share_legged_controllers))
 496 
 497     for raw_sequence_path in sequence_paths:
 498         resolved_sequence = resolve_action_sequence_yaml_path(raw_sequence_path, package_share_legged_controllers)
 499         if not resolved_sequence or not os.path.isfile(resolved_sequence):
 500             raise ValueError(
 501                 f'action sequence referenced by keyboard/joystick map is missing: '
 502                 f'{raw_sequence_path!r} (resolved: {resolved_sequence!r})'
 503             )
 504         has_imitation, policy = _parse_sequence_yaml(resolved_sequence, package_share_legged_controllers)
 505         if has_imitation:
 506             if not policy:
 507                 raise ValueError(
 508                     f'Action sequence {resolved_sequence} has rl_imitation phase(s) but '
 509                     'sequence.imitation_policy is missing.'
 510                 )
 511             policies.add(policy)
 512             rl_imitation_sequences.append(resolved_sequence)
 513 
 514     sorted_policies = sorted(policies)
 515     skipped_policies: List[str] = []
 516     if sorted_policies and robot_type and policy_root_path:
 517         available_policies: List[str] = []
 518         for policy in sorted_policies:
 519             try:
 520                 _validate_imitation_policy_assets(policy, robot_type, policy_root_path)
 521             except (RuntimeError, ValueError, OSError) as exc:
 522                 skipped_policies.append(f"{policy}: {exc}")
 523                 continue
 524             available_policies.append(policy)
 525         sorted_policies = available_policies
 526 
 527     return ActionSequenceManifest(
 528         sequence_paths=sorted(set(sequence_paths)),
 529         imitation_policies=sorted_policies,
 530         preload_imitation_policies=sorted_policies,
 531         rl_imitation_sequences=sorted(set(rl_imitation_sequences)),
 532         skipped_imitation_policies=skipped_policies,
 533     )
