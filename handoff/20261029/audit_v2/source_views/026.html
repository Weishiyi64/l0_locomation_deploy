  31 bool LeggedController::startPolicyHandoff(const PolicyInputs& start, double duration_sec, const char* label) {
  32   if (duration_sec <= 1e-9 || start.q_snapshot.empty()) return false;
  33   const std::size_t count = jointNames_.size();
  34   std::vector<bool> mask(count, false);
  35   std::vector<JointCommandSample> commands(count);
  36   const std::size_t available = std::min(count, start.q_snapshot.size());
  37   for (std::size_t joint = 0; joint < available; ++joint)
  38   {
  39     mask[joint] = true;
  40     commands[joint].q = start.q_snapshot[joint];
  41     commands[joint].dq = 0.0;
  42     commands[joint].kp = joint < start.gains.kps.size() ? start.gains.kps[joint] : 0.0;
  43     commands[joint].kd = joint < start.gains.kds.size() ? start.gains.kds[joint] : 0.0;
  44     commands[joint].tau_ff = 0.0;
  45   }
  46   double wall_sec = last_controller_wall_sec_;
  47   if (wall_sec <= 0.0) wall_sec = nh_->now().seconds();
  48   JointCommandTransition& handoff = sequenceRuntime_.releaseHandoff();
  49   handoff.clear();
  50   if (!handoff.start(mask, commands, wall_sec, duration_sec)) return false;
  51   RCLCPP_INFO(nh_->get_logger(), "[WBC] %s handoff blend %.3fs", label, duration_sec);
  52   return true;
  53 }

  55 bool LeggedController::startWbcMotion(
  56     const std::string& motion_id, int64_t timestamp_ms,
  57     std::shared_ptr<luvbot_control_interfaces::srv::ControlWbcMotion::Response> response) {
  58   if (!wbc_enabled_)
  59   {
  60     response->message = "WBC is disabled; launch with wbc_enabled:=true";
  61     return false;
  62   }
  63   // Keep the preparation slot until cancellation is acknowledged, even after a timeout.
  64   const auto cancel_preparation = [this, &motion_id]() {
  65     const auto cancelled = policyMux_.prepareWbcMotion(
  66         luvbot_control_interfaces::srv::ControlWbcMotion::Request::CANCEL, motion_id, wbc_command_timeout_sec_);
  67     wbc_cleanup_pending_ = !cancelled || !cancelled->success;
  68     if (wbc_cleanup_pending_)
  69       RCLCPP_WARN(nh_->get_logger(), "[WBC] Worker cancellation unconfirmed; refusing new preparation until cleanup");
  70   };
  71   if (wbc_cleanup_pending_)
  72   {
  73     cancel_preparation();
  74     if (wbc_cleanup_pending_)
  75     {
  76       response->message = "WBC worker cleanup has not been acknowledged";
  77       return false;
  78     }
  79   }
  80   uint64_t generation;
  81   geometry_msgs::msg::Quaternion q;
  82   geometry_msgs::msg::Twist velocity;
  83   {
  84     std::lock_guard<std::mutex> lock(wbc_transition_mutex_);
  85     if (!wbc_start_allowed_ || wbc_return_pending_.load() || wbc_handoff_in_progress_.load())
  86     {
  87       response->message = "controller state does not allow a WBC motion";
  88       return false;
  89     }
  90     generation = wbc_generation_;
  91     q = wbc_orientation_;
  92     velocity = wbc_velocity_;
  93   }
  94   if (policyMux_.wbcActiveCommanded() || policyMux_.source() == PolicyCmdSource::kWbc)
  95   {
  96     response->message = "a WBC motion is already active";
  97     return false;
  98   }
  99   const double roll = std::atan2(2.0 * (q.w * q.x + q.y * q.z), 1.0 - 2.0 * (q.x * q.x + q.y * q.y));
 100   const double pitch = std::asin(std::max(-1.0, std::min(1.0, 2.0 * (q.w * q.y - q.z * q.x))));
 101   if (safetyChecker_ && !safetyChecker_->checkOrientation(roll, pitch).ok)
 102   {
 103     response->message = "current roll/pitch violates the safety posture limits";
 104     return false;
 105   }
 106   if (!std::isfinite(velocity.linear.x) || !std::isfinite(velocity.linear.y) || !std::isfinite(velocity.angular.z) ||
 107       std::abs(velocity.linear.x) > 0.05 || std::abs(velocity.linear.y) > 0.05 || std::abs(velocity.angular.z) > 0.10)
 108   {
 109     response->message = "WBC requires |vx|<=0.05, |vy|<=0.05 and |wz|<=0.10";
 110     return false;
 111   }
 112 
 113   const PolicyInputs amp_start = policyMux_.selectInputs(true);
 114   if (policyMux_.source() != PolicyCmdSource::kRl || !amp_start.valid)
 115   {
 116     response->message = "a fresh AMP command is required before WBC start";
 117     return false;
 118   }
 119   const uint64_t previous_wbc_revision = policyMux_.wbcCommandRevision();
 120   auto prepared = policyMux_.prepareWbcMotion(luvbot_control_interfaces::srv::ControlWbcMotion::Request::START,
 121                                               motion_id, wbc_command_timeout_sec_);
 122   if (!prepared)
 123   {
 124     cancel_preparation();
 125     response->message = "WBC worker did not respond";
 126     return false;
 127   }
 128   response->motion_id = prepared->motion_id;
 129   response->robot_cfg_path = prepared->robot_cfg_path;
 130   response->duration_sec = prepared->duration_sec;
 131   if (!prepared->success)
 132   {
 133     cancel_preparation();
 134     response->message = prepared->message;
 135     return false;
 136   }
 137   if (!policyMux_.loadWbcWalkingGains(prepared->robot_cfg_path, prepared->motion_id))
 138   {
 139     cancel_preparation();
 140     response->message = "failed to load WBC gains";
 141     return false;
 142   }
 143   if (!policyMux_.waitForWbcCommandAfter(previous_wbc_revision, wbc_command_timeout_sec_))
 144   {
 145     cancel_preparation();
 146     response->message = "WBC pre-inference command was not received";
 147     return false;
 148   }
 149 
 150   {
 151     std::lock_guard<std::mutex> lock(wbc_transition_mutex_);
 152     const PolicyInputs current_amp = policyMux_.selectInputs(true);
 153     if (generation == wbc_generation_ && wbc_start_allowed_ && policyMux_.source() == PolicyCmdSource::kRl &&
 154         current_amp.valid && policyMux_.waitForWbcCommandAfter(previous_wbc_revision, 0.0))
 155     {
 156       wbc_handoff_start_ = current_amp;
 157       wbc_handoff_pending_ = true;
 158       wbc_handoff_in_progress_.store(true);
 159       policyMux_.publishWbcActive(true, "[WBC] Activate worker");
 160       wbc_handoff_target_ = PolicyCmdSource::kWbc;
 161       std::lock_guard<std::mutex> state_lock(wbc_action_state_mutex_);
 162       wbc_action_active_ = true;
 163       wbc_action_timestamp_ms_ = timestamp_ms;
 164       wbc_action_id_ = motion_id;
 165       publishWbcActionFeedback(timestamp_ms, motion_id, "started");
 166       response->success = true;
 167       response->message = "WBC motion started";
 168       return true;
 169     }
 170   }
 171   cancel_preparation();
 172   response->message = "controller state changed or policy command expired during WBC preparation";
 173   return false;
 174 }

 176 bool LeggedController::finishWbcMotion(const char* reason, std::string& message) {
 177   uint64_t generation;
 178   PolicyInputs wbc_start;
 179   uint64_t previous_rl_revision;
 180   {
 181     std::lock_guard<std::mutex> lock(wbc_transition_mutex_);
 182     if (!policyMux_.wbcActiveCommanded() && policyMux_.source() != PolicyCmdSource::kWbc)
 183     {
 184       message = "no WBC motion is active";
 185       return false;
 186     }
 187     generation = wbc_generation_;
 188     wbc_start = policyMux_.selectInputs(true);
 189     previous_rl_revision = policyMux_.rlCommandRevision();
 190     wbc_return_pending_.store(true);
 191     policyMux_.publishWbcActive(false, reason);
 192   }
 193   if (!wbc_start.valid)
 194   {
 195     message = "WBC command expired; entering emergency damping";
 196     triggerEmergencyStop();
 197     return false;
 198   }
 199   const auto cancelled = policyMux_.prepareWbcMotion(
 200       luvbot_control_interfaces::srv::ControlWbcMotion::Request::CANCEL, "", wbc_command_timeout_sec_);
 201   wbc_cleanup_pending_ = !cancelled || !cancelled->success;
 202   publishRlObsReset();
 203   if (!policyMux_.waitForRlCommandAfter(previous_rl_revision, wbc_command_timeout_sec_))
 204   {
 205     message = "fresh AMP command timed out; entering emergency damping";
 206     RCLCPP_ERROR(nh_->get_logger(), "[WBC] %s", message.c_str());
 207     wbc_return_pending_.store(false);
 208     triggerEmergencyStop();
 209     return false;
 210   }
 211   std::unique_lock<std::mutex> lock(wbc_transition_mutex_);
 212   if (generation != wbc_generation_ || !wbc_controller_walking_ ||
 213       !policyMux_.waitForRlCommandAfter(previous_rl_revision, 0.0))
 214   {
 215     wbc_return_pending_.store(false);
 216     message = "controller left Walking or AMP command expired while returning";
 217     lock.unlock();
 218     triggerEmergencyStop();
 219     return false;
 220   }
 221   wbc_handoff_start_ = wbc_start;
 222   wbc_handoff_pending_ = true;
 223   wbc_handoff_in_progress_.store(true);
 224   wbc_handoff_target_ = PolicyCmdSource::kRl;
 225   message = "WBC motion stopped; blending to AMP";
 226   return true;
 227 }

 229 void LeggedController::abortWbcImmediately(const char* reason) {
 230   std::lock_guard<std::mutex> lock(wbc_transition_mutex_);
 231   ++wbc_generation_;
 232   wbc_start_allowed_ = false;
 233   wbc_controller_walking_ = false;
 234   wbc_handoff_pending_ = false;
 235   wbc_handoff_clear_pending_ = true;
 236   wbc_handoff_in_progress_.store(false);
 237   if (!policyMux_.wbcActiveCommanded() && policyMux_.source() != PolicyCmdSource::kWbc && !wbc_return_pending_.load())
 238     return;
 239   policyMux_.publishWbcActive(false, reason);
 240   wbc_return_pending_.store(false);
 241   setPolicySource(PolicyCmdSource::kRl, "[PolicyMux] WBC aborted; AMP source restored");
 242   publishRlObsReset();
 243   publishActiveWbcActionTerminal("failed");
 244 }

 264 void LeggedController::wbcActionCommandCallback(const std_msgs::msg::String::ConstSharedPtr msg) {
 265   int64_t timestamp_ms = 0;
 266   std::string action_id;
 267   try
 268   {
 269     const YAML::Node command = YAML::Load(msg->data);
 270     if (!command.IsMap() || !command["timestamp_ms"] || !command["action_id"])
 271       throw std::runtime_error("expected timestamp_ms and action_id");
 272     timestamp_ms = command["timestamp_ms"].as<int64_t>();
 273     action_id = command["action_id"].as<std::string>();
 274   } catch (const std::exception& exception)
 275   {
 276     RCLCPP_WARN(nh_->get_logger(), "[WBC Action] Rejected malformed command: %s", exception.what());
 277     publishWbcActionFeedback(0, "", "rejected");
 278     return;
 279   }
 280 
 281   if (!validActionId(action_id))
 282   {
 283     RCLCPP_WARN(nh_->get_logger(), "[WBC Action] Rejected invalid action_id");
 284     publishWbcActionFeedback(timestamp_ms, "", "rejected");
 285     return;
 286   }
 287 
 288   const int64_t now_ms = unixTimeMs();
 289   if (timestamp_ms <= 0 || timestamp_ms < now_ms - kWbcActionMaxAgeMs || timestamp_ms > now_ms + kWbcActionMaxFutureMs)
 290   {
 291     RCLCPP_WARN(nh_->get_logger(), "[WBC Action] Rejected timestamp %lld at %lld", static_cast<long long>(timestamp_ms),
 292                 static_cast<long long>(now_ms));
 293     publishWbcActionFeedback(timestamp_ms, action_id, "rejected");
 294     return;
 295   }
 296 
 297   const auto received_at = std::chrono::steady_clock::now();
 298   const std::string request_key = std::to_string(timestamp_ms) + ":" + action_id;
 299   {
 300     std::lock_guard<std::mutex> lock(wbc_action_state_mutex_);
 301     while (!recent_wbc_action_requests_.empty() &&
 302            received_at - recent_wbc_action_requests_.front().second > std::chrono::seconds(6))
 303       recent_wbc_action_requests_.pop_front();
 304     const auto duplicate = std::find_if(recent_wbc_action_requests_.begin(), recent_wbc_action_requests_.end(),
 305                                         [&request_key](const auto& request) { return request.first == request_key; });
 306     if (duplicate != recent_wbc_action_requests_.end())
 307     {
 308       RCLCPP_WARN(nh_->get_logger(), "[WBC Action] Rejected duplicate request %s", request_key.c_str());
 309       publishWbcActionFeedback(timestamp_ms, action_id, "rejected");
 310       return;
 311     }
 312     if (recent_wbc_action_requests_.size() >= kMaxRecentWbcActionRequests) recent_wbc_action_requests_.pop_front();
 313     recent_wbc_action_requests_.emplace_back(request_key, received_at);
 314   }
 315 
 316   std::unique_lock<std::mutex> lock(auxiliary_action_trigger_mutex_, std::try_to_lock);
 317   if (!lock.owns_lock())
 318   {
 319     publishWbcActionFeedback(timestamp_ms, action_id, "rejected");
 320     RCLCPP_WARN(nh_->get_logger(), "[WBC Action] Rejected %s: another action request is being handled", action_id.c_str());
 321     return;
 322   }
 323   auto response = std::make_shared<luvbot_control_interfaces::srv::ControlWbcMotion::Response>();
 324   if (startWbcMotion(action_id, timestamp_ms, response)) return;
 325   RCLCPP_WARN(nh_->get_logger(), "[WBC Action] Rejected %s: %s", action_id.c_str(), response->message.c_str());
 326   publishWbcActionFeedback(timestamp_ms, action_id, "rejected");
 327 }

 329 void LeggedController::wbcMotionDoneCallback(const std_msgs::msg::Bool::ConstSharedPtr msg) {
 330   std::lock_guard<std::mutex> lock(auxiliary_action_trigger_mutex_);
 331   if (!policyMux_.wbcActiveCommanded()) return;
 332   if (!msg->data)
 333   {
 334     RCLCPP_ERROR(nh_->get_logger(), "[WBC Action] Worker failed; entering emergency damping");
 335     triggerEmergencyStop();
 336     return;
 337   }
 338   std::string message;
 339   const bool returned_to_amp = finishWbcMotion("[WBC] Motion completed", message);
 340   if (!returned_to_amp) RCLCPP_WARN(nh_->get_logger(), "[WBC Action] %s", message.c_str());
 341   publishActiveWbcActionTerminal(returned_to_amp ? "completed" : "failed");
 342 }

 344 void LeggedController::textWbcCancelCallback(const std_msgs::msg::String::ConstSharedPtr msg) {
 345   if (msg->data.compare(0, 2, "T_") != 0 || msg->data.size() > 32) return;
 346   std::lock_guard<std::mutex> lock(auxiliary_action_trigger_mutex_);
 347   {
 348     std::lock_guard<std::mutex> state_lock(wbc_action_state_mutex_);
 349     if (!wbc_action_active_ || wbc_action_id_ != msg->data) return;
 350   }
 351   std::string message;
 352   const bool returning = finishWbcMotion("[Text WBC] Cancelled", message);
 353   publishActiveWbcActionTerminal(returning ? "cancelled" : "failed");
 354 }

 356 void LeggedController::updateWbcControlState() {
 357   if (!wbc_enabled_) return;
 358   std::lock_guard<std::mutex> lock(wbc_transition_mutex_);
 359   if (!wbc_handoff_pending_ && !sequenceRuntime_.releaseHandoffActive(last_controller_wall_sec_))
 360     wbc_handoff_in_progress_.store(false);
 361   wbc_controller_walking_ = fsm_ && fsm_->isWalking();
 362   wbc_start_allowed_ = wbc_controller_walking_ && !sequenceRuntime_.isActive() &&
 363                        !policyMux_.imitationActiveCommanded() && !wbc_handoff_pending_ &&
 364                        !sequenceRuntime_.releaseHandoffActive(last_controller_wall_sec_);
 365   const auto& q = policy_observation_.imu.orientation;
 366   const double roll = std::atan2(2.0 * (q.w * q.x + q.y * q.z), 1.0 - 2.0 * (q.x * q.x + q.y * q.y));
 367   const double pitch = std::asin(std::max(-1.0, std::min(1.0, 2.0 * (q.w * q.y - q.z * q.x))));
 368   const auto& velocity = policy_observation_.velocity_command;
 369   wbc_start_allowed_ = wbc_start_allowed_ && std::isfinite(roll) && std::isfinite(pitch) &&
 370                        (!safetyChecker_ || safetyChecker_->checkOrientation(roll, pitch).ok) &&
 371                        std::isfinite(velocity.linear.x) && std::isfinite(velocity.linear.y) &&
 372                        std::isfinite(velocity.angular.z) && std::abs(velocity.linear.x) <= 0.05 &&
 373                        std::abs(velocity.linear.y) <= 0.05 && std::abs(velocity.angular.z) <= 0.10;
 374   wbc_orientation_ = policy_observation_.imu.orientation;
 375   wbc_velocity_ = policy_observation_.velocity_command;
 376   if (wbc_handoff_clear_pending_)
 377   {
 378     sequenceRuntime_.releaseHandoff().clear();
 379     wbc_handoff_clear_pending_ = false;
 380   }
 381   if (wbc_handoff_pending_)
 382   {
 383     if (wbc_controller_walking_)
 384     {
 385       startPolicyHandoff(wbc_handoff_start_, wbc_transition_sec_, "policy switch");
 386       setPolicySource(wbc_handoff_target_, "[PolicyMux] WBC handoff applied");
 387     }
 388     wbc_return_pending_.store(false);
 389     wbc_handoff_pending_ = false;
 390   }

 102   const uint64_t previous_wbc_revision = policyMux_.wbcCommandRevision();
 103   auto prepared = policyMux_.prepareWbcMotion(luvbot_control_interfaces::srv::ControlWbcMotion::Request::START,
 104                                               motion_id, wbc_command_timeout_sec_);
 105   if (!prepared)
 106   {
 107     policyMux_.publishWbcActive(false, "[WBC] Preparation timed out");
 108     response->message = "WBC worker did not respond";
 109     return false;
 110   }
 111   response->motion_id = prepared->motion_id;
 112   response->robot_cfg_path = prepared->robot_cfg_path;
 113   response->duration_sec = prepared->duration_sec;
 114   if (!prepared->success)
 115   {
 116     response->message = prepared->message;
 117     return false;
 118   }
 119   if (!policyMux_.loadWbcWalkingGains(prepared->robot_cfg_path, prepared->motion_id))
 120   {
 121     policyMux_.prepareWbcMotion(luvbot_control_interfaces::srv::ControlWbcMotion::Request::CANCEL, motion_id,
 122                                 wbc_command_timeout_sec_);
 123     response->message = "failed to load WBC gains";
 124     return false;
 125   }
 126   if (!policyMux_.waitForWbcCommandAfter(previous_wbc_revision, wbc_command_timeout_sec_))
 127   {
 128     policyMux_.prepareWbcMotion(luvbot_control_interfaces::srv::ControlWbcMotion::Request::CANCEL, motion_id,
 129                                 wbc_command_timeout_sec_);
 130     response->message = "WBC pre-inference command was not received";
 131     return false;
 132   }
 133 
 134   {
 135     std::lock_guard<std::mutex> lock(wbc_transition_mutex_);
 136     const PolicyInputs current_amp = policyMux_.selectInputs(true);
 137     if (generation == wbc_generation_ && wbc_start_allowed_ && policyMux_.source() == PolicyCmdSource::kRl &&
 138         current_amp.valid && policyMux_.waitForWbcCommandAfter(previous_wbc_revision, 0.0))
 139     {
 140       wbc_handoff_start_ = current_amp;
 141       wbc_handoff_pending_ = true;
 142       wbc_handoff_in_progress_.store(true);
 143       policyMux_.publishWbcActive(true, "[WBC] Activate worker");
 144       wbc_handoff_target_ = PolicyCmdSource::kWbc;
 145       std::lock_guard<std::mutex> state_lock(wbc_action_state_mutex_);
 146       wbc_action_active_ = true;
 147       wbc_action_timestamp_ms_ = timestamp_ms;
 148       wbc_action_id_ = motion_id;
 149       publishWbcActionFeedback(timestamp_ms, motion_id, "started");
 150       response->success = true;
 151       response->message = "WBC motion started";
 152       return true;
 153     }
 154   }
 155   policyMux_.prepareWbcMotion(luvbot_control_interfaces::srv::ControlWbcMotion::Request::CANCEL, motion_id,
 156                               wbc_command_timeout_sec_);
 157   response->message = "controller state changed or policy command expired during WBC preparation";
 158   return false;
 159 }

 161 bool LeggedController::finishWbcMotion(const char* reason, std::string& message) {
 162   uint64_t generation;
 163   PolicyInputs wbc_start;
 164   uint64_t previous_rl_revision;
 165   {
 166     std::lock_guard<std::mutex> lock(wbc_transition_mutex_);
 167     if (!policyMux_.wbcActiveCommanded() && policyMux_.source() != PolicyCmdSource::kWbc)
 168     {
 169       message = "no WBC motion is active";
 170       return false;
 171     }
 172     generation = wbc_generation_;
 173     wbc_start = policyMux_.selectInputs(true);
 174     previous_rl_revision = policyMux_.rlCommandRevision();
 175     wbc_return_pending_.store(true);
 176     policyMux_.publishWbcActive(false, reason);
 177   }
 178   if (!wbc_start.valid)
 179   {
 180     message = "WBC command expired; entering emergency damping";
 181     triggerEmergencyStop();
 182     return false;
 183   }
 184   (void)policyMux_.prepareWbcMotion(luvbot_control_interfaces::srv::ControlWbcMotion::Request::CANCEL, "",
 185                                     wbc_command_timeout_sec_);
 186   publishRlObsReset();
 187   if (!policyMux_.waitForRlCommandAfter(previous_rl_revision, wbc_command_timeout_sec_))
 188   {
 189     message = "fresh AMP command timed out; entering emergency damping";
 190     RCLCPP_ERROR(nh_->get_logger(), "[WBC] %s", message.c_str());
 191     wbc_return_pending_.store(false);
 192     triggerEmergencyStop();
 193     return false;
 194   }
 195   std::unique_lock<std::mutex> lock(wbc_transition_mutex_);
 196   if (generation != wbc_generation_ || !wbc_controller_walking_ ||
 197       !policyMux_.waitForRlCommandAfter(previous_rl_revision, 0.0))
 198   {
 199     wbc_return_pending_.store(false);
 200     message = "controller left Walking or AMP command expired while returning";
 201     lock.unlock();
 202     triggerEmergencyStop();
 203     return false;
 204   }
 205   wbc_handoff_start_ = wbc_start;
 206   wbc_handoff_pending_ = true;
 207   wbc_handoff_in_progress_.store(true);
 208   wbc_handoff_target_ = PolicyCmdSource::kRl;
 209   message = "WBC motion stopped; blending to AMP";
 210   return true;
 211 }

 213 void LeggedController::abortWbcImmediately(const char* reason) {
 214   std::lock_guard<std::mutex> lock(wbc_transition_mutex_);
 215   ++wbc_generation_;
 216   wbc_start_allowed_ = false;
 217   wbc_controller_walking_ = false;
 218   wbc_handoff_pending_ = false;
 219   wbc_handoff_clear_pending_ = true;
 220   wbc_handoff_in_progress_.store(false);
 221   if (!policyMux_.wbcActiveCommanded() && policyMux_.source() != PolicyCmdSource::kWbc && !wbc_return_pending_.load())
 222     return;
 223   policyMux_.publishWbcActive(false, reason);
 224   wbc_return_pending_.store(false);
 225   setPolicySource(PolicyCmdSource::kRl, "[PolicyMux] WBC aborted; AMP source restored");
 226   publishRlObsReset();
 227   publishActiveWbcActionTerminal("failed");
 228 }

 230 void LeggedController::publishWbcActionFeedback(int64_t timestamp_ms, const std::string& action_id,
 231                                                 const char* status) {
 232   if (!wbcActionFeedbackPub_) return;
 233   std_msgs::msg::String feedback;
 234   feedback.data = "{\"timestamp_ms\":" + std::to_string(timestamp_ms) + ",\"action_id\":\"" + action_id +
 235                   "\",\"status\":\"" + status + "\"}";
 236   wbcActionFeedbackPub_->publish(feedback);
 237 }

 239 void LeggedController::publishActiveWbcActionTerminal(const char* status) {
 240   std::lock_guard<std::mutex> lock(wbc_action_state_mutex_);
 241   if (!wbc_action_active_) return;
 242   publishWbcActionFeedback(wbc_action_timestamp_ms_, wbc_action_id_, status);
 243   wbc_action_active_ = false;
 244   wbc_action_timestamp_ms_ = 0;
 245   wbc_action_id_.clear();
 246 }

 248 void LeggedController::wbcActionCommandCallback(const std_msgs::msg::String::ConstSharedPtr msg) {
 249   int64_t timestamp_ms = 0;
 250   std::string action_id;
 251   try
 252   {
 253     const YAML::Node command = YAML::Load(msg->data);
 254     if (!command.IsMap() || !command["timestamp_ms"] || !command["action_id"])
 255       throw std::runtime_error("expected timestamp_ms and action_id");
 256     timestamp_ms = command["timestamp_ms"].as<int64_t>();
 257     action_id = command["action_id"].as<std::string>();
 258   } catch (const std::exception& exception)
 259   {
 260     RCLCPP_WARN(nh_->get_logger(), "[WBC Action] Rejected malformed command: %s", exception.what());
 261     publishWbcActionFeedback(0, "", "rejected");
 262     return;
 263   }
 264 
 265   if (!validActionId(action_id))
 266   {
 267     RCLCPP_WARN(nh_->get_logger(), "[WBC Action] Rejected invalid action_id");
 268     publishWbcActionFeedback(timestamp_ms, "", "rejected");
 269     return;
 270   }
 271 
 272   const int64_t now_ms = unixTimeMs();
 273   if (timestamp_ms <= 0 || timestamp_ms < now_ms - kWbcActionMaxAgeMs || timestamp_ms > now_ms + kWbcActionMaxFutureMs)
 274   {
 275     RCLCPP_WARN(nh_->get_logger(), "[WBC Action] Rejected timestamp %lld at %lld", static_cast<long long>(timestamp_ms),
 276                 static_cast<long long>(now_ms));
 277     publishWbcActionFeedback(timestamp_ms, action_id, "rejected");
 278     return;
 279   }
 280 
 281   const auto received_at = std::chrono::steady_clock::now();
 282   const std::string request_key = std::to_string(timestamp_ms) + ":" + action_id;
 283   {
 284     std::lock_guard<std::mutex> lock(wbc_action_state_mutex_);
 285     while (!recent_wbc_action_requests_.empty() &&
 286            received_at - recent_wbc_action_requests_.front().second > std::chrono::seconds(6))
 287       recent_wbc_action_requests_.pop_front();
 288     const auto duplicate = std::find_if(recent_wbc_action_requests_.begin(), recent_wbc_action_requests_.end(),
 289                                         [&request_key](const auto& request) { return request.first == request_key; });
 290     if (duplicate != recent_wbc_action_requests_.end())
 291     {
 292       RCLCPP_WARN(nh_->get_logger(), "[WBC Action] Rejected duplicate request %s", request_key.c_str());
 293       publishWbcActionFeedback(timestamp_ms, action_id, "rejected");
 294       return;
 295     }
 296     if (recent_wbc_action_requests_.size() >= kMaxRecentWbcActionRequests) recent_wbc_action_requests_.pop_front();
 297     recent_wbc_action_requests_.emplace_back(request_key, received_at);
 298   }
 299 
 300   std::lock_guard<std::mutex> lock(auxiliary_action_trigger_mutex_);
 301   auto response = std::make_shared<luvbot_control_interfaces::srv::ControlWbcMotion::Response>();
 302   if (startWbcMotion(action_id, timestamp_ms, response)) return;
 303   RCLCPP_WARN(nh_->get_logger(), "[WBC Action] Rejected %s: %s", action_id.c_str(), response->message.c_str());
 304   publishWbcActionFeedback(timestamp_ms, action_id, "rejected");
 305 }

 307 void LeggedController::wbcMotionDoneCallback(const std_msgs::msg::Bool::ConstSharedPtr msg) {
 308   std::lock_guard<std::mutex> lock(auxiliary_action_trigger_mutex_);
 309   if (!policyMux_.wbcActiveCommanded()) return;
 310   if (!msg->data)
 311   {
 312     RCLCPP_ERROR(nh_->get_logger(), "[WBC Action] Worker failed; entering emergency damping");
 313     triggerEmergencyStop();
 314     return;
 315   }
 316   std::string message;
 317   const bool returned_to_amp = finishWbcMotion("[WBC] Motion completed", message);
 318   if (!returned_to_amp) RCLCPP_WARN(nh_->get_logger(), "[WBC Action] %s", message.c_str());
 319   publishActiveWbcActionTerminal(returned_to_amp ? "completed" : "failed");
 320 }

 322 void LeggedController::updateWbcControlState() {
 323   if (!wbc_enabled_) return;
 324   std::lock_guard<std::mutex> lock(wbc_transition_mutex_);
 325   if (!wbc_handoff_pending_ && !sequenceRuntime_.releaseHandoffActive(last_controller_wall_sec_))
 326     wbc_handoff_in_progress_.store(false);
 327   wbc_controller_walking_ = fsm_ && fsm_->isWalking();
 328   wbc_start_allowed_ = wbc_controller_walking_ && !sequenceRuntime_.isActive() &&
 329                        !policyMux_.imitationActiveCommanded() && !wbc_handoff_pending_ &&
 330                        !sequenceRuntime_.releaseHandoffActive(last_controller_wall_sec_);
 331   const auto& q = policy_observation_.imu.orientation;
 332   const double roll = std::atan2(2.0 * (q.w * q.x + q.y * q.z), 1.0 - 2.0 * (q.x * q.x + q.y * q.y));
 333   const double pitch = std::asin(std::max(-1.0, std::min(1.0, 2.0 * (q.w * q.y - q.z * q.x))));
 334   const auto& velocity = policy_observation_.velocity_command;
 335   wbc_start_allowed_ = wbc_start_allowed_ && std::isfinite(roll) && std::isfinite(pitch) &&
 336                        (!safetyChecker_ || safetyChecker_->checkOrientation(roll, pitch).ok) &&
 337                        std::isfinite(velocity.linear.x) && std::isfinite(velocity.linear.y) &&
 338                        std::isfinite(velocity.angular.z) && std::abs(velocity.linear.x) <= 0.05 &&
 339                        std::abs(velocity.linear.y) <= 0.05 && std::abs(velocity.angular.z) <= 0.10;
 340   wbc_orientation_ = policy_observation_.imu.orientation;
 341   wbc_velocity_ = policy_observation_.velocity_command;
 342   if (wbc_handoff_clear_pending_)
 343   {
 344     sequenceRuntime_.releaseHandoff().clear();
 345     wbc_handoff_clear_pending_ = false;
 346   }
 347   if (wbc_handoff_pending_)
 348   {
 349     if (wbc_controller_walking_)
 350     {
 351       startPolicyHandoff(wbc_handoff_start_, wbc_transition_sec_, "policy switch");
 352       setPolicySource(wbc_handoff_target_, "[PolicyMux] WBC handoff applied");
 353     }
 354     wbc_return_pending_.store(false);
 355     wbc_handoff_pending_ = false;
 356   }
 357 }
