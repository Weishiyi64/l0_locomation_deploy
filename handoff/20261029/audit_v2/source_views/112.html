  28 MANAGED_NAMES=()

  34 function process_group_alive() {
  35     local pid="$1"
  36     [[ -n "$pid" ]] && kill -0 -- "-$pid" 2>/dev/null
  37 }
  38 
  39 function signal_process_group() {
  40     local pid="$1"
  41     local signal="$2"
  42     kill "-$signal" -- "-$pid" 2>/dev/null || true
  43 }
  44 
  45 function wait_for_process_group_exit() {
  46     local pid="$1"
  47     local timeout_sec="$2"
  48     local deadline=$((SECONDS + timeout_sec))
  49 
  50     while process_group_alive "$pid" && (( SECONDS < deadline )); do
  51         sleep 0.2
  52     done
  53 
  54     ! process_group_alive "$pid"
  55 }

  57 function start_managed_process() {
  58     local name="$1"
  59     shift
  60

  68     local exit_code=$?
  69

  71         return "$exit_code"
  72     fi
  73     CLEANUP_STARTED=true
  74     trap '' INT TERM

  76

  80         echo "Repeated Ctrl+C/SIGTERM will be ignored until cleanup completes."
  81     fi
  82

  84         local pid="${MANAGED_PIDS[$idx]}"

  98             fi
  99         fi

 103 
 104     return "$exit_code"
