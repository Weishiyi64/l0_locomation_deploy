  20 class GoalToPlanner(Node):
  21     def __init__(self):
  22         super().__init__('goal_to_planner')
  23 
  24         def param(name, default):
  25             return self.declare_parameter(name, default).value
  26 
  27         self.frame = param('odom_frame', 'device0/odom')
  28         self.navigation_map = param('navigation_map', 'flat')
  29         self.speed = float(param('nav_speed', 0.5))
  30         self.turn_speed = float(param('turn_speed', 2.0))
  31         self.timeout = float(param('odom_timeout', 0.5))
  32         self.goal = self.pose = None
  33         self.route = self.map_planner = None
  34         self.planar_speed = 0.0
  35         self.route_path = self.create_publisher(
  36             PathMsg, '/navigation_path', QoSProfile(depth=1, durability=DurabilityPolicy.TRANSIENT_LOCAL)
  37         )
  38         self.remaining_path = self.create_publisher(
  39             PathMsg, '/navigation_remaining_path', QoSProfile(depth=1, durability=DurabilityPolicy.TRANSIENT_LOCAL)
  40         )
  41         self.create_subscription(Float32MultiArray, '/navigation_goal', self.on_navigation_goal, 10)
  42         self.wait_reason = None
  43         self.accepted_goal = self.create_publisher(
  44             PoseArray,
  45             '/goal_to_planner/accepted_goal',
  46             QoSProfile(depth=1, durability=DurabilityPolicy.TRANSIENT_LOCAL),
  47         )
  48         self.odom_time = -float('inf')
  49         self.input_pub = self.create_publisher(Float32MultiArray, '/planner_input', 1)
  50         self.status = self.create_publisher(String, '/goal_to_planner/state', 10)
  51         self.create_subscription(PoseStamped, param('goal_topic', '/goal_pose'), self.on_goal, 10)
  52         self.create_subscription(
  53             Odometry, param('odom_topic', '/manifold/ODIN2/device0/odometry'), self.on_odom, qos_profile_sensor_data
  54         )
  55         # Match the trained high-level task update period (5 Hz).
  56         self.create_timer(0.2, self.tick)

  58     def publish_accepted_goal(self):
  59         message = PoseArray()
  60         message.header.frame_id = self.frame
  61         message.header.stamp = self.get_clock().now().to_msg()
  62         if self.goal is not None:
  63             pose = Pose()
  64             pose.position.x, pose.position.y = float(self.goal.x), float(self.goal.y)
  65             yaw = self.goal.heading or 0.0
  66             pose.orientation.z = float(np.sin(yaw / 2))
  67             pose.orientation.w = float(np.cos(yaw / 2))
  68             message.poses = [pose]
  69         self.accepted_goal.publish(message)
  70 
  71     @staticmethod
  72     def quaternion(pose):
  73         q = pose.orientation
  74         return [q.x, q.y, q.z, q.w]

  76     def publish_route(self, remaining=False):
  77         message = PathMsg()
  78         message.header.frame_id = self.frame
  79         message.header.stamp = self.get_clock().now().to_msg()
  80         if self.route is not None and self.route.path is not None:
  81             points = self.route.path[self.route.index :] if remaining else self.route.path
  82             for x, y in points:
  83                 pose = PoseStamped()
  84                 pose.header = message.header
  85                 pose.pose.position.x, pose.pose.position.y = float(x), float(y)
  86                 pose.pose.orientation.w = 1.0
  87                 message.poses.append(pose)
  88         (self.remaining_path if remaining else self.route_path).publish(message)

  90     def start_navigation(self, goal):
  91         if self.goal is not None and self.route is not None and self.goal.keep_position_for(goal):
  92             # A nearby goal does not reopen completed navigation. Heading and
  93             # speed settings can still change; preserve the completed XY anchor.
  94             if goal.heading != self.goal.heading:
  95                 self.goal.heading_locked = False
  96                 self.route.complete = False
  97                 self.route.hold_steps = 0
  98             self.goal.heading = goal.heading
  99             self.goal.speed, self.goal.turn_speed = goal.speed, goal.turn_speed
 100             return
 101         if not self.navigation_map:
 102             raise ValueError('Set navigation_map to a map registered in device0/odom before using A*')
 103         if self.map_planner is None:
 104             self.map_planner = load_house_planner(Path(self.navigation_map))
 105         self.goal = goal
 106         self.route = NavigationRoute(self.map_planner, goal)
 107 
 108     def on_navigation_goal(self, msg):
 109         try:
 110             self.start_navigation(absolute_goal(msg.data))
 111         except (ValueError, OSError, RuntimeError) as error:
 112             self.route = self.goal = None
 113             self.get_logger().error(str(error))
 114         self.publish_route()
 115         self.publish_accepted_goal()
 116 
 117     def on_goal(self, msg):
 118         try:
 119             if msg.header.frame_id != self.frame:
 120                 raise ValueError(f'Goal frame must be {self.frame}')
 121             q = self.quaternion(msg.pose)
 122             # An omitted (all-zero) quaternion means position only.
 123             heading = None if all(v == 0 for v in q) else yaw_xyzw(q)
 124             self.start_navigation(
 125                 GoalTask(msg.pose.position.x, msg.pose.position.y, heading, self.speed, self.turn_speed)
 126             )
 127         except (ValueError, OSError, RuntimeError) as error:
 128             self.route = self.goal = None
 129             self.get_logger().warning(str(error))
 130         self.publish_route()
 131         self.publish_accepted_goal()
 132 
 133     def on_odom(self, msg):
 134         try:
 135             if msg.header.frame_id != self.frame or msg.child_frame_id != 'device0/base_link':
 136                 raise ValueError('Odometry must describe device0/base_link in the configured odom frame')
 137             p = msg.pose.pose
 138             self.pose = (p.position.x, p.position.y, yaw_xyzw(self.quaternion(p)))
 139             v = msg.twist.twist.linear
 140             self.planar_speed = float(np.hypot(v.x, v.y))
 141             if not np.isfinite(self.planar_speed):
 142                 raise ValueError('Non-finite odometry speed')
 143             if not np.isfinite(self.pose).all():
 144                 raise ValueError('Non-finite odometry')
 145             self.odom_time = time.monotonic()
 146         except ValueError as error:
 147             self.odom_time = -float('inf')
 148             self.get_logger().warning(str(error), throttle_duration_sec=2.0)
 149 
 150     def tick(self):
 151         command, mode = np.zeros(5), 'Hold'
 152         preview = np.zeros(4)
 153         if (
 154             self.goal is not None
 155             and self.route is not None
 156             and self.pose is not None
 157             and time.monotonic() - self.odom_time <= self.timeout
 158         ):
 159             revision = self.route.revision
 160             command, mode = self.route.update(self.pose, self.planar_speed, self.get_clock().now().nanoseconds / 1e9)
 161             preview = self.route.preview if mode == 'Nav' else np.zeros(4)
 162             if revision != self.route.revision:
 163                 self.publish_route()
 164                 if self.route.error:
 165                     self.get_logger().warning('A* unavailable; Hold and retry: ' + self.route.error)
 166             x, y, yaw = self.pose
 167             distance = float(np.hypot(self.goal.x - x, self.goal.y - y))
 168             self.get_logger().info(
 169                 f'Goal tracking: mode={mode}, odom=({x:.3f},{y:.3f}), '
 170                 f'yaw={yaw:.3f}, distance={distance:.3f} m, '
 171                 f'planner_input={np.r_[command, preview].tolist()}',
 172                 throttle_duration_sec=2.0,
 173             )
 174         if self.route is not None and (self.pose is None or time.monotonic() - self.odom_time > self.timeout):
 175             self.route.arrived_since = None
 176         reason = (
 177             'waiting for odometry'
 178             if self.goal is not None and (self.pose is None or time.monotonic() - self.odom_time > self.timeout)
 179             else None
 180         )
 181         if reason != self.wait_reason:
 182             if reason:
 183                 self.get_logger().warning('Goal accepted but waiting for fresh odometry; output Hold')
 184             self.wait_reason = reason
 185         self.publish_route(remaining=True)
 186         self.input_pub.publish(Float32MultiArray(data=np.r_[command, preview].tolist()))
 187         self.status.publish(String(data=mode))
