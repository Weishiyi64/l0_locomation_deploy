  19 class TextMotionNode(Node):
  20     def __init__(self):
  21         super().__init__("text_motion_generator")
  22         self.declare_parameter("model_manifest", "")
  23         self.declare_parameter("mapping_file", "")
  24         self.declare_parameter("backend", "lite2")
  25         self.limits = load_limits(self.get_parameter("mapping_file").value)
  26         self.pipeline = open_pipeline(self.get_parameter("model_manifest").value,
  27                                       self.get_parameter("backend").value)
  28         self.lock = threading.Lock()
  29         self.wake = threading.Event()
  30         self.closing = False
  31         self.pose = None
  32         self.request = None
  33         self.cancelled = False
  34         self.sent = False
  35         self.deadline = 0.0
  36         self.timeout_reported = False
  37         self.recent = []
  38         self.cancel_requests = []
  39         self.status_pub = self.create_publisher(String, "/text_motion/status", 20)
  40         self.trajectory_pub = self.create_publisher(String, "/text_motion/trajectory", 1)
  41         self.cancel_pub = self.create_publisher(String, "/text_motion/cancel", 10)
  42         self.create_subscription(String, "/text_motion/request", self.on_request, 10)
  43         self.create_subscription(String, "/text_motion/playback_status", self.on_playback_status, 20)
  44         self.create_subscription(PolicyObservation, "/control/policy/observation", self.on_observation,
  45                                  QoSProfile(depth=1, reliability=ReliabilityPolicy.BEST_EFFORT))
  46         self.create_timer(0.1, self.check_timeout)
  47         self.worker = threading.Thread(target=self.work, name="text_motion_inference", daemon=False)
  48         self.worker.start()
  49         self.get_logger().info("Current ONNX delivery loaded; backend=" + self.pipeline.backend)

  51     def status(self, identifier, state, detail=""):
  52         if not rclpy.ok():
  53             return
  54         message = String()
  55         message.data = json.dumps({"request_id": identifier, "state": state, "detail": detail},
  56                                   ensure_ascii=False, allow_nan=False)
  57         self.status_pub.publish(message)
  58 
  59     def cancel(self, identifier):
  60         if not rclpy.ok():
  61             return
  62         message = String()
  63         message.data = identifier
  64         self.cancel_pub.publish(message)
  65 
  66     def on_request(self, message):
  67         identifier = ""
  68         try:
  69             if len(message.data.encode("utf-8")) > 16384:
  70                 raise ValueError("Request exceeds 16 KiB")
  71             request = json.loads(message.data)
  72             if not isinstance(request, dict) or not valid_id(request.get("request_id")):
  73                 raise ValueError("Invalid request_id")
  74             identifier = request["request_id"]
  75             if request.get("command") == "cancel":
  76                 with self.lock:
  77                     self.cancel_requests = (self.cancel_requests + [identifier])[-256:]
  78                     self.recent = (self.recent + [identifier])[-256:]
  79                     if self.request and self.request["request_id"] == identifier:
  80                         self.cancelled = True
  81                     self.cancel(identifier)
  82                 return
  83             with self.lock:
  84                 if identifier in self.recent:
  85                     self.status(identifier, "duplicate", "Request ID has already been handled")
  86                     return
  87                 validate_request(request)
  88                 if self.request is not None:
  89                     raise ValueError("Generator is busy; concurrent requests are refused")
  90                 if self.trajectory_pub.get_subscription_count() < 1 or \
  91                         self.count_publishers("/text_motion/playback_status") != 1:
  92                     raise ValueError("Exactly one enabled text-motion controller is required")
  93                 self.fresh_pose()
  94                 self.recent = (self.recent + [identifier])[-256:]
  95                 self.request = request
  96                 self.cancelled = False
  97                 self.sent = False
  98                 self.timeout_reported = False
  99                 self.status(identifier, "generating")
 100                 self.wake.set()
 101         except Exception as exc:
 102             self.status(identifier, "rejected", str(exc))
 103 
 104     def on_observation(self, message):
 105         pose = None
 106         joints = message.joints
 107         stamp = message.header.stamp.sec * 1_000_000_000 + message.header.stamp.nanosec
 108         age = (self.get_clock().now().nanoseconds - stamp) * 1e-9
 109         if message.inference_enabled and 0 <= age <= 0.2 and len(joints.name) == len(joints.position) and \
 110                 len(set(joints.name)) == len(joints.name):
 111             by_name = dict(zip(joints.name, joints.position))
 112             try:
 113                 pose = check_pose([by_name[name] for name in JOINTS], self.limits)
 114             except (KeyError, ValueError):
 115                 pass
 116         with self.lock:
 117             self.pose = None if pose is None else (pose, time.monotonic())
 118             if pose is None:
 119                 self.cancel_generation_if_needed("Observation gate closed during generation")
 120 
 121     def cancel_generation_if_needed(self, detail):
 122         # Caller holds lock. Keep cancellation sticky across stop/resume and
 123         # retain the slot until the uninterruptible NPU call has returned.
 124         if self.request and not self.sent and not self.cancelled:
 125             self.cancelled = True
 126             identifier = self.request["request_id"]
 127             self.cancel(identifier)
 128             self.status(identifier, "cancelling", detail)
 129 
 130     def fresh_pose(self):
 131         # Caller holds lock. Receive time adds a steady-clock freshness check.
 132         if self.pose is None or time.monotonic() - self.pose[1] > 0.2:
 133             raise ValueError("No fresh calibrated Walking observation")
 134         return self.pose[0].copy()
 135 
 136     def on_playback_status(self, message):
 137         try:
 138             event = json.loads(message.data)
 139             identifier, state = event["request_id"], event["state"]
 140             with self.lock:
 141                 matches = self.request and self.request["request_id"] == identifier
 142                 if matches and not self.sent:
 143                     # A cancel-before-delivery acknowledgement must not free
 144                     # the generator slot while the old NPU call still runs.
 145                     return
 146                 if not matches and identifier not in self.cancel_requests:
 147                     return
 148                 if matches and state in TERMINAL:
 149                     self.request = None
 150                     self.sent = False
 151                 if state in TERMINAL and identifier in self.cancel_requests:
 152                     self.cancel_requests.remove(identifier)
 153             self.status(identifier, state, event.get("detail", ""))
 154         except (ValueError, KeyError, TypeError):
 155             self.get_logger().error("Invalid controller text-motion feedback")
 156 
 157     def check_timeout(self):
 158         with self.lock:
 159             if self.pose is None or time.monotonic() - self.pose[1] > 0.2:
 160                 self.cancel_generation_if_needed("Observation stream expired during generation")
 161             if self.request and self.sent and time.monotonic() >= self.deadline and not self.timeout_reported:
 162                 self.timeout_reported = True
 163                 self.cancelled = True
 164                 identifier = self.request["request_id"]
 165                 self.cancel(identifier)
 166                 # Retain the slot until controller terminal feedback. Never start
 167                 # another motion when cancellation/execution state is unknown.
 168                 self.status(identifier, "cancelling", "Playback timeout; waiting for controller cancellation")
 169 
 170     def work(self):
 171         while True:
 172             self.wake.wait()
 173             self.wake.clear()
 174             with self.lock:
 175                 if self.closing:
 176                     return
 177                 request = self.request
 178             if request is None:
 179                 continue
 180             identifier = request["request_id"]
 181             try:
 182                 with self.lock:
 183                     if self.cancelled:
 184                         self.request = None
 185                         self.status(identifier, "cancelled", "Cancelled before inference")
 186                         continue
 187                     initial = self.fresh_pose()
 188                 trajectory = generate(self.pipeline, request, initial, self.limits)
 189                 with self.lock:
 190                     if self.closing or self.cancelled:
 191                         state, detail = "cancelled", "Generation cancelled; no trajectory sent"
 192                     elif trajectory is None:
 193                         state, detail = "no_motion", "All model body-part tokens are zero"
 194                     else:
 195                         latest = self.fresh_pose()
 196                         mask = np.asarray(trajectory["active_joints"], dtype=bool)
 197                         if np.max(np.abs(latest[mask] - initial[mask])) > 0.15:
 198                             raise ValueError("Robot pose moved during generation; generate again from its current pose")
 199                         if self.trajectory_pub.get_subscription_count() < 1 or \
 200                                 self.count_publishers("/text_motion/playback_status") != 1:
 201                             raise ValueError("Controller connection changed during generation")
 202                         trajectory["expires_at_ms"] = int(time.time() * 1000) + 2000
 203                         message = String()
 204                         message.data = json.dumps(trajectory, ensure_ascii=False, separators=(",", ":"), allow_nan=False)
 205                         if len(message.data.encode("utf-8")) > 2 * 1024 * 1024:
 206                             raise ValueError("Complete trajectory exceeds 2 MiB transport limit")
 207                         self.sent = True
 208                         self.deadline = time.monotonic() + (trajectory["frame_count"] - 1) / 50 + 10
 209                         self.trajectory_pub.publish(message)
 210                         self.status(identifier, "submitted", "Complete trajectory sent for controller validation")
 211                         continue
 212                     self.request = None
 213                     self.status(identifier, state, detail)
 214             except Exception as exc:
 215                 with self.lock:
 216                     if not self.request or self.request["request_id"] != identifier:
 217                         continue
 218                     if self.sent:
 219                         # A publishing exception does not prove DDS sent nothing.
 220                         # Cancel and retain the slot until execution is known.
 221                         self.cancelled = True
 222                         self.deadline = time.monotonic() + 2
 223                         self.cancel(identifier)
 224                         self.status(identifier, "cancelling", "Send state uncertain: " + str(exc))
 225                     else:
 226                         self.request = None
 227                         self.status(identifier, "failed", str(exc))
 228 
 229     def close(self):
 230         with self.lock:
 231             self.closing = True
 232             self.cancelled = True
 233             if self.request:
 234                 self.cancel(self.request["request_id"])
 235         self.wake.set()
 236         self.worker.join()
 237         self.pipeline.close()
