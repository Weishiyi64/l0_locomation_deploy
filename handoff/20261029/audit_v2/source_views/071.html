  22 def bundle_contract(path, base_action):
  23     path = Path(path).resolve()
  24     graph = json.loads(path.read_text(encoding="utf-8"))
  25     if graph.get("format") != "wbc_lora_graph_v1":
  26         raise ValueError("Expected the same shared WBC manifest used by LoRA")
  27     action = next(a for a in graph["actions"] if a["action_id"] == base_action)
  28     if action.get("parameter_source") != "base":
  29         raise ValueError("Text WBC requires a complete Base parameter pack, not an action adapter")
  30     config = (path.parent / action["config"]).resolve()
  31     if not config.is_relative_to(path.parent) or hashlib.sha256(config.read_bytes()).hexdigest() != action["config_sha256"]:
  32         raise ValueError("Invalid Base config path/hash")
  33     names = yaml.safe_load(config.read_text())["RobotCfg"]["joint_order_mj"]
  34     if len(names) != 21 or len(set(names)) != 21:
  35         raise ValueError("Base config must use the 21-joint WBC ABI")
  36     return graph, names

  39 def full_limits(mapping, names):
  40     motors = yaml.safe_load(Path(mapping).read_text())["motors"]
  41     result = []
  42     for name in names:
  43         matched = [m for m in motors if m["joint_name"] == name]
  44         if len(matched) != 1 or matched[0]["enabled"] is not True:
  45             raise ValueError("Missing/duplicate/disabled mapping joint: " + name)
  46         lo, hi = float(matched[0]["joint_lower"]), float(matched[0]["joint_upper"])
  47         if not np.isfinite([lo, hi]).all() or lo >= hi:
  48             raise ValueError("Invalid mapping limits")
  49         result.append((lo, hi))
  50     return np.asarray(result, dtype=np.float64)

  53 def check_snapshot(snapshot, names, limits):
  54     if snapshot["joint_names"] != names:
  55         raise ValueError("Snapshot joint order differs from WBC Base")
  56     joints = np.asarray(snapshot["joint_positions"], dtype=np.float64)
  57     position = np.asarray(snapshot["base_position"], dtype=np.float64)
  58     orientation = np.asarray(snapshot["base_quaternion_wxyz"], dtype=np.float64)
  59     if joints.shape != (21,) or position.shape != (3,) or orientation.shape != (4,) or \
  60             not all(np.isfinite(x).all() for x in (joints, position, orientation)):
  61         raise ValueError("Invalid full-body snapshot dimensions/values")
  62     if np.any(joints < limits[:, 0] - 1e-6) or np.any(joints > limits[:, 1] + 1e-6):
  63         raise ValueError("Captured joint pose exceeds robot limits")
  64     # Ordinary WBC accepts a finite observation position, including z=0 when
  65     # odometry is absent. Its reference model supplies the target root pose.
  66     norm_squared = float(orientation @ orientation)
  67     if norm_squared <= 1e-12 or abs(norm_squared - 1) > 0.1:
  68         raise ValueError("Invalid IMU quaternion under the standard WBC observation contract")
  69     # Same roll/pitch-only reference convention as prepare_json_batch. The
  70     # existing WBC yaw anchor restores the robot's actual heading at prepare.
  71     w, x, y, z = orientation / np.linalg.norm(orientation)
  72     roll = np.arctan2(2 * (w*x + y*z), 1 - 2 * (x*x + y*y))
  73     pitch = np.arcsin(np.clip(2 * (w*y - z*x), -1, 1))
  74     cr, sr, cp, sp = np.cos(roll/2), np.sin(roll/2), np.cos(pitch/2), np.sin(pitch/2)
  75     result = dict(snapshot)
  76     result["joint_positions"] = np.clip(joints, limits[:, 0], limits[:, 1]).tolist()
  77     result["base_position"] = position.tolist()
  78     result["base_quaternion_wxyz"] = [float(cr*cp), float(sr*cp), float(cr*sp), float(-sr*sp)]
  79     return result

  82 def make_document(trajectory, snapshot, graph):
  83     if trajectory is None:
  84         return None
  85     return {
  86         "request_id": trajectory["request_id"], "format": "gmr_upper_body_joint_trajectory_v1",
  87         "robot": "l0_v5", "angle_unit": "rad", "fps": 50,
  88         "coordinate_contract": "robot_anatomical_v1", "joint_names": trajectory["joint_names"],
  89         "frame_count": trajectory["frame_count"],
  90         "frames": [{"frame_index": i, "time": i / 50, "joint_positions": row}
  91                    for i, row in enumerate(trajectory["positions"])],
  92         "source_zip_sha256": SOURCE_SHA256,
  93         "wbc_base_checkpoint_sha256": graph["base_checkpoint_sha256"],
  94         "wbc_normalization_sha256": graph["normalization_sha256"],
  95         "wbc_start_state": snapshot,
  96         "root_reference_source": snapshot.get("root_reference_source", "snapshot"),
  97         "wbc_base_action": snapshot.get("wbc_base_action", ""),
  98         "model_diagnostics": dict(trajectory["model_diagnostics"], token_ids=trajectory["token_ids"],
  99                                   material_ids=trajectory["material_ids"], backend=trajectory["backend"],
 100                                   inference_ms=trajectory["inference_ms"]),
 101         "generation_request": trajectory["generation_request"],
 102         "safety_adjustments": trajectory["limit_adjustments"],
 103         "expires_at_ms": int(time.time() * 1000) + 5000,
 104     }

 108     def __init__(self, names, root_height, base_action="A-C04_mod"):
 109         import rclpy
 110         from rclpy.executors import SingleThreadedExecutor
 111         from rclpy.qos import QoSProfile, ReliabilityPolicy
 112         from rclpy.signals import SignalHandlerOptions
 113         from std_msgs.msg import String
 114         from luvbot_control_interfaces.msg import PolicyObservation
 115         self.rclpy, self.String = rclpy, String
 116         rclpy.init(args=[], signal_handler_options=SignalHandlerOptions.NO)
 117         self.node = rclpy.create_node("text_wbc_transaction_" + uuid.uuid4().hex[:8])
 118         self.names, self.root_height, self.base_action = names, root_height, base_action
 119         self.lock = threading.Lock()
 120         self.snapshot = None
 121         self.invalidated = False
 122         self.events = queue.Queue()
 123         self.json_pub = self.node.create_publisher(String, "/text_motion/json", 1)
 124         self.brain_pub = self.node.create_publisher(String, "/algorithm/brain/wbc_action_cmd", 10)
 125         self.cancel_pub = self.node.create_publisher(String, "/text_motion/cancel", 10)
 126         for topic, source in [("/text_motion/wbc_status", "worker"),
 127                               ("/system/state_machine/wbc_action_fdb", "controller")]:
 128             self.node.create_subscription(String, topic, lambda msg, src=source: self.on_status(src, msg), 20)
 129         self.node.create_subscription(PolicyObservation, "/control/policy/observation", self.on_observation,
 130                                      QoSProfile(depth=1, reliability=ReliabilityPolicy.BEST_EFFORT))
 131         self.executor = SingleThreadedExecutor()
 132         self.node.create_timer(0.05, self.check_observation_timeout)
 133         self.executor.add_node(self.node)
 134         self.thread = threading.Thread(target=self.executor.spin, daemon=False)
 135         self.thread.start()

 137     def on_status(self, source, message):
 138         try:
 139             self.events.put((source, json.loads(message.data)))
 140         except ValueError:
 141             pass

 143     def on_observation(self, message):
 144         stamp = message.header.stamp.sec * 1_000_000_000 + message.header.stamp.nanosec
 145         age = (self.node.get_clock().now().nanoseconds - stamp) * 1e-9
 146         valid = message.inference_enabled and 0 <= age <= 0.2 and \
 147             len(message.joints.name) == len(message.joints.position) and \
 148             len(set(message.joints.name)) == len(message.joints.name)
 149         snapshot = None
 150         if valid:
 151             try:
 152                 by_name = dict(zip(message.joints.name, message.joints.position))
 153                 position = message.base_pose.position
 154                 height = position.z
 155                 if height <= 0.05 and self.root_height is not None:
 156                     height = self.root_height
 157                 q = message.imu.orientation
 158                 snapshot = {"joint_names": self.names, "joint_positions": [by_name[n] for n in self.names],
 159                             "base_position": [position.x, position.y, height],
 160                             "base_quaternion_wxyz": [q.w, q.x, q.y, q.z],
 161                             "captured_stamp_ns": stamp, "root_height_override": self.root_height,
 162                             "root_reference_source": "wbc_base_action" if self.root_height is None else "snapshot",
 163                             "wbc_base_action": self.base_action}
 164             except KeyError:
 165                 pass
 166         with self.lock:
 167             if snapshot is None:
 168                 self.invalidated = True
 169             self.snapshot = None if snapshot is None else (snapshot, time.monotonic())

 171     def current_snapshot(self):
 172         with self.lock:
 173             if self.snapshot is None or time.monotonic() - self.snapshot[1] > 0.2:
 174                 raise RuntimeError("No fresh enabled policy observation")
 175             return self.snapshot[0]

 177     def check_observation_timeout(self):
 178         with self.lock:
 179             if self.snapshot is None or time.monotonic() - self.snapshot[1] > 0.2:
 180                 self.invalidated = True
 181 
 182     def reset_observation_gate(self):
 183         with self.lock:
 184             self.invalidated = False

 186     def check_observation_gate(self):
 187         with self.lock:
 188             if self.invalidated:
 189                 raise RuntimeError("Observation gate closed during generation; late result discarded")

 191     def wait_endpoints(self, timeout=10):
 192         deadline = time.monotonic() + timeout
 193         while time.monotonic() < deadline:
 194             if self.json_pub.get_subscription_count() > 0 and self.brain_pub.get_subscription_count() > 0 and \
 195                     self.node.count_publishers("/text_motion/wbc_status") == 1 and \
 196                     self.node.count_publishers("/system/state_machine/wbc_action_fdb") == 1:
 197                 try:
 198                     return self.current_snapshot()
 199                 except RuntimeError:
 200                     pass
 201             time.sleep(0.05)
 202         raise RuntimeError("Text JSON worker, controller feedback and live observation must all be available")

 204     def publish(self, publisher, payload):

 207         publisher.publish(message)

 209     def execute(self, document, timeout):
 210         identifier = document["request_id"]
 211         self.publish(self.json_pub, document)
 212         ready_deadline = time.monotonic() + 7
 213         deadline = time.monotonic() + timeout
 214         commanded = False
 215         timestamp = 0
 216         while time.monotonic() < deadline:
 217             try:
 218                 source, event = self.events.get(timeout=0.1)
 219             except queue.Empty:
 220                 if not commanded and time.monotonic() > ready_deadline:
 221                     raise RuntimeError("Complete JSON admission timed out")
 222                 continue
 223             if event.get("request_id", event.get("action_id")) != identifier:
 224                 continue
 225             print(json.dumps(dict(event, source=source), ensure_ascii=False), flush=True)
 226             state = event.get("state", event.get("status"))
 227             if source == "worker" and state == "ready" and not commanded:
 228                 self.check_observation_gate()
 229                 self.current_snapshot()
 230                 timestamp = int(time.time() * 1000)
 231                 self.publish(self.brain_pub, {"timestamp_ms": timestamp, "action_id": identifier})
 232                 commanded = True
 233             elif source == "controller" and event.get("timestamp_ms") == timestamp:
 234                 if state in ("completed", "cancelled"):
 235                     return state
 236                 if state in ("rejected", "failed"):
 237                     raise RuntimeError("Controller terminal state: " + state)
 238             elif source == "worker" and state == "duplicate" and commanded:
 239                 continue
 240             elif source == "worker" and state == "cancelled" and commanded:
 241                 continue  # Wait for the controller's actual WBC -> AMP terminal feedback.
 242             elif source == "worker" and state in ("rejected", "cancelled", "duplicate"):
 243                 raise RuntimeError("JSON admission failed: " + state + ": " + event.get("detail", ""))
 244         raise RuntimeError("WBC completion timed out")

 246     def cancel(self, identifier):
 247         self.publish(self.cancel_pub, identifier)
 248         from rclpy.duration import Duration
 249         self.cancel_pub.wait_for_all_acked(Duration(seconds=2))

 252         self.executor.shutdown()
 253         self.thread.join()

 258 def main():
 259     parser = argparse.ArgumentParser(description=__doc__)
 260     parser.add_argument("--model", type=Path, required=True)
 261     parser.add_argument("--bundle", type=Path, required=True)
 262     parser.add_argument("--base-action", default="A-C04_mod")
 263     parser.add_argument("--mapping", type=Path, required=True)
 264     parser.add_argument("--text", default="")
 265     parser.add_argument("--tokens", nargs=3, type=int)
 266     parser.add_argument("--seed", type=int)
 267     parser.add_argument("--speed", type=float, default=1.2)
 268     parser.add_argument("--return-frames", type=int, choices=(-1, 30, 60, 90), default=-1)
 269     parser.add_argument("--no-return", action="store_true")
 270     parser.add_argument("--limit-policy", choices=("reject", "clip"), default="reject")
 271     parser.add_argument("--backend", choices=("lite2", "onnx-reference"), default="lite2")
 272     parser.add_argument("--snapshot", type=Path, help="Offline generation only: explicit full-body snapshot JSON")
 273     parser.add_argument("--root-height", type=float, help="Optional explicit reference height; default reuses the WBC Base reference")
 274     parser.add_argument("--execute", action="store_true", help="After writing JSON, request the standard WBC controller lifecycle")
 275     parser.add_argument("--timeout", type=float, default=130)
 276     parser.add_argument("--output", type=Path, required=True)
 277     args = parser.parse_args()
 278     if args.output.exists():
 279         parser.error("output already exists; choose a new JSON path")
 280     if args.execute and (args.snapshot is not None or args.backend != "lite2"):
 281         parser.error("execution requires fresh ROS observations and board Lite2 inference")
 282     if not math.isfinite(args.timeout) or args.timeout <= 0:
 283         parser.error("timeout must be finite and positive")
 284     if args.root_height is not None and (not math.isfinite(args.root_height) or not 0.05 < args.root_height < 2):
 285         parser.error("root-height must be finite and between 0.05 and 2 metres")
 286     graph, names = bundle_contract(args.bundle, args.base_action)
 287     limits = full_limits(args.mapping, names)
 288     identifier = "T_" + uuid.uuid4().hex[:28]
 289     transaction = None
 290     submitted = False
 291     try:
 292         if args.snapshot:
 293             initial = check_snapshot(json.loads(args.snapshot.read_text()), names, limits)
 294         else:
 295             transaction = RosTransaction(names, args.root_height, args.base_action)
 296             initial = check_snapshot(transaction.wait_endpoints(), names, limits)
 297             transaction.reset_observation_gate()
 298         request = {"request_id": identifier, "mode": "web" if args.tokens is not None else "text",
 299                    "text": args.text, "token_ids": args.tokens, "seed": args.seed, "speed": args.speed,
 300                    "return_home": not args.no_return, "return_frames": args.return_frames,
 301                    "limit_policy": args.limit_policy}
 302         print(json.dumps({"request_id": identifier, "state": "generating", "backend": args.backend,
 303                           "python_executable": sys.executable, "python_version": sys.version.split()[0],
 304                           "rknn_runtime_library": os.environ.get("RKNN_RUNTIME_LIB")}), flush=True)
 305         with open_pipeline(args.model, args.backend) as pipeline:
 306             trajectory = generate(pipeline, request, initial["joint_positions"][12:], load_limits(args.mapping))
 307         if transaction is not None:
 308             transaction.check_observation_gate()
 309             latest = check_snapshot(transaction.current_snapshot(), names, limits)
 310             if np.max(np.abs(np.asarray(latest["joint_positions"]) - initial["joint_positions"])) > 0.15:
 311                 raise RuntimeError("Robot moved during generation; generate again from current state")
 312         document = make_document(trajectory, initial, graph)
 313         if document is None:
 314             print(json.dumps({"request_id": identifier, "state": "no_motion"}), flush=True)
 315             return
 316         encoded = json.dumps(document, ensure_ascii=False, separators=(",", ":"), allow_nan=False)
 317         if len(encoded.encode("utf-8")) > 4 * 1024 * 1024:
 318             raise ValueError("Complete JSON exceeds the WBC transport limit")
 319         args.output.parent.mkdir(parents=True, exist_ok=True)
 320         with tempfile.NamedTemporaryFile(mode="w", encoding="utf-8", dir=args.output.parent,
 321                                          prefix=".text_wbc_", delete=False) as stream:
 322             temporary = Path(stream.name)
 323             stream.write(encoded + "\n")
 324         try:
 325             os.link(temporary, args.output)  # Atomic exclusive publication; never overwrite another writer.
 326         finally:
 327             temporary.unlink(missing_ok=True)
 328         print(json.dumps({"request_id": identifier, "state": "generated", "file": str(args.output),
 329                           "frames": document["frame_count"], "base": graph["base_checkpoint_sha256"]}), flush=True)
 330         if args.execute:
 331             submitted = True
 332             result = transaction.execute(document, args.timeout)
 333             submitted = False
 334             print(json.dumps({"request_id": identifier, "state": result}), flush=True)
 335     finally:
 336         if transaction is not None:
 337             try:
 338                 if submitted:
 339                     transaction.cancel(identifier)
 340             finally:
 341                 transaction.close()
