   8 void LeggedController::processJoystickActions(const JoystickCommand& command) {

  23 void LeggedController::processJoystickBackEmergencySequence(const JoystickCommand& command) {
  24   if (!joystick_back_emergency_sequence_enabled_ || !command.back_emergency_sequence_pressed) return;
  25   if (!fsm_ || !fsm_->isWalking()) return;
  26 
  27   const std::string action_label = "Back emergency sequence";
  28   std::lock_guard<std::mutex> lock(auxiliary_action_trigger_mutex_);
  29   if (!canStartAmpDerivedAction("[BackEmergency]", action_label, true, true))
  30   {
  31     return;
  32   }
  33 
  34   if (joystick_back_emergency_sequence_path_.empty())
  35   {
  36     RCLCPP_ERROR(nh_->get_logger(), "[BackEmergency] Ignoring trigger: sequence path is empty");
  37     return;
  38   }
  39 
  40   if (!fullCsvVelocityAllowed())
  41   {
  42     logFullCsvVelocityBlocked("[BackEmergency]", joystick_back_emergency_sequence_path_);
  43     return;
  44   }
  45 
  46   const auto log_trigger = [this]() {
  47     RCLCPP_WARN(nh_->get_logger(), "[BackEmergency] Back button -> %s; EmergencyStop after finish",
  48                 displayControllerPath(joystick_back_emergency_sequence_path_).c_str());
  49   };
  50   if (triggerSequenceFromPath(joystick_back_emergency_sequence_path_, "[BackEmergency] Trigger", log_trigger))
  51   {
  52     sequence_finish_emergency_pending_ = true;
  53   }
  54 }

  67 void LeggedController::triggerEmergencyStop() {
  68   if (fsm_->isCalibration())
  69   {
  70     RCLCPP_WARN(nh_->get_logger(), "[EMERGENCY] Ignored while hardware calibration mode is active");
  71     return;
  72   }
  73 
  74   const bool had_active_sequence = sequenceRuntime_.isActive();
  75   sequence_finish_emergency_pending_ = false;
  76   sequenceRuntime_.resetForAbort();
  77   if (had_active_sequence)
  78   {
  79     RCLCPP_WARN(nh_->get_logger(), "[EMERGENCY] Active action sequence aborted");
  80   }
  81   publishImitationActive(false, "[EMERGENCY] Deactivate imitation worker");
  82   abortWbcImmediately("[EMERGENCY] Deactivate WBC worker");
  83   setPolicySource(walkStartPolicySource(), "[EMERGENCY] Policy source restored to default");
  84 
  85   if (!fsm_->isEmergency()) fsm_->emergencyStop();
  86 
  87   kp_standby_ = 0;
  88   printEmergencyStopBanner(nh_->get_logger());
  89   RCLCPP_WARN(nh_->get_logger(), "[EMERGENCY] System stopped for safety!");
  90 }

 102 void LeggedController::resetController() {
 103   if (!fsm_->isEmergency()) return;
 104 
 105   fsm_->resetFromEmergency();
 106   kp_standby_ = 0;
 107   orientation_violation_count_ = 0;
 108   sequenceRuntime_.resetForAbort();
 109   publishImitationActive(false, "[RESET] Deactivate imitation worker");
 110   abortWbcImmediately("[RESET] Deactivate WBC worker");
 111   policyMux_.clearCommands(0.0);
 112   RCLCPP_INFO(nh_->get_logger(), "[RESET] Standby targets cleared to zero pose");
 113   setPolicySource(walkStartPolicySource(), "[RESET] Policy source restored to startup default");
 114   publishRlObsReset();
 115 
 116   logResetBanner(nh_->get_logger());
 117 
 118   joyFeedbackArray_.array[0].intensity = 0.0;
 119   joyFeedbackArray_.array[1].intensity = 0.0;
 120   joyFeedbackPub_->publish(joyFeedbackArray_);
 121   RCLCPP_INFO(nh_->get_logger(), "[RESET] Controller reset - Ready for operation!");
 122 }

 124 void LeggedController::adjustStandbyKp(double delta) {
 125   if (fsm_->isStandby())
 126   {
 127     kp_standby_ += delta;
 128     RCLCPP_INFO(nh_->get_logger(), "[KP] Standby gain %s: %.1f", delta >= 0.0 ? "increased" : "decreased", kp_standby_);
 129     if (kp_standby_ > 80)
 130     {
 131       kp_standby_ = 80;
 132       RCLCPP_INFO(nh_->get_logger(), "[KP] Limit reached - max: 80");
 133     }
 134     else if (kp_standby_ < 0)
 135     {
 136       kp_standby_ = 0;
 137       RCLCPP_INFO(nh_->get_logger(), "[KP] Limit reached - min: 0");
 138     }
 139   }
 140 }

 142 void LeggedController::toggleCalibrationMode() {
 143   if (isSim_)
 144   {
 145     RCLCPP_WARN(nh_->get_logger(), "[CALIBRATION] Calibration mode is not supported in simulation");
 146     return;
 147   }
 148 
 149   if (fsm_->isEmergency())
 150   {
 151     if (!robot_hw_->set_calibration_mode(true))
 152     {

 156     fsm_->enterCalibrationMode();

 161   {
 162     if (!robot_hw_->set_calibration_mode(false))

 164       RCLCPP_WARN(nh_->get_logger(), "[CALIBRATION] Failed to disable hardware calibration mode");
 165       return;
 166     }
 167     fsm_->exitCalibrationMode();
 168     printCalibrationDisabledBanner(nh_->get_logger());
 169     RCLCPP_INFO(nh_->get_logger(), "[CALIBRATION] Calibration mode deactivated, back to Emergency Stop");
 170   }
 171   else
 172   {
 173     RCLCPP_WARN(nh_->get_logger(), "[CALIBRATION] Can only enter calibration mode from Emergency Stop state");
 174   }

 196 bool LeggedController::canStartAmpDerivedAction(const char* context, const std::string& action_label,
 197                                                 bool require_walking, bool block_imitation_active) const {
 198   const char* tag = context != nullptr ? context : "[ActionTrigger]";
 199   const std::string label = action_label.empty() ? "unnamed" : action_label;
 200 
 201   if (require_walking && (!fsm_ || !fsm_->isWalking()))
 202   {
 203     RCLCPP_WARN(nh_->get_logger(), "%s Ignoring action '%s': robot must be in Walking mode", tag, label.c_str());
 204     return false;
 205   }
 206   if (fsm_ && (fsm_->isEmergency() || fsm_->isCalibration()))
 207   {
 208     RCLCPP_WARN(nh_->get_logger(), "%s Ignoring action '%s': controller is not in a safe state", tag, label.c_str());
 209     return false;
 210   }
 211   if (sequenceRuntime_.isActive())
 212   {
 213     RCLCPP_WARN(nh_->get_logger(), "%s Ignoring action '%s': action sequence is already active", tag, label.c_str());
 214     return false;
 215   }
 216   if (block_imitation_active && imitationActionActive())
 217   {
 218     RCLCPP_WARN(nh_->get_logger(), "%s Ignoring action '%s': another policy action or WBC handoff is already active",
 219                 tag, label.c_str());
 220     return false;
 221   }
 222   return true;
 223 }

 196 bool LeggedController::canStartAmpDerivedAction(const char* context, const std::string& action_label,
 197                                                 bool require_walking, bool block_imitation_active) const {
 198   const char* tag = context != nullptr ? context : "[ActionTrigger]";
 199   const std::string label = action_label.empty() ? "unnamed" : action_label;
 200   if (textMotionCtrl_.busy())
 201   {
 202     RCLCPP_WARN(nh_->get_logger(), "%s Ignoring action '%s': text motion owns auxiliary joints", tag, label.c_str());
 203     return false;
 204   }
 205 
 206   if (require_walking && (!fsm_ || !fsm_->isWalking()))
 207   {
 208     RCLCPP_WARN(nh_->get_logger(), "%s Ignoring action '%s': robot must be in Walking mode", tag, label.c_str());
 209     return false;
 210   }
 211   if (fsm_ && (fsm_->isEmergency() || fsm_->isCalibration()))
 212   {
 213     RCLCPP_WARN(nh_->get_logger(), "%s Ignoring action '%s': controller is not in a safe state", tag, label.c_str());
 214     return false;
 215   }
 216   if (sequenceRuntime_.isActive())
 217   {
 218     RCLCPP_WARN(nh_->get_logger(), "%s Ignoring action '%s': action sequence is already active", tag, label.c_str());
 219     return false;
 220   }
 221   if (block_imitation_active && imitationActionActive())
 222   {
 223     RCLCPP_WARN(nh_->get_logger(), "%s Ignoring action '%s': another policy action or WBC handoff is already active",
 224                 tag, label.c_str());
 225     return false;
 226   }
 227   return true;
