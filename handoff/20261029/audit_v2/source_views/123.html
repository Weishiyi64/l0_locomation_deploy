  25 bool validId(const std::string& id) {
  26   return !id.empty() && id.size() <= 64 && std::all_of(id.begin(), id.end(), [](unsigned char c) {
  27     return (c >= 'a' && c <= 'z') || (c >= 'A' && c <= 'Z') || (c >= '0' && c <= '9') || c == '_' || c == '-';
  28   });
  29 }

  31 double smooth(double x) {
  32   const double u = std::max(0.0, std::min(1.0, x));
  33   return u * u * (3.0 - 2.0 * u);

  36 std::string jsonString(const std::string& value) {
  37   static const char hex[] = "0123456789abcdef";
  38   std::string out = "\"";
  39   for (const unsigned char c : value)
  40   {
  41     if (c == '\\' || c == '"')
  42     {
  43       out += '\\';
  44       out += static_cast<char>(c);
  45     }
  46     else if (c < 0x20)
  47     {
  48       out += "\\u00";
  49       out += hex[c >> 4];
  50       out += hex[c & 0xf];
  51     }
  52     else out += static_cast<char>(c);
  53   }
  54   return out + '"';
  55 }

  59 bool TextMotionController::configure(const std::vector<std::string>& joints, const std::string& mapping_file,
  60                                      std::string& error) {
  61   try
  62   {
  63     const auto motors = YAML::LoadFile(mapping_file)["motors"];
  64     if (!motors.IsSequence()) throw std::runtime_error("mapping_file has no motors sequence");
  65     for (std::size_t j = 0; j < kJoints.size(); ++j)
  66     {
  67       const auto found = std::find(joints.begin(), joints.end(), kJoints[j]);
  68       if (found == joints.end() || std::count(joints.begin(), joints.end(), kJoints[j]) != 1)
  69         throw std::runtime_error("missing/duplicate text-motion joint: " + kJoints[j]);
  70       indices_[j] = static_cast<std::size_t>(found - joints.begin());
  71       unsigned matches = 0;
  72       for (const auto& motor : motors)
  73       {
  74         if (motor["joint_name"].as<std::string>("") != kJoints[j]) continue;
  75         ++matches;
  76         lower_[j] = motor["joint_lower"].as<double>();
  77         upper_[j] = motor["joint_upper"].as<double>();
  78         if (!motor["enabled"].as<bool>(false) || !std::isfinite(lower_[j]) || !std::isfinite(upper_[j]) ||
  79             lower_[j] >= upper_[j]) throw std::runtime_error("invalid/disabled joint in text-motion mapping");
  80       }
  81       if (matches != 1) throw std::runtime_error("missing/duplicate joint limit in text-motion mapping");
  82     }
  83   }
  84   catch (const std::exception& exc)
  85   {
  86     error = exc.what();
  87     return false;
  88   }
  89   return true;
  90 }

 101 TextMotionController::Event TextMotionController::submitJson(const std::string& payload, bool allowed) {
 102   const auto gate_generation = gate_generation_.load();
 103   Event event{"", "rejected", ""};
 104   try
 105   {
 106     if (payload.empty() || payload.size() > kMaxBytes) throw std::runtime_error("trajectory exceeds payload limit");
 107     const YAML::Node root = YAML::Load(payload);
 108     event.request_id = root["request_id"].as<std::string>("");
 109     if (!validId(event.request_id))
 110     {
 111       event.request_id.clear();
 112       throw std::runtime_error("invalid request_id");
 113     }
 114     if (!allowed) throw std::runtime_error("controller must be in unobstructed Walking with a fresh AMP command");
 115     if (root["format"].as<std::string>("") != "l0_text_motion_v2" ||
 116         root["robot"].as<std::string>("") != "l0_v5" || root["angle_unit"].as<std::string>("") != "rad" ||
 117         root["fps"].as<double>(0.0) != 50.0 ||
 118         root["coordinate_contract"].as<std::string>("") != "robot_anatomical_v1" ||
 119         root["source_zip_sha256"].as<std::string>("") !=
 120             "48485ea7f14fd7e52920f7f59e4da20515b9f043e0aa388ee2a3e43130a84119")
 121       throw std::runtime_error("trajectory contract mismatch");
 122     auto incoming = std::make_shared<Trajectory>();
 123     incoming->id = event.request_id;
 124     incoming->expires_at_ms = root["expires_at_ms"].as<std::int64_t>();
 125     const auto names = root["joint_names"];
 126     const auto mask = root["active_joints"];
 127     if (!names.IsSequence() || names.size() != 9 || !mask.IsSequence() || mask.size() != 9)
 128       throw std::runtime_error("expected nine joint names and explicit mask");
 129     for (std::size_t j = 0; j < 9; ++j)
 130     {
 131       if (names[j].as<std::string>() != kJoints[j]) throw std::runtime_error("incorrect joint order");
 132       const auto flag = mask[j].as<std::string>();
 133       if (flag != "true" && flag != "false") throw std::runtime_error("active_joints must be booleans");
 134       incoming->mask[j] = flag == "true";
 135     }
 136     for (std::size_t j = 0; j < 9; j += 3)
 137       if (incoming->mask[j] != incoming->mask[j + 1] || incoming->mask[j] != incoming->mask[j + 2])
 138         throw std::runtime_error("ownership must cover complete body parts");
 139     if (std::none_of(incoming->mask.begin(), incoming->mask.end(), [](bool value) { return value; }))
 140       throw std::runtime_error("no active body part");
 141     const auto frames = root["positions"];
 142     const auto count = root["frame_count"].as<std::size_t>(0);
 143     if (!frames.IsSequence() || count != frames.size() || count < 2 || count > kMaxFrames)
 144       throw std::runtime_error("invalid frame count");
 145     incoming->frames.reserve(count);
 146     for (const auto& frame : frames)
 147     {
 148       if (!frame.IsSequence() || frame.size() != 9) throw std::runtime_error("invalid frame dimensions");
 149       Pose row;
 150       for (std::size_t j = 0; j < 9; ++j)
 151       {
 152         row[j] = frame[j].as<double>();
 153         if (!std::isfinite(row[j]) || row[j] < lower_[j] - 1e-6 || row[j] > upper_[j] + 1e-6)
 154           throw std::runtime_error("nonfinite/out-of-limit joint: " + kJoints[j]);
 155         row[j] = std::max(lower_[j], std::min(upper_[j], row[j]));
 156       }
 157       incoming->frames.push_back(row);
 158     }
 159     const auto age = incoming->expires_at_ms - unixTimeMs();
 160     if (age <= 0 || age > 5000) throw std::runtime_error("expired or excessive trajectory TTL");
 161     std::lock_guard<std::mutex> lock(mutex_);
 162     if (!allowed_.load() || gate_generation != gate_generation_.load())
 163       throw std::runtime_error("controller gate changed during trajectory validation");
 164     if (seen(incoming->id))
 165       return {incoming->id, "duplicate", "request already received or cancelled"};
 166     if (busy_.load()) throw std::runtime_error("another text motion is preparing/playing/releasing");
 167     remember(incoming->id);
 168     slot_id_ = incoming->id;
 169     pending_ = std::move(incoming);
 170     cancelled_ = false;
 171     busy_.store(true);
 172     event.state = "ready";
 173     event.detail = "complete trajectory validated";
 174   }
 175   catch (const std::exception& exc)
 176   {
 177     event.detail = exc.what();
 178   }
 179   return event;
 180 }

 182 TextMotionController::Event TextMotionController::cancel(const std::string& request_id) {
 183   if (!validId(request_id)) return {"", "rejected", "invalid cancellation request_id"};
 184   std::lock_guard<std::mutex> lock(mutex_);
 185   // Remember cancellation even if inference/validation has not delivered a motion yet.
 186   remember(request_id);
 187   if (slot_id_ == request_id)
 188   {
 189     cancelled_ = true;
 190     return {request_id, "cancelling", "cancellation queued for control loop"};
 191   }
 192   return {request_id, "cancelled", "cancelled before trajectory admission"};
 193 }

 211 TextMotionController::Event TextMotionController::step(const std::vector<double>& base,
 212     const std::vector<JointMotorCommand>& previous, bool allowed) {
 213   if (!busy_.load()) return {};
 214   std::unique_lock<std::mutex> lock(mutex_, std::try_to_lock);
 215   if (!lock.owns_lock()) return {};  // Never wait for a ROS callback on the control loop.
 216   const auto now = std::chrono::steady_clock::now();
 217   for (const auto index : indices_)
 218     if (index >= base.size() || !std::isfinite(base[index])) allowed = false;
 219   if (!allowed) return finish("stopped", "Walking/AMP command/safety gate no longer permits playback");
 220   Event event;
 221   if (pending_)
 222   {
 223     if (cancelled_) return finish("cancelled", "cancelled before playback");
 224     if (pending_->expires_at_ms <= unixTimeMs()) return finish("rejected", "trajectory expired before playback");
 225     current_ = std::move(pending_);
 226     for (std::size_t j = 0; j < 9; ++j)
 227     {
 228       const auto index = indices_[j];
 229       // Blend from the last composed command, including head/arm overlays.
 230       entry_[j] = index < previous.size() ? previous[index].q_des : base[index];
 231       if (!std::isfinite(entry_[j]) || entry_[j] < lower_[j] - 1e-6 || entry_[j] > upper_[j] + 1e-6)
 232         return finish("rejected", "previous joint command is invalid");
 233       q_[j] = entry_[j];
 234     }
 235     started_ = now;
 236     releasing_ = false;
 237     release_alpha_ = 0.0;
 238     active_.store(true);
 239     event = {slot_id_, "started", "upper-body playback; AMP continues controlling legs"};
 240   }
 241   if (!current_) return {};
 242   const double elapsed = std::chrono::duration<double>(now - started_).count();
 243   const double duration = static_cast<double>(current_->frames.size() - 1) / 50.0;
 244   if (!releasing_ && (cancelled_ || elapsed >= kEntrySec + duration))
 245   {
 246     releasing_ = true;
 247     cancelled_release_ = cancelled_;
 248     releasing_at_ = now;
 249     release_ = cancelled_ ? q_ : current_->frames.back();
 250   }
 251   if (releasing_)
 252   {
 253     cancelled_release_ = cancelled_release_ || cancelled_;
 254     const double u = std::chrono::duration<double>(now - releasing_at_).count() / kExitSec;
 255     release_alpha_ = smooth(u);
 256     if (u >= 1.0) return finish(cancelled_release_ ? "cancelled" : "completed", "released to current background command");
 257     return event;
 258   }
 259   if (elapsed < kEntrySec)
 260   {
 261     const double alpha = smooth(elapsed / kEntrySec);
 262     for (std::size_t j = 0; j < 9; ++j)
 263       q_[j] = entry_[j] + alpha * (current_->frames.front()[j] - entry_[j]);
 264     return event;
 265   }
 266   const double frame = std::min((elapsed - kEntrySec) * 50.0, static_cast<double>(current_->frames.size() - 1));
 267   const auto first = static_cast<std::size_t>(std::floor(frame));
 268   const auto second = std::min(first + 1, current_->frames.size() - 1);
 269   const double alpha = frame - first;
 270   for (std::size_t j = 0; j < 9; ++j)
 271     q_[j] = current_->frames[first][j] + alpha * (current_->frames[second][j] - current_->frames[first][j]);
 272   return event;
 273 }

 275 bool TextMotionController::getJointPosition(std::size_t index, double& q) const {
 276   // Called only from the same control thread as step().
 277   if (!active_.load() || !current_) return false;
 278   for (std::size_t j = 0; j < 9; ++j)
 279     if (indices_[j] == index && current_->mask[j])
 280     {
 281       // Composer supplies the current background command (AMP plus any allowed
 282       // existing overlay). This also makes the last release target continuous.
 283       q = releasing_ ? release_[j] + release_alpha_ * (q - release_[j]) : q_[j];
 284       return true;
 285     }
 286   return false;
 287 }
