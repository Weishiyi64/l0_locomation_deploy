  55 def convert_manifest(args):
  56     import importlib.metadata
  57     manifest_path = args.manifest.resolve()
  58     root = manifest_path.parent
  59     if root != args.root.resolve():
  60         raise ValueError("--manifest must be inside the selected root directory")
  61     graph = load_graph_manifest(manifest_path)
  62     check_graph(asset_path(root, graph["model"]["onnx"]), manifest_path)
  63     commands = [(asset_path(root, graph["model"]["onnx"]), "convert_onnx_to_rknn_rk3588_wbc_policy.py",
  64                  asset_path(root, graph["model"]["rknn"]), manifest_path)]
  65     if graph["kind"] == "policy":
  66         if not graph.get("actions") or len({a["action_id"] for a in graph["actions"]}) != len(graph["actions"]):
  67             raise ValueError("LoRA manifest requires unique non-empty actions")
  68         for action in graph["actions"]:
  69             load_adapter(asset_path(root, action["adapter_manifest"]), graph, action["action_id"])
  70             for key, digest in (("config", "config_sha256"), ("reference_onnx", "reference_onnx_sha256")):
  71                 if sha256(asset_path(root, action[key])) != action[digest]:
  72                     raise ValueError(f"Action source checksum mismatch: {action['action_id']}")
  73             if not args.policy_only:
  74                 commands.append((asset_path(root, action["reference_onnx"]),
  75                                  "convert_onnx_to_rknn_rk3588_wbc_reference.py",
  76                                  asset_path(root, action["reference_rknn"]), None))
  77     if args.include_probes:
  78         for relative in graph.get("probes", []):
  79             probe_path = asset_path(root, relative)
  80             probe = load_graph_manifest(probe_path)
  81             if probe["kind"] != "probe" or any(probe[k] != graph[k] for k in
  82                     ("base_checkpoint_sha256", "normalization_sha256", "parameter_schema_sha256", "actions")):
  83                 raise ValueError(f"Probe is not from this policy bundle: {probe_path}")
  84             check_graph(asset_path(root, probe["model"]["onnx"]), probe_path)
  85             commands.append((asset_path(root, probe["model"]["onnx"]),
  86                              "convert_onnx_to_rknn_rk3588_wbc_policy.py",
  87                              asset_path(root, probe["model"]["rknn"]), probe_path))
  88     if len({str(item[2]) for item in commands}) != len(commands):
  89         raise ValueError("Duplicate RKNN output destinations in manifest")
  90     for source, converter, _, _ in commands:
  91         if not source.is_file() or not Path(__file__).with_name(converter).is_file():
  92             raise ValueError(f"Missing ONNX or converter: {source} / {converter}")
  93     log_dir = args.log_dir.resolve() if args.log_dir else None
  94     if log_dir and not args.dry_run:
  95         log_dir.mkdir(parents=True, exist_ok=False)
  96     summary = {"manifest": str(manifest_path), "graph_manifest_sha256": sha256(manifest_path),
  97                "rknn_config": rknn_build_config(),
  98                "target_platform": "rk3588", "build_only": args.build_only,
  99                "formal_runtime": "not_run", "planned": len(commands), "models": []}
 100     failures = 0
 101     for index, (source, converter, output, graph_path) in enumerate(commands, 1):
 102         skip = False
 103         provenance = Path(str(output) + ".json")
 104         if args.skip_existing and output.is_file() and provenance.is_file():
 105             try:
 106                 record = read_json(provenance)
 107             except (ValueError, OSError):
 108                 record = {}
 109             try:
 110                 toolkit = importlib.metadata.version("rknn-toolkit2")
 111             except importlib.metadata.PackageNotFoundError:
 112                 toolkit = "unknown"
 113             skip = (record.get("format") == "wbc_rknn_conversion_v1" and record.get("do_quantization") is False and
 114                     record.get("rknn_config") == rknn_build_config() and
 115                     record.get("source_sha256") == sha256(source) and
 116                     record.get("rknn_sha256") == sha256(output) and record.get("precision") == "float16" and
 117                     record.get("toolkit_version") == toolkit and toolkit != "unknown" and
 118                     record.get("target_platform") == "rk3588" and
 119                     record.get("converter_sha256") == sha256(Path(__file__).with_name(converter)) and
 120                     record.get("asset_helper_sha256") == sha256(Path(__file__).with_name("wbc_lora_assets.py")) and
 121                     record.get("target") == args.target and record.get("device_id") == args.device_id)
 122             if not args.build_only:
 123                 skip = skip and record.get("verification_backend") in ("target", "toolkit_simulator")
 124             if graph_path:
 125                 skip = skip and record.get("graph_manifest_sha256") == sha256(graph_path)
 126                 expected = [(a["action_id"], sha256(asset_path(root, a["adapter_manifest"]))) for a in graph["actions"]]
 127                 verified = [(a["action_id"], a["adapter_manifest_sha256"]) for a in (record.get("verification") or {}).get("actions", [])]
 128                 if not args.build_only:
 129                     skip = skip and expected == verified
 130         if skip:
 131             print(f"[SKIP matching conversion; validation={record.get('verification_backend')}] {output}", flush=True)
 132             summary["models"].append({"output": str(output), "status": "skipped_matching_record"})
 133             continue
 134         command = [sys.executable, "-B", str(Path(__file__).resolve().parent / converter), str(source), "-o", str(output)]
 135         if args.build_only:
 136             command += ["--build-only"]
 137         if graph_path:
 138             command += ["--manifest", str(graph_path)]
 139         if args.target:
 140             command += ["--target", args.target]
 141         if args.device_id:
 142             command += ["--device-id", args.device_id]
 143         print(f"[{'DRY-RUN' if args.dry_run else 'RUN'}] {shlex.join(command)}", flush=True)
 144         if args.dry_run:
 145             continue
 146         entry = {"source": str(source), "output": str(output), "command": command}
 147         try:
 148             if log_dir:
 149                 log_path = log_dir / f"{index:02d}_{output.stem}.log"
 150                 print(f"[LOG] {log_path}", flush=True)
 151                 with log_path.open("x") as stream:
 152                     result = subprocess.run(command, stdout=stream, stderr=subprocess.STDOUT, check=False)
 153                 entry["log"] = str(log_path)
 154                 log_path.with_suffix(".exitcode").write_text(str(result.returncode) + "\n")
 155             else:
 156                 result = subprocess.run(command, check=False)
 157             entry.update(exit_code=result.returncode, status="converted" if result.returncode == 0 else "failed")
 158         except OSError as error:
 159             entry.update(exit_code=1, status="failed", error=str(error))
 160         summary["models"].append(entry)
 161         if entry["exit_code"]:
 162             failures += 1
 163             print(f"[FAIL] {source}; remaining models not attempted", file=sys.stderr, flush=True)
 164             break
 165     summary["failed"] = failures
 166     summary["not_attempted"] = 0 if args.dry_run else len(commands) - len(summary["models"])
 167     if log_dir and not args.dry_run:
 168         (log_dir / "conversion_summary.json").write_text(json.dumps(summary, indent=2) + "\n")
 169     print(f"LoRA manifest conversions={len(commands)}, failed={failures}, dry_run={args.dry_run}")
 170     return 1 if failures else 0

 173 def main():
 174     args = parse_args()
 175     if args.device_id and not args.target:
 176         raise ValueError("--device-id requires --target rk3588")
 177     if args.manifest:
 178         return convert_manifest(args)
 179     root = args.root.resolve()
 180     script_dir = Path(__file__).resolve().parent
 181     scripts = {name: script_dir / filename for name, filename in CONVERTERS}
 182     for name, script in scripts.items():
 183         if not script.is_file():
 184             print(f"ERROR: converter for {name} not found: {script}", file=sys.stderr)
 185             return 2
 186 
 187     found = succeeded = failed = skipped = 0
 188     for directory, subdirs, filenames in os.walk(root):
 189         subdirs.sort()
 190         current_dir = Path(directory)
 191         if current_dir == root:  # Only process subfolders (including nested ones).
 192             continue
 193         present = set(filenames)
 194         for onnx_name, _ in CONVERTERS:
 195             if onnx_name not in present:
 196                 continue
 197             onnx_path = current_dir / onnx_name
 198             if not onnx_path.is_file():
 199                 continue
 200             found += 1
 201             output_path = onnx_path.with_suffix(".rknn")
 202             if args.skip_existing and output_path.is_file():
 203                 skipped += 1
 204                 print(f"[SKIP] {output_path} already exists", flush=True)
 205                 continue
 206 
 207             command = [sys.executable, str(scripts[onnx_name]), str(onnx_path),
 208                        "-o", str(output_path)]
 209             if args.build_only:
 210                 command.append("--build-only")
 211             if args.target:
 212                 command.extend(["--target", args.target])
 213             if args.device_id:
 214                 command.extend(["--device-id", args.device_id])
 215             print(f"\n[{'DRY-RUN' if args.dry_run else 'RUN'}] {shlex.join(command)}", flush=True)
 216             if args.dry_run:
 217                 continue
 218 
 219             try:
 220                 result = subprocess.run(command, check=False)
 221             except OSError as exc:
 222                 print(f"[FAIL] {onnx_path}: {exc}", file=sys.stderr, flush=True)
 223                 failed += 1
 224                 continue
 225             if result.returncode == 0:
 226                 succeeded += 1
 227                 print(f"[OK] {output_path}", flush=True)
 228             else:
 229                 failed += 1
 230                 print(f"[FAIL] {onnx_path} (exit code {result.returncode})",
 231                       file=sys.stderr, flush=True)
 232 
 233     print(f"\nSummary: found={found}, succeeded={succeeded}, "
 234           f"failed={failed}, skipped={skipped}, dry_run={args.dry_run}")
 235     if not found:
 236         print("No policy.onnx or reference_motion.onnx found in subfolders.", file=sys.stderr)
 237         return 1
 238     return 1 if failed else 0

 243         sys.exit(main())
