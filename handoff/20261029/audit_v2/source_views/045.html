   9 def launch_setup(context, *args, **kwargs):
  10     # All launch args are supplied by sequential_launch_real.sh (from sim2real.launch.py); no in-file defaults.
  11     cfg = context.launch_configurations
  12     robot_type = cfg['robot_type']
  13     subpolicy = cfg['subpolicy']
  14 
  15     legged_controllers_share = get_package_share_directory('legged_controllers')
  16     command_arbiter_share = get_package_share_directory('system_command_arbiter')
  17     imu_enabled = cfg['imu_enabled'] == 'true'
  18     hpm_mode = cfg.get('hpm_mode', 'real')
  19     if hpm_mode not in ('real', 'isolated'):
  20         raise RuntimeError('hpm_mode must be real or isolated')
  21     if robot_type not in ('l0_v3', 'l0_v4', 'l0_v5'):
  22         raise RuntimeError(f'Unsupported HPM robot_type: {robot_type}')
  23     mapping_file = os.path.abspath(os.path.join(command_arbiter_share, 'config', f'{robot_type}.yaml'))
  24     head_script_enabled = cfg['head_script_enabled'] == 'true'
  25     head_script_config = cfg['head_script_config']
  26     joystick_imitation_enabled = cfg.get('joystick_imitation_enabled', 'false') == 'true'
  27     joystick_imitation_config_path = cfg.get('joystick_imitation_config_path', '')
  28     joystick_sequence_enabled = cfg.get('joystick_sequence_enabled', 'false') == 'true'
  29     joystick_sequence_config_path = cfg.get('joystick_sequence_config_path', '')
  30     radar_control_enabled = cfg.get('radar_control_enabled', 'false') == 'true'
  31     emergency_damping_kd = float(cfg.get('emergency_damping_kd', '5.0'))
  32     joystick_back_emergency_sequence_enabled = cfg.get('joystick_back_emergency_sequence_enabled', 'true') == 'true'
  33     joystick_back_emergency_sequence_path = cfg.get(
  34         'joystick_back_emergency_sequence_path',
  35         'action_sequences/walking_back_emergency.yaml',
  36     )
  37     arm_swing_enabled = cfg.get('arm_swing_enabled', 'true') == 'true'
  38     safety_orientation_roll_limit_deg = float(cfg['safety_orientation_roll_limit_deg'])
  39     safety_orientation_pitch_limit_deg = float(cfg['safety_orientation_pitch_limit_deg'])
  40     safety_orientation_consecutive_violation_count = int(cfg['safety_orientation_consecutive_violation_count'])
  41 
  42     legged_deploy_share = get_package_share_directory('legged_deploy')
  43     robot_cfg_path = os.path.abspath(
  44         os.path.join(legged_deploy_share, 'policy', robot_type, subpolicy, 'rl_config.yaml')
  45     )
  46 
  47     joy_cfg_path = os.path.abspath(
  48         os.path.join(legged_deploy_share, 'policy', robot_type, subpolicy, 'joy_config.yaml')
  49     )
  50 
  51     legged_controller_config_path = os.path.abspath(
  52         os.path.join(legged_controllers_share, 'config', 'legged_controller_config.yaml')
  53     )
  54 
  55     rviz = cfg['rviz'] == 'true'
  56     legged_real_params = {
  57         'mapping_file': mapping_file,
  58         'robot_type': robot_type,
  59         'robot_cfg_path': robot_cfg_path,
  60         'legged_controller_config_path': legged_controller_config_path,
  61         'joy_cfg_path': joy_cfg_path,
  62         'rviz': rviz,
  63         'isSim': False,
  64         '/RobotHw/imu_enabled': imu_enabled,
  65         'head_script_enabled': head_script_enabled,
  66         'head_script_config': head_script_config,
  67         'joystick_imitation_enabled': joystick_imitation_enabled,
  68         'joystick_imitation_config_path': joystick_imitation_config_path,
  69         'joystick_sequence_enabled': joystick_sequence_enabled,
  70         'joystick_sequence_config_path': joystick_sequence_config_path,
  71         'radar_control_enabled': radar_control_enabled,
  72         'face_follow_enabled': True,
  73         'wbc_enabled': cfg['wbc_enabled'] == 'true',
  74         'emergency.damping_kd': emergency_damping_kd,
  75         'joystick_back_emergency_sequence_enabled': joystick_back_emergency_sequence_enabled,
  76         'joystick_back_emergency_sequence_path': joystick_back_emergency_sequence_path,
  77         'arm_swing.enabled': arm_swing_enabled,
  78         'arm_swing.stop_vx_threshold': float(cfg.get('arm_swing_stop_vx_threshold', '0.05')),
  79         'arm_swing.high_vx_threshold': float(cfg.get('arm_swing_high_vx_threshold', '0.8')),
  80         'arm_swing.vx_hysteresis': float(cfg.get('arm_swing_vx_hysteresis', '0.03')),
  81         'arm_swing.blend_sec': float(cfg.get('arm_swing_blend_sec', '0.35')),
  82         'safety.orientation.roll_limit_deg': safety_orientation_roll_limit_deg,
  83         'safety.orientation.pitch_limit_deg': safety_orientation_pitch_limit_deg,
  84         'safety.orientation.consecutive_violation_count': safety_orientation_consecutive_violation_count,
  85     }
  86     legged_real_node = Node(
  87         package='legged_real',
  88         executable='topic_hardware_adapter',
  89         name='topic_hardware_adapter',
  90         output='screen',
  91         parameters=[legged_real_params],
  92         remappings=[
  93             ('emergency_stop', '/system/safety/emergency_stop'),
  94             ('hardware_fault', '/system/safety/hardware_fault'),
  95             ('motors_ready', '/hardware_status/motors_ready'),
  96         ] + ([
  97             ('actuator/motor/state', '/wbc_isolated/actuator/motor/state'),
  98             ('actuator/motor/command', '/wbc_isolated/actuator/motor/command'),
  99             ('actuator/motor/enable', '/wbc_isolated/actuator/motor/enable'),
 100             ('actuator/motor/set_zero', '/wbc_isolated/actuator/motor/set_zero'),
 101             ('imu/data', '/wbc_isolated/imu/data'),
 102             ('odom', '/wbc_isolated/odom'),
 103         ] if hpm_mode == 'isolated' else []),
 104     )
 105 
 106     return [legged_real_node]
