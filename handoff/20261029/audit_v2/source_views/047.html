   1 #!/usr/bin/env python3
   2 """Stationary HPM-protocol endpoint for the unchanged real deployment control loop.
   3 
   4 Only /wbc_isolated endpoints are created. No motor transport is imported or opened.
   5 Motor readiness, watchdogs and command composition remain in LeggedReal.
   6 """
   7 import math
   8 
   9 import rclpy
  10 from rclpy.node import Node
  11 from rclpy.qos import QoSProfile, ReliabilityPolicy, DurabilityPolicy, qos_profile_sensor_data
  12 from robot_interfaces.msg import MotorCommand, MotorEnableCommand, MotorState, MotorStateArray
  13 from robot_interfaces.srv import MotorSetZero
  14 from sensor_msgs.msg import Imu
  15 import yaml
  16 
  17 
  18 class IsolatedHpm(Node):
  19     def __init__(self):
  20         super().__init__('isolated_hpm')
  21         self.declare_parameter('mapping_file', '')
  22         self.declare_parameter('robot_cfg_path', '')
  23         self.declare_parameter('feedback_enabled', True)
  24         self.declare_parameter('imu_feedback_enabled', True)
  25         with open(self.get_parameter('mapping_file').value, encoding='utf-8') as stream:
  26             mapping = yaml.safe_load(stream)
  27         with open(self.get_parameter('robot_cfg_path').value, encoding='utf-8') as stream:
  28             cfg = yaml.safe_load(stream)['RobotCfg']
  29         joints, positions = cfg['joint_order_mj'], cfg['default_joint_angles']
  30         if len(joints) != 21 or len(positions) != 21 or len(mapping['motors']) != 21:
  31             raise ValueError('Isolated HPM requires the same 21-joint hardware configuration')
  32         self.state = MotorStateArray()
  33         self.enabled = [False] * 21
  34         for index, (name, position, entry) in enumerate(zip(joints, positions, mapping['motors'])):
  35             model = mapping['models'][entry['model']]
  36             direction = float(entry['motor_direction']) * float(model['rotation_direction'])
  37             reduction = float(entry['reduction'])
  38             if name != entry['joint_name'] or not math.isfinite(position) or abs(direction) != 1 or reduction <= 0:
  39                 raise ValueError('Joint/mapping mismatch in isolated HPM configuration')
  40             motor_q = (position - float(entry['zero_offset'])) * direction * reduction
  41             if not math.isfinite(motor_q) or not model['p_min'] <= motor_q <= model['p_max'] or \
  42                     not entry['joint_lower'] <= position <= entry['joint_upper']:
  43                 raise ValueError(f'Isolated default pose is outside mapping limits: {name}')
  44             state = MotorState()
  45             state.route.motor_type = index
  46             state.q = motor_q
  47             state.valid = True
  48             state.operating_mode = MotorState.OPERATING_MODE_UNKNOWN
  49             self.state.states.append(state)
  50             self.state.feedback_mask |= 1 << index
  51         reliable = QoSProfile(depth=10, reliability=ReliabilityPolicy.RELIABLE,
  52                               durability=DurabilityPolicy.VOLATILE)
  53         self.state_pub = self.create_publisher(MotorStateArray, '/wbc_isolated/actuator/motor/state', qos_profile_sensor_data)
  54         self.imu_pub = self.create_publisher(Imu, '/wbc_isolated/imu/data', qos_profile_sensor_data)
  55         self.enable_sub = self.create_subscription(MotorEnableCommand, '/wbc_isolated/actuator/motor/enable',
  56                                                    self.enable, reliable)
  57         self.command_sub = self.create_subscription(MotorCommand, '/wbc_isolated/actuator/motor/command',
  58                                                     self.command, reliable)
  59         self.zero_service = self.create_service(MotorSetZero, '/wbc_isolated/actuator/motor/set_zero', self.reject_zero)
  60         self.command_count = 0
  61         self.last_summary_ns = 0
  62         self.timer = self.create_timer(0.002, self.publish_feedback)
  63         self.get_logger().warning('HPM ISOLATED: stationary feedback; commands are consumed without motor transport')
  64 
  65     def enable(self, msg):
  66         if len(msg.motor_types) != len(msg.enabled) or any(i >= 21 for i in msg.motor_types):
  67             self.get_logger().error('Invalid isolated enable request')
  68             return
  69         for index, enabled in zip(msg.motor_types, msg.enabled):
  70             self.enabled[index] = enabled
  71         self.get_logger().info(f'Isolated enable sequence={msg.sequence_id}; enabled joints={sum(self.enabled)}')
  72 
  73     def command(self, msg):
  74         for target in msg.commands:
  75             if target.route.motor_type >= 21 or not all(math.isfinite(x) for x in
  76                                                        (target.q, target.dq, target.tau, target.kp, target.kd)):
  77                 self.get_logger().error('Invalid isolated motor command')
  78                 return
  79         self.command_count += 1
  80         now = self.get_clock().now().nanoseconds
  81         if now - self.last_summary_ns >= 5_000_000_000:
  82             self.last_summary_ns = now
  83             self.get_logger().info(f'Isolated sink received={self.command_count}, sequence={msg.sequence_id}, '
  84                                    f'joints={len(msg.commands)}; stationary feedback')
  85 
  86     @staticmethod
  87     def reject_zero(request, response):
  88         response.motor_types = request.motor_types
  89         response.results = [MotorSetZero.Response.REJECTED] * len(request.motor_types)
  90         response.messages = ['Calibration is unavailable in isolated mode'] * len(request.motor_types)
  91         return response
  92 
  93     def publish_feedback(self):
  94         stamp = self.get_clock().now().to_msg()
  95         if self.get_parameter('feedback_enabled').value:
  96             self.state.stamp = stamp
  97             self.state.sequence_id = (self.state.sequence_id + 1) & 0xFFFFFFFF
  98             self.state_pub.publish(self.state)
  99         if self.get_parameter('imu_feedback_enabled').value:
 100             imu = Imu()
 101             imu.header.stamp = stamp
 102             imu.header.frame_id = 'imu_link'
 103             imu.orientation.w = 1.0
 104             imu.linear_acceleration.z = 9.81
 105             self.imu_pub.publish(imu)
