  42 bool supportedResidentFloat(const rknn_tensor_attr& attr) {
  43   // RKNN can tag FP16 I/O as AFFINE. Rockchip's rknpu2 examples explicitly
  44   // exclude FP16 from their AFFINE quantization check. Native FP16 bytes still
  45   // represent IEEE half values: do not apply integer scale/zp or rewrite attrs.
  46   // Keep FP32/NONE support; integer, DFP and unknown encodings remain rejected.
  47   if (attr.type == RKNN_TENSOR_FLOAT16)
  48     return attr.qnt_type == RKNN_TENSOR_QNT_NONE || attr.qnt_type == RKNN_TENSOR_QNT_AFFINE_ASYMMETRIC;
  49   return attr.type == RKNN_TENSOR_FLOAT32 && attr.qnt_type == RKNN_TENSOR_QNT_NONE;
  50 }

  52 // IEEE 754 binary16, round-to-nearest, ties-to-even, including subnormals.
  53 // No ARM-specific casts or FPCR-dependent rounding in the parameter loader.
  54 std::uint16_t toHalf(float value) {
  55   std::uint32_t bits;
  56   std::memcpy(&bits, &value, sizeof(bits));
  57   const auto sign = static_cast<std::uint16_t>((bits >> 16U) & 0x8000U);
  58   const int exponent = static_cast<int>((bits >> 23U) & 255U) - 127;
  59   const std::uint32_t fraction = bits & 0x7fffffU;
  60   if (exponent < -25) return sign;
  61   const unsigned shift = exponent < -14 ? static_cast<unsigned>(-exponent - 1) : 13U;
  62   const std::uint32_t significand = exponent < -14 ? fraction | 0x800000U : fraction;
  63   std::uint32_t rounded = significand >> shift;
  64   const std::uint32_t remainder = significand & ((1U << shift) - 1U);
  65   const std::uint32_t halfway = 1U << (shift - 1U);
  66   if (remainder > halfway || (remainder == halfway && (rounded & 1U))) ++rounded;
  67   return static_cast<std::uint16_t>(sign | (exponent < -14 ? rounded :
  68       (static_cast<std::uint32_t>(exponent + 15) << 10U) + rounded));
  69 }
  70 
  71 float fromHalf(std::uint16_t bits) {
  72   const int exponent = (bits >> 10U) & 31U;
  73   if (exponent == 31) return std::numeric_limits<float>::quiet_NaN();
  74   const int fraction = bits & 1023U;
  75   const float value = std::ldexp(static_cast<float>(exponent ? 1024 + fraction : fraction),
  76                                 exponent ? exponent - 25 : -24);
  77   return bits & 0x8000U ? -value : value;
  78 }

  92 RknnModel::RknnModel(const std::string& model_path, InputMode mode) : input_mode_(mode) {
  93   std::ifstream file(model_path, std::ios::binary | std::ios::ate);
  94   if (!file.good()) throw std::runtime_error("RKNN model not found: " + model_path);
  95 
  96   const std::streamsize size = file.tellg();
  97   if (size <= 0 || static_cast<std::uint64_t>(size) > std::numeric_limits<std::uint32_t>::max())
  98     throw std::runtime_error("RKNN model has invalid size: " + model_path);
  99 
 100   std::vector<char> data(static_cast<std::size_t>(size));
 101   file.seekg(0, std::ios::beg);
 102   if (!file.read(data.data(), size)) throw std::runtime_error("Failed to read RKNN model: " + model_path);
 103 
 104   // These flags belong only to the persistent context. Copy-mode AMP/static/reference
 105   // contexts retain the SDK's original cache and inputs_set/outputs_get behavior.
 106   const std::uint32_t flags = mode == InputMode::ResidentWbc ?
 107       RKNN_FLAG_DISABLE_FLUSH_INPUT_MEM_CACHE | RKNN_FLAG_DISABLE_FLUSH_OUTPUT_MEM_CACHE : 0;
 108   checkRknn(rknn_init(&context_, data.data(), static_cast<std::uint32_t>(data.size()), flags, nullptr),
 109             "rknn_init(" + model_path + ")");
 110 
 111   try
 112   {
 113     rknn_input_output_num io_num{};
 114     checkRknn(rknn_query(context_, RKNN_QUERY_IN_OUT_NUM, &io_num, sizeof(io_num)), "RKNN query I/O count");
 115     if (io_num.n_input == 0 || io_num.n_output == 0)
 116       throw std::runtime_error("RKNN models must have at least one input and one output");
 117 
 118     input_attrs_.resize(io_num.n_input);
 119     input_names_.reserve(io_num.n_input);
 120     for (std::uint32_t i = 0; i < io_num.n_input; ++i)
 121     {
 122       input_attrs_[i].index = i;
 123       checkRknn(rknn_query(context_, RKNN_QUERY_INPUT_ATTR, &input_attrs_[i], sizeof(input_attrs_[i])),
 124                 "RKNN query input " + std::to_string(i));
 125       input_names_.emplace_back(input_attrs_[i].name);
 126     }
 127 
 128     output_names_.reserve(io_num.n_output);
 129     output_buffers_.reserve(io_num.n_output);
 130     outputs_.resize(io_num.n_output);
 131     for (std::uint32_t i = 0; i < io_num.n_output; ++i)
 132     {
 133       rknn_tensor_attr output_attr{};
 134       output_attr.index = i;
 135       checkRknn(rknn_query(context_, RKNN_QUERY_OUTPUT_ATTR, &output_attr, sizeof(output_attr)),
 136                 "RKNN query output " + std::to_string(i));
 137       output_names_.emplace_back(output_attr.name);
 138       output_attrs_.push_back(output_attr);
 139       output_buffers_.emplace_back(output_attr.n_elems, 0.0f);
 140       outputs_[i].index = i;
 141       outputs_[i].want_float = 1;
 142     }
 143     if (mode == InputMode::ResidentWbc) initializeResident();
 144   } catch (...)
 145   {
 146     releaseResident();
 147     rknn_destroy(context_);
 148     context_ = 0;
 149     throw;
 150   }
 151 }

 217 bool RknnModel::run(const std::vector<const float*>& inputs, const std::vector<std::size_t>& element_counts) {
 218   last_error_.clear();
 219   if (input_mode_ != InputMode::Copy) {
 220     last_error_ = "inputs_set is forbidden on a resident context";
 221     return false;
 222   }
 223   if (inputs.size() != inputCount() || element_counts.size() != inputCount())
 224   {
 225     last_error_ =
 226         "input count mismatch: expected " + std::to_string(inputCount()) + ", got " + std::to_string(inputs.size());
 227     return false;
 228   }
 229 
 230   std::vector<rknn_input> rknn_inputs(inputCount());
 231   for (std::size_t i = 0; i < inputCount(); ++i)
 232   {
 233     if (inputs[i] == nullptr || element_counts[i] != inputElements(i))
 234     {
 235       last_error_ = "input '" + inputName(i) + "' element count mismatch: expected " +
 236                     std::to_string(inputElements(i)) + ", got " + std::to_string(element_counts[i]);
 237       return false;
 238     }
 239     rknn_inputs[i].index = static_cast<std::uint32_t>(i);
 240     rknn_inputs[i].buf = const_cast<float*>(inputs[i]);
 241     rknn_inputs[i].size = static_cast<std::uint32_t>(element_counts[i] * sizeof(float));
 242     rknn_inputs[i].type = RKNN_TENSOR_FLOAT32;
 243     rknn_inputs[i].fmt = input_attrs_[i].fmt;
 244   }
 245   int ret = rknn_inputs_set(context_, static_cast<std::uint32_t>(rknn_inputs.size()), rknn_inputs.data());
 246   if (ret != RKNN_SUCC)
 247   {
 248     last_error_ = "rknn_inputs_set failed, ret=" + std::to_string(ret);
 249     return false;
 250   }
 251 
 252   ret = rknn_run(context_, nullptr);
 253   if (ret != RKNN_SUCC)
 254   {
 255     last_error_ = "rknn_run failed, ret=" + std::to_string(ret);
 256     return false;
 257   }
 258 
 259   for (std::size_t i = 0; i < outputs_.size(); ++i)
 260   {
 261     outputs_[i].index = static_cast<std::uint32_t>(i);
 262     outputs_[i].want_float = 1;
 263     outputs_[i].buf = nullptr;
 264     outputs_[i].size = 0;
 265     outputs_[i].is_prealloc = 0;
 266   }
 267   ret = rknn_outputs_get(context_, static_cast<std::uint32_t>(outputs_.size()), outputs_.data(), nullptr);
 268   if (ret != RKNN_SUCC)
 269   {
 270     last_error_ = "rknn_outputs_get failed, ret=" + std::to_string(ret);
 271     return false;
 272   }
 273 
 274   bool valid = true;
 275   for (std::size_t i = 0; i < outputs_.size(); ++i)
 276   {
 277     const std::size_t expected_bytes = output_buffers_[i].size() * sizeof(float);
 278     if (outputs_[i].buf == nullptr || outputs_[i].size < expected_bytes)
 279     {
 280       last_error_ = "RKNN output '" + output_names_[i] + "' has an invalid buffer size";
 281       valid = false;
 282       break;
 283     }
 284     std::memcpy(output_buffers_[i].data(), outputs_[i].buf, expected_bytes);
 285     if (!std::all_of(output_buffers_[i].begin(), output_buffers_[i].end(),
 286                      [](float value) { return std::isfinite(value); }))
 287     {
 288       last_error_ = "RKNN output '" + output_names_[i] + "' contains a non-finite value";
 289       valid = false;
 290       break;
 291     }
 292   }
 293   const int release_ret = rknn_outputs_release(context_, static_cast<std::uint32_t>(outputs_.size()), outputs_.data());
 294   if (valid && release_ret != RKNN_SUCC)
 295   {
 296     last_error_ = "rknn_outputs_release failed, ret=" + std::to_string(release_ret);
 297     valid = false;
 298   }
 299   return valid;
 300 }

 306 RknnModel::ResidentTensor RknnModel::residentTensor(const rknn_tensor_attr& logical, rknn_query_cmd query) {
 307   ResidentTensor tensor;
 308   auto& attr = tensor.attr;
 309   attr.index = logical.index;
 310   checkRknn(rknn_query(context_, query, &attr, sizeof(attr)), "RKNN native query " + std::string(logical.name));
 311   const auto reject = [&] {
 312     throw std::runtime_error("Unsupported resident WBC layout; logical: " + describe(logical) +
 313                              "; native: " + describe(attr));
 314   };
 315   // The exported WBC ABI consists solely of vectors and matrices. Do not interpret
 316   // a blocked/image layout (or padding) as a change to the manifest's logical ABI.
 317   // Such a model must be inspected on the board before adding an explicit mapping.
 318   if (attr.n_dims < 1 || attr.n_dims > 2 || logical.n_dims < 1 || logical.n_dims > RKNN_MAX_DIMS ||
 319       attr.fmt != RKNN_TENSOR_UNDEFINED || !supportedResidentFloat(attr) ||
 320       !supportedResidentFloat(logical) || attr.h_stride != 0)
 321     reject();
 322   std::vector<std::uint32_t> logical_shape, native_shape;
 323   for (std::uint32_t i = 0; i < logical.n_dims; ++i) {
 324     if (!logical.dims[i]) reject();
 325     if (logical.dims[i] != 1) logical_shape.push_back(logical.dims[i]);
 326   }
 327   std::uint64_t native_elements = 1;
 328   for (std::uint32_t i = 0; i < attr.n_dims; ++i) {
 329     if (!attr.dims[i]) reject();
 330     if (attr.dims[i] != 1) native_shape.push_back(attr.dims[i]);
 331     native_elements *= attr.dims[i];
 332   }
 333   if (logical_shape.empty() || logical_shape.size() > 2 || logical_shape.size() != native_shape.size()) reject();
 334   std::uint64_t elements = 1;
 335   for (std::size_t i = 0; i < logical_shape.size(); ++i) {
 336     elements *= logical_shape[i];
 337     // Only the last planar dimension may be padded. Never guess an axis permutation.
 338     if (i + 1 == logical_shape.size()) {
 339       if (native_shape[i] < logical_shape[i]) reject();
 340     } else if (native_shape[i] != logical_shape[i]) reject();
 341   }
 342   if (elements != logical.n_elems || attr.n_elems < elements) reject();
 343   tensor.elements = logical.n_elems;
 344   const std::size_t native_width = attr.dims[attr.n_dims - 1];
 345   tensor.width = native_width == 1 ? 1 : logical_shape.back();
 346   tensor.stride = attr.w_stride ? attr.w_stride : native_width;
 347   tensor.element_bytes = attr.type == RKNN_TENSOR_FLOAT16 ? 2 : sizeof(float);
 348   const std::uint64_t capacity = attr.size_with_stride ? attr.size_with_stride : attr.size;
 349   if (tensor.stride < native_width || attr.size < elements * tensor.element_bytes ||
 350       capacity < attr.size || capacity > 16U * 1024U * 1024U ||
 351       native_elements > capacity / tensor.element_bytes || attr.n_elems > capacity / tensor.element_bytes ||
 352       tensor.stride > capacity / tensor.element_bytes / (elements / tensor.width))
 353     reject();
 354   attr.size_with_stride = static_cast<std::uint32_t>(capacity);
 355   attr.pass_through = 1;  // Bytes written below are genuinely in this native dtype/layout.
 356   return tensor;
 357 }

 359 void RknnModel::initializeResident() {
 360   policy_index_ = inputIndex("policy");
 361   command_index_ = inputIndex("command_short");
 362   if (inputCount() != 33 || outputCount() != 1 || policy_index_ < 0 || command_index_ < 0 ||
 363       inputElements(policy_index_) != 399 || inputElements(command_index_) != 240 ||
 364       outputIndex("action") != 0 || outputElements(0) != 21)
 365     throw std::runtime_error("Resident mode requires the WBC 33-input / action[21] contract");
 366   // Query/validate every layout before allocating any device memory.
 367   for (const auto& attr : input_attrs_) resident_inputs_.push_back(residentTensor(attr, RKNN_QUERY_NATIVE_INPUT_ATTR));
 368   for (const auto& attr : output_attrs_) resident_outputs_.push_back(residentTensor(attr, RKNN_QUERY_NATIVE_OUTPUT_ATTR));
 369   const auto allocate = [&](ResidentTensor& tensor, int slots) {
 370     for (int slot = 0; slot < slots; ++slot) {
 371       auto*& memory = tensor.memory[slot];
 372       memory = rknn_create_mem2(context_, tensor.attr.size_with_stride, RKNN_FLAG_MEMORY_CACHEABLE);
 373       if (!memory || !memory->virt_addr || memory->size < tensor.attr.size_with_stride)
 374         throw std::runtime_error("RKNN persistent allocation failed: " + describe(tensor.attr));
 375       resident_stats_.allocated_bytes += memory->size;
 376       resident_stats_.layout_bytes += tensor.attr.size_with_stride;
 377       std::memset(memory->virt_addr, 0, memory->size);
 378     }
 379   };
 380   for (std::size_t i = 0; i < resident_inputs_.size(); ++i) allocate(resident_inputs_[i], isObservation(i) ? 1 : 2);
 381   for (auto& tensor : resident_outputs_) allocate(tensor, 1);
 382   for (auto& tensor : resident_inputs_) {
 383     checkRknn(rknn_mem_sync(context_, tensor.memory[0], RKNN_MEMORY_SYNC_TO_DEVICE), "RKNN initial input sync");
 384     checkRknn(rknn_set_io_mem(context_, tensor.memory[0], &tensor.attr), "RKNN initial input bind");
 385   }
 386   for (auto& tensor : resident_outputs_) {
 387     // Padding was CPU-initialized; flush it once before the device owns the output.
 388     checkRknn(rknn_mem_sync(context_, tensor.memory[0], RKNN_MEMORY_SYNC_TO_DEVICE), "RKNN initial output sync");
 389     checkRknn(rknn_set_io_mem(context_, tensor.memory[0], &tensor.attr), "RKNN persistent output bind");
 390   }
 391   // Slot 0 is a complete rollback binding, never a usable all-zero adapter.
 392   // resident_slot_ remains -1 until a validated parameter package is committed.
 393   resident_stats_.context_initializations = 1;
 394   rknn_mem_size sizes{};
 395   resident_stats_.sdk_memory_query = rknn_query(context_, RKNN_QUERY_MEM_SIZE, &sizes, sizeof(sizes));
 396   if (resident_stats_.sdk_memory_query == RKNN_SUCC) {
 397     resident_stats_.sdk_weight_bytes = sizes.total_weight_size;
 398     resident_stats_.sdk_internal_bytes = sizes.total_internal_size;
 399     resident_stats_.sdk_dma_bytes = sizes.total_dma_allocated_size;
 400   }
 401 }

 403 void RknnModel::releaseResident() noexcept {
 404   for (auto* tensors : {&resident_inputs_, &resident_outputs_}) {
 405     for (auto& tensor : *tensors) {
 406       for (auto*& memory : tensor.memory) {
 407         if (memory) rknn_destroy_mem(context_, memory);
 408         memory = nullptr;
 409       }
 410     }
 411   }

 414 void RknnModel::writeResident(ResidentTensor& tensor, int slot, const float* values, std::size_t count) {
 415   if (!values || count != tensor.elements) throw std::runtime_error("Resident input size mismatch");
 416   // Validate before touching a buffer; FP16 overflow must never silently become Inf.
 417   for (std::size_t i = 0; i < count; ++i) {
 418     if (!std::isfinite(values[i]) || (tensor.attr.type == RKNN_TENSOR_FLOAT16 && std::abs(values[i]) > 65504.0f))
 419       throw std::runtime_error("Resident input not representable: " + std::string(tensor.attr.name));
 420   }
 421   auto* data = static_cast<unsigned char*>(tensor.memory[slot]->virt_addr);
 422   // Initialize all padding, including allocation alignment, on each prepare/write.
 423   std::memset(data, 0, tensor.memory[slot]->size);
 424   for (std::size_t i = 0; i < count; ++i) {
 425     const auto offset = (i / tensor.width * tensor.stride + i % tensor.width) * tensor.element_bytes;
 426     if (tensor.attr.type == RKNN_TENSOR_FLOAT16) {
 427       const auto half = toHalf(values[i]);
 428       std::memcpy(data + offset, &half, sizeof(half));
 429     } else std::memcpy(data + offset, values + i, sizeof(float));
 430   }
 431 }

 433 void RknnModel::stageResidentParameters(int slot, const std::vector<ParameterInput>& parameters) {
 434   std::lock_guard<std::mutex> lock(resident_mutex_);
 435   if (input_mode_ != InputMode::ResidentWbc || !resident_healthy_ || slot < 0 || slot > 1 ||
 436       slot == (resident_slot_ < 0 ? 0 : resident_slot_))
 437     throw std::runtime_error("Cannot write active/unavailable resident parameter slot");
 438   staged_[slot] = false;
 439   if (parameters.size() != inputCount() - 2) throw std::runtime_error("Incomplete resident parameter set");
 440   std::set<std::size_t> seen;
 441   for (const auto& value : parameters) {
 442     if (value.index >= inputCount() || isObservation(value.index) || !seen.insert(value.index).second ||
 443         !value.data || value.elements != inputElements(value.index))
 444       throw std::runtime_error("Invalid resident parameter mapping");
 445   }
 446   ++resident_stats_.parameter_prepares;
 447   const auto started = std::chrono::steady_clock::now();
 448   try {
 449     for (const auto& value : parameters) {
 450       auto& tensor = resident_inputs_[value.index];
 451       writeResident(tensor, slot, value.data, value.elements);
 452       resident_stats_.parameter_write_bytes += tensor.memory[slot]->size;
 453       if (tensor.attr.type == RKNN_TENSOR_FLOAT16) resident_stats_.parameter_converted_elements += value.elements;
 454     }
 455     staged_[slot] = true;
 456   } catch (...) {
 457     resident_stats_.pack_ms += millisecondsSince(started);
 458     throw;
 459   }
 460   resident_stats_.pack_ms += millisecondsSince(started);

 463 bool RknnModel::bindResidentSlot(int slot) {
 464   bool success = true;
 465   // Attempt the whole set even after a failure, particularly during rollback.
 466   for (std::size_t i = 0; i < resident_inputs_.size(); ++i) {
 467     if (isObservation(i)) continue;
 468     auto& tensor = resident_inputs_[i];
 469     ++resident_stats_.parameter_bind_calls;
 470     const int ret = rknn_set_io_mem(context_, tensor.memory[slot], &tensor.attr);
 471     if (ret != RKNN_SUCC) {
 472       if (success) last_error_ = "Resident bind failed for " + inputName(i) + ", ret=" + std::to_string(ret);
 473       success = false;
 474     }
 475   }
 476   return success;
 477 }

 484     last_error_ += "; complete rollback failed; resident inference disabled until restart";
 485     return false;

 487   resident_slot_ = slot;

 491 bool RknnModel::commitResidentParameters(int slot) {
 492   std::lock_guard<std::mutex> lock(resident_mutex_);
 493   last_error_.clear();
 494   if (input_mode_ != InputMode::ResidentWbc || !resident_healthy_ || slot < 0 || slot > 1 ||
 495       !staged_[slot] || slot == (resident_slot_ < 0 ? 0 : resident_slot_)) {
 496     last_error_ = "Resident commit requires a staged spare slot and a healthy context";
 497     return false;
 498   }
 499   const auto started = std::chrono::steady_clock::now();
 500   // Sync all new tensors before changing any binding. No SDK operation runs in stage.
 501   for (std::size_t i = 0; i < resident_inputs_.size(); ++i) {
 502     if (isObservation(i)) continue;
 503     ++resident_stats_.parameter_sync_calls;
 504     const int ret = rknn_mem_sync(context_, resident_inputs_[i].memory[slot], RKNN_MEMORY_SYNC_TO_DEVICE);
 505     if (ret != RKNN_SUCC) {
 506       last_error_ = "Resident parameter sync failed for " + inputName(i) + ", ret=" + std::to_string(ret);
 507       resident_stats_.commit_ms += millisecondsSince(started);
 508       return false;  // Old complete binding is untouched.
 509     }
 510   }
 511   if (!bindResidentSlot(slot)) {
 512     const auto failure = last_error_;
 513     if (restoreResidentSlot(resident_slot_)) last_error_ = failure + "; previous complete binding restored";
 514     resident_stats_.commit_ms += millisecondsSince(started);
 515     return false;
 516   }
 517   resident_slot_ = slot;
 518   ++resident_stats_.commits;
 519   resident_stats_.commit_ms += millisecondsSince(started);
 520   return true;
 521 }
 522 
 523 bool RknnModel::restoreResidentParameters(int slot) {
 524   std::lock_guard<std::mutex> lock(resident_mutex_);
 525   if (input_mode_ != InputMode::ResidentWbc || !resident_healthy_ || slot < -1 || slot > 1 ||
 526       (slot >= 0 && !staged_[slot])) {
 527     last_error_ = "Resident rollback unavailable; inference disabled until restart";
 528     resident_healthy_ = false;
 529     return false;
 530   }
 531   return restoreResidentSlot(slot);
 532 }

 534 bool RknnModel::runResident(const float* policy, std::size_t policy_size,
 535                             const float* command, std::size_t command_size) {
 536   std::lock_guard<std::mutex> lock(resident_mutex_);
 537   last_error_.clear();
 538   if (input_mode_ != InputMode::ResidentWbc || !resident_healthy_ || resident_slot_ < 0) {
 539     last_error_ = "Resident inference requires a complete prepared parameter set";
 540     return false;
 541   }
 542   try {
 543     const auto observation_started = std::chrono::steady_clock::now();
 544     writeResident(resident_inputs_[policy_index_], 0, policy, policy_size);
 545     writeResident(resident_inputs_[command_index_], 0, command, command_size);
 546     for (int index : {policy_index_, command_index_}) {
 547       ++resident_stats_.observation_sync_calls;
 548       checkRknn(rknn_mem_sync(context_, resident_inputs_[index].memory[0], RKNN_MEMORY_SYNC_TO_DEVICE),
 549                 "RKNN observation sync");
 550     }
 551     const double observation_ms = millisecondsSince(observation_started);
 552     resident_stats_.observation_ms += observation_ms;
 553     resident_stats_.observation_max_ms = std::max(resident_stats_.observation_max_ms, observation_ms);
 554     const auto run_started = std::chrono::steady_clock::now();
 555     ++resident_stats_.runs;
 556     // Synchronous set_io_mem API, no ASYNC_MASK, non-block or external fences.
 557     // rknn_run includes this frame's completion wait; never use outputs_get/release here.
 558     const int ret = rknn_run(context_, nullptr);
 559     const double execution_ms = millisecondsSince(run_started);
 560     resident_stats_.execution_ms += execution_ms;
 561     resident_stats_.execution_max_ms = std::max(resident_stats_.execution_max_ms, execution_ms);
 562     if (ret != RKNN_SUCC) {
 563       // No asynchronous submission is used. Treat any execution/driver error as
 564       // terminal, and report the extra wait result instead of retrying the frame.
 565       const int wait_ret = rknn_wait(context_, nullptr);
 566       resident_healthy_ = false;
 567       checkRknn(ret, "RKNN resident run (context disabled; wait_ret=" + std::to_string(wait_ret) + ")");
 568     }
 569     const auto output_started = std::chrono::steady_clock::now();
 570     for (std::size_t index = 0; index < resident_outputs_.size(); ++index) {
 571       const auto& tensor = resident_outputs_[index];
 572       ++resident_stats_.output_sync_calls;
 573       checkRknn(rknn_mem_sync(context_, tensor.memory[0], RKNN_MEMORY_SYNC_FROM_DEVICE), "RKNN output sync");
 574       const auto* bytes = static_cast<const unsigned char*>(tensor.memory[0]->virt_addr);
 575       for (std::size_t i = 0; i < tensor.elements; ++i) {
 576         const auto offset = (i / tensor.width * tensor.stride + i % tensor.width) * tensor.element_bytes;
 577         float value;
 578         if (tensor.attr.type == RKNN_TENSOR_FLOAT16) {
 579           std::uint16_t half;
 580           std::memcpy(&half, bytes + offset, sizeof(half));
 581           value = fromHalf(half);
 582         } else std::memcpy(&value, bytes + offset, sizeof(value));
 583         if (!std::isfinite(value)) throw std::runtime_error("Non-finite resident output");
 584         output_buffers_[index][i] = value;
 585       }
 586     }
 587     const double output_ms = millisecondsSince(output_started);
 588     resident_stats_.output_ms += output_ms;
 589     resident_stats_.output_max_ms = std::max(resident_stats_.output_max_ms, output_ms);
 590     return true;
 591   } catch (const std::exception& exception) {
 592     last_error_ = exception.what();
 593     return false;
 594   }
 595 }

 139 bool RknnModel::run(const std::vector<const float*>& inputs, const std::vector<std::size_t>& element_counts) {
 140   last_error_.clear();
 141   if (inputs.size() != inputCount() || element_counts.size() != inputCount())
 142   {
 143     last_error_ =
 144         "input count mismatch: expected " + std::to_string(inputCount()) + ", got " + std::to_string(inputs.size());
 145     return false;
 146   }
 147 
 148   std::vector<rknn_input> rknn_inputs(inputCount());
 149   for (std::size_t i = 0; i < inputCount(); ++i)
 150   {
 151     if (inputs[i] == nullptr || element_counts[i] != inputElements(i))
 152     {
 153       last_error_ = "input '" + inputName(i) + "' element count mismatch: expected " +
 154                     std::to_string(inputElements(i)) + ", got " + std::to_string(element_counts[i]);
 155       return false;
 156     }
 157     rknn_inputs[i].index = static_cast<std::uint32_t>(i);
 158     rknn_inputs[i].buf = const_cast<float*>(inputs[i]);
 159     rknn_inputs[i].size = static_cast<std::uint32_t>(element_counts[i] * sizeof(float));
 160     rknn_inputs[i].type = RKNN_TENSOR_FLOAT32;
 161     rknn_inputs[i].fmt = input_attrs_[i].fmt;
 162   }
 163   int ret = rknn_inputs_set(context_, static_cast<std::uint32_t>(rknn_inputs.size()), rknn_inputs.data());
 164   if (ret != RKNN_SUCC)
 165   {
 166     last_error_ = "rknn_inputs_set failed, ret=" + std::to_string(ret);
 167     return false;
 168   }
 169 
 170   ret = rknn_run(context_, nullptr);
 171   if (ret != RKNN_SUCC)
 172   {
 173     last_error_ = "rknn_run failed, ret=" + std::to_string(ret);
 174     return false;
 175   }
 176 
 177   for (std::size_t i = 0; i < outputs_.size(); ++i)
 178   {
 179     outputs_[i].index = static_cast<std::uint32_t>(i);
 180     outputs_[i].want_float = 1;
 181     outputs_[i].buf = nullptr;
 182     outputs_[i].size = 0;
 183     outputs_[i].is_prealloc = 0;
 184   }
 185   ret = rknn_outputs_get(context_, static_cast<std::uint32_t>(outputs_.size()), outputs_.data(), nullptr);
 186   if (ret != RKNN_SUCC)
 187   {
 188     last_error_ = "rknn_outputs_get failed, ret=" + std::to_string(ret);
 189     return false;
 190   }
 191 
 192   bool valid = true;
 193   for (std::size_t i = 0; i < outputs_.size(); ++i)
 194   {
 195     const std::size_t expected_bytes = output_buffers_[i].size() * sizeof(float);
 196     if (outputs_[i].buf == nullptr || outputs_[i].size < expected_bytes)
 197     {
 198       last_error_ = "RKNN output '" + output_names_[i] + "' has an invalid buffer size";
 199       valid = false;
 200       break;
 201     }
 202     std::memcpy(output_buffers_[i].data(), outputs_[i].buf, expected_bytes);
 203     if (!std::all_of(output_buffers_[i].begin(), output_buffers_[i].end(),
 204                      [](float value) { return std::isfinite(value); }))
 205     {
 206       last_error_ = "RKNN output '" + output_names_[i] + "' contains a non-finite value";
 207       valid = false;
 208       break;
 209     }
 210   }
 211   const int release_ret = rknn_outputs_release(context_, static_cast<std::uint32_t>(outputs_.size()), outputs_.data());
 212   if (valid && release_ret != RKNN_SUCC)
 213   {
 214     last_error_ = "rknn_outputs_release failed, ret=" + std::to_string(release_ret);
 215     valid = false;
 216   }
 217   return valid;
