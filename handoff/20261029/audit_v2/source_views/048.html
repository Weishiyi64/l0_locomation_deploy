 166     mapped_state.dq = state.dq;

 177                            feedback_wrap_[i].error_samples, system_command_arbiter::kMotorErrorFaultSamples);

 232 bool LeggedReal::read(rclcpp::Time time, rclcpp::Duration) {
 233   std::vector<system_command_arbiter::JointFeedback> feedback;
 234   std::vector<bool> updated;
 235   bool reset_feedback_watchdogs = false;
 236   bool motors_enabled_at_snapshot = false;
 237   {
 238     std::lock_guard<std::mutex> lock(feedback_mutex_);
 239     reset_feedback_watchdogs = reset_feedback_watchdogs_.exchange(false);
 240     motors_enabled_at_snapshot = motors_enabled_.load();
 241     feedback = latest_feedback_;
 242     updated = feedback_updated_;
 243     control_feedback_ready_ = true;
 244     for (std::size_t i = 0; i < feedback_wrap_.size(); ++i)
 245     {
 246       control_wrap_offsets_[i] = feedback_wrap_[i].offset;
 247       if (motor_mapping_->entries()[i].enabled && !feedback_wrap_[i].ready()) control_feedback_ready_ = false;
 248     }
 249     std::fill(feedback_updated_.begin(), feedback_updated_.end(), false);
 250   }
 251   if (reset_feedback_watchdogs)
 252   {
 253     std::fill(consecutive_missing_feedback_.begin(), consecutive_missing_feedback_.end(), 0);
 254     std::fill(feedback_fault_.begin(), feedback_fault_.end(), false);
 255   }
 256 
 257   bool motor_fault = false;
 258   for (std::size_t i = 0; i < feedback.size(); ++i)
 259   {
 260     const auto& mapping = motor_mapping_->entries()[i];
 261     if (!mapping.enabled)
 262     {
 263       joint_motor_states_[i].q = default_joint_positions_[i];
 264       joint_motor_states_[i].dq = 0.0;
 265       joint_motor_states_[i].tau = 0.0;
 266       continue;
 267     }
 268 
 269     if (updated[i] && feedback[i].valid && feedback[i].error == 0U)
 270     {
 271       const bool recovered = feedback_fault_[i];
 272       feedback_fault_[i] = false;
 273       consecutive_missing_feedback_[i] = 0;
 274       joint_data_[i].pos_ = feedback[i].q;
 275       joint_data_[i].vel_ =
 276           joint_filter_coeff_real_ * feedback[i].dq + (1.0 - joint_filter_coeff_real_) * joint_data_[i].prev_vel_;
 277       joint_data_[i].tau_ = feedback[i].tau;
 278       joint_data_[i].prev_vel_ = joint_data_[i].vel_;
 279       if (recovered)
 280         RCLCPP_INFO(nh_->get_logger(), "Motor %s feedback recovered; motor commands may resume",
 281                     joint_names_[i].c_str());
 282     }
 283     else if (updated[i] && feedback[i].error_pending)
 284     {
 285       // Count received error frames in the callback, not control cycles; keep the last good observation.
 286       consecutive_missing_feedback_[i] = 0;
 287     }
 288     else if (updated[i] && feedback[i].initializing)
 289     {
 290       // An uncalibrated joint can be online without a usable control observation.
 291       const bool recovered = feedback_fault_[i];
 292       feedback_fault_[i] = false;
 293       consecutive_missing_feedback_[i] = 0;
 294       if (recovered)
 295         RCLCPP_INFO(nh_->get_logger(),
 296                     "Motor %s feedback received again; joint position/turn still unconfirmed",
 297                     joint_names_[i].c_str());
 298     }
 299     else if (updated[i])
 300     {
 301       const bool newly_faulted = !feedback_fault_[i];
 302       feedback_fault_[i] = true;
 303       consecutive_missing_feedback_[i] = 0;
 304       if (newly_faulted)
 305       {
 306         RCLCPP_ERROR(nh_->get_logger(),
 307                      "\033[1;31mMotor %s (HPM index %zu) feedback fault: valid=%s error_code=%u; switching to "
 308                      "emergency damping\033[0m",
 309                      joint_names_[i].c_str(), i, feedback[i].valid ? "true" : "false",
 310                      static_cast<unsigned>(feedback[i].error));
 311       }
 312     }
 313     else if (motors_enabled_at_snapshot && !calibration_mode_.load())
 314     {
 315       const int previous_missing = consecutive_missing_feedback_[i];
 316       consecutive_missing_feedback_[i] = std::min(previous_missing + 1, max_consecutive_missing_);
 317       if (consecutive_missing_feedback_[i] == std::max(1, max_consecutive_missing_ / 2))
 318         RCLCPP_WARN(nh_->get_logger(),
 319                     "\033[1;33mMotor %s (HPM index %zu) no new motor feedback for %d control cycles; "
 320                     "watchdog grace period remains active\033[0m",
 321                     joint_names_[i].c_str(), i, consecutive_missing_feedback_[i]);
 322       if (previous_missing < max_consecutive_missing_ && consecutive_missing_feedback_[i] == max_consecutive_missing_)
 323       {
 324         feedback_fault_[i] = true;
 325         RCLCPP_ERROR(
 326             nh_->get_logger(),
 327             "\033[1;31mMotor %s (HPM index %zu) no new motor feedback for %d consecutive control cycles; "
 328             "switching to emergency damping\033[0m",
 329             joint_names_[i].c_str(), i, max_consecutive_missing_);
 330       }
 331     }
 332 
 333     joint_motor_states_[i].q = joint_data_[i].pos_;
 334     joint_motor_states_[i].dq = joint_data_[i].vel_;
 335     joint_motor_states_[i].tau = joint_data_[i].tau_;
 336   }
 337 
 338   motor_fault = std::any_of(feedback_fault_.begin(), feedback_fault_.end(), [](bool value) { return value; });
 339   motor_fault_.store(motor_fault);
 340   emergency_stop_triggered_.store(motor_fault);
 341   publishMotorsReady(motors_enabled_.load() && !motor_fault);
 342   publishHardwareFault();
 343 
 344   {
 345     std::lock_guard<std::mutex> lock(sensor_mutex_);
 346     imu_qx_ = imu_data_.ori[0];
 347     imu_qy_ = imu_data_.ori[1];
 348     imu_qz_ = imu_data_.ori[2];
 349     imu_qw_ = imu_data_.ori[3];
 350     imu_gyx_ = imu_data_.angular_vel[0];
 351     imu_gyy_ = imu_data_.angular_vel[1];
 352     imu_gyz_ = imu_data_.angular_vel[2];
 353     if (odom_received_)
 354     {
 355       base_px_ = odom_position_[0];
 356       base_py_ = odom_position_[1];
 357       base_pz_ = odom_position_[2];
 358       base_qx_ = odom_orientation_[0];
 359       base_qy_ = odom_orientation_[1];
 360       base_qz_ = odom_orientation_[2];
 361       base_qw_ = odom_orientation_[3];
 362     }
 363     else
 364     {
 365       base_qx_ = imu_qx_;
 366       base_qy_ = imu_qy_;
 367       base_qz_ = imu_qz_;
 368       base_qw_ = imu_qw_;
 369     }
 370   }
 371   timestamp_ns_ = static_cast<double>(time.nanoseconds());
 372   return true;
 373 }

 375 void LeggedReal::imuCallback(const sensor_msgs::msg::Imu::SharedPtr msg) {
 376   const double quaternion_norm_squared =
 377       msg->orientation.x * msg->orientation.x + msg->orientation.y * msg->orientation.y +
 378       msg->orientation.z * msg->orientation.z + msg->orientation.w * msg->orientation.w;
 379   const double values[]{msg->angular_velocity.x, msg->angular_velocity.y, msg->angular_velocity.z,
 380                         quaternion_norm_squared};
 381   if (!std::all_of(std::begin(values), std::end(values), [](double value) { return std::isfinite(value); }) ||
 382       quaternion_norm_squared < 1e-12)
 383   {
 384     RCLCPP_WARN_THROTTLE(nh_->get_logger(), *nh_->get_clock(), 1000, "Ignoring invalid IMU sample");
 385     return;
 386   }
 387   const double inverse_norm = 1.0 / std::sqrt(quaternion_norm_squared);
 388   std::lock_guard<std::mutex> lock(sensor_mutex_);
 389   imu_connection_status_ = true;
 390   last_imu_received_time_ = nh_->now();
 391   for (int i = 0; i < 3; ++i)
 392   {
 393     imu_data_.angular_vel[i] = imu_first_read_ ? values[i]
 394                                                : imu_filter_coeff_real_ * values[i] +
 395                                                      (1.0 - imu_filter_coeff_real_) * imu_data_.prev_angular_vel[i];
 396     imu_data_.prev_angular_vel[i] = imu_data_.angular_vel[i];
 397   }
 398   imu_first_read_ = false;
 399   imu_data_.ori[0] = msg->orientation.x * inverse_norm;
 400   imu_data_.ori[1] = msg->orientation.y * inverse_norm;
 401   imu_data_.ori[2] = msg->orientation.z * inverse_norm;
 402   imu_data_.ori[3] = msg->orientation.w * inverse_norm;
 403 }

 405 void LeggedReal::odomCallback(const nav_msgs::msg::Odometry::SharedPtr msg) {
 406   const auto& position = msg->pose.pose.position;
 407   const auto& orientation = msg->pose.pose.orientation;
 408   const double norm_squared = orientation.x * orientation.x + orientation.y * orientation.y +
 409                               orientation.z * orientation.z + orientation.w * orientation.w;
 410   if (msg->header.frame_id.empty() || msg->child_frame_id.empty() || !std::isfinite(position.x) ||
 411       !std::isfinite(position.y) || !std::isfinite(position.z) || !std::isfinite(norm_squared) || norm_squared < 1e-12)
 412   {
 413     RCLCPP_WARN_THROTTLE(nh_->get_logger(), *nh_->get_clock(), 1000, "Ignoring invalid odometry pose");
 414     return;
 415   }
 416   const double inverse_norm = 1.0 / std::sqrt(norm_squared);
 417   std::lock_guard<std::mutex> lock(sensor_mutex_);
 418   odom_position_[0] = position.x;
 419   odom_position_[1] = position.y;
 420   odom_position_[2] = position.z;
 421   odom_orientation_[0] = orientation.x * inverse_norm;
 422   odom_orientation_[1] = orientation.y * inverse_norm;
 423   odom_orientation_[2] = orientation.z * inverse_norm;
 424   odom_orientation_[3] = orientation.w * inverse_norm;
 425   odom_received_ = true;
 426 }

 428 bool LeggedReal::write(rclcpp::Time, rclcpp::Duration) {
 429   bool imu_failed = false;
 430   const rclcpp::Time current = nh_->now();
 431   const bool motors_enabled = motors_enabled_.load();
 432   {
 433     std::lock_guard<std::mutex> lock(sensor_mutex_);
 434     // No actuator command is published before lifecycle enable completes, so
 435     // the IMU startup watchdog must not expire while its backend is still absent.
 436     if (imu_enabled_ && motors_enabled && !imu_connection_status_)
 437       imu_failed = (current - imu_startup_time_).seconds() >= imu_startup_timeout_;
 438     else if (imu_enabled_ && motors_enabled && (current - last_imu_received_time_).seconds() > imu_connection_timeout_)
 439     {
 440       imu_connection_status_ = false;
 441       imu_failed = true;
 442     }
 443   }
 444 
 445   imu_fault_.store(imu_failed);
 446   publishHardwareFault();
 447   const bool stop = emergency_stop_triggered_.load() || imu_failed || lifecycle_fault_.load();
 448   if (stop != emergency_published_)
 449   {
 450     std_msgs::msg::Bool message;
 451     message.data = stop;
 452     emergency_pub_->publish(message);
 453     emergency_published_ = stop;
 454     if (stop)
 455       RCLCPP_ERROR(nh_->get_logger(),
 456                    "\033[1;31mMotor commands switched to emergency damping: emergency_stop=%s motor_feedback_fault=%s "
 457                    "imu_fault=%s lifecycle_fault=%s\033[0m",
 458                    emergency_stop_triggered_.load() ? "true" : "false", motor_fault_.load() ? "true" : "false",
 459                    imu_failed ? "true" : "false", lifecycle_fault_.load() ? "true" : "false");
 460     else
 461       RCLCPP_INFO(nh_->get_logger(), "Hardware fault cleared; controller state determines motor commands");
 462   }
 463   // Serialize publication with disable, calibration transitions and SETZERO requests.
 464   std::lock_guard<std::mutex> lifecycle_lock(lifecycle_mutex_);
 465   // Serialize turn selection with publication, including the first powered command.
 466   std::lock_guard<std::mutex> feedback_lock(feedback_mutex_);
 467   bool zero_gain_command = !motors_enabled_.load() || calibration_mode_.load() || lifecycle_active_ ||
 468                            lifecycle_fault_.load();
 469   bool active_command = false;
 470   bool damping_command = stop || zero_gain_command;
 471   bool feedback_ready = control_feedback_ready_;
 472   for (std::size_t i = 0; i < feedback_wrap_.size(); ++i)
 473   {
 474     if (!motor_mapping_->entries()[i].enabled) continue;
 475     damping_command = damping_command || feedback_wrap_[i].fault;
 476     const auto& command = joint_motor_commands_[i];
 477     active_command = active_command || command.kp != 0.0 || command.dq_des != 0.0 || command.tau_ff != 0.0;
 478     feedback_ready =
 479         feedback_ready && feedback_wrap_[i].ready() && control_wrap_offsets_[i] == feedback_wrap_[i].offset;
 480   }
 481   if (!damping_command && !feedback_ready)
 482   {
 483     RCLCPP_WARN_THROTTLE(nh_->get_logger(), *nh_->get_clock(), 1000,
 484                          "Publishing zero-gain MIT until all feedback turns are confirmed in the control snapshot");
 485     zero_gain_command = true;
 486     damping_command = true;
 487   }
 488   robot_interfaces::msg::MotorCommand output;
 489   output.stamp = current;
 490   output.sequence_id = ++command_sequence_;
 491   output.control_mode = robot_interfaces::msg::MotorCommand::CONTROL_MODE_MIT;
 492   output.commands.reserve(motor_mapping_->enabledSize());
 493   try
 494   {
 495     for (std::size_t i = 0; i < joint_motor_commands_.size(); ++i)
 496     {
 497       const auto& mapping = motor_mapping_->entries()[i];
 498       if (!mapping.enabled) continue;
 499       const auto& command = joint_motor_commands_[i];
 500       // Disabled/calibration states take precedence over emergency damping.
 501       const auto mapped = motor_mapping_->commandFor(
 502           mapping, damping_command ? joint_motor_states_[i].q : command.q_des,
 503           damping_command ? 0.0 : command.dq_des, damping_command ? 0.0 : command.tau_ff,
 504           damping_command ? 0.0 : command.kp,
 505           zero_gain_command ? 0.0 : (damping_command ? emergency_damping_kd_ : command.kd),
 506           feedback_wrap_[i].offset);
 507       robot_interfaces::msg::MotorTarget target;
 508       target.route.motor_type = static_cast<uint8_t>(i);
 509       target.q = mapped.q;
 510       target.dq = mapped.dq;
 511       target.tau = mapped.tau;
 512       target.kp = mapped.kp;
 513       target.kd = mapped.kd;
 514       output.commands.push_back(target);
 515     }
 516   } catch (const std::exception& error)
 517   {
 518     emergency_stop_triggered_.store(true);
 519     RCLCPP_ERROR_THROTTLE(nh_->get_logger(), *nh_->get_clock(), 1000, "\033[1;31mRejected motor command: %s\033[0m",
 520                           error.what());
 521     return false;
 522   }
 523   if (command_publisher_->get_subscription_count() == 0U)
 524     RCLCPP_WARN_THROTTLE(nh_->get_logger(), *nh_->get_clock(), 1000,
 525                          "\033[1;33mPublishing motor commands without an HPM subscriber\033[0m");
 526   if (!damping_command && active_command)
 527     for (auto& wrap : feedback_wrap_) wrap.locked = true;
 528   command_publisher_->publish(output);
 529   return true;
 530 }

 584 void LeggedReal::lifecycleTick() {
 585   std::lock_guard<std::mutex> lock(lifecycle_mutex_);
 586   if (!lifecycle_active_) return;
 587   const rclcpp::Time current = nh_->now();
 588   if (enable_publisher_->get_subscription_count() == 0U)
 589   {
 590     if ((current - lifecycle_started_).seconds() > 20.0)
 591     {
 592       lifecycle_active_ = false;
 593       motors_enabled_.store(false);
 594       lifecycle_fault_.store(true);
 595       publishMotorsReady(false);
 596       publishHardwareFault();
 597       RCLCPP_ERROR(nh_->get_logger(),
 598                    "\033[1;31mTimed out waiting for /actuator/motor/enable subscriber; motors remain disabled\033[0m");
 599     }
 600     else
 601       RCLCPP_WARN_THROTTLE(nh_->get_logger(), *nh_->get_clock(), 1000,
 602                            "\033[1;33mWaiting for HPM subscriber on /actuator/motor/enable\033[0m");
 603     return;
 604   }
 605 
 606   robot_interfaces::msg::MotorEnableCommand command;
 607   command.stamp = current;
 608   command.sequence_id = ++enable_sequence_;
 609   command.motor_types.reserve(motor_mapping_->enabledSize());
 610   command.enabled.reserve(motor_mapping_->enabledSize());
 611   for (std::size_t i = 0; i < motor_mapping_->size(); ++i)
 612   {
 613     if (!motor_mapping_->entries()[i].enabled) continue;
 614     command.motor_types.push_back(static_cast<uint8_t>(i));
 615     command.enabled.push_back(lifecycle_target_enabled_);
 616   }
 617   lifecycle_active_ = false;
 618   {
 619     // Do not count feedback received before this enable publication toward the
 620     // first complete post-enable feedback round.
 621     std::lock_guard<std::mutex> feedback_lock(feedback_mutex_);
 622     if (lifecycle_target_enabled_)
 623     {
 624       std::fill(feedback_updated_.begin(), feedback_updated_.end(), false);
 625       for (auto& wrap : feedback_wrap_)
 626       {
 627         wrap.stable_samples = 0U;
 628         wrap.error_code = 0U;
 629         wrap.error_samples = 0U;
 630       }
 631       for (auto& feedback : latest_feedback_) feedback.valid = false;
 632       reset_feedback_watchdogs_.store(true);
 633     }
 634     else
 635     {
 636       for (auto& wrap : feedback_wrap_) wrap = system_command_arbiter::FeedbackWrapState{};
 637       for (auto& feedback : latest_feedback_) feedback.valid = false;
 638       std::fill(feedback_updated_.begin(), feedback_updated_.end(), false);
 639     }
 640     enable_publisher_->publish(command);
 641     motors_enabled_.store(lifecycle_target_enabled_);
 642   }
 643   if (lifecycle_target_enabled_)
 644   {
 645     std::lock_guard<std::mutex> sensor_lock(sensor_mutex_);
 646     if (!imu_connection_status_) imu_startup_time_ = current;
 647   }
 648   lifecycle_fault_.store(false);
 649   publishHardwareFault();
 650   RCLCPP_INFO(nh_->get_logger(), "Published HPM %s command for %zu motors",
 651               lifecycle_target_enabled_ ? "enable" : "disable", motor_mapping_->enabledSize());
 652 }

 234   bool motors_enabled_at_snapshot = false;

 237     reset_feedback_watchdogs = reset_feedback_watchdogs_.exchange(false);

 240     updated = feedback_updated_;

 246     }

 258     const auto& mapping = motor_mapping_->entries()[i];
 259     if (!mapping.enabled)

 262       joint_motor_states_[i].dq = 0.0;

 272       joint_data_[i].pos_ = feedback[i].q;
 273       joint_data_[i].vel_ =
 274           joint_filter_coeff_real_ * feedback[i].dq + (1.0 - joint_filter_coeff_real_) * joint_data_[i].prev_vel_;
 275       joint_data_[i].tau_ = feedback[i].tau;
 276       joint_data_[i].prev_vel_ = joint_data_[i].vel_;
 277       if (recovered)
 278         RCLCPP_INFO(nh_->get_logger(), "Motor %s feedback recovered; motor commands may resume",
 279                     joint_names_[i].c_str());
 280     }
 281     else if (updated[i] && feedback[i].error_pending)
 282     {
 283       // Count received error frames in the callback, not control cycles; keep the last good observation.
 284       consecutive_missing_feedback_[i] = 0;
 285     }
 286     else if (updated[i] && feedback[i].initializing)
 287     {
 288       // An uncalibrated joint can be online without a usable control observation.
 289       const bool recovered = feedback_fault_[i];
 290       feedback_fault_[i] = false;
 291       consecutive_missing_feedback_[i] = 0;
 292       if (recovered)
 293         RCLCPP_INFO(nh_->get_logger(),
 294                     "Motor %s feedback received again; joint position/turn still unconfirmed",
 295                     joint_names_[i].c_str());
 296     }
 297     else if (updated[i])
 298     {
 299       const bool newly_faulted = !feedback_fault_[i];
 300       feedback_fault_[i] = true;
 301       consecutive_missing_feedback_[i] = 0;
 302       if (newly_faulted)
 303       {
 304         RCLCPP_ERROR(nh_->get_logger(),
 305                      "\033[1;31mMotor %s (HPM index %zu) feedback fault: valid=%s error_code=%u; switching to "
 306                      "emergency damping\033[0m",
 307                      joint_names_[i].c_str(), i, feedback[i].valid ? "true" : "false",
 308                      static_cast<unsigned>(feedback[i].error));
 309       }
 310     }
 311     else if (motors_enabled_at_snapshot && !calibration_mode_.load())
 312     {
 313       const int previous_missing = consecutive_missing_feedback_[i];
 314       consecutive_missing_feedback_[i] = std::min(previous_missing + 1, max_consecutive_missing_);
 315       if (consecutive_missing_feedback_[i] == std::max(1, max_consecutive_missing_ / 2))
 316         RCLCPP_WARN(nh_->get_logger(),
 317                     "\033[1;33mMotor %s (HPM index %zu) no new motor feedback for %d control cycles; "
 318                     "watchdog grace period remains active\033[0m",
 319                     joint_names_[i].c_str(), i, consecutive_missing_feedback_[i]);
 320       if (previous_missing < max_consecutive_missing_ && consecutive_missing_feedback_[i] == max_consecutive_missing_)
 321       {
 322         feedback_fault_[i] = true;
 323         RCLCPP_ERROR(
 324             nh_->get_logger(),
 325             "\033[1;31mMotor %s (HPM index %zu) no new motor feedback for %d consecutive control cycles; "
 326             "switching to emergency damping\033[0m",
 327             joint_names_[i].c_str(), i, max_consecutive_missing_);
 328       }
 329     }
 330 
 331     joint_motor_states_[i].q = joint_data_[i].pos_;
 332     joint_motor_states_[i].dq = joint_data_[i].vel_;
 333     joint_motor_states_[i].tau = joint_data_[i].tau_;
 334   }
 335 
 336   motor_fault = std::any_of(feedback_fault_.begin(), feedback_fault_.end(), [](bool value) { return value; });
 337   motor_fault_.store(motor_fault);
 338   emergency_stop_triggered_.store(motor_fault);
 339   publishMotorsReady(motors_enabled_.load() && !motor_fault);
 340   publishHardwareFault();
 341 
 342   {
 343     std::lock_guard<std::mutex> lock(sensor_mutex_);
 344     imu_qx_ = imu_data_.ori[0];
 345     imu_qy_ = imu_data_.ori[1];
 346     imu_qz_ = imu_data_.ori[2];
 347     imu_qw_ = imu_data_.ori[3];
 348     imu_gyx_ = imu_data_.angular_vel[0];
 349     imu_gyy_ = imu_data_.angular_vel[1];
 350     imu_gyz_ = imu_data_.angular_vel[2];
 351     if (odom_received_)
 352     {
 353       base_px_ = odom_position_[0];
 354       base_py_ = odom_position_[1];
 355       base_pz_ = odom_position_[2];
 356       base_qx_ = odom_orientation_[0];
 357       base_qy_ = odom_orientation_[1];
 358       base_qz_ = odom_orientation_[2];
 359       base_qw_ = odom_orientation_[3];
 360     }
 361     else
 362     {
 363       base_qx_ = imu_qx_;
 364       base_qy_ = imu_qy_;
 365       base_qz_ = imu_qz_;
 366       base_qw_ = imu_qw_;
 367     }
 368   }
 369   timestamp_ns_ = static_cast<double>(time.nanoseconds());
 370   return true;
 371 }

 373 void LeggedReal::imuCallback(const sensor_msgs::msg::Imu::SharedPtr msg) {
 374   const double quaternion_norm_squared =
 375       msg->orientation.x * msg->orientation.x + msg->orientation.y * msg->orientation.y +
 376       msg->orientation.z * msg->orientation.z + msg->orientation.w * msg->orientation.w;
 377   const double values[]{msg->angular_velocity.x, msg->angular_velocity.y, msg->angular_velocity.z,
 378                         quaternion_norm_squared};
 379   if (!std::all_of(std::begin(values), std::end(values), [](double value) { return std::isfinite(value); }) ||
 380       quaternion_norm_squared < 1e-12)
 381   {
 382     RCLCPP_WARN_THROTTLE(nh_->get_logger(), *nh_->get_clock(), 1000, "Ignoring invalid IMU sample");
 383     return;
 384   }
 385   const double inverse_norm = 1.0 / std::sqrt(quaternion_norm_squared);
 386   std::lock_guard<std::mutex> lock(sensor_mutex_);
 387   imu_connection_status_ = true;
 388   last_imu_received_time_ = nh_->now();
 389   for (int i = 0; i < 3; ++i)
 390   {
 391     imu_data_.angular_vel[i] = imu_first_read_ ? values[i]
 392                                                : imu_filter_coeff_real_ * values[i] +
 393                                                      (1.0 - imu_filter_coeff_real_) * imu_data_.prev_angular_vel[i];
 394     imu_data_.prev_angular_vel[i] = imu_data_.angular_vel[i];
 395   }
 396   imu_first_read_ = false;
 397   imu_data_.ori[0] = msg->orientation.x * inverse_norm;
 398   imu_data_.ori[1] = msg->orientation.y * inverse_norm;
 399   imu_data_.ori[2] = msg->orientation.z * inverse_norm;
 400   imu_data_.ori[3] = msg->orientation.w * inverse_norm;
 401 }

 403 void LeggedReal::odomCallback(const nav_msgs::msg::Odometry::SharedPtr msg) {
 404   const auto& position = msg->pose.pose.position;
 405   const auto& orientation = msg->pose.pose.orientation;
 406   const double norm_squared = orientation.x * orientation.x + orientation.y * orientation.y +
 407                               orientation.z * orientation.z + orientation.w * orientation.w;
 408   if (msg->header.frame_id.empty() || msg->child_frame_id.empty() || !std::isfinite(position.x) ||
 409       !std::isfinite(position.y) || !std::isfinite(position.z) || !std::isfinite(norm_squared) || norm_squared < 1e-12)
 410   {
 411     RCLCPP_WARN_THROTTLE(nh_->get_logger(), *nh_->get_clock(), 1000, "Ignoring invalid odometry pose");
 412     return;
 413   }
 414   const double inverse_norm = 1.0 / std::sqrt(norm_squared);
 415   std::lock_guard<std::mutex> lock(sensor_mutex_);
 416   odom_position_[0] = position.x;
 417   odom_position_[1] = position.y;
 418   odom_position_[2] = position.z;
 419   odom_orientation_[0] = orientation.x * inverse_norm;
 420   odom_orientation_[1] = orientation.y * inverse_norm;
 421   odom_orientation_[2] = orientation.z * inverse_norm;
 422   odom_orientation_[3] = orientation.w * inverse_norm;
 423   odom_received_ = true;
 424 }

 426 bool LeggedReal::write(rclcpp::Time, rclcpp::Duration) {
 427   bool imu_failed = false;
 428   const rclcpp::Time current = nh_->now();
 429   const bool motors_enabled = motors_enabled_.load();
 430   {
 431     std::lock_guard<std::mutex> lock(sensor_mutex_);
 432     // No actuator command is published before lifecycle enable completes, so
 433     // the IMU startup watchdog must not expire while its backend is still absent.
 434     if (imu_enabled_ && motors_enabled && !imu_connection_status_)
 435       imu_failed = (current - imu_startup_time_).seconds() >= imu_startup_timeout_;
 436     else if (imu_enabled_ && motors_enabled && (current - last_imu_received_time_).seconds() > imu_connection_timeout_)
 437     {
 438       imu_connection_status_ = false;
 439       imu_failed = true;
 440     }
 441   }
 442 
 443   imu_fault_.store(imu_failed);
 444   publishHardwareFault();
 445   const bool stop = emergency_stop_triggered_.load() || imu_failed || lifecycle_fault_.load();
 446   if (stop != emergency_published_)
 447   {
 448     std_msgs::msg::Bool message;
 449     message.data = stop;
 450     emergency_pub_->publish(message);
 451     emergency_published_ = stop;
 452     if (stop)
 453       RCLCPP_ERROR(nh_->get_logger(),
 454                    "\033[1;31mMotor commands switched to emergency damping: emergency_stop=%s motor_feedback_fault=%s "
 455                    "imu_fault=%s lifecycle_fault=%s\033[0m",
 456                    emergency_stop_triggered_.load() ? "true" : "false", motor_fault_.load() ? "true" : "false",
 457                    imu_failed ? "true" : "false", lifecycle_fault_.load() ? "true" : "false");
 458     else
 459       RCLCPP_INFO(nh_->get_logger(), "Hardware fault cleared; controller state determines motor commands");
 460   }
 461   // Serialize publication with disable, calibration transitions and SETZERO requests.
 462   std::lock_guard<std::mutex> lifecycle_lock(lifecycle_mutex_);
 463   // Serialize turn selection with publication, including the first powered command.
 464   std::lock_guard<std::mutex> feedback_lock(feedback_mutex_);
 465   bool zero_gain_command = !motors_enabled_.load() || calibration_mode_.load() || lifecycle_active_ ||
 466                            lifecycle_fault_.load();
 467   bool active_command = false;
 468   bool damping_command = stop || zero_gain_command;
 469   bool feedback_ready = control_feedback_ready_;
 470   for (std::size_t i = 0; i < feedback_wrap_.size(); ++i)
 471   {
 472     if (!motor_mapping_->entries()[i].enabled) continue;
 473     damping_command = damping_command || feedback_wrap_[i].fault;
 474     const auto& command = joint_motor_commands_[i];
 475     active_command = active_command || command.kp != 0.0 || command.dq_des != 0.0 || command.tau_ff != 0.0;
 476     feedback_ready =
 477         feedback_ready && feedback_wrap_[i].ready() && control_wrap_offsets_[i] == feedback_wrap_[i].offset;
 478   }
 479   if (!damping_command && !feedback_ready)
 480   {
 481     RCLCPP_WARN_THROTTLE(nh_->get_logger(), *nh_->get_clock(), 1000,
 482                          "Publishing zero-gain MIT until all feedback turns are confirmed in the control snapshot");
 483     zero_gain_command = true;
 484     damping_command = true;
 485   }
 486   robot_interfaces::msg::MotorCommand output;
 487   output.stamp = current;
 488   output.sequence_id = ++command_sequence_;
 489   output.control_mode = robot_interfaces::msg::MotorCommand::CONTROL_MODE_MIT;
 490   output.commands.reserve(motor_mapping_->enabledSize());
 491   try
 492   {
 493     for (std::size_t i = 0; i < joint_motor_commands_.size(); ++i)
 494     {
 495       const auto& mapping = motor_mapping_->entries()[i];
 496       if (!mapping.enabled) continue;
 497       const auto& command = joint_motor_commands_[i];
 498       // Disabled/calibration states take precedence over emergency damping.
 499       const auto mapped = motor_mapping_->commandFor(
 500           mapping, damping_command ? joint_motor_states_[i].q : command.q_des,
 501           damping_command ? 0.0 : command.dq_des, damping_command ? 0.0 : command.tau_ff,
 502           damping_command ? 0.0 : command.kp,
 503           zero_gain_command ? 0.0 : (damping_command ? emergency_damping_kd_ : command.kd),
 504           feedback_wrap_[i].offset);
 505       robot_interfaces::msg::MotorTarget target;
 506       target.route.motor_type = static_cast<uint8_t>(i);
 507       target.q = mapped.q;
 508       target.dq = mapped.dq;
 509       target.tau = mapped.tau;
 510       target.kp = mapped.kp;
 511       target.kd = mapped.kd;
 512       output.commands.push_back(target);
 513     }
 514   } catch (const std::exception& error)
 515   {
 516     emergency_stop_triggered_.store(true);
 517     RCLCPP_ERROR_THROTTLE(nh_->get_logger(), *nh_->get_clock(), 1000, "\033[1;31mRejected motor command: %s\033[0m",
 518                           error.what());
 519     return false;
 520   }
 521   if (command_publisher_->get_subscription_count() == 0U)
 522     RCLCPP_WARN_THROTTLE(nh_->get_logger(), *nh_->get_clock(), 1000,
 523                          "\033[1;33mPublishing motor commands without an HPM subscriber\033[0m");
 524   if (!damping_command && active_command)
 525     for (auto& wrap : feedback_wrap_) wrap.locked = true;
 526   command_publisher_->publish(output);
 527   return true;
 528 }

 530 void LeggedReal::publishHardwareFault() {
 531   const bool fault = motor_fault_.load() || imu_fault_.load() || lifecycle_fault_.load();
 532   std::lock_guard<std::mutex> lock(status_mutex_);
 533   if (fault == hardware_fault_published_) return;
 534   std_msgs::msg::Bool message;
 535   message.data = fault;
 536   hardware_fault_pub_->publish(message);
 537   hardware_fault_published_ = fault;
 538 }

 540 void LeggedReal::publishMotorsReady(bool ready) {
 541   std::lock_guard<std::mutex> lock(status_mutex_);
 542   if (ready == motors_ready_state_) return;
 543   std_msgs::msg::Bool message;
 544   message.data = ready;
 545   motors_ready_pub_->publish(message);
 546   motors_ready_state_ = ready;
 547   if (ready)
 548     RCLCPP_INFO(nh_->get_logger(),
 549                 "Motor command path ready for %zu enabled motors after publishing the HPM enable command",
 550                 motor_mapping_->enabledSize());
 551   else if (motors_enabled_.load())
 552     RCLCPP_ERROR(nh_->get_logger(), "\033[1;31mMotor readiness lost; inspect preceding per-motor errors\033[0m");
 553   else
 554     RCLCPP_INFO(nh_->get_logger(), "Motors disabled; readiness is false");
 555 }

 557 bool LeggedReal::requestAllMotorsEnabled(bool enabled) {
 558   if (!motor_mapping_)
 559   {
 560     RCLCPP_ERROR(nh_->get_logger(), "Motor lifecycle request blocked before motor mapping is configured");
 561     return false;
 562   }
 563   std::lock_guard<std::mutex> lock(lifecycle_mutex_);
 564   if (enabled && (calibration_mode_.load() || zero_pending_ || zero_failed_))
 565   {
 566     RCLCPP_ERROR(nh_->get_logger(), "Motor enable blocked by active or failed calibration");
 567     return false;
 568   }
 569   lifecycle_target_enabled_ = enabled;
 570   lifecycle_active_ = true;
 571   lifecycle_started_ = nh_->now();
 572   lifecycle_fault_.store(false);
 573   if (!enabled)
 574   {
 575     motors_enabled_.store(false);
 576     publishMotorsReady(false);
 577   }
 578   publishHardwareFault();
 579   return true;
 580 }

 582 void LeggedReal::lifecycleTick() {
 583   std::lock_guard<std::mutex> lock(lifecycle_mutex_);
 584   if (!lifecycle_active_) return;
 585   const rclcpp::Time current = nh_->now();
 586   if (enable_publisher_->get_subscription_count() == 0U)
 587   {
 588     if ((current - lifecycle_started_).seconds() > 20.0)
 589     {
 590       lifecycle_active_ = false;
 591       motors_enabled_.store(false);
 592       lifecycle_fault_.store(true);
 593       publishMotorsReady(false);
 594       publishHardwareFault();
 595       RCLCPP_ERROR(nh_->get_logger(),
 596                    "\033[1;31mTimed out waiting for /actuator/motor/enable subscriber; motors remain disabled\033[0m");
 597     }
 598     else
 599       RCLCPP_WARN_THROTTLE(nh_->get_logger(), *nh_->get_clock(), 1000,
 600                            "\033[1;33mWaiting for HPM subscriber on /actuator/motor/enable\033[0m");
 601     return;
 602   }
 603 
 604   robot_interfaces::msg::MotorEnableCommand command;
 605   command.stamp = current;
 606   command.sequence_id = ++enable_sequence_;
 607   command.motor_types.reserve(motor_mapping_->enabledSize());
 608   command.enabled.reserve(motor_mapping_->enabledSize());
 609   for (std::size_t i = 0; i < motor_mapping_->size(); ++i)
 610   {
 611     if (!motor_mapping_->entries()[i].enabled) continue;
 612     command.motor_types.push_back(static_cast<uint8_t>(i));
 613     command.enabled.push_back(lifecycle_target_enabled_);
 614   }
 615   lifecycle_active_ = false;
 616   {
 617     // Do not count feedback received before this enable publication toward the
 618     // first complete post-enable feedback round.
 619     std::lock_guard<std::mutex> feedback_lock(feedback_mutex_);
 620     if (lifecycle_target_enabled_)
 621     {
 622       std::fill(feedback_updated_.begin(), feedback_updated_.end(), false);
 623       for (auto& wrap : feedback_wrap_)
 624       {
 625         wrap.stable_samples = 0U;
 626         wrap.error_code = 0U;
 627         wrap.error_samples = 0U;
 628       }
 629       for (auto& feedback : latest_feedback_) feedback.valid = false;
 630       reset_feedback_watchdogs_.store(true);
 631     }
 632     else
 633     {
 634       for (auto& wrap : feedback_wrap_) wrap = system_command_arbiter::FeedbackWrapState{};
 635       for (auto& feedback : latest_feedback_) feedback.valid = false;
 636       std::fill(feedback_updated_.begin(), feedback_updated_.end(), false);
 637     }
 638     enable_publisher_->publish(command);
 639     motors_enabled_.store(lifecycle_target_enabled_);
 640   }
 641   if (lifecycle_target_enabled_)
 642   {
 643     std::lock_guard<std::mutex> sensor_lock(sensor_mutex_);
 644     if (!imu_connection_status_) imu_startup_time_ = current;
 645   }
 646   lifecycle_fault_.store(false);
 647   publishHardwareFault();
 648   RCLCPP_INFO(nh_->get_logger(), "Published HPM %s command for %zu motors",
 649               lifecycle_target_enabled_ ? "enable" : "disable", motor_mapping_->enabledSize());
 650 }
