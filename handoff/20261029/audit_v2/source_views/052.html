  36 def static_shape(item, kind, onnx):
  37     if item.type.tensor_type.elem_type != onnx.TensorProto.FLOAT:
  38         raise ValueError(f"{kind} '{item.name}' must be float32")
  39     dims = []
  40     for dim in item.type.tensor_type.shape.dim:
  41         if dim.dim_value <= 0:
  42             raise ValueError(f"{kind} '{item.name}' must have a fully static shape")
  43         dims.append(dim.dim_value)
  44     return tuple(dims)

  47 def check_reference_model(path):
  48     import onnx
  49 
  50     model = onnx.load(str(path))
  51     onnx.checker.check_model(model)
  52     inputs = list(model.graph.input)
  53     outputs = list(model.graph.output)
  54     if len(inputs) != 1 or inputs[0].name != "time_step" or static_shape(inputs[0], "input", onnx) != (1, 1):
  55         raise ValueError("reference model must have float32 time_step[1,1]")
  56     if len(outputs) != 6 or {item.name for item in outputs} != set(REFERENCE_OUTPUTS):
  57         raise ValueError(f"reference model must have outputs {list(REFERENCE_OUTPUTS)}")
  58     for item in outputs:
  59         if static_shape(item, "output", onnx) != REFERENCE_OUTPUTS[item.name]:
  60             raise ValueError(f"reference output {item.name} must be float32 {REFERENCE_OUTPUTS[item.name]}")
  61     meta = {item.key: item.value for item in model.metadata_props}
  62     if meta.get("quaternion_order") != "wxyz" or \
  63             meta.get("time_step_semantics") != "zero_based_frame_index_clamped" or \
  64             float(meta.get("fps", 0)) != 50 or not 0 < int(meta.get("num_frames", 0)) <= 100000:
  65         raise ValueError("reference model needs valid 50 Hz frame metadata and clamped wxyz semantics")
  66     joints = meta["joint_names"].split(",")
  67     bodies = meta["body_names"].split(",")
  68     if len(joints) != 21 or len(set(joints)) != 21 or len(bodies) != 23 or len(set(bodies)) != 23 or \
  69             "base_link" not in bodies:
  70         raise ValueError("invalid reference joint/body metadata")
  71     return [("time_step", (1, 1))], [(item.name, REFERENCE_OUTPUTS[item.name]) for item in outputs], meta

  74 def pack_reference_gathers(model):
  75     import numpy as np
  76     from onnx import helper, numpy_helper
  77 
  78     count = int(next(item.value for item in model.metadata_props if item.key == "num_frames"))
  79     tables = {item.name: numpy_helper.to_array(item) for item in model.graph.initializer}
  80     gathers = [node for node in model.graph.node if node.op_type == "Gather"]
  81     by_output = {node.output[0]: node for node in gathers if len(node.output) == 1}
  82     if len(gathers) != 6 or set(by_output) != set(REFERENCE_OUTPUTS):
  83         raise ValueError("reference compatibility rewrite requires six direct table Gather outputs")
  84     indices = {node.input[1] for node in gathers}
  85     if len(indices) != 1:
  86         raise ValueError("reference Gather branches must share the same frame index")
  87     packed, widths, table_names = [], [], set()
  88     for name, shape in REFERENCE_OUTPUTS.items():
  89         node = by_output[name]
  90         if node.input[0] not in tables or any(attr.name == "axis" and attr.i != 0 for attr in node.attribute):
  91             raise ValueError(f"{name}: expected a constant frame table on axis 0")
  92         table = tables[node.input[0]]
  93         if table.dtype != np.float32 or table.shape != (count,) + shape[1:] or not np.isfinite(table).all():
  94             raise ValueError(f"{name}: invalid frame table shape, type or value")
  95         packed.append(table.reshape(count, -1))
  96         widths.append(math.prod(shape))
  97         table_names.add(node.input[0])
  98 
  99     # Avoid Toolkit2 2.3.2 merging parallel Gather branches with different widths (3 vs 4).
 100     prefix = "wbc_rknn_reference_"
 101     existing_names = {name for node in model.graph.node for name in list(node.input) + list(node.output)}
 102     if any(name.startswith(prefix) for name in existing_names):
 103         raise ValueError("reference graph already uses compatibility rewrite names")
 104     initializers = [item for item in model.graph.initializer if item.name not in table_names]
 105     initializers.extend([
 106         numpy_helper.from_array(np.concatenate(packed, axis=1), prefix + "table"),
 107         numpy_helper.from_array(np.asarray(widths, dtype=np.int64), prefix + "widths"),
 108     ])
 109     nodes = [node for node in model.graph.node if node.op_type != "Gather"]
 110     nodes.append(helper.make_node("Gather", [prefix + "table", next(iter(indices))],
 111                                   [prefix + "frame"], axis=0, name=prefix + "gather"))
 112     split_names = [prefix + name for name in REFERENCE_OUTPUTS]
 113     nodes.append(helper.make_node("Split", [prefix + "frame", prefix + "widths"],
 114                                   split_names, axis=1, name=prefix + "split"))
 115     for name, shape in REFERENCE_OUTPUTS.items():
 116         shape_name = prefix + name + "_shape"
 117         initializers.append(numpy_helper.from_array(np.asarray(shape, dtype=np.int64), shape_name))
 118         nodes.append(helper.make_node("Reshape", [prefix + name, shape_name], [name], name=prefix + name + "_reshape"))
 119     del model.graph.node[:]
 120     model.graph.node.extend(nodes)
 121     del model.graph.initializer[:]
 122     model.graph.initializer.extend(initializers)
 123     del model.graph.value_info[:]
 124     print(f"packed six reference Gather branches into one Gather ({sum(widths)} floats/frame)")

 148 def verify_reference(rknn, onnx_path, output_specs, meta, args):
 149     import numpy as np
 150     import onnxruntime as ort
 151 
 152     session = ort.InferenceSession(str(onnx_path), providers=["CPUExecutionProvider"])
 153     ret = rknn.init_runtime(**runtime_options(args))
 154     if ret != 0:
 155         raise RuntimeError(f"rknn.init_runtime failed, ret={ret}")
 156     count = int(meta["num_frames"])
 157     names = [name for name, _ in output_specs]
 158     maximum_errors = np.zeros(len(names), dtype=np.float64)
 159     error_sums = np.zeros(len(names), dtype=np.float64)
 160     worst_latency = 0.0
 161     # Check every frame, plus the clamped boundary behavior used by the controller.
 162     steps = [-8, -1] + list(range(count)) + [count, count + 8]
 163     for step in steps:
 164         time_step = np.array([[step]], dtype=np.float32)
 165         expected = session.run(names, {"time_step": time_step})
 166         started = time.perf_counter()
 167         actual = rknn.inference(inputs=[time_step])
 168         worst_latency = max(worst_latency, time.perf_counter() - started)
 169         if actual is None or len(actual) != len(names):
 170             raise RuntimeError("reference RKNN must return all six outputs")
 171         for i, (name, shape) in enumerate(output_specs):
 172             value = np.asarray(actual[i], dtype=np.float32)
 173             reference = np.asarray(expected[i], dtype=np.float32)
 174             if value.size != math.prod(shape) or not np.isfinite(value).all() or \
 175                     reference.shape != shape or not np.isfinite(reference).all():
 176                 raise RuntimeError(f"invalid reference output {name} at frame {step}")
 177             error = np.abs(value.reshape(shape) - reference)
 178             maximum_errors[i] = max(maximum_errors[i], float(error.max()))
 179             error_sums[i] += float(error.mean())
 180             if error.max() > args.max_abs_error:
 181                 raise RuntimeError(f"reference verification failed: frame={step}, output={name}, max_abs={error.max()}")
 182         if step % 50 == 0 or step == steps[-1]:
 183             print(f"  reference frame={step}, maximum errors={maximum_errors.tolist()}")
 184     for i, name in enumerate(names):
 185         print(f"{name}: max_abs={maximum_errors[i]:.6g}, mean_abs={error_sums[i] / len(steps):.6g}")
 186     print(f"reference verification passed: {len(steps)} queries, worst inference ms={worst_latency * 1000:.3f}")
 187     return {"queries": len(steps), "max_abs_per_output": dict(zip(names, maximum_errors.tolist())),
 188             "worst_ms": worst_latency * 1000.0}

 191 def main():
 192     args = parse_args()
 193     onnx_path = args.onnx.resolve()
 194     output_path = (args.output or onnx_path.with_suffix(".rknn")).resolve()
 195     if not onnx_path.is_file() or onnx_path.suffix.lower() != ".onnx":
 196         raise ValueError(f"ONNX model not found: {onnx_path}")
 197     if output_path.suffix.lower() != ".rknn":
 198         raise ValueError("output filename must end with .rknn")
 199     if not math.isfinite(args.max_abs_error) or args.max_abs_error <= 0:
 200         raise ValueError("--max-abs-error must be finite and positive")
 201     runtime_options(args)
 202     input_specs, output_specs, reference_meta = check_reference_model(onnx_path)
 203     print("inputs: " + ", ".join(f"{name}{shape}" for name, shape in input_specs))
 204     print("outputs: " + ", ".join(f"{name}{shape}" for name, shape in output_specs))
 205 
 206     with rknn_compatible_model(onnx_path) as rknn_onnx_path:
 207         from rknn.api import RKNN
 208 
 209         rknn = RKNN(verbose=True)
 210         try:
 211             print(f"build: config={rknn_build_config()} do_quantization=False; output={output_path}")
 212             ret = rknn.config(**rknn_build_config())
 213             if ret != 0:
 214                 raise RuntimeError(f"rknn.config failed, ret={ret}")
 215             ret = rknn.load_onnx(model=str(rknn_onnx_path))
 216             if ret != 0:
 217                 raise RuntimeError(f"rknn.load_onnx failed, ret={ret}")
 218             ret = rknn.build(do_quantization=False)
 219             if ret != 0:
 220                 raise RuntimeError(f"rknn.build failed, ret={ret}")
 221             output_path.parent.mkdir(parents=True, exist_ok=True)
 222             with tempfile.TemporaryDirectory(prefix="wbc_rknn_", dir=output_path.parent) as directory:
 223                 candidate = Path(directory) / output_path.name
 224                 ret = rknn.export_rknn(str(candidate))
 225                 if ret != 0:
 226                     raise RuntimeError(f"rknn.export_rknn failed, ret={ret}")
 227                 verification = None if args.build_only else verify_reference(rknn, onnx_path, output_specs, reference_meta, args)
 228                 candidate.replace(output_path)
 229                 conversion_record(onnx_path, output_path, args, verification)
 230             print(f"exported: {output_path}; validation={'pending C++ deployment' if args.build_only else 'Python verification completed'}")
 231         finally:
 232             rknn.release()
 233     return 0
