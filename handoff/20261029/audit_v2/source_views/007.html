 174     if imitation_policies:
 175         dynamic_imitation_actions.append(
 176             LogInfo(
 177                 msg=(
 178                     '[one_start] Starting dynamic imitation worker '
 179                     f'({len(imitation_policies)} policies, {len(preload_imitation_policies)} preloaded)'
 180                 )
 181             )
 182         )
 183         dynamic_imitation_actions.append(
 184             Node(
 185                 package='legged_deploy',
 186                 executable='rl_controller',
 187                 name='rl_controller_imitation',
 188                 output='screen',
 189                 parameters=[
 190                     {'robot_type': robot_type},
 191                     {
 192                         'robot_cfg_path': PathJoinSubstitution(
 193                             [FindPackageShare('legged_deploy'), 'policy', robot_type, subpolicy, 'rl_config.yaml']
 194                         )
 195                     },
 196                     {'subpolicy': ''},
 197                     {'policy_type': 'rl_imitation'},
 198                     {'dynamic_policy_switch_enabled': True},
 199                     {'dynamic_imitation_idle_frequency': 100},
 200                     {'policy_root_path': policy_root_path},
 201                     {
 202                         'rl_controller_config_path': PathJoinSubstitution(
 203                             [FindPackageShare('legged_deploy'), 'config', 'rl_controller_config.yaml']
 204                         )
 205                     },
 206                     {'available_imitation_policies': imitation_policies},
 207                     {'preload_imitation_policies': preload_imitation_policies},
 208                 ],
 209             )
 210         )
 211     else:
 212         dynamic_imitation_actions.append(
 213             LogInfo(
 214                 msg=(
 215                     '[one_start] No enabled joystick/keyboard rl_imitation action '
 216                     f'({len(action_manifest.sequence_paths)} action sequence file(s) validated).'
 217                 )
 218             )
 219         )
 220 
 221     bag_record = GroupAction(
 222         condition=IfCondition(bag_enabled),
 223         actions=[
 224             ExecuteProcess(
 225                 cmd=[
 226                     'bash',
 227                     '-c',
 228                     PythonExpression(
 229                         [
 230                             "'mkdir -p ",
 231                             bag_dir,
 232                             " && cd ",
 233                             bag_dir,
 234                             " && ros2 bag record -o ",
 235                             bag_prefix,
 236                             " ",
 237                             bag_topics,
 238                             "'",
 239                         ]
 240                     ),
 241                 ],
 242                 output='screen',
 243             )
 244         ],
 245     )
 246 
 247     manifest_log = LogInfo(
 248         msg=(
 249             '[one_start] Validated '
 250             f'{len(action_manifest.sequence_paths)} action sequence file(s); '
 251             f'{len(action_manifest.rl_imitation_sequences)} use rl_imitation.'
 252         )
 253     )
 254 
 255     return [
 256         manifest_log,
 257         rviz_launch,
 258         joy_launch,
 259     #    radar_driver_launch,
 260         radar_planner_launch,
 261         rl_launch,
 262         *wbc_actions,
 263         *dynamic_imitation_actions,
 264         bag_record,
 265     ]
