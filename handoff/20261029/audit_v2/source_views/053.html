  42 def statistics(state):
  43     result = {}
  44     for name in ("policy", "command_short"):
  45         count = state[name + ".count"]
  46         if not np.isfinite(float(count)) or float(count) <= 0:
  47             raise ValueError("Invalid normalization count")
  48         mean = state[name + ".sum"] / count
  49         std = (state[name + ".ssq"] / count - mean.square()).clamp_min(1e-5).sqrt()
  50         result[name] = (array(mean), array(std))
  51     return result

  58 def actor_forward(actor, linear_names, norm_names, stats, policy, command):
  59     import torch
  60     from torch.nn import functional as F
  61     p = (torch.from_numpy(policy) - torch.from_numpy(stats["policy"][0])) / torch.from_numpy(stats["policy"][1])
  62     c = (torch.from_numpy(command) - torch.from_numpy(stats["command_short"][0])) / torch.from_numpy(stats["command_short"][1])
  63     value, norm_index = torch.cat((p, c)), 0
  64     for index, name in enumerate(linear_names):
  65         if index == 3:
  66             value = torch.cat((p, value))
  67         value = F.linear(value, actor[name + ".weight"], actor[name + ".bias"])
  68         if index in NORMALIZED_LAYERS:
  69             norm = norm_names[norm_index]
  70             value = F.mish(F.layer_norm(value, (1024,), actor[norm + ".weight"], actor[norm + ".bias"], eps=1e-5))
  71             norm_index += 1
  72     return value.detach().numpy()

  75 def topology(base, model):
  76     from onnx import numpy_helper
  77     actor = base["actor"]
  78     linears = [k[:-7] for k, v in actor.items() if k.endswith(".weight") and v.ndim == 2]
  79     norms = [k[:-7] for k, v in actor.items() if k.endswith(".weight") and v.ndim == 1]
  80     if [tuple(actor[k + ".weight"].shape) for k in linears] != DIMS or len(norms) != 5:
  81         raise ValueError("Unsupported native L0 actor topology")
  82     initializers = {x.name: numpy_helper.to_array(x) for x in model.graph.initializer}
  83     matmuls = [n for n in model.graph.node if n.op_type == "MatMul"]
  84     ln_nodes = [n for n in model.graph.node if n.op_type == "LayerNormalization"]
  85     if len(matmuls) != 7 or len(ln_nodes) != 5 or len([n for n in model.graph.node if n.op_type == "Mish"]) != 5:
  86         raise ValueError("Expected the native seven-Linear / five-LayerNorm / five-Mish graph")
  87     bindings, norm_bindings = [], []
  88     for name in linears:
  89         weight = array(actor[name + ".weight"]).T
  90         matches = [n for n in matmuls if n.input[1] in initializers and
  91                    np.array_equal(initializers[n.input[1]], weight)]
  92         if len(matches) != 1:
  93             raise ValueError(f"Cannot uniquely bind ONNX Linear to checkpoint: {name}")
  94         node = matches[0]
  95         consumers = [n for n in model.graph.node if node.output[0] in n.input]
  96         if len(consumers) != 1 or consumers[0].op_type != "Add":
  97             raise ValueError("Unsupported Linear bias topology")
  98         bias = consumers[0]
  99         other = next(x for x in bias.input if x != node.output[0])
 100         if other not in initializers or not np.array_equal(initializers[other], array(actor[name + ".bias"])):
 101             raise ValueError("Base ONNX bias differs from checkpoint")
 102         bindings.append((node.name, bias.name, node.output[0]))
 103     if any(not n.name for n in matmuls + ln_nodes) or len({n.name for n in model.graph.node}) != len(model.graph.node):
 104         raise ValueError("Native graph node names must be unique and non-empty")
 105     for norm_index, (name, node) in enumerate(zip(norms, ln_nodes)):
 106         bias_name = bindings[NORMALIZED_LAYERS[norm_index]][1]
 107         bias_node = next(n for n in model.graph.node if n.name == bias_name)
 108         if node.input[0] != bias_node.output[0]:
 109             raise ValueError("LayerNorm/Linear connectivity differs from the native actor")
 110         for index, suffix in ((1, ".weight"), (2, ".bias")):
 111             if not np.array_equal(initializers[node.input[index]], array(actor[name + suffix])):
 112                 raise ValueError("Base ONNX LayerNorm differs from checkpoint")
 113         norm_bindings.append(node.name)
 114     stats = statistics(base["vecnorms"])
 115     for name, (mean, std) in stats.items():
 116         sub = next(n for n in model.graph.node if n.op_type == "Sub" and n.input[0] == name)
 117         div = next(n for n in model.graph.node if n.op_type == "Div" and n.input[0] == sub.output[0])
 118         if not np.array_equal(initializers[sub.input[1]], mean) or not np.array_equal(initializers[div.input[1]], std):
 119             raise ValueError("Base ONNX normalization differs from checkpoint")
 120     return linears, norms, bindings, norm_bindings, stats

 123 def schema_for(linears, norms, actor):
 124     result = []
 125     for i, name in enumerate(linears):
 126         for suffix, short in ((".lora_A", "a"), (".lora_B", "b"), (".bias", "bias")):
 127             shape = [16, DIMS[i][1]] if short == "a" else [DIMS[i][0], 16] if short == "b" else [DIMS[i][0]]
 128             result.append({"name": f"lora_{i}_{short}", "actor_key": name + suffix, "shape": shape})
 129     for i, name in enumerate(norms):
 130         for suffix, short in ((".weight", "gamma"), (".bias", "beta")):
 131             result.append({"name": f"norm_{i}_{short}", "actor_key": name + suffix, "shape": list(actor[name + suffix].shape)})
 132     return result

 135 def residual_nodes(prefix, value, base_output, index):
 136     from onnx import helper
 137     a, b = f"lora_{index}_a", f"lora_{index}_b"
 138     return [
 139         helper.make_node("Transpose", [a], [prefix + "at"], perm=[1, 0]),
 140         helper.make_node("Transpose", [b], [prefix + "bt"], perm=[1, 0]),
 141         helper.make_node("MatMul", [value, prefix + "at"], [prefix + "low"]),
 142         helper.make_node("MatMul", [prefix + "low", prefix + "bt"], [prefix + "delta"]),
 143         helper.make_node("Add", [base_output, prefix + "delta"], [prefix + "sum"]),
 144     ]

 147 def affine_norm_nodes(prefix, input_value, output_value, index, original=None):
 148     from onnx import helper
 149     node = copy.deepcopy(original) if original is not None else helper.make_node(
 150         "LayerNormalization", [input_value, prefix + "ones", prefix + "zeros"], [prefix + "normalized"], epsilon=1e-5)
 151     node.input[:] = [input_value, prefix + "ones", prefix + "zeros"]
 152     node.output[0] = prefix + "normalized"
 153     return [node, helper.make_node("Mul", [node.output[0], f"norm_{index}_gamma"], [prefix + "scaled"]),
 154             helper.make_node("Add", [prefix + "scaled", f"norm_{index}_beta"], [output_value])]

 157 def dynamic_model(source, schema, bindings, norm_bindings):
 158     from onnx import helper, numpy_helper
 159     model = copy.deepcopy(source)
 160     linear_by_node = {entry[0]: (i, entry) for i, entry in enumerate(bindings)}
 161     bias_by_node = {entry[1]: (i, entry) for i, entry in enumerate(bindings)}
 162     norms = {name: i for i, name in enumerate(norm_bindings)}
 163     nodes, constants = [], []
 164     for node in model.graph.node:
 165         if node.name in linear_by_node:
 166             i, _ = linear_by_node[node.name]
 167             nodes.append(node)
 168             nodes.extend(residual_nodes(f"wbc_lora_{i}_", node.input[0], node.output[0], i))
 169         elif node.name in bias_by_node:
 170             i, entry = bias_by_node[node.name]
 171             node.input[:] = [f"wbc_lora_{i}_sum", f"lora_{i}_bias"]
 172             nodes.append(node)
 173         elif node.name in norms:
 174             i = norms[node.name]
 175             prefix = f"wbc_norm_{i}_"
 176             constants += [numpy_helper.from_array(np.ones(1024, np.float32), prefix + "ones"),
 177                           numpy_helper.from_array(np.zeros(1024, np.float32), prefix + "zeros")]
 178             nodes.extend(affine_norm_nodes(prefix, node.input[0], node.output[0], i, node))
 179         else:
 180             nodes.append(node)
 181     del model.graph.node[:]
 182     model.graph.node.extend(nodes)
 183     used = {x for n in nodes for x in n.input}
 184     kept = [x for x in model.graph.initializer if x.name in used] + constants
 185     del model.graph.initializer[:]
 186     model.graph.initializer.extend(kept)
 187     del model.graph.value_info[:]
 188     for p in schema:
 189         model.graph.input.append(helper.make_tensor_value_info(p["name"], 1, p["shape"]))
 190     return model

 193 def probe_model(base_actor, linears, index, schema):
 194     from onnx import helper, numpy_helper
 195     prefix, value = f"probe_{index}_", "x"
 196     nodes = [helper.make_node("MatMul", [value, "base_weight"], [prefix + "base"])]
 197     nodes += residual_nodes(prefix, value, prefix + "base", index)
 198     nodes += [helper.make_node("Add", [prefix + "sum", f"lora_{index}_bias"], [prefix + "linear"])]
 199     constants = [numpy_helper.from_array(array(base_actor[linears[index] + ".weight"]).T.copy(), "base_weight")]
 200     names = [f"lora_{index}_{s}" for s in ("a", "b", "bias")]
 201     if index in NORMALIZED_LAYERS:
 202         norm_index = NORMALIZED_LAYERS.index(index)
 203         names += [f"norm_{norm_index}_{s}" for s in ("gamma", "beta")]
 204         constants += [numpy_helper.from_array(np.ones(1024, np.float32), prefix + "ones"),
 205                       numpy_helper.from_array(np.zeros(1024, np.float32), prefix + "zeros")]
 206         nodes += affine_norm_nodes(prefix, prefix + "linear", prefix + "affine", norm_index)
 207         nodes += [helper.make_node("Mish", [prefix + "affine"], ["action"])]
 208     else:
 209         nodes += [helper.make_node("Identity", [prefix + "linear"], ["action"])]
 210     inputs = [{"name": "x", "shape": [DIMS[index][1]], "dtype": "float32", "role": "observation"}]
 211     inputs += [{"name": p["name"], "shape": p["shape"], "dtype": "float32", "role": "adapter"}
 212                for p in schema if p["name"] in names]
 213     graph = helper.make_graph(nodes, f"WBC LoRA layer {index}",
 214         [helper.make_tensor_value_info(x["name"], 1, x["shape"]) for x in inputs],
 215         [helper.make_tensor_value_info("action", 1, [DIMS[index][0]])], constants)
 216     model = helper.make_model(graph, opset_imports=[helper.make_opsetid("", 19)])
 217     model.ir_version = 9
 218     return model, inputs

 221 def adapter_state(adapter, base, base_hash, linears, schema):
 222     import torch
 223     if adapter.get("format") != "mimic_lite_lora_adapter_v2" or adapter.get("base_sha256") != base_hash:
 224         raise ValueError("Adapter format/Base checksum mismatch")
 225     if adapter["rank"] != 16 or adapter["alpha"] != 16 or adapter["layers"] != linears:
 226         raise ValueError("Adapter layers/rank/alpha mismatch")
 227     values = adapter["actor"]
 228     expected = {p["actor_key"] for p in schema}
 229     extra = set(values) - expected
 230     if any(not name.endswith("actor_std") for name in extra) or not expected <= set(values):
 231         raise ValueError("Unexpected/missing actor adapter parameters")
 232     for p in schema:
 233         if list(values[p["actor_key"]].shape) != p["shape"]:
 234             raise ValueError("Adapter tensor shape mismatch")
 235         array(values[p["actor_key"]])
 236     if normalization_hash(statistics(adapter["vecnorms"])) != normalization_hash(statistics(base["vecnorms"])):
 237         raise ValueError("Adapter normalization differs from the shared Base graph")
 238     merged = {k: v.detach().cpu().clone() for k, v in base["actor"].items()}
 239     for p in schema:
 240         if not p["actor_key"].endswith((".lora_A", ".lora_B")):
 241             merged[p["actor_key"]] = values[p["actor_key"]].detach().cpu().clone()
 242     for name in linears:
 243         merged[name + ".weight"] += values[name + ".lora_B"] @ values[name + ".lora_A"]
 244     return merged

 247 def export_adapter(directory, action, adapter, graph, parameter_source="lora"):
 248     payload, tensors = bytearray(), []
 249     for p in graph["parameters"]:
 250         payload.extend(b"\0" * ((-len(payload)) % 64))
 251         value = array(adapter["actor"][p["actor_key"]])
 252         with np.errstate(over="ignore"):
 253             value = value.astype("<f2")
 254         if not np.isfinite(value).all():
 255             raise ValueError("Adapter value is not representable in FP16")
 256         data = value.tobytes(order="C")
 257         tensors.append(dict(p, offset_bytes=len(payload), size_bytes=len(data)))
 258         payload.extend(data)
 259     directory.mkdir(parents=True)
 260     (directory / "weights.bin").write_bytes(payload)
 261     manifest = {"format": ADAPTER_FORMAT, "action_id": action, "rank": 16, "alpha": 16,
 262                 "parameter_source": parameter_source,
 263                 **{k: graph[k] for k in ("base_checkpoint_sha256", "normalization_sha256", "parameter_schema_sha256")},
 264                 "weights": {"path": "weights.bin", "dtype": "float16-le", "size_bytes": len(payload),
 265                             "sha256": sha256(directory / "weights.bin")}, "tensors": tensors}
 266     write_json(directory / "manifest.json", manifest)

 291 def fp16_initializers(model):
 292     """Store native float initializers in FP16 while retaining the FP32 graph ABI."""
 293     import onnx
 294     from onnx import helper, numpy_helper
 295 
 296     model = copy.deepcopy(model)
 297     names = {x.name for x in model.graph.initializer}
 298     names.update(x for n in model.graph.node for x in list(n.input) + list(n.output))
 299     prefix = "wbc_fp16_storage_"
 300     if any(name.startswith(prefix) for name in names):
 301         raise ValueError("Model already contains FP16 storage casts")
 302     if any(a.type in (onnx.AttributeProto.GRAPH, onnx.AttributeProto.GRAPHS)
 303            for node in model.graph.node for a in node.attribute):
 304         raise ValueError("FP16 storage conversion does not support subgraphs")
 305     casts = []
 306     for index, tensor in enumerate(model.graph.initializer):
 307         if tensor.data_location == onnx.TensorProto.EXTERNAL:
 308             raise ValueError("External tensor data is not supported")
 309         if tensor.data_type != onnx.TensorProto.FLOAT:
 310             continue
 311         original_name = tensor.name
 312         values = numpy_helper.to_array(tensor)
 313         with np.errstate(over="ignore"):
 314             reduced = values.astype(np.float16)
 315         if not np.isfinite(values).all() or not np.isfinite(reduced).all():
 316             raise ValueError(f"Initializer cannot be represented in FP16: {original_name}")
 317         storage_name = prefix + str(index)
 318         tensor.CopyFrom(numpy_helper.from_array(reduced, storage_name))
 319         casts.append(helper.make_node("Cast", [storage_name], [original_name],
 320                                       name=storage_name + "_cast", to=onnx.TensorProto.FLOAT))
 321     nodes = casts + list(model.graph.node)
 322     del model.graph.node[:]
 323     model.graph.node.extend(nodes)
 324     onnx.checker.check_model(model, full_check=True)
 325     return model

 328 def finalize_fp16(stage, graph, report):
 329     """Convert and verify the final delivery against the existing FP32 oracles."""
 330     import onnx
 331     import onnxruntime as ort
 332     from convert_onnx_to_rknn_rk3588_wbc_reference import check_reference_model
 333     from wbc_lora_assets import check_graph, read_json
 334 
 335     options = ort.SessionOptions()
 336     options.intra_op_num_threads = options.inter_op_num_threads = 1
 337     final_report = {"initializer_dtype": "float16", "application_dtype": "float32",
 338                     "policy_max_abs_error": 0.05, "reference_max_abs_error": 0.02,
 339                     "graphs": [], "references": []}
 340     for manifest_name, current in [("base_manifest.json", graph)] + [
 341             (name, read_json(stage / name)) for name in graph["probes"]]:
 342         path = asset_path(stage, current["model"]["onnx"])
 343         onnx.save(fp16_initializers(onnx.load(str(path), load_external_data=False)), str(path))
 344         current["model"]["sha256"] = sha256(path)
 345         current["initializer_dtype"] = "float16"
 346         write_json(stage / manifest_name, current)
 347         check_graph(path, stage / manifest_name)
 348         session = ort.InferenceSession(str(path), options, providers=["CPUExecutionProvider"])
 349         errors, first_feed, first_output = [], None, None
 350         with np.load(stage / current["verification_data"]) as observations, \
 351                 np.load(stage / current["verification_expected"]) as expected:
 352             for action in current["actions"]:
 353                 action_id = action["action_id"]
 354                 packed = load_adapter(asset_path(stage, action["adapter_manifest"]), current, action_id)
 355                 maximum = 0.0
 356                 for i, target in enumerate(expected[action_id]):
 357                     feed = graph_feeds(current, {key: observations[key][i] for key in observations.files}, packed)
 358                     actual = session.run(["action"], feed)[0]
 359                     error = float(np.max(np.abs(actual - target)))
 360                     if not np.isfinite(actual).all() or not np.isfinite(error) or error > 0.05:
 361                         raise ValueError(f"Final FP16 policy/probe tolerance exceeded: {manifest_name}/{action_id}: {error}")
 362                     maximum = max(maximum, error)
 363                     if first_feed is None:
 364                         first_feed, first_output = feed, actual.copy()
 365                 errors.append({"action_id": action_id, "max_abs": maximum})
 366         np.testing.assert_array_equal(session.run(["action"], first_feed)[0], first_output)
 367         final_report["graphs"].append({"manifest": manifest_name, "actions": errors,
 368                                        "switch_repeatability": "passed"})
 369         print(f"FP16 verified: {manifest_name}, {len(errors)} actions", flush=True)
 370     for action in graph["actions"]:
 371         path = asset_path(stage, action["reference_onnx"])
 372         original = ort.InferenceSession(str(path), options, providers=["CPUExecutionProvider"])
 373         onnx.save(fp16_initializers(onnx.load(str(path), load_external_data=False)), str(path))
 374         _, _, meta = check_reference_model(path)
 375         converted = ort.InferenceSession(str(path), options, providers=["CPUExecutionProvider"])
 376         count, maximum = int(meta["num_frames"]), 0.0
 377         for frame in [-1] + list(range(count)) + [count]:
 378             feed = {"time_step": np.asarray([[frame]], dtype=np.float32)}
 379             for actual, target in zip(converted.run(None, feed), original.run(None, feed)):
 380                 error = float(np.max(np.abs(actual - target)))
 381                 if not np.isfinite(actual).all() or not np.isfinite(error) or error > 0.02:
 382                     raise ValueError(f"Final FP16 reference tolerance exceeded: {action['action_id']}/{frame}: {error}")
 383                 maximum = max(maximum, error)
 384         action["reference_onnx_sha256"] = sha256(path)
 385         final_report["references"].append({"action_id": action["action_id"], "frames_checked": count + 2,
 386                                             "max_abs": maximum})
 387     # Probe catalogs share the same action/reference identities as the policy.
 388     for name in graph["probes"]:
 389         probe = read_json(stage / name)
 390         probe["actions"] = copy.deepcopy(graph["actions"])
 391         write_json(stage / name, probe)
 392     report["fp16_delivery"] = final_report

 395 def export_delivery(source, output, actions, samples=33, base_actions=(), fp16=False):
 396     import onnx
 397     import onnxruntime as ort
 398     import torch
 399     from convert_onnx_to_rknn_rk3588_wbc_reference import check_reference_model
 400     source, output = Path(source).resolve(), Path(output).resolve()
 401     index = json.loads((source / "emotion_index.json").read_text(encoding="utf-8"))
 402     if actions is None:
 403         actions = [entry["motion"] for entry in index]
 404     if not actions or len(set(actions)) != len(actions) or any(not ID_PATTERN.fullmatch(x) for x in actions):
 405         raise ValueError("Select unique valid action IDs")
 406     if len(set(base_actions)) != len(base_actions) or not set(base_actions) <= set(actions):
 407         raise ValueError("--base-actions must be a unique subset of --actions")
 408     base_actions = set(base_actions)
 409     if samples < 20:
 410         raise ValueError("At least 20 verification samples are required")
 411     if output.exists() and (not output.is_dir() or any(output.iterdir())):
 412         raise ValueError("Output must be absent or an empty directory")
 413     entries = {x["motion"]: x for x in index}
 414     if len(entries) != len(index):
 415         raise ValueError("Duplicate source action IDs")
 416     if any(x not in entries for x in actions):
 417         raise ValueError("Unknown source action ID")
 418     base_path, onnx_path = source / "base/base_ckpt.pt", source / "base/policy.onnx"
 419     base_hash = sha256(base_path)
 420     base = checkpoint(base_path)["policy"]
