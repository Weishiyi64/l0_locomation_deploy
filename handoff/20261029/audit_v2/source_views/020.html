  17 def sample_guidance_path(vertices, planner, spacing=1.0):
  18     """Use a fixed-distance lookahead, shortening only for corners/the goal.
  19 
  20     Intersect the remaining polyline with a circle around the current anchor.
  21     Only skip bends if the direct connector passes the original clearance test.
  22     Unlike equal subdivision, a 1.2 m straight leg starts with a 1 m target.
  23     """
  24     remaining = np.asarray(vertices, dtype=np.float64)
  25     result = [remaining[0]]
  26     while len(remaining) > 1:
  27         anchor = remaining[0]
  28         candidate = remaining[-1]
  29         segment_end = len(remaining) - 1
  30         for index in range(1, len(remaining)):
  31             if np.linalg.norm(remaining[index] - anchor) < spacing - 1e-9:
  32                 continue
  33             start, end = remaining[index - 1 : index + 1]
  34             direction, offset = end - start, start - anchor
  35             a = float(np.dot(direction, direction))
  36             if a < 1e-16:
  37                 continue
  38             b = float(np.dot(offset, direction))
  39             c = float(np.dot(offset, offset) - spacing**2)
  40             fraction = np.clip((-b + math.sqrt(max(0.0, b * b - a * c))) / a, 0.0, 1.0)
  41             candidate = start + fraction * direction
  42             segment_end = index
  43             break
  44         if not planner.segment_is_free(anchor, candidate):
  45             # A mandatory bend may be closer than the lookahead. Never turn a
  46             # spacing preference into a connector through an occupied corner.
  47             for index in range(segment_end - 1, 0, -1):
  48                 if planner.segment_is_free(anchor, remaining[index]):
  49                     candidate, segment_end = remaining[index], index
  50                     break
  51             else:
  52                 raise ValueError("无法在保持净空的同时采样前视路径点")
  53         if np.linalg.norm(candidate - anchor) < 1e-8:
  54             raise ValueError("前视路径点未沿路线推进")
  55         result.append(candidate)
  56         tail = remaining[segment_end:]
  57         # The intersection can coincide with an existing vertex (or the goal).
  58         if len(tail) and np.linalg.norm(tail[0] - candidate) < 1e-8:
  59             tail = tail[1:]
  60         remaining = np.concatenate((candidate[None], tail))
  61     return np.asarray(result, dtype=np.float32)

  64 def refresh_committed_path(path_xy, waypoint_index, start, planner, spacing=1.0):
  65     """Re-anchor a still-valid committed route at the live robot position.
  66 
  67     Periodic replanning used to discard the entire route and solve an equally
  68     valid A* homotopy from scratch.  Around a doorway, small tracking errors can
  69     then make successive solutions alternate between opposite obstacle sides.
  70     Keep only the unvisited suffix and search a fresh full-distance connector
  71     into it before falling back to a necessary short bend. Returning
  72     ``None`` asks the caller for a genuine A* replan/recovery when the committed
  73     corridor is no longer safely reachable.
  74     """
  75     path = np.asarray(path_xy, dtype=np.float64)
  76     start = np.asarray(start, dtype=np.float64)
  77     if (
  78         path.ndim != 2
  79         or path.shape[1:] != (2,)
  80         or len(path) < 2
  81         or not 1 <= waypoint_index < len(path)
  82         or not np.isfinite(path).all()
  83         or not np.isfinite(start).all()
  84         or not planner.start_is_free(start)
  85     ):
  86         return None
  87 
  88     # Retain the corridor, not the exact old sampling vertices. A vertex-only
  89     # reconnect leaves a 0.2--0.3 m target at a doorway even when a checked
  90     # two-segment connection through a fresh 1 m target is available.
  91     suffix = path[waypoint_index:]
  92     if np.linalg.norm(suffix[-1] - start) <= spacing and planner.segment_is_free(start, suffix[-1]):
  93         return np.asarray((start, suffix[-1]), dtype=np.float32)
  94     remaining_lengths = np.r_[np.cumsum(np.linalg.norm(np.diff(suffix, axis=0), axis=1)[::-1])[::-1], 0.0]
  95     direction = suffix[0] - start
  96     bearing = math.atan2(direction[1], direction[0])
  97     angles = np.r_[bearing, np.linspace(-math.pi, math.pi, 128, endpoint=False)]
  98     endpoints = start + spacing * np.stack((np.cos(angles), np.sin(angles)), axis=1)
  99     best = None
 100     for endpoint in endpoints:
 101         if not planner.segment_is_free(start, endpoint):
 102             continue
 103         for index in range(len(suffix) - 1, -1, -1):
 104             outgoing = suffix[index] - endpoint
 105             # Do not create a new target beyond a corner and then turn back
 106             # towards the old point. A genuine tight turn uses the fallback.
 107             if np.dot(outgoing, endpoint - start) < -1.0e-8:
 108                 continue
 109             if not planner.segment_is_free(endpoint, suffix[index]):
 110                 continue
 111             cost = np.linalg.norm(outgoing) + remaining_lengths[index]
 112             if best is None or cost < best[0]:
 113                 best = (cost, endpoint.copy(), index)
 114             break
 115     if best is not None:
 116         _, endpoint, index = best
 117         vertices = np.concatenate((endpoint[None], suffix[index:]))
 118         keep = np.r_[True, np.linalg.norm(np.diff(vertices, axis=0), axis=1) > 1.0e-7]
 119         vertices = vertices[keep]
 120         try:
 121             tail = sample_guidance_path(vertices, planner, spacing)
 122         except ValueError:
 123             return None
 124         refreshed = np.concatenate((start[None], tail)).astype(np.float32)
 125         if all(planner.segment_is_free(a, b) for a, b in zip(refreshed[:-1], refreshed[1:])):
 126             return refreshed
 127         return None
 128 
 129     # Never reconnect to a visited point.  Among the outstanding points, the
 130     # farthest visible one is guaranteed to make route progress and removes
 131     # redundant points made visible by the robot's actual motion.
 132     reconnect = None
 133     for index in range(len(path) - 1, waypoint_index - 1, -1):
 134         if planner.segment_is_free(start, path[index]):
 135             reconnect = index
 136             break
 137     if reconnect is None:
 138         return None
 139 
 140     vertices = np.concatenate((start[None], path[reconnect:]))
 141     keep = np.r_[True, np.any(np.diff(vertices, axis=0) != 0.0, axis=1)]
 142     vertices = vertices[keep]
 143     if len(vertices) < 2:
 144         return None
 145     try:
 146         refreshed = sample_guidance_path(vertices, planner, spacing)
 147     except ValueError:
 148         return None
 149     if len(refreshed) < 2 or not all(planner.segment_is_free(a, b) for a, b in zip(refreshed[:-1], refreshed[1:])):
 150         return None
 151     return refreshed

 154 class RecoveryAStarPlanner(AStarPlanner):
 155     def __init__(self, obstacles, terrain_size, resolution=0.1):
 156         self.terrain_size = np.asarray(terrain_size, dtype=np.float64)
 157         self.robot_radius = 0.2
 158         clearance = self.robot_radius + resolution / math.sqrt(2.0)
 159         width, height = terrain_size
 160         origin = (-width / 2, -height / 2)
 161         xs = origin[0] + (np.arange(math.ceil(width / resolution)) + 0.5) * resolution
 162         ys = origin[1] + (np.arange(math.ceil(height / resolution)) + 0.5) * resolution
 163         xx, yy = np.meshgrid(xs, ys)
 164         # Separate layers let recovery relax only the constraints occupied at
 165         # the real start, never an entire neighborhood or the normal A* map.
 166         self.layers = []
 167         for axis, coordinate in enumerate((xx, yy)):
 168             half = self.terrain_size[axis] / 2
 169             for sign in (-1, 1):
 170                 mask = sign * coordinate >= half - clearance
 171                 self.layers.append((dict(shape="boundary", axis=axis, sign=sign), mask))
 172         for obstacle in obstacles:
 173             if obstacle.get("shape") == "box":
 174                 mask = (np.abs(xx - obstacle["x"]) <= obstacle["size_x"] / 2 + clearance) & (
 175                     np.abs(yy - obstacle["y"]) <= obstacle["size_y"] / 2 + clearance
 176                 )
 177             else:
 178                 mask = (xx - obstacle["x"]) ** 2 + (yy - obstacle["y"]) ** 2 <= (obstacle["radius"] + clearance) ** 2
 179             self.layers.append((dict(obstacle), mask))
 180         grid = np.logical_or.reduce([mask for _, mask in self.layers])
 181         super().__init__(grid, resolution, origin, prevent_corner_cutting=True)
 182         self.layer_planners = [AStarPlanner(mask, resolution, origin) for _, mask in self.layers]
 183         # Binary A* has no preference inside a free corridor and therefore hugs
 184         # an inflated obstacle boundary.  The local policy's tracking error can
 185         # then alternate between the two inflation layers of a narrow passage.
 186         # Keep the same hard occupancy map, but add a soft cost within 0.50 m of
 187         # it so the global guide stays near the passage centre when space allows.
 188         self.preferred_clearance_m = 0.50
 189         self.clearance_cost_weight = 8.0
 190         self.clearance_m = self._distance_to_occupied()
 191         self._goal_cost_cache = {}
 192 
 193     def _distance_to_occupied(self):
 194         """Return an 8-connected distance field without a SciPy dependency."""
 195         distances = np.full(self.grid.shape, np.inf, dtype=np.float64)
 196         pending = []
 197         for row, col in np.argwhere(self.grid != 0):
 198             cell = int(row), int(col)
 199             distances[cell] = 0.0
 200             pending.append((0.0, cell))
 201         heapq.heapify(pending)
 202         while pending:
 203             distance, current = heapq.heappop(pending)
 204             if distance != distances[current]:
 205                 continue
 206             for dr, dc, move_cost in self.neighbors:
 207                 nxt = current[0] + dr, current[1] + dc
 208                 if not (0 <= nxt[0] < self.h and 0 <= nxt[1] < self.w):
 209                     continue
 210                 candidate = distance + move_cost * self.resolution
 211                 if candidate < distances[nxt]:
 212                     distances[nxt] = candidate
 213                     heapq.heappush(pending, (candidate, nxt))
 214         return distances
 215 
 216     def _step_cost(self, current, nxt, move_cost):
 217         """Symmetric path cost: geometric length plus a near-obstacle premium."""
 218         preferred = self.preferred_clearance_m
 219         current_deficit = max(0.0, preferred - self.clearance_m[current]) / preferred
 220         next_deficit = max(0.0, preferred - self.clearance_m[nxt]) / preferred
 221         multiplier = 1.0 + self.clearance_cost_weight * 0.5 * (
 222             current_deficit * current_deficit + next_deficit * next_deficit
 223         )
 224         return move_cost * multiplier

 226     def plan(self, start_xy, goal_xy):
 227         """A* on the unchanged hard grid with a deterministic clearance cost."""
 228         start = self.world_to_grid(start_xy)
 229         goal = self.world_to_grid(goal_xy)
 230         if not self.valid(start):
 231             raise ValueError(f"start is invalid: rc={start}, xy={start_xy}")
 232         if not self.valid(goal):
 233             raise ValueError(f"goal is invalid: rc={goal}, xy={goal_xy}")
 234 
 235         pending = [(0.0, start)]
 236         came_from = {}
 237         costs = {start: 0.0}
 238         closed = set()
 239         while pending:
 240             _, current = heapq.heappop(pending)
 241             if current in closed:
 242                 continue
 243             closed.add(current)
 244             if current == goal:
 245                 return self._reconstruct(came_from, current)
 246             for dr, dc, move_cost in self.neighbors:
 247                 nxt = current[0] + dr, current[1] + dc
 248                 if not self.valid(nxt):
 249                     continue
 250                 if (
 251                     dr
 252                     and dc
 253                     and (not self.valid((current[0] + dr, current[1])) or not self.valid((current[0], current[1] + dc)))
 254                 ):
 255                     continue
 256                 candidate = costs[current] + self._step_cost(current, nxt, move_cost)
 257                 if candidate < costs.get(nxt, np.inf):
 258                     came_from[nxt] = current
 259                     costs[nxt] = candidate
 260                     score = candidate + self.heuristic(nxt, goal)
 261                     heapq.heappush(pending, (score, nxt))
 262         return None

 264     def _cost_to_goal(self, goal):
 265         """Return an exact grid cost-to-go map for the static obstacle layer."""
 266         goal_cell = self.world_to_grid(goal)
 267         if not self.valid(goal_cell):
 268             return np.full(self.grid.shape, np.inf, dtype=np.float64)
 269         cached = self._goal_cost_cache.get(goal_cell)
 270         if cached is not None:
 271             return cached
 272         costs = np.full(self.grid.shape, np.inf, dtype=np.float64)
 273         costs[goal_cell] = 0.0
 274         pending = [(0.0, goal_cell)]
 275         while pending:
 276             cost, current = heapq.heappop(pending)
 277             if cost != costs[current]:
 278                 continue
 279             for dr, dc, move_cost in self.neighbors:
 280                 nxt = current[0] + dr, current[1] + dc
 281                 if not self.valid(nxt):
 282                     continue
 283                 if (
 284                     dr
 285                     and dc
 286                     and (not self.valid((current[0] + dr, current[1])) or not self.valid((current[0], current[1] + dc)))
 287                 ):
 288                     continue
 289                 candidate = cost + self._step_cost(current, nxt, move_cost)
 290                 if candidate < costs[nxt]:
 291                     costs[nxt] = candidate
 292                     heapq.heappush(pending, (candidate, nxt))
 293         # The final goal remains fixed during a replay; retaining a small cache
 294         # also supports interactive replacement without recomputing every 1 Hz.
 295         if len(self._goal_cost_cache) >= 4:
 296             self._goal_cost_cache.pop(next(iter(self._goal_cost_cache)))
 297         self._goal_cost_cache[goal_cell] = costs
 298         return costs

 300     def safe_lookahead_endpoint(self, start, goal, distance=1.0, preferred_direction=None):
 301         """Choose an exactly ``distance``-away point inside the A* corridor.
 302 
 303         A shortest grid path can contain a mandatory bend only a few centimetres
 304         ahead. Publishing that bend directly makes a 5 Hz consumer overrun it.
 305         Search the lookahead circle for a collision-free connector, but only
 306         accept candidates that remain close to the actual shortest path.  An
 307         unconstrained circle search can otherwise erase a necessary turn by
 308         choosing a globally cheaper point across the inside of a corner.  When
 309         no full-distance point satisfies both constraints, return ``None`` so
 310         the caller publishes the mandatory clearance-preserving bend instead.
 311         """
 312         start = np.asarray(start, dtype=np.float64)
 313         goal = np.asarray(goal, dtype=np.float64)
 314         if distance <= 0.0 or not np.isfinite((start, goal)).all():
 315             return None
 316         goal_delta = goal - start
 317         goal_distance = float(np.linalg.norm(goal_delta))
 318         if goal_distance <= distance and self.segment_is_free(start, goal):
 319             return goal.astype(np.float32)
 320 
 321         dense_path = self.plan(start, goal)
 322         if dense_path is None:
 323             return None
 324         dense_path = np.concatenate((start[None], dense_path, goal[None]))
 325         keep = np.r_[True, np.any(np.diff(dense_path, axis=0) != 0.0, axis=1)]
 326         dense_path = dense_path[keep].astype(np.float64)
 327         path_starts = dense_path[:-1]
 328         path_vectors = np.diff(dense_path, axis=0)
 329         path_length_squared = np.sum(path_vectors * path_vectors, axis=1)
 330 
 331         def distance_to_shortest_path(point):
 332             fractions = np.clip(
 333                 np.sum((point - path_starts) * path_vectors, axis=1) / np.maximum(path_length_squared, 1.0e-12),
 334                 0.0,
 335                 1.0,
 336             )
 337             projections = path_starts + fractions[:, None] * path_vectors
 338             return float(np.min(np.linalg.norm(point - projections, axis=1)))
 339 
 340         goal_cost = self._cost_to_goal(goal)
 341         start_cell = self.world_to_grid(start)
 342         start_cost = goal_cost[start_cell] if self.valid(start_cell) else np.inf
 343         goal_bearing = math.atan2(goal_delta[1], goal_delta[0])
 344         # Keep adjacent samples no farther apart than half a grid cell on the
 345         # circle; explicitly add the goal bearing so straight routes stay exact.
 346         count = max(72, int(math.ceil(4.0 * math.pi * distance / self.resolution)))
 347         angles = np.concatenate(
 348             (
 349                 np.array([goal_bearing], dtype=np.float64),
 350                 np.linspace(-math.pi, math.pi, count, endpoint=False, dtype=np.float64),
 351             )
 352         )
 353         candidates = start + distance * np.stack((np.cos(angles), np.sin(angles)), axis=-1)
 354         preferred = None
 355         if preferred_direction is not None:
 356             preferred = np.asarray(preferred_direction, dtype=np.float64)
 357             preferred_norm = float(np.linalg.norm(preferred))
 358             preferred = preferred / preferred_norm if preferred_norm > 1.0e-8 else None
 359         options = []
 360         # One and a half cells absorbs raster-centre quantisation, while still
 361         # rejecting the 0.3--0.4 m corner cuts observed in the house map.
 362         corridor_radius = 1.5 * self.resolution
 363         for angle, endpoint in zip(angles, candidates):
 364             cell = self.world_to_grid(endpoint)
 365             if not self.valid(cell) or not np.isfinite(goal_cost[cell]):
 366                 continue
 367             # A full-distance candidate must advance on the static global cost
 368             # field. If no such endpoint exists, the caller retains the short
 369             # mandatory bend from the actual shortest path instead of looping.
 370             if np.isfinite(start_cost) and goal_cost[cell] >= start_cost - 0.25:
 371                 continue
 372             if not self.segment_is_free(start, endpoint):
 373                 continue
 374             route_deviation = distance_to_shortest_path(endpoint)
 375             if route_deviation > corridor_radius + 1.0e-9:
 376                 continue
 377             cell_center = self.grid_to_world(cell)
 378             cost = float(goal_cost[cell] * self.resolution + np.linalg.norm(endpoint - cell_center))
 379             bearing_error = abs(math.atan2(math.sin(angle - goal_bearing), math.cos(angle - goal_bearing)))
 380             alignment = float(np.dot((endpoint - start) / distance, preferred)) if preferred is not None else 1.0
 381             continuity_cost = 0.15 * (1.0 - alignment)
 382             options.append(
 383                 (
 384                     (cost + continuity_cost, route_deviation, cost, bearing_error),
 385                     endpoint.copy(),
 386                 )
 387             )
 388         if not options:
 389             return None
 390         # Continuity only breaks near-ties; global progress remains dominant.
 391         best = min(options, key=lambda option: option[0])
 392         return best[1].astype(np.float32)

 398     def recovery_segment_is_safe(self, start, end):
 399         """Check whether a cached target safely exits current inflation layers.
 400 
 401         Entering the conservative inflated grid does not invalidate a target
 402         that is already leading monotonically out of exactly those layers.
 403         Keeping that target avoids replacing a nearly reached corner with a
 404         newly sampled exit on the opposite side of the robot.
 405         """
 406         start = np.asarray(start, dtype=np.float64)
 407         end = np.asarray(end, dtype=np.float64)
 408         if not np.isfinite((start, end)).all() or not self.start_is_free(end):
 409             return False
 410         initial = [not layer.segment_is_free(start, start) for layer in self.layer_planners]
 411         if not any(initial):
 412             return self.segment_is_free(start, end)
 413         for occupied, layer in zip(initial, self.layer_planners):
 414             if not occupied and not layer.segment_is_free(start, end):
 415                 return False
 416         return all(
 417             self._moves_outward(obstacle, start, end)
 418             for occupied, (obstacle, _) in zip(initial, self.layers)
 419             if occupied
 420         )
 421 
 422     def _moves_outward(self, obstacle, start, end):
 423         """Analytic nondecreasing physical clearance along a straight segment.
 424 
 425         For a convex box/cylinder, a nonnegative outward directional derivative
 426         makes distance nondecreasing along the entire ray. Contact is allowed to
 427         separate; starting with the base center inside a solid is not certified.
 428         """
 429         direction = end - start
 430         if obstacle["shape"] == "boundary":
 431             axis, sign = obstacle["axis"], obstacle["sign"]
 432             return sign * direction[axis] < -1e-8
 433         center = np.array([obstacle["x"], obstacle["y"]])
 434         if obstacle.get("shape") == "box":
 435             half = np.array([obstacle["size_x"], obstacle["size_y"]]) / 2
 436             relative = start - center
 437             outside = np.abs(relative) - half
 438             if np.max(outside) < -1e-8:
 439                 return False
 440             separation = relative - np.clip(relative, -half, half)
 441             if np.linalg.norm(separation) <= 1e-8:
 442                 # At a physical face/corner, at least one active face must
 443                 # strictly separate; do not traverse the solid or its surface.
 444                 active = outside >= -1e-8
 445                 return bool(np.any(active & (np.sign(relative) * direction > 1e-8)))
 446         else:
 447             separation = start - center
 448             if np.linalg.norm(separation) < obstacle["radius"] - 1e-8:
 449                 return False
 450         return float(np.dot(separation, direction)) >= -1e-10

 452     def recovery_exits(self, start, goal, max_distance=1.0, preferred_direction=None):
 453         """Rank checked exits by full route cost, allowing a necessary short bend.
 454 
 455         The checked connector must be followed before any normal route point.
 456         No robot pose is snapped or moved. Empty results mean no certified exit.
 457         """
 458         start, goal = np.asarray(start, dtype=np.float64), np.asarray(goal, dtype=np.float64)
 459         if not np.isfinite(start).all() or np.any(np.abs(start) >= self.terrain_size / 2):
 460             return []
 461         initial = [not layer.segment_is_free(start, start) for layer in self.layer_planners]
 462         if not any(initial):
 463             return []
 464         blocked = np.zeros_like(self.grid)
 465         for is_initial, (_, mask) in zip(initial, self.layers):
 466             if not is_initial:
 467                 blocked |= mask
 468         connector_planner = AStarPlanner(blocked, self.resolution, tuple(self.origin_xy))
 469         # One reverse flood avoids running an unsuccessful A* for hundreds of
 470         # exits when a wall disconnects the goal. Map connectivity stays strict.
 471         component = self._goal_component(goal)
 472         cells = np.argwhere(component)
 473         centers = self.origin_xy + (cells[:, ::-1] + 0.5) * self.resolution
 474         distances = np.linalg.norm(centers - start, axis=1)
 475         # Include both the normal lookahead ring and nearer free cells. The
 476         # route-cost ordering below decides whether the full distance is safe
 477         # and useful or a shorter clearance bend is required.
 478         nearby = (distances > 1e-8) & (distances <= max_distance + self.resolution)
 479         ring = start + (centers[nearby] - start) * (max_distance / distances[nearby, None])
 480         ring = [
 481             point for point in ring if self.valid(self.world_to_grid(point)) and component[self.world_to_grid(point)]
 482         ]
 483         # Avoid placing the escape target inside the usual 0.2 m switching ball.
 484         keep = (distances >= 0.35) & (distances <= max_distance)
 485         candidates = np.concatenate((np.asarray(ring).reshape(-1, 2), centers[keep]))
 486         short_goal = 1e-8 < np.linalg.norm(goal - start) <= max_distance
 487         if short_goal:
 488             candidates = np.concatenate((goal[None], candidates))
 489         if len(candidates) == 0:
 490             return []
 491         distances = np.linalg.norm(candidates - start, axis=1)
 492         preferred = None
 493         if preferred_direction is not None:
 494             preferred = np.asarray(preferred_direction, dtype=np.float64)
 495             norm = float(np.linalg.norm(preferred))
 496             preferred = preferred / norm if norm > 1.0e-8 else None
 497         alignment_cost = np.zeros(len(candidates), dtype=np.float64)
 498         if preferred is not None:
 499             directions = (candidates - start) / np.maximum(distances[:, None], 1.0e-8)
 500             alignment_cost = 1.0 - directions @ preferred
 501         # Recovery is a necessary clearance bend, not a normal lookahead that
 502         # must be stretched to 1 m. Preferring the outer ring can carry the
 503         # target across a narrow doorway and place it beside the opposite wall;
 504         # the next recovery then reverses direction. Rank exits by the complete
 505         # safe route cost instead: checked connector plus weighted free-space
 506         # cost to the unchanged final goal. Continuity only breaks close ties.
 507         goal_cost = self._cost_to_goal(goal)
 508         cells_rc = np.asarray([self.world_to_grid(point) for point in candidates])
 509         costs_to_goal = goal_cost[cells_rc[:, 0], cells_rc[:, 1]] * self.resolution
 510         route_cost = distances + costs_to_goal
 511         order = np.lexsort(
 512             (
 513                 -distances,
 514                 alignment_cost,
 515                 np.round(route_cost, 6),
 516             )
 517         )
 518         exits = []
 519         for endpoint in candidates[order]:
 520             if not self.start_is_free(endpoint):
 521                 continue
 522             if not connector_planner.segment_is_free(start, endpoint):
 523                 continue
 524             if not all(
 525                 self._moves_outward(obstacle, start, endpoint)
 526                 for is_initial, (obstacle, _) in zip(initial, self.layers)
 527                 if is_initial
 528             ):
 529                 continue
 530             exits.append(endpoint.astype(np.float32))
 531         return exits

 533     def _goal_component(self, goal):
 534         reachable = np.zeros_like(self.grid, dtype=bool)
 535         cell = self.world_to_grid(goal)
 536         if not self.valid(cell):
 537             return reachable
 538         pending = deque([cell])
 539         reachable[cell] = True
 540         while pending:
 541             row, col = pending.popleft()
 542             for dr, dc, _ in self.neighbors:
 543                 nxt = row + dr, col + dc
 544                 if not self.valid(nxt) or reachable[nxt]:
 545                     continue
 546                 if dr and dc and (not self.valid((row + dr, col)) or not self.valid((row, col + dc))):
 547                     continue
 548                 reachable[nxt] = True
 549                 pending.append(nxt)
 550         return reachable
