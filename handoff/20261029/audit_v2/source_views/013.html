  95     def message(self, text):
  96         self.status.set_text(text)
  97         self.figure.canvas.draw_idle()

  99     def free(self, x, y):
 100         if self.flat:
 101             return math.isfinite(x) and math.isfinite(y)
 102         # Training selector's clearance is applied to all obstacle footprints.
 103         margin = 0.35
 104         return (
 105             -5 + margin < x < 5 - margin
 106             and -5 + margin < y < 5 - margin
 107             and all(not (a - margin <= x <= c + margin and b - margin <= y <= d + margin) for a, b, c, d in self.boxes)
 108         )
 109 
 110     def draw_goal(self):
 111         x, y = self.goal
 112         self.marker.set_data([x], [y])
 113         if self.arrow is not None:
 114             self.arrow.remove()
 115         self.arrow = FancyArrowPatch(
 116             (x, y),
 117             (x + 0.65 * math.cos(self.heading), y + 0.65 * math.sin(self.heading)),
 118             arrowstyle='-|>',
 119             mutation_scale=18,
 120             color='red',
 121             linewidth=2,
 122         )
 123         self.axes.add_patch(self.arrow)
 124         self.figure.canvas.draw_idle()
 125 
 126     def press(self, event):
 127         if event.button != 1 or event.inaxes is not self.axes:
 128             return
 129         if self.figure.canvas.toolbar and self.figure.canvas.toolbar.mode:
 130             return
 131         if not self.free(event.xdata, event.ydata):
 132             self.message('Invalid goal:\nobstacle or insufficient clearance.')
 133             return
 134         self.goal = (event.xdata, event.ydata)
 135         self.dragging = True
 136         self.draw_goal()
 137         self.message('Drag to set heading;\nthen click Apply goal.')
 138 
 139     def motion(self, event):
 140         if not self.dragging or event.inaxes is not self.axes:
 141             return
 142         dx, dy = event.xdata - self.goal[0], event.ydata - self.goal[1]
 143         if math.hypot(dx, dy) > 0.001:
 144             self.heading = math.atan2(dy, dx)
 145             self.draw_goal()
 146 
 147     def release(self, event):
 148         if event.button == 1:
 149             self.motion(event)
 150             self.dragging = False
 151 
 152     def apply(self, _event):
 153         if self.goal is None:
 154             self.message('Select a goal first.')
 155             return
 156         try:
 157             speed, turn = float(self.linear.text), float(self.angular.text)
 158             if not (0.3 <= speed <= 1.2 and 1.57 <= turn <= 3.14):
 159                 raise ValueError()
 160         except ValueError:
 161             self.message('Speed: 0.3–1.2 m/s\nTurn: 1.57–3.14 rad/s')
 162             return
 163         if self.publisher.get_subscription_count() == 0:
 164             self.message('Goal manager unavailable.\nWait and click Apply again.')
 165             return
 166         if time.monotonic() - self.odom_received > 0.5:
 167             self.message('No fresh odometry.\nCheck robot connection.')
 168             return
 169         command = [float(self.goal[0]), float(self.goal[1]), speed, self.heading, turn, 1.0]
 170         self.publisher.publish(Float32MultiArray(data=command))
 171         self.message(
 172             f'Sent ({command[0]:.2f}, {command[1]:.2f})\nyaw {math.degrees(self.heading):.1f} deg\nExplicit policy start still required.'
 173         )
 174 
 175     def on_path(self, message):
 176         if message.header.frame_id != 'device0/odom':
 177             return
 178         self.route_line.set_data([p.pose.position.x for p in message.poses], [p.pose.position.y for p in message.poses])
 179         self.figure.canvas.draw_idle()
 180 
 181     def on_inference_enabled(self, message):
 182         self.active = bool(message.inference_enabled)
 183         self.state_received = time.monotonic()
 184 
 185     def on_task_state(self, message):
 186         self.task_mode = message.data
 187 
 188     def odometry(self, message):
 189         if message.header.frame_id == 'device0/odom':
 190             p = message.pose.pose.position
 191             if not (math.isfinite(p.x) and math.isfinite(p.y)):
 192                 return
 193             self.odom_received = time.monotonic()
 194             if self.flat and not self.centered:
 195                 self.axes.set(xlim=(p.x - 1.5, p.x + 1.5), ylim=(p.y - 1.5, p.y + 1.5))
 196                 self.centered = True
 197             self.robot.set_data([p.x], [p.y])
 198             self.figure.canvas.draw_idle()
 199 
 200     def poll(self):
 201         if rclpy.ok():
 202             # One callback per 50 ms cannot service all high-rate subscriptions.
 203             # Drain a bounded batch on the GUI thread; depth=1 keeps sensor data fresh.
 204             for _ in range(16):
 205                 rclpy.spin_once(self, timeout_sec=0)
 206             now = time.monotonic()
 207             if (
 208                 now - self.odom_received <= 0.5
 209                 and self.status.get_text() == 'No fresh odometry.\nCheck robot connection.'
 210             ):
 211                 self.message('Odometry recovered.\nClick Apply goal again.')
 212             if now - self.odom_received > 0.5 or self.state_received == -math.inf:
 213                 text = 'No fresh robot state.\nCheck launch / ROS network.'
 214             elif not self.active:
 215                 text = 'Policy NOT activated.\nApply only sets the goal.'
 216             else:
 217                 text = f'Policy active | {self.task_mode}'
 218             if self.runtime_status.get_text() != text:
 219                 self.runtime_status.set_text(text)
 220                 self.runtime_status.set_color(
 221                     'green' if self.active and now - self.odom_received <= 0.5 else 'darkorange'
 222                 )
 223                 self.figure.canvas.draw_idle()
 224         else:
 225             plt.close(self.figure)
