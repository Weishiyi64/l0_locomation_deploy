  25 wbc_environment() {
  26   local config=$1
  27   [[ -f /opt/ros/humble/setup.bash ]] || wbc_die '缺少 ROS Humble 环境'
  28   [[ -f "$wbc_workspace/install/setup.bash" ]] || wbc_die '工作空间尚未构建'
  29   [[ -f "$config" ]] || wbc_die "缺少模型配置：$config"
  30   # ROS setup files are not nounset-safe. Preserve the caller's DDS environment.
  31   source /opt/ros/humble/setup.bash
  32   source "$wbc_workspace/install/setup.bash"
  33   unset robot_type amp_policy wbc_bundle wbc_actions
  34   source "$config"
  35   [[ ${robot_type:-} =~ ^[A-Za-z0-9_-]+$ ]] || wbc_die 'robot_type 无效'
  36   [[ ${amp_policy:-} =~ ^[A-Za-z0-9_-]+$ ]] || wbc_die 'amp_policy 无效'
  37   [[ ${wbc_bundle:-} =~ ^[A-Za-z0-9_-]+$ ]] || wbc_die 'wbc_bundle 无效'
  38   [[ -n ${wbc_actions:-} ]] || wbc_die 'wbc_actions 为空'
  39   wbc_prefix=$(ros2 pkg prefix legged_deploy)
  40   [[ "$wbc_prefix" == "$wbc_workspace/install" || "$wbc_prefix" == "$wbc_workspace/install/"* ]] ||
  41     wbc_die 'legged_deploy 来自其他工作空间'
  42   wbc_installed="$wbc_prefix/share/legged_deploy/policy/$robot_type/$wbc_bundle"
  43   wbc_actions=$(/usr/bin/python3 -B - "$wbc_installed" "$wbc_actions" <<'PY'
  44 import json, re, sys
  45 from pathlib import Path
  46 import yaml
  47 root = Path(sys.argv[1])
  48 selected = yaml.safe_load(sys.argv[2])
  49 if (not isinstance(selected, list) or not selected or
  50         any(not isinstance(a, str) or not re.fullmatch(r'[A-Za-z0-9_-]{1,32}', a) for a in selected) or
  51         len(selected) != len(set(selected))):
  52     raise SystemExit('wbc_actions 必须是非空、无重复的动作 ID 列表')
  53 manifest = json.loads((root / 'base_manifest.json').read_text())
  54 available = {a['action_id'] for a in manifest['actions']}
  55 if not set(selected) <= available:
  56     raise SystemExit('配置包含未安装动作：' + repr(set(selected) - available))
  57 for name in ('INSTALLATION.json', 'base_lora.rknn'):
  58     if not (root / name).is_file():
  59         raise SystemExit('缺少运行资产：' + str(root / name))
  60 print(json.dumps(selected))
  61 PY
  62   ) || wbc_die '模型配置检查失败'
  63 }

  65 wbc_current_run() {
  66   wbc_run_dir=$(/usr/bin/python3 -B - "$wbc_run_root" <<'PY'
  67 from pathlib import Path
  68 import sys
  69 root = Path(sys.argv[1]).resolve()

  71 if run.parent != root or not (run / 'environment.env').is_file():
  72     raise SystemExit('无效的 current_run_dir；先运行 start_sim2real.sh')
  73 if (run / 'LOGS.sha256').exists():
  74     raise SystemExit('本次记录已封存；请重新启动生成新的运行目录')
  75 print(run)
  76 PY
  77   ) || wbc_die '找不到可写的当前运行记录'
  78   wbc_environment "$wbc_run_dir/environment.env"
  79   cmp -s "$wbc_run_dir/base_manifest.json" "$wbc_installed/base_manifest.json" ||
  80     wbc_die '安装 manifest 与运行快照不一致，请重新启动部署'
  81   cmp -s "$wbc_run_dir/INSTALLATION.json" "$wbc_installed/INSTALLATION.json" ||
  82     wbc_die '安装清单与运行快照不一致，请重新启动部署'
  83   export ROS_LOG_DIR="$wbc_run_dir/ros"

  88 wbc_logged_command() {
  89   local log=$1 child logger log_fd result
  90   shift
  91   exec {log_fd}> >(trap '' INT TERM; exec tee "$log")
  92   logger=$!
  93   setsid "$@" >&$log_fd 2>&1 &
  94   child=$!
  95   exec {log_fd}>&-
  96   # Let ROS and its existing sequential launcher shut down their own children.
  97   trap 'kill -INT -- "-$child" 2>/dev/null || true' INT TERM
  98   while true; do
  99     if wait "$child"; then result=0; else result=$?; fi
 100     kill -0 "$child" 2>/dev/null || break
 101   done
 102   wait "$logger" || true
 103   trap - INT TERM
 104   return "$result"
