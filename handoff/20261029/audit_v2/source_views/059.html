  37 def prepare(args):
  38     replay = Path(args.replay).resolve()
  39     metadata = read_json(replay.with_suffix('.json'))
  40     segments_path = replay.parent / 'segments.json'
  41     if sha256(segments_path) != metadata['input_files_sha256']['segments.json']:
  42         raise ValueError('Segments changed since the baseline replay')
  43     segments = json.loads(segments_path.read_text())
  44     graph = load_graph_manifest(args.manifest)
  45     if (graph['model']['sha256'] != metadata['base_onnx_sha256'] or
  46             graph['base_checkpoint_sha256'] != metadata['base_checkpoint_sha256']):
  47         raise ValueError('Model identity differs from the baseline replay')
  48     catalog = {x['action_id']: x for x in graph['actions']}
  49     root = Path(args.manifest).resolve().parent
  50     data = np.load(replay, allow_pickle=False)
  51     indices, cases, assets, scales = [], [], {}, []
  52     for block, action_id in enumerate(args.actions):
  53         if not ID_PATTERN.fullmatch(action_id):
  54             raise ValueError(f'Invalid action ID: {action_id}')
  55         si, segment = next((i, s) for i, s in enumerate(segments) if s['action_id'] == action_id)
  56         start, end = segment['start_index'], segment['end_index']
  57         # Keep preparation plus samples after the approximate initial history.
  58         if args.samples < 2 or args.samples - 1 > end - start - 25:
  59             raise ValueError('Need 2..(segment length - 24) samples per action')
  60         chosen = np.r_[start, np.linspace(start + 25, end - 1, args.samples - 1, dtype=int)]
  61         config_path = asset_path(root, catalog[action_id]['config'])
  62         if sha256(config_path) != metadata['action_assets'][action_id]['config_sha256']:
  63             raise ValueError(f'Config changed: {action_id}')
  64         cfg = yaml.safe_load(config_path.read_text())['RobotCfg']
  65         if list(cfg['joint_order_isaac']) != data['joint_names'].tolist():
  66             raise ValueError('Joint order differs from baseline outputs')
  67         assets[action_id] = metadata['action_assets'][action_id]
  68         for index in chosen:
  69             cutoff = int(data['command_stamp'][index]) - 10_000_000
  70             if int(data['observation_stamp'][index]) > cutoff or int(data['segment_id'][index]) != si:
  71                 raise ValueError('Replay row violates the 10 ms cutoff or action identity')
  72             cases.append(dict(case_index=len(cases), action_id=action_id, block=block,
  73                               replay_index=int(index), segment=si,
  74                               command_stamp_ns=int(data['command_stamp'][index]),
  75                               observation_stamp_ns=int(data['observation_stamp'][index]),
  76                               observation_cutoff_ns=cutoff,
  77                               reference_frame=int(data['reference_frame'][index])))
  78             indices.append(index)
  79             scales.append(cfg['action_scale'])
  80     if len(cases) > 8192:
  81         raise ValueError('At most 8192 cases are supported')
  82     arrays = {key: data[key][indices] for key in ('policy', 'command_short', 'action_fp16', 'action_fp32')}
  83     arrays['action_scale'] = np.asarray(scales, dtype=np.float64)
  84     arrays['joint_names'] = data['joint_names']
  85     if not all(np.isfinite(v).all() for k, v in arrays.items() if k != 'joint_names'):
  86         raise ValueError('Non-finite input or baseline')
  87     output = new_directory(args.output_dir)
  88     for key in ('policy', 'command_short'):
  89         np.asarray(arrays[key], dtype='<f4').tofile(output / (key + '.f32'))
  90     np.savez_compressed(output / 'baseline.npz', **arrays)
  91     request = dict(format='wbc_policy_io_cases_v1', cases=cases,
  92                    input_rule=metadata['input_rule'],
  93                    scope='Frozen inputs for policy-only numerical comparison; not live input-builder validation.',
  94                    fp16_definition=metadata['model_fp16'], fp32_definition=metadata['model_fp32'],
  95                    reconstruction_caveats=metadata['caveats'],
  96                    base_onnx_sha256=metadata['base_onnx_sha256'],
  97                    base_checkpoint_sha256=metadata['base_checkpoint_sha256'], action_assets=assets,
  98                    source_replay_sha256=sha256(replay), source_metadata_sha256=sha256(replay.with_suffix('.json')),
  99                    files_sha256={name: sha256(output / name) for name in
 100                                  ('policy.f32', 'command_short.f32', 'baseline.npz')})
 101     save_json(output / 'cases.json', request)
 102     print(f'Prepared {len(cases)} cases in {output}', flush=True)

 105 def load_cases(directory):
 106     directory = Path(directory).resolve()
 107     request = read_json(directory / 'cases.json')
 108     if request['format'] != 'wbc_policy_io_cases_v1' or not 0 < len(request['cases']) <= 8192:
 109         raise ValueError('Invalid case package')
 110     for name in ('policy.f32', 'command_short.f32', 'baseline.npz'):
 111         if sha256(directory / name) != request['files_sha256'][name]:
 112             raise ValueError(f'Case checksum mismatch: {name}')
 113     with np.load(directory / 'baseline.npz', allow_pickle=False) as data:
 114         arrays = dict(data)
 115     count = len(request['cases'])
 116     for key, width in (('policy', 399), ('command_short', 240), ('action_fp16', 21),
 117                        ('action_fp32', 21), ('action_scale', 21)):
 118         if arrays[key].shape != (count, width) or not np.isfinite(arrays[key]).all():
 119             raise ValueError(f'Invalid tensor: {key}')
 120     if arrays['joint_names'].shape != (21,):
 121         raise ValueError('Invalid joint names')
 122     for key in ('policy', 'command_short'):
 123         binary = np.fromfile(directory / (key + '.f32'), dtype='<f4')
 124         if not np.array_equal(binary, arrays[key].ravel()):
 125             raise ValueError(f'Binary input differs from baseline: {key}')
 126     for i, item in enumerate(request['cases']):
 127         if item['case_index'] != i or not ID_PATTERN.fullmatch(item['action_id']):
 128             raise ValueError('Invalid case index/action ID')
 129     return request, arrays

 132 def validate_assets(manifest, request):
 133     manifest = Path(manifest).resolve()
 134     graph = load_graph_manifest(manifest)
 135     if (graph['model']['sha256'] != request['base_onnx_sha256'] or
 136             graph['base_checkpoint_sha256'] != request['base_checkpoint_sha256']):
 137         raise ValueError('Requested baseline and deployed model differ')
 138     catalog = {x['action_id']: x for x in graph['actions']}
 139     for action_id in {c['action_id'] for c in request['cases']}:
 140         item = catalog[action_id]
 141         adapter = asset_path(manifest.parent, item['adapter_manifest'])
 142         load_adapter(adapter, graph, action_id)  # Reuse full tensor/hash validation.
 143         expected = request['action_assets'][action_id]
 144         if read_json(adapter)['weights']['sha256'] != expected['weights_fp16_sha256']:
 145             raise ValueError(f'Adapter differs from baseline: {action_id}')
 146         if sha256(asset_path(manifest.parent, item['config'])) != expected['config_sha256']:
 147             raise ValueError(f'Action scale/config differs from baseline: {action_id}')
 148     return graph, catalog

 151 def error_stats(error):
 152     error = np.asarray(error, dtype=np.float64)
 153     return dict(rmse=float(np.sqrt(np.mean(error ** 2))), max_abs=float(np.max(abs(error))),
 154                 p99_abs=float(np.percentile(abs(error), 99)))

 157 def compare(output, request, arrays, actions):
 158     if actions.shape != arrays['action_fp16'].shape or not np.isfinite(actions).all():
 159         raise ValueError('Incomplete/non-finite inference output; retain raw logs for diagnosis')
 160     residual16 = actions.astype(np.float64) - arrays['action_fp16']
 161     residual32 = actions.astype(np.float64) - arrays['action_fp32']
 162     scaled16 = residual16 * arrays['action_scale']
 163     scaled32 = residual32 * arrays['action_scale']
 164     np.savez_compressed(output / 'outputs.npz', action=actions, action_fp16=arrays['action_fp16'],
 165                         action_fp32=arrays['action_fp32'], residual_fp16=residual16,
 166                         residual_fp32=residual32, residual_fp16_scaled_rad=scaled16,
 167                         residual_fp32_scaled_rad=scaled32, joint_names=arrays['joint_names'])
 168     report = dict(samples=len(actions), output='Raw policy action before clipping/scaling',
 169                   scaled_residual='(output - baseline) * action_scale, radians before clipping',
 170                   vs_local_fp16=dict(raw_action=error_stats(residual16), scaled_rad=error_stats(scaled16)),
 171                   vs_original_fp32=dict(raw_action=error_stats(residual32), scaled_rad=error_stats(scaled32)))
 172     report['per_action'] = {}
 173     for action_id in dict.fromkeys(c['action_id'] for c in request['cases']):
 174         mask = np.array([c['action_id'] == action_id for c in request['cases']])
 175         report['per_action'][action_id] = dict(samples=int(mask.sum()),
 176             vs_fp16_scaled_rad=error_stats(scaled16[mask]), vs_fp32_scaled_rad=error_stats(scaled32[mask]))
 177     repeats, first = [], {}
 178     for i, c in enumerate(request['cases']):
 179         key = (c['action_id'], c['replay_index'])
 180         if key in first:
 181             j = first[key]
 182             if not all(np.array_equal(arrays[k][i], arrays[k][j]) for k in ('policy', 'command_short')):
 183                 raise ValueError('Repeated case has different input')
 184             repeats.append(actions[i].astype(np.float64) - actions[j])
 185         else:
 186             first[key] = i
 187     report['repeated_identical_inputs'] = dict(pairs=len(repeats),
 188         raw_action=error_stats(repeats) if repeats else None)
 189     with (output / 'per_joint.csv').open('w', newline='') as stream:
 190         writer = csv.writer(stream)
 191         writer.writerow(['case_index', 'action_id', 'replay_index', 'joint', 'output', 'local_fp16',
 192                          'original_fp32', 'residual_fp16', 'residual_fp32',
 193                          'residual_fp16_scaled_rad', 'residual_fp32_scaled_rad'])
 194         for i, c in enumerate(request['cases']):
 195             for j, name in enumerate(arrays['joint_names']):
 196                 writer.writerow([i, c['action_id'], c['replay_index'], name, actions[i, j],
 197                                  arrays['action_fp16'][i, j], arrays['action_fp32'][i, j],
 198                                  residual16[i, j], residual32[i, j], scaled16[i, j], scaled32[i, j]])
 199     return report

 202 def run_local(args, output, request, arrays, graph, catalog):
 203     import onnxruntime as ort
 204     root = Path(args.manifest).resolve().parent
 205     model = asset_path(root, graph['model']['onnx'])
 206     check_graph(model, args.manifest)
 207     options = ort.SessionOptions()
 208     options.intra_op_num_threads = options.inter_op_num_threads = 1
 209     session = ort.InferenceSession(str(model), options, providers=['CPUExecutionProvider'])
 210     actions, active, parameters = [], None, None
 211     with (output / 'actions.f32').open('wb') as binary, (output / 'frames.csv').open('w', newline='') as stream:
 212         writer = csv.writer(stream)
 213         writer.writerow(['case_index', 'action_id', 'inference_ms'])
 214         for i, c in enumerate(request['cases']):
 215             if active != c['action_id']:
 216                 active = c['action_id']
 217                 parameters = load_adapter(asset_path(root, catalog[active]['adapter_manifest']), graph, active)
 218             feeds = graph_feeds(graph, {k: arrays[k][i] for k in ('policy', 'command_short')}, parameters)
 219             started = time.perf_counter()
 220             action = np.asarray(session.run(['action'], feeds)[0], dtype='<f4').reshape(21)
 221             elapsed_ms = (time.perf_counter() - started) * 1000
 222             if not np.isfinite(action).all():
 223                 raise ValueError(f'Non-finite local output at case {i}')
 224             binary.write(action.tobytes())
 225             binary.flush()
 226             writer.writerow([i, active, elapsed_ms])
 227             stream.flush()
 228             actions.append(action)
 229     return np.asarray(actions), dict(onnxruntime=ort.__version__, providers=session.get_providers(),
 230         arithmetic='CPU FP32 computation with FP16 stored weights; not native RKNN FP16')

 233 def run_board(args, output, request, arrays, graph, catalog):
 234     executable = Path(args.executable).resolve(strict=True)
 235     command = [str(executable), '--wbc-io-probe', str(Path(args.manifest).resolve()),
 236                str(Path(args.cases).resolve()), str(output / 'resident')]
 237     model_root = Path(args.manifest).resolve().parent
 238     model_path = asset_path(model_root, graph['model']['rknn'])
 239     invocation = dict(command=command, executable_sha256=sha256(executable),
 240                       model_sha256=sha256(model_path),
 241                       conversion_record_sha256=sha256(Path(str(model_path) + '.json')))
 242     save_json(output / 'invocation.json', invocation)
 243     # The regular rl_controller executable retains its installed RPATH/runtime.
 244     # No LD_LIBRARY_PATH override and no alternative Python RKNN loader.
 245     with (output / 'runtime.log').open('w') as log:
 246         process = subprocess.Popen(command, stdout=log, stderr=subprocess.STDOUT, start_new_session=True)
 247         try:
 248             returncode = process.wait(timeout=args.timeout_seconds)
 249         except BaseException:
 250             if process.poll() is None:
 251                 os.killpg(process.pid, signal.SIGINT)
 252                 try:
 253                     process.wait(timeout=10)
 254                 except subprocess.TimeoutExpired:
 255                     os.killpg(process.pid, signal.SIGKILL)
 256                     process.wait()
 257             raise
 258     if returncode:
 259         raise RuntimeError(f'rl_controller returned {returncode}; see runtime.log and resident/runtime.yaml')
 260     runtime = yaml.safe_load((output / 'resident/runtime.yaml').read_text())
 261     count = len(request['cases'])
 262     switches = 1 + sum(a['action_id'] != b['action_id'] for a, b in zip(request['cases'], request['cases'][1:]))
 263     if (runtime['status'] != 'completed' or runtime['completed_samples'] != count or
 264             runtime['context_initializations'] != 1 or runtime['runs'] != count or
 265             runtime['parameter_prepares'] != switches or runtime['commits'] != switches or runtime['rollbacks'] != 0):
 266         raise ValueError('Incomplete run or unexpected resident context/adapter counters')
 267     with (output / 'resident/frames.csv').open() as stream:
 268         rows = list(csv.DictReader(stream))
 269     if len(rows) != count or any(int(row['case_index']) != i or row['action_id'] != request['cases'][i]['action_id']
 270                                  for i, row in enumerate(rows)):
 271         raise ValueError('Frame log does not match requested cases')
 272     raw = np.fromfile(output / 'resident/actions.f32', dtype='<f4')
 273     if raw.size != count * 21:
 274         raise ValueError('Output byte count mismatch')
 275     return raw.reshape(count, 21), runtime

 278 def main():
 279     parser = argparse.ArgumentParser(description=__doc__)
 280     sub = parser.add_subparsers(dest='mode', required=True)
 281     prepare_parser = sub.add_parser('prepare', help='Freeze the audited 10 ms replay inputs and FP16/FP32 baselines')
 282     prepare_parser.add_argument('--replay', required=True)
 283     prepare_parser.add_argument('--actions', nargs='+', default=['L0_peace_new_02_A', 'L0_angry_7', 'L0_strange_1', 'L0_peace_new_02_A'])
 284     prepare_parser.add_argument('--samples', type=int, default=33)
 285     for mode in ('local', 'board'):
 286         p = sub.add_parser(mode, help='Run fixed inputs using ' + ('ONNX CPU baseline' if mode == 'local' else 'the existing C++ RKNN loader'))
 287         p.add_argument('--cases', required=True)
 288         if mode == 'board':
 289             p.add_argument('--executable', required=True, help='Installed RKNN rl_controller with --wbc-io-probe')
 290             p.add_argument('--timeout-seconds', type=float, default=300)
 291     for p in sub.choices.values():
 292         p.add_argument('--manifest', required=True, help='base_manifest.json')
 293         p.add_argument('--output-dir', required=True, help='New directory; existing results are never overwritten')
 294     args = parser.parse_args()
 295     if args.mode == 'prepare':
 296         prepare(args)
 297         return 0
 298     output = new_directory(args.output_dir)
 299     report = dict(format='wbc_policy_io_result_v1', mode=args.mode, status='initializing',
 300                   python=sys.executable, python_version=platform.python_version(), numpy=np.__version__,
 301                   script_sha256=sha256(Path(__file__)), asset_helper_sha256=sha256(Path(__file__).with_name('wbc_lora_assets.py')))
 302     save_json(output / 'report.json', report)
 303     try:
 304         if args.mode == 'board' and (not np.isfinite(args.timeout_seconds) or args.timeout_seconds <= 0):
 305             raise ValueError('Timeout must be finite and positive')
 306         request, arrays = load_cases(args.cases)
 307         graph, catalog = validate_assets(args.manifest, request)
 308         report.update(cases_sha256=sha256(Path(args.cases) / 'cases.json'),
 309                       manifest_sha256=sha256(args.manifest), input_files_sha256=request['files_sha256'],
 310                       numerical_acceptance='Metrics only; no engineering tolerance has been specified.')
 311         runner = run_local if args.mode == 'local' else run_board
 312         actions, runtime = runner(args, output, request, arrays, graph, catalog)
 313         report.update(runtime=runtime, comparison=compare(output, request, arrays, actions), status='completed')
 314     except (Exception, KeyboardInterrupt) as error:
 315         report.update(status='failed', error=f'{type(error).__name__}: {error}')
 316         print(report['error'], file=sys.stderr)
 317     save_json(output / 'report.json', report)
 318     print(json.dumps(dict(status=report['status'], output_dir=str(output), comparison=report.get('comparison')), indent=2))
 319     return 0 if report['status'] == 'completed' else 1
