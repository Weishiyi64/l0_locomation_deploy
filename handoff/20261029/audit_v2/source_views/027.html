  17 bool SequenceRuntime::loadKeyboardActionMap(const std::string& path) {
  18   const std::string resolved_map_path = resolvePackagePath(trimCopy(path), package_share_dir_);
  19   if (resolved_map_path.empty() || !pathExists(resolved_map_path))
  20   {
  21     RCLCPP_ERROR(logger(), "KeyboardAction: map path is missing or not a file: %s", path.c_str());
  22     return false;
  23   }
  24 
  25   YAML::Node root;
  26   try
  27   { root = YAML::LoadFile(resolved_map_path); } catch (const std::exception& e)
  28   {
  29     RCLCPP_ERROR(logger(), "KeyboardAction: failed to load map '%s': %s", resolved_map_path.c_str(), e.what());
  30     return false;
  31   }
  32 
  33   YAML::Node cfg = root["keyboard_actions"] ? root["keyboard_actions"] : root;
  34   const YAML::Node actions = cfg["actions"];
  35   if (!actions || !actions.IsSequence())
  36   {
  37     RCLCPP_ERROR(logger(), "KeyboardAction: '%s' must contain keyboard_actions.actions", resolved_map_path.c_str());
  38     return false;
  39   }
  40 
  41   std::string audio_topic =
  42       cfg["audio_topic"] ? trimCopy(cfg["audio_topic"].as<std::string>("")) : kDefaultKeyboardActionAudioTopic;
  43   if (audio_topic.empty()) audio_topic = kDefaultKeyboardActionAudioTopic;
  44 
  45   std::unordered_map<std::string, KeyboardActionBinding> bindings;
  46   for (const auto& action : actions)
  47   {
  48     if (!action || !action.IsMap())
  49     {
  50       RCLCPP_ERROR(logger(), "KeyboardAction: every actions[] entry must be a map");
  51       return false;
  52     }
  53 
  54     KeyboardActionBinding binding;
  55     binding.name = action["name"] ? trimCopy(action["name"].as<std::string>("")) : "";
  56     binding.key = action["key"] ? trimCopy(action["key"].as<std::string>("")) : "";
  57     binding.sequence_path = action["sequence"] ? trimCopy(action["sequence"].as<std::string>("")) : "";
  58     if (binding.sequence_path.empty() && action["sequence_path"])
  59     {
  60       binding.sequence_path = trimCopy(action["sequence_path"].as<std::string>(""));
  61     }
  62     binding.audio_cmd = action["audio_cmd"] ? trimCopy(action["audio_cmd"].as<std::string>("")) : "";
  63 
  64     if (binding.name.empty()) binding.name = binding.key;
  65     if (binding.name.empty() || binding.key.empty() || binding.sequence_path.empty())
  66     {
  67       RCLCPP_ERROR(logger(),
  68                    "KeyboardAction: each action needs non-empty name/key/sequence (name may be omitted if key exists)");
  69       return false;
  70     }
  71     if (bindings.find(binding.name) != bindings.end())
  72     {
  73       RCLCPP_ERROR(logger(), "KeyboardAction: duplicate action name '%s'", binding.name.c_str());
  74       return false;
  75     }
  76 
  77     const std::string resolved_sequence_path = resolvePackagePath(binding.sequence_path, package_share_dir_);
  78     if (!pathExists(resolved_sequence_path))
  79     {
  80       RCLCPP_ERROR(logger(), "KeyboardAction: action '%s' sequence does not exist: %s", binding.name.c_str(),
  81                    resolved_sequence_path.c_str());
  82       return false;
  83     }
  84 
  85     bool uses_rl_imitation = false;
  86     bool requires_walking_policy = false;
  87     std::string imitation_policy;
  88     if (!inspectSequenceImitation(resolved_sequence_path, logger(), "KeyboardAction", uses_rl_imitation,
  89                                   requires_walking_policy, imitation_policy))
  90     {
  91       return false;
  92     }
  93     if (uses_rl_imitation)
  94     {
  95       binding.uses_rl_imitation = true;
  96       binding.imitation_policy = imitation_policy;
  97     }
  98     binding.requires_walking_policy = requires_walking_policy;
  99     binding.sequence_path = resolved_sequence_path;
 100     bindings[binding.name] = std::move(binding);
 101   }
 102 
 103   if (bindings.empty())
 104   {
 105     RCLCPP_ERROR(logger(), "KeyboardAction: '%s' has no usable actions", resolved_map_path.c_str());
 106     return false;
 107   }
 108 
 109   std::lock_guard<std::recursive_mutex> lock(mutex_);
 110   keyboard_action_config_path_ = resolved_map_path;
 111   keyboard_action_audio_topic_ = audio_topic;
 112   keyboard_action_bindings_ = std::move(bindings);
 113   RCLCPP_INFO(logger(), "KeyboardAction: loaded %zu actions from %s (audio_topic=%s)", keyboard_action_bindings_.size(),
 114               keyboard_action_config_path_.c_str(), keyboard_action_audio_topic_.c_str());
 115   return true;
 116 }

 118 const KeyboardActionBinding* SequenceRuntime::findKeyboardAction(const std::string& action_name) const {
 119   auto it = keyboard_action_bindings_.find(action_name);
 120   if (it == keyboard_action_bindings_.end()) return nullptr;
 121   return &it->second;
 122 }
