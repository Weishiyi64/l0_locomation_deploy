  14 def sha256(path):
  15     digest = hashlib.sha256()
  16     with Path(path).open("rb") as stream:
  17         for block in iter(lambda: stream.read(1024 * 1024), b""):
  18             digest.update(block)
  19     return digest.hexdigest()

  52 def load_graph_manifest(path):
  53     manifest = read_json(path)
  54     if manifest.get("format") != GRAPH_FORMAT or manifest.get("kind") not in ("policy", "probe"):
  55         raise ValueError("Unsupported WBC LoRA graph manifest")
  56     parameters = manifest.get("parameters")
  57     if not isinstance(parameters, list) or len(parameters) != 31:
  58         raise ValueError("Expected the complete 31-tensor WBC adapter schema")
  59     for key in ("base_checkpoint_sha256", "normalization_sha256", "parameter_schema_sha256", "source_onnx_sha256"):
  60         if not isinstance(manifest.get(key), str) or not re.fullmatch(r"[0-9a-f]{64}", manifest[key]):
  61             raise ValueError(f"Missing/invalid required graph fingerprint: {key}")
  62     if manifest.get("target_precision") != "float16":
  63         raise ValueError("The target RKNN precision must be float16")
  64     if fingerprint(parameters) != manifest.get("parameter_schema_sha256"):
  65         raise ValueError("Adapter schema fingerprint mismatch")
  66     names = [p["name"] for p in parameters]
  67     if len(set(names)) != len(names) or manifest.get("rank") != 16 or manifest.get("alpha") != 16:
  68         raise ValueError("Invalid parameter names or unsupported rank/alpha")
  69     dimensions = [(1024, 639), (1024, 1024), (1024, 1024), (1024, 1423), (1024, 1024), (1024, 1024), (21, 1024)]
  70     expected = {}
  71     for i, (out_width, in_width) in enumerate(dimensions):
  72         expected.update({f"lora_{i}_a": [16, in_width], f"lora_{i}_b": [out_width, 16], f"lora_{i}_bias": [out_width]})
  73     for i in range(5):
  74         expected.update({f"norm_{i}_gamma": [1024], f"norm_{i}_beta": [1024]})
  75     if {p["name"]: p["shape"] for p in parameters} != expected:
  76         raise ValueError("Adapter schema does not match the native L0 31-tensor ABI")
  77     for param in parameters:
  78         tensor_shape(param["shape"])
  79         if not isinstance(param.get("actor_key"), str) or not param["actor_key"]:
  80             raise ValueError("Missing actor parameter mapping")
  81     inputs = manifest.get("inputs", [])
  82     if not inputs or len({x["name"] for x in inputs}) != len(inputs):
  83         raise ValueError("Invalid graph inputs")
  84     for item in inputs + manifest.get("outputs", []):
  85         tensor_shape(item["shape"])
  86         if item.get("dtype") != "float32":
  87             raise ValueError("The application ABI must remain float32")
  88     obs = {x["name"]: x["shape"] for x in inputs if x.get("role") == "observation"}
  89     if manifest["kind"] == "policy":
  90         if obs != {"policy": [399], "command_short": [240]} or len(inputs) != 33:
  91             raise ValueError("Policy graph requires 399/240 observations and 31 dynamic parameters")
  92         if manifest["outputs"] != [{"name": "action", "shape": [21], "dtype": "float32"}]:
  93             raise ValueError("Invalid WBC policy output")
  94     declared = {p["name"]: p for p in parameters}
  95     for item in inputs:
  96         if item["role"] == "adapter":
  97             if item["name"] not in declared or item["shape"] != declared[item["name"]]["shape"]:
  98                 raise ValueError(f"Graph parameter does not match schema: {item['name']}")
  99         elif item["role"] != "observation":
 100             raise ValueError("Unknown graph input role")
 101     return manifest

 104 def load_adapter(path, graph, expected_action_id=None):
 105     path = Path(path).resolve()
 106     manifest = read_json(path)
 107     if manifest.get("format") != ADAPTER_FORMAT or not ID_PATTERN.fullmatch(manifest.get("action_id", "")):
 108         raise ValueError("Invalid adapter format or action ID")
 109     if expected_action_id is not None and manifest["action_id"] != expected_action_id:
 110         raise ValueError("Adapter action ID does not match its catalog entry")
 111     for key in ("base_checkpoint_sha256", "parameter_schema_sha256", "normalization_sha256", "rank", "alpha"):
 112         if manifest.get(key) != graph.get(key):
 113             raise ValueError(f"Adapter/graph mismatch: {key}")
 114     weights = asset_path(path.parent, manifest["weights"]["path"])
 115     info = manifest["weights"]
 116     if info.get("dtype") != "float16-le" or weights.stat().st_size != info.get("size_bytes"):
 117         raise ValueError("Adapter dtype/size mismatch")
 118     if sha256(weights) != info.get("sha256"):
 119         raise ValueError("Adapter payload checksum mismatch")
 120     records = manifest.get("tensors", [])
 121     if len(records) != len(graph["parameters"]):
 122         raise ValueError("Incomplete adapter tensor set")
 123     payload = weights.read_bytes()
 124     result, previous_end = {}, 0
 125     for spec, record in zip(graph["parameters"], records):
 126         shape = tensor_shape(spec["shape"])
 127         size = int(np.prod(shape)) * 2
 128         offset = record.get("offset_bytes")
 129         if any(record.get(k) != spec[k] for k in ("name", "actor_key", "shape")):
 130             raise ValueError(f"Adapter tensor mismatch: {spec['name']}")
 131         if type(offset) is not int or offset < previous_end or offset % 64 or record.get("size_bytes") != size:
 132             raise ValueError("Invalid/overlapping adapter tensor range")
 133         if offset + size > len(payload):
 134             raise ValueError("Truncated adapter tensor")
 135         value = np.frombuffer(payload, dtype="<f2", count=size // 2, offset=offset).astype(np.float32).reshape(shape)
 136         if not np.isfinite(value).all():
 137             raise ValueError("Non-finite adapter data")
 138         result[spec["name"]] = value
 139         previous_end = offset + size
 140     if previous_end != len(payload):
 141         raise ValueError("Unexpected trailing adapter data")
 142     return result

 151 def check_graph(path, manifest_path):
 152     import onnx
 153 
 154     path, manifest_path = Path(path).resolve(), Path(manifest_path).resolve()
 155     graph = load_graph_manifest(manifest_path)
 156     if path != asset_path(manifest_path.parent, graph["model"]["onnx"]) or sha256(path) != graph["model"]["sha256"]:
 157         raise ValueError("ONNX path/checksum does not match graph manifest")
 158     model = onnx.load(str(path), load_external_data=False)
 159     if any(t.data_location == onnx.TensorProto.EXTERNAL for t in model.graph.initializer):
 160         raise ValueError("External ONNX tensor data is not supported")
 161     onnx.checker.check_model(model)
 162     initializer_names = {t.name for t in model.graph.initializer}
 163     def specs(items):
 164         values = []
 165         for item in items:
 166             tensor = item.type.tensor_type
 167             shape = [d.dim_value for d in tensor.shape.dim]
 168             if tensor.elem_type != onnx.TensorProto.FLOAT:
 169                 raise ValueError("WBC application inputs/outputs must be float32")
 170             tensor_shape(shape)
 171             values.append({"name": item.name, "shape": shape, "dtype": "float32"})
 172         return values
 173     expected_inputs = [{k: x[k] for k in ("name", "shape", "dtype")} for x in graph["inputs"]]
 174     if specs(model.graph.input) != expected_inputs or specs(model.graph.output) != graph["outputs"]:
 175         raise ValueError("ONNX I/O does not match the manifest ABI")
 176     if initializer_names.intersection(x["name"] for x in graph["inputs"]):
 177         raise ValueError("Dynamic adapter inputs must not be ONNX initializers")
 178     return graph
