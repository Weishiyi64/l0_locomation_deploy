  35   const double norm = std::sqrt(q[0] * q[0] + q[1] * q[1] + q[2] * q[2] + q[3] * q[3]);

  60 void appendRotationFirstTwoRows(const std::array<double, 4>& input, std::vector<float>& out) {
  61   const auto q = normalize(input);
  62   const double w = q[0], x = q[1], y = q[2], z = q[3];
  63   out.push_back(static_cast<float>(1.0 - 2.0 * (y * y + z * z)));
  64   out.push_back(static_cast<float>(2.0 * (x * y - z * w)));
  65   out.push_back(static_cast<float>(2.0 * (x * z + y * w)));
  66   out.push_back(static_cast<float>(2.0 * (x * y + z * w)));
  67   out.push_back(static_cast<float>(1.0 - 2.0 * (x * x + z * z)));
  68   out.push_back(static_cast<float>(2.0 * (y * z - x * w)));

 182 void RLWbc::loadConfig(Motion& motion, const std::string& path) {
 183   const YAML::Node root = YAML::LoadFile(path);
 184   const YAML::Node robot = root["RobotCfg"];
 185   const YAML::Node wbc = root["WbcCfg"];
 186   requireNode(robot, "RobotCfg");
 187   requireNode(wbc, "WbcCfg");
 188   const auto joints = asStringList(robot["joint_order_mj"], "RobotCfg.joint_order_mj");
 189   if (!joint_names_.empty() && joints != joint_names_)
 190     throw std::runtime_error("All WBC motions must have identical observation joint names/order");
 191   if (joint_names_.empty()) joint_names_ = joints;
 192   for (const char* key : {"kps", "kds"})
 193   {
 194     const auto gains = asDoubleList(robot[key], std::string("RobotCfg.") + key);
 195     if (gains.size() != kJointCount || !std::all_of(gains.begin(), gains.end(), [](double value) {
 196           return std::isfinite(value) && value >= 0.0;
 197         }))
 198       throw std::runtime_error(std::string("Invalid WBC gains: ") + key);
 199   }
 200   const auto defaults = asDoubleList(robot["default_joint_angles"], "RobotCfg.default_joint_angles");
 201   const auto scales = asDoubleList(robot["action_scale"], "RobotCfg.action_scale");
 202   if (joint_names_.size() != kJointCount || defaults.size() != kJointCount || scales.size() != kJointCount)
 203     throw std::runtime_error("WBC joint/default/action_scale arrays must each contain 21 values");
 204   if (std::set<std::string>(joint_names_.begin(), joint_names_.end()).size() != joint_names_.size())
 205     throw std::runtime_error("WBC policy joint names must be unique");
 206   motion.default_joint_positions_.assign(defaults.begin(), defaults.end());
 207   motion.action_scale_.assign(scales.begin(), scales.end());
 208   if (!finiteVector(motion.default_joint_positions_) || !finiteVector(motion.action_scale_) ||
 209       std::any_of(motion.action_scale_.begin(), motion.action_scale_.end(), [](float value) { return value <= 0.0f; }))
 210     throw std::runtime_error("WBC defaults must be finite and action scales finite and positive");
 211 
 212   motion.control_dt_ = wbc["control_dt"].as<double>();
 213   motion.start_hold_sec_ = wbc["start_hold_sec"].as<double>();
 214   motion.history_steps_ = wbc["history_steps"].as<std::vector<int>>();
 215   motion.future_steps_ = wbc["future_steps"].as<std::vector<int>>();
 216   motion.previous_action_steps_ = wbc["previous_action_steps"].as<int>();
 217   motion.policy_input_name_ = wbc["policy_input_name"].as<std::string>();
 218   motion.command_input_name_ = wbc["command_input_name"].as<std::string>();
 219   motion.output_name_ = wbc["output_name"].as<std::string>();
 220   if (!std::isfinite(motion.control_dt_) || !std::isfinite(motion.start_hold_sec_) ||
 221       std::abs(motion.control_dt_ - 0.02) > 1e-9 || motion.start_hold_sec_ < 0.0 ||
 222       robot["policy_frequency"].as<int>() != 50 || robot["num_actions"].as<int>() != 21 ||
 223       motion.policy_input_name_ != "policy" ||
 224       (motion.command_input_name_ != "command_short" && motion.command_input_name_ != "command") ||
 225       motion.output_name_ != "action" ||
 226       motion.history_steps_ != std::vector<int>({0, 1, 2, 3, 4, 8, 16}) ||
 227       motion.future_steps_ != std::vector<int>({-8, -4, -2, 0, 1, 2, 3, 4}) || motion.previous_action_steps_ != 3)
 228     throw std::runtime_error("WBC config does not match the exported 399/240 model ABI");
 229 
 230   motion.motion_id_ = wbc["motion_id"].as<std::string>();
 231   if (motion.motion_id_.empty() || motion.motion_id_.size() > 32 ||
 232       !std::all_of(motion.motion_id_.begin(), motion.motion_id_.end(), [](char c) {
 233         return (c >= 'A' && c <= 'Z') || (c >= 'a' && c <= 'z') || (c >= '0' && c <= '9') || c == '-' || c == '_';
 234       }))
 235     throw std::runtime_error("Invalid WBC motion id");
 236   motion.reference_num_frames_ = wbc["reference_num_frames"].as<int>();
 237   motion.adapter_id_ = motion.motion_id_;
 238   motion.reference_root_body_index_ = wbc["reference_root_body_index"].as<int>();
 239   if (motion.reference_num_frames_ <= 0 || motion.reference_num_frames_ > 100000 || motion.reference_root_body_index_ < 0 ||
 240       motion.reference_root_body_index_ >= static_cast<int>(kReferenceBodyCount))
 241     throw std::runtime_error("Invalid WBC reference frame count or root body index");
 242 }

 244 void RLWbc::initializeModel(Motion& motion) {
 245   if (lora_assets_) motion.policy_ = shared_policy_.get();
 246   else {
 247     motion.model_ = std::make_unique<RknnModel>(rknnModelPath(joinPath(motion.config_dir_, "policy.onnx")));
 248     motion.policy_ = motion.model_.get();
 249   }
 250   if (motion.policy_->inputCount() != (lora_assets_ ? 33U : 2U) || motion.policy_->outputCount() != 1)
 251     throw std::runtime_error("WBC RKNN model input/output count does not match selected mode");
 252   motion.policy_input_index_ = motion.policy_->inputIndex(motion.policy_input_name_);
 253   motion.command_input_index_ = motion.policy_->inputIndex(motion.command_input_name_);
 254   if (motion.command_input_index_ < 0)
 255     motion.command_input_index_ =
 256         motion.policy_->inputIndex(motion.command_input_name_ == "command_short" ? "command" : "command_short");
 257   motion.output_index_ = motion.policy_->outputIndex(motion.output_name_);
 258   if (motion.policy_input_index_ < 0 || motion.command_input_index_ < 0 ||
 259       motion.policy_input_index_ == motion.command_input_index_ || motion.output_index_ < 0)
 260     throw std::runtime_error("WBC RKNN input/output names do not match policy, command_short, action");
 261   if (motion.policy_->inputElements(motion.policy_input_index_) != kPolicyInputSize ||
 262       motion.policy_->inputElements(motion.command_input_index_) != kCommandInputSize ||
 263       motion.policy_->outputElements(motion.output_index_) != kJointCount)
 264     throw std::runtime_error("WBC RKNN tensor sizes do not match policy[399], command_short[240], action[21]");
 265 
 266   if (motion.json_reference_) return;
 267   motion.reference_model_ = std::make_unique<RknnModel>(lora_assets_ ? lora_assets_->action(motion.adapter_id_).reference :
 268       rknnModelPath(joinPath(motion.config_dir_, "reference_motion.onnx")));
 269   if (motion.reference_model_->inputCount() != 1 || motion.reference_model_->inputIndex("time_step") != 0 ||
 270       motion.reference_model_->inputElements() != 1 || motion.reference_model_->outputCount() != kReferenceOutputs.size())
 271     throw std::runtime_error("WBC reference RKNN must have time_step[1,1] and six reference outputs");
 272   for (std::size_t i = 0; i < kReferenceOutputs.size(); ++i)
 273   {
 274     const int index = motion.reference_model_->outputIndex(kReferenceOutputs[i]);
 275     if (index < 0 || motion.reference_model_->outputElements(index) != kReferenceOutputSizes[i])
 276       throw std::runtime_error(std::string("WBC reference RKNN output size/name mismatch: ") + kReferenceOutputs[i]);
 277     motion.reference_output_indices_[i] = index;
 278   }
 279 }

 281 bool RLWbc::captureFrame(RobotFrame& frame, std::array<double, 4>* orientation_wxyz, std::string& error,
 282                          int64_t* observation_stamp) {
 283   luvbot_control_interfaces::msg::PolicyObservation sensor;
 284   {
 285     std::lock_guard<std::mutex> lock(observation_mutex_);
 286     const double age =
 287         std::chrono::duration<double>(std::chrono::steady_clock::now() - observation_received_at_).count();
 288     const double stamp_age = (node_->now().nanoseconds() - observation_stamp_ns_) * 1e-9;
 289     if (!observation_received_ || age > sensor_timeout_sec_ || stamp_age < 0.0 || stamp_age > sensor_timeout_sec_ ||
 290         !policy_observation_.inference_enabled)
 291     {
 292       error = "fresh, inference-enabled PolicyObservation is required";
 293       return false;
 294     }
 295     sensor = policy_observation_;
 296     if (observation_stamp) *observation_stamp = observation_stamp_ns_;
 297   }
 298   try
 299   {
 300     const std::array<double, 4> orientation = normalize(
 301         {{sensor.imu.orientation.w, sensor.imu.orientation.x, sensor.imu.orientation.y, sensor.imu.orientation.z}});
 302     const auto gravity = rotateInverse(orientation, {{0.0, 0.0, -1.0}});
 303     const double angular_velocity[] = {sensor.imu.angular_velocity.x, sensor.imu.angular_velocity.y,
 304                                        sensor.imu.angular_velocity.z};
 305     for (std::size_t i = 0; i < 3; ++i)
 306     {
 307       frame.angular_velocity[i] = static_cast<float>(angular_velocity[i]);
 308       frame.projected_gravity[i] = static_cast<float>(gravity[i]);
 309     }
 310     frame.joint_position.resize(kJointCount);
 311     frame.joint_velocity.resize(kJointCount);
 312     for (std::size_t i = 0; i < kJointCount; ++i)
 313     {
 314       frame.joint_position[i] = static_cast<float>(sensor.joints.position[i]) - current_->default_joint_positions_[i];
 315       frame.joint_velocity[i] = static_cast<float>(sensor.joints.velocity[i]);
 316     }
 317     if (orientation_wxyz) *orientation_wxyz = orientation;
 318   } catch (const std::exception& exception)
 319   {
 320     error = exception.what();
 321     return false;
 322   }
 323   if (!finiteVector(frame.joint_position) || !finiteVector(frame.joint_velocity))
 324   {
 325     error = "non-finite WBC sensor input";
 326     return false;
 327   }
 328   return true;
 329 }

 335 RLWbc::ReferenceFrame RLWbc::inferReferenceFrame(const Motion& motion, int index) {
 336   if (motion.json_reference_) {
 337     const auto started = std::chrono::steady_clock::now();
 338     const auto& reference = *motion.json_reference_;
 339     ReferenceFrame frame;
 340     const auto& joints = reference.frame(index);
 341     frame.joint_position.assign(joints.begin(), joints.end());
 342     frame.root_position = reference.rootPosition();
 343     frame.root_orientation = reference.rootOrientation();
 344     ++reference_queries_;
 345     reference_sec_ += std::chrono::duration<double>(std::chrono::steady_clock::now() - started).count();
 346     return frame;
 347   }
 348   const float time_step = static_cast<float>(std::max(0, std::min(index, motion.reference_num_frames_ - 1)));
 349   const auto started = std::chrono::steady_clock::now();
 350   ++reference_queries_;
 351   const bool inferred = motion.reference_model_->run(&time_step, 1);
 352   reference_sec_ += std::chrono::duration<double>(std::chrono::steady_clock::now() - started).count();
 353   if (!inferred)
 354     throw std::runtime_error("WBC reference RKNN inference failed: " + motion.reference_model_->lastError());
 355   ReferenceFrame frame;
 356   frame.joint_position = motion.reference_model_->output(motion.reference_output_indices_[0]);
 357   const auto& positions = motion.reference_model_->output(motion.reference_output_indices_[2]);
 358   const auto& orientations = motion.reference_model_->output(motion.reference_output_indices_[3]);
 359   for (std::size_t i = 0; i < 3; ++i) frame.root_position[i] = positions[motion.reference_root_body_index_ * 3 + i];
 360   double norm_squared = 0.0;
 361   for (std::size_t i = 0; i < 4; ++i)
 362   {
 363     frame.root_orientation[i] = orientations[motion.reference_root_body_index_ * 4 + i];
 364     norm_squared += frame.root_orientation[i] * frame.root_orientation[i];
 365   }
 366   if (!std::isfinite(norm_squared) || std::abs(norm_squared - 1.0) > 0.1)
 367     throw std::runtime_error("WBC reference RKNN returned an invalid root quaternion");
 368   frame.root_orientation = normalize(frame.root_orientation);
 369   return frame;
 370 }

 372 std::vector<float> RLWbc::makePolicyInput() const {
 373   std::vector<float> input;
 374   input.reserve(kPolicyInputSize);
 375   for (int field = 0; field < 4; ++field)
 376   {
 377     for (int offset : current_->history_steps_)
 378     {
 379       const RobotFrame& frame = state_history_.at(static_cast<std::size_t>(offset));
 380       if (field == 0)
 381         input.insert(input.end(), frame.angular_velocity.begin(), frame.angular_velocity.end());
 382       else if (field == 1)
 383         input.insert(input.end(), frame.projected_gravity.begin(), frame.projected_gravity.end());
 384       else if (field == 2)
 385         input.insert(input.end(), frame.joint_position.begin(), frame.joint_position.end());
 386       else
 387         input.insert(input.end(), frame.joint_velocity.begin(), frame.joint_velocity.end());
 388     }
 389   }
 390   for (const auto& action : action_history_) input.insert(input.end(), action.begin(), action.end());
 391   if (input.size() != kPolicyInputSize) throw std::runtime_error("constructed WBC policy input is not 399 elements");
 392   return input;
 393 }

 395 std::vector<float> RLWbc::makeCommandInput(const std::array<double, 4>& robot_orientation, int reference_index,
 396                                          bool reset_anchor) {
 397   std::vector<int> indices;
 398   std::vector<ReferenceFrame> references;
 399   indices.reserve(current_->future_steps_.size());
 400   references.reserve(current_->future_steps_.size());
 401   for (int offset : current_->future_steps_)
 402   {
 403     const int index = std::max(0, std::min(reference_index + offset, current_->reference_num_frames_ - 1));
 404     const auto previous = std::find(indices.begin(), indices.end(), index);
 405     // Reuse only duplicate clamped frames within this control cycle.
 406     if (previous == indices.end())
 407       references.push_back(inferReferenceFrame(index));
 408     else
 409       references.push_back(references[static_cast<std::size_t>(previous - indices.begin())]);
 410     indices.push_back(index);
 411   }
 412   std::vector<float> input;
 413   input.reserve(kCommandInputSize);
 414   const auto& current = references[3];  // future_steps[3] is the current reference frame.
 415   if (reset_anchor)
 416     yaw_anchor_wxyz_ = multiply(yawQuaternion(robot_orientation), conjugate(yawQuaternion(current.root_orientation)));
 417   const auto reference_yaw = yawQuaternion(current.root_orientation);
 418   // Signal-major layout: 8 positions, 8 rotation matrices' first two rows, 8 joint vectors.
 419   for (const auto& future : references)
 420   {
 421     const auto local = rotateInverse(reference_yaw,
 422         {{future.root_position[0] - current.root_position[0],
 423           future.root_position[1] - current.root_position[1], future.root_position[2]}});
 424     for (double value : local) input.push_back(static_cast<float>(value));
 425   }
 426   const auto robot_inverse = conjugate(normalize(robot_orientation));
 427   for (const auto& future : references)
 428   {
 429     appendRotationFirstTwoRows(multiply(multiply(robot_inverse, yaw_anchor_wxyz_), future.root_orientation), input);
 430   }
 431   for (const auto& future : references)
 432   {
 433     const auto& joints = future.joint_position;
 434     input.insert(input.end(), joints.begin(), joints.end());
 435   }
 436   if (input.size() != kCommandInputSize) throw std::runtime_error("constructed WBC command input is not 240 elements");
 437   return input;
 438 }

 440 bool RLWbc::inferAndPublish(double elapsed_sec, bool commit_action, std::string& error) {
 441   const auto inference_started = std::chrono::steady_clock::now();
 442   reference_sec_ = 0.0;
 443   policy_sec_ = 0.0;
 444   reference_queries_ = 0;
 445   int64_t input_stamp = 0;
 446   const bool success = [&]() {
 447     RobotFrame frame;
 448     std::array<double, 4> orientation;
 449     if (!captureFrame(frame, &orientation, error, &input_stamp)) return false;
 450     state_history_.push_front(std::move(frame));
 451     while (state_history_.size() > 17) state_history_.pop_back();
 452 
 453     try
 454     {
 455       const std::vector<float> policy_input = makePolicyInput();
 456       const int reference_index = std::min(current_->reference_num_frames_ - 1,
 457           static_cast<int>(std::max(0.0, elapsed_sec - current_->start_hold_sec_) / current_->control_dt_));
 458       const std::vector<float> command_input = makeCommandInput(orientation, reference_index, !commit_action);
 459       const auto policy_started = std::chrono::steady_clock::now();
 460       bool inferred;
 461       if (lora_assets_) {
 462         inferred = shared_policy_->runResident(policy_input.data(), policy_input.size(),
 463                                                command_input.data(), command_input.size());
 464       } else {
 465         inputs_[current_->policy_input_index_] = policy_input.data();
 466         input_sizes_[current_->policy_input_index_] = policy_input.size();
 467         inputs_[current_->command_input_index_] = command_input.data();
 468         input_sizes_[current_->command_input_index_] = command_input.size();
 469         inferred = current_->policy_->run(inputs_, input_sizes_);
 470       }
 471       policy_sec_ = std::chrono::duration<double>(std::chrono::steady_clock::now() - policy_started).count();
 472       if (!inferred)
 473       {
 474         error = current_->policy_->lastError();
 475         return false;
 476       }
 477       auto action = current_->policy_->output(current_->output_index_);
 478       if (action.size() != kJointCount || !finiteVector(action))
 479       {
 480         error = "WBC policy returned an invalid action";
 481         return false;
 482       }
 483       // Reject late results and observations invalidated during inference.
 484       RobotFrame latest_frame;
 485       if (!captureFrame(latest_frame, nullptr, error)) return false;
 486       const double inference_sec =
 487           std::chrono::duration<double>(std::chrono::steady_clock::now() - inference_started).count();
 488       const double input_age = (node_->now().nanoseconds() - input_stamp) * 1e-9;
 489       if (input_age < 0.0 || input_age > sensor_timeout_sec_ ||
 490           inference_sec > std::min(current_->control_dt_, std::min(sensor_timeout_sec_, command_validity_ms_ / 1000.0)))
 491       {
 492         error = input_age < 0.0 || input_age > sensor_timeout_sec_ ?
 493             "WBC input timestamp expired/invalid" : "WBC inference exceeded command validity budget";
 494         return false;
 495       }
 496       posCmd_.target_positions.resize(kJointCount);
 497       // Bound Qreference in radians after scaling; preserve raw actor history.
 498       for (std::size_t i = 0; i < kJointCount; ++i)
 499         posCmd_.target_positions[i] = std::clamp(
 500             static_cast<double>(action[i]) * current_->action_scale_[i] + current_->default_joint_positions_[i],
 501             -10.0, 10.0);
 502       publishPolicyCommand();
 503       if (commit_action)
 504       {
 505         action_history_.push_front(action);
 506         while (action_history_.size() > static_cast<std::size_t>(current_->previous_action_steps_))
 507           action_history_.pop_back();
 508       }
 509       return true;
 510     } catch (const std::exception& exception)
 511     {
 512       error = exception.what();
 513       return false;
 514     }
 515   }();
 516   const double cycle_sec = std::chrono::duration<double>(std::chrono::steady_clock::now() - inference_started).count();
 517   ++timing_.samples;
 518   if (cycle_sec > std::min(current_->control_dt_, std::min(sensor_timeout_sec_, command_validity_ms_ / 1000.0)))
 519     ++timing_.timeouts;
 520   timing_.reference_total += reference_sec_;
 521   timing_.reference_max = std::max(timing_.reference_max, reference_sec_);
 522   timing_.policy_total += policy_sec_;
 523   timing_.policy_max = std::max(timing_.policy_max, policy_sec_);
 524   timing_.cycle_total += cycle_sec;
 525   timing_.cycle_max = std::max(timing_.cycle_max, cycle_sec);
 526   const auto bucket = [](double seconds) {
 527     return static_cast<std::size_t>(std::min(1000.0, std::max(0.0, seconds * 10000.0)));
 528   };
 529   ++timing_.reference_hist[bucket(reference_sec_)];
 530   ++timing_.policy_hist[bucket(policy_sec_)];
 531   ++timing_.cycle_hist[bucket(cycle_sec)];
 532   if (!success) {
 533     const double budget = std::min(current_->control_dt_, std::min(sensor_timeout_sec_, command_validity_ms_ / 1000.0));
 534     error += "; cycle_ms=" + std::to_string(cycle_sec * 1000.0) + " budget_ms=" + std::to_string(budget * 1000.0) +
 535              " input_age_ms=" + (input_stamp > 0 ? std::to_string((node_->now().nanoseconds() - input_stamp) * 1e-6) : "unavailable") +
 536              " reference_queries=" + std::to_string(reference_queries_);
 537   }
 538   return success;
 539 }

 541 void RLWbc::prepareCallback(const std::shared_ptr<luvbot_control_interfaces::srv::ControlWbcMotion::Request> request,
 542                             std::shared_ptr<luvbot_control_interfaces::srv::ControlWbcMotion::Response> response) {
 543   const auto started = std::chrono::steady_clock::now();
 544   response->motion_id = request->motion_id;
 545   if (request->command == luvbot_control_interfaces::srv::ControlWbcMotion::Request::CANCEL) {
 546     std::lock_guard<std::mutex> lock(mutex_);
 547     ++prepare_generation_;
 548     preparing_ = prepared_ = active_ = false;
 549     response->success = true;
 550     response->message = "WBC cancelled; in-flight loading result invalidated";
 551     return;
 552   }
 553   if (request->command != luvbot_control_interfaces::srv::ControlWbcMotion::Request::START) {
 554     response->message = "unsupported WBC command";
 555     return;
 556   }
 557   // The executor callback is a non-realtime loader. Observation/active callbacks remain runnable.
 558   // A cancelled read can finish, but a new request cannot write the same spare slot until it does.
 559   std::unique_lock<std::mutex> loading_lock(loading_mutex_, std::try_to_lock);
 560   if (!loading_lock.owns_lock()) {
 561     response->message = "previous adapter load is still retiring";
 562     return;
 563   }
 564   Motion* next = nullptr;
 565   int slot;
 566   uint64_t generation;
 567   {
 568     std::lock_guard<std::mutex> lock(mutex_);
 569     if (prepared_ && rclcpp::Time(posCmd_.valid_until) <= node_->now()) prepared_ = false;
 570     if (active_ || prepared_ || preparing_) {
 571       response->message = "a WBC motion is preparing, prepared or active";
 572       return;
 573     }
 574     const auto motion = motions_.find(request->motion_id);
 575     if (motion == motions_.end()) {
 576       response->message = "unknown WBC motion_id";
 577       return;
 578     }
 579     next = motion->second.get();
 580     if (next->json_reference_) {
 581       if (next->json_consumed_ || next->json_reference_->expired()) {
 582         response->message = "JSON reference already consumed/cancelled or expired";
 583         return;
 584       }
 585       next->json_consumed_ = true;  // A failed preparation cannot later replay this request.
 586     }
 587     slot = 1 - active_buffer_;
 588     generation = ++prepare_generation_;
 589     preparing_ = true;
 590   }
 591   std::string error;
 592   const auto load_started = std::chrono::steady_clock::now();
 593   try {
 594     if (lora_assets_) {
 595       lora_assets_->loadAdapter(lora_assets_->action(next->adapter_id_), adapter_buffers_[slot]);
 596       shared_policy_->stageResidentParameters(slot, lora_assets_->parameterInputs(adapter_buffers_[slot]));
 597     }
 598   } catch (const std::exception& exception) {
 599     error = exception.what();
 600   }
 601   const double load_ms = std::chrono::duration<double, std::milli>(std::chrono::steady_clock::now() - load_started).count();
 602   std::lock_guard<std::mutex> lock(mutex_);
 603   if (next->json_reference_ && std::find(json_cancelled_ids_.begin(), json_cancelled_ids_.end(), request->motion_id) !=
 604       json_cancelled_ids_.end() && generation == prepare_generation_) preparing_ = false;
 605   if (generation != prepare_generation_ || !preparing_) {
 606     response->message = "WBC preparation was cancelled";
 607     return;
 608   }
 609   preparing_ = false;
 610   const auto elapsed = [&] { return std::chrono::duration<double>(std::chrono::steady_clock::now() - started).count(); };
 611   if (!error.empty() || elapsed() >= prepare_timeout_sec_) {
 612     response->message = error.empty() ? "adapter load exceeded preparation budget" : error;
 613     RCLCPP_WARN(node_->get_logger(), "WBC prepare %s rejected: %s", request->motion_id.c_str(), response->message.c_str());
 614     return;
 615   }
 616   // No inference is in flight while mutex_ is held. Never mutate the active adapter buffer.
 617   Motion* previous = current_;
 618   const int previous_slot = active_buffer_;
 619   const int previous_resident_slot = shared_policy_ ? shared_policy_->residentSlot() : -1;
 620   // Do allocating snapshots before committing the new device bindings.
 621   const auto previous_command = posCmd_;
 622   if (shared_policy_ && !shared_policy_->commitResidentParameters(slot)) {
 623     response->message = shared_policy_->lastError();
 624     RCLCPP_ERROR(node_->get_logger(), "WBC prepare %s rejected: %s", request->motion_id.c_str(), response->message.c_str());
 625     return;
 626   }
 627   // Keep the complete inactive motion state for rollback after first-frame failure.
 628   const auto previous_anchor = yaw_anchor_wxyz_;
 629   auto previous_states = std::move(state_history_);
 630   auto previous_actions = std::move(action_history_);
 631   const auto previous_timing = timing_;
 632   const bool previous_done = done_published_;
 633   current_ = next;
 634   active_buffer_ = slot;
 635   timing_ = TimingStats{};
 636   yaw_anchor_wxyz_ = {{1.0, 0.0, 0.0, 0.0}};
 637   RobotFrame initial_frame;
 638   try {
 639     if (elapsed() < prepare_timeout_sec_ && captureFrame(initial_frame, nullptr, error)) {
 640       // Sensor/start validity is decided by the same captureFrame path for
 641       // ordinary and JSON motions. Only stale JSON payload matching is extra.
 642       if (current_->json_reference_) {
 643         const auto& initial = current_->json_reference_->start();
 644         for (std::size_t j = 0; j < kJointCount; ++j) {
 645           const float observed = initial_frame.joint_position[j] + current_->default_joint_positions_[j];
 646           if (std::abs(observed - initial[j]) > 0.15)
 647             throw std::runtime_error("robot pose changed since JSON generation");
 648         }
 649       }
 650       state_history_.assign(17, initial_frame);
 651       action_history_.assign(static_cast<std::size_t>(current_->previous_action_steps_), std::vector<float>(kJointCount, 0.0f));
 652       done_published_ = false;
 653       if (inferAndPublish(0.0, false, error) && elapsed() < prepare_timeout_sec_) {
 654         response->robot_cfg_path = current_->robot_cfg_path_;
 655         response->duration_sec = current_->start_hold_sec_ + current_->reference_num_frames_ * current_->control_dt_;
 656         response->message = "WBC motion prepared";
 657         logTiming("prepare");
 658         if (elapsed() < prepare_timeout_sec_) {
 659           timing_ = TimingStats{};
 660           prepared_ = true;
 661           response->success = true;
 662           RCLCPP_INFO(node_->get_logger(), "WBC prepared action=%s generation=%llu slot=%d shared_context=%s %.2f ms load_pack=%.2f ms",
 663                       request->motion_id.c_str(), static_cast<unsigned long long>(generation), slot,
 664                       lora_assets_ ? "reused" : "static", elapsed() * 1000.0, load_ms);
 665           return;
 666         }
 667         error = "preparation deadline after diagnostics";
 668       }
 669     }
 670   } catch (const std::exception& exception) {
 671     error = exception.what();
 672   }
 673   logTiming("prepare_failed");
 674   if (shared_policy_ && !shared_policy_->restoreResidentParameters(previous_resident_slot))
 675     error += "; " + shared_policy_->lastError();
 676   current_ = previous;
 677   active_buffer_ = previous_slot;
 678   yaw_anchor_wxyz_ = previous_anchor;
 679   state_history_ = std::move(previous_states);
 680   action_history_ = std::move(previous_actions);
 681   timing_ = previous_timing;
 682   done_published_ = previous_done;
 683   posCmd_ = previous_command;
 684   response->message = "WBC pre-inference failed: " + (error.empty() ? std::string("preparation deadline") : error);
 685   RCLCPP_WARN(node_->get_logger(), "%s", response->message.c_str());
 686 }

 708   prepared_ = false;

 793 void RLWbc::runLoop() {
 794   rclcpp::WallRate rate(50.0);
 795   while (loop_running_.load() && rclcpp::ok())
 796   {
 797     {
 798       std::lock_guard<std::mutex> lock(mutex_);
 799       if (active_)
 800       {
 801         const double elapsed = std::chrono::duration<double>(std::chrono::steady_clock::now() - motion_start_).count();
 802         const double duration = current_->start_hold_sec_ + current_->reference_num_frames_ * current_->control_dt_;
 803         RobotFrame current_frame;
 804         std::string state_error;
 805         if (!captureFrame(current_frame, nullptr, state_error))
 806         {
 807           active_ = false;
 808           publishDone(false, state_error.c_str());
 809         }
 810         else if (elapsed >= duration)
 811         {
 812           active_ = false;
 813           publishDone(true, "completed");
 814         }
 815         else
 816         {
 817           std::string error;
 818           if (!inferAndPublish(elapsed, true, error))
 819           {
 820             active_ = false;
 821             RCLCPP_ERROR(node_->get_logger(), "WBC inference failed: %s", error.c_str());
 822             publishDone(false, "runtime failure");
 823           }
 824         }
 825       }
 826     }
 827     rate.sleep();
 828   }

  47   return {{a[0] * b[0] - a[1] * b[1] - a[2] * b[2] - a[3] * b[3], a[0] * b[1] + a[1] * b[0] + a[2] * b[3] - a[3] * b[2],
  48            a[0] * b[2] - a[1] * b[3] + a[2] * b[0] + a[3] * b[1],
  49            a[0] * b[3] + a[1] * b[2] - a[2] * b[1] + a[3] * b[0]}};
  50 }
  51 
  52 std::array<double, 3> rotateInverse(const std::array<double, 4>& orientation, const std::array<double, 3>& vector) {
  53   const auto q = normalize(orientation);
  54   const auto rotated = multiply(multiply(conjugate(q), {{0.0, vector[0], vector[1], vector[2]}}), q);
  55   return {{rotated[1], rotated[2], rotated[3]}};
  56 }
  57 
  58 std::array<double, 4> yawQuaternion(const std::array<double, 4>& orientation) {
  59   const auto q = normalize(orientation);
  60   const double yaw = std::atan2(2.0 * (q[0] * q[3] + q[1] * q[2]), 1.0 - 2.0 * (q[2] * q[2] + q[3] * q[3]));
  61   return {{std::cos(yaw * 0.5), 0.0, 0.0, std::sin(yaw * 0.5)}};
  62 }
  63 
  64 void appendRotationFirstTwoRows(const std::array<double, 4>& input, std::vector<float>& out) {
  65   const auto q = normalize(input);
  66   const double w = q[0], x = q[1], y = q[2], z = q[3];
  67   out.push_back(static_cast<float>(1.0 - 2.0 * (y * y + z * z)));
  68   out.push_back(static_cast<float>(2.0 * (x * y - z * w)));
  69   out.push_back(static_cast<float>(2.0 * (x * z + y * w)));
  70   out.push_back(static_cast<float>(2.0 * (x * y + z * w)));
  71   out.push_back(static_cast<float>(1.0 - 2.0 * (x * x + z * z)));
  72   out.push_back(static_cast<float>(2.0 * (y * z - x * w)));
  73 }

 126 void RLWbc::loadConfig(const std::string& path) {
 127   const YAML::Node root = YAML::LoadFile(path);
 128   const YAML::Node robot = root["RobotCfg"];
 129   const YAML::Node wbc = root["WbcCfg"];
 130   requireNode(robot, "RobotCfg");
 131   requireNode(wbc, "WbcCfg");
 132   joint_names_ = asStringList(robot["joint_order_mj"], "RobotCfg.joint_order_mj");
 133   const auto defaults = asDoubleList(robot["default_joint_angles"], "RobotCfg.default_joint_angles");
 134   const auto scales = asDoubleList(robot["action_scale"], "RobotCfg.action_scale");
 135   if (joint_names_.size() != kJointCount || defaults.size() != kJointCount || scales.size() != kJointCount)
 136     throw std::runtime_error("WBC joint/default/action_scale arrays must each contain 21 values");
 137   if (std::set<std::string>(joint_names_.begin(), joint_names_.end()).size() != joint_names_.size())
 138     throw std::runtime_error("WBC policy joint names must be unique");
 139   default_joint_positions_.assign(defaults.begin(), defaults.end());
 140   action_scale_.assign(scales.begin(), scales.end());
 141   if (!finiteVector(default_joint_positions_) || !finiteVector(action_scale_) ||
 142       std::any_of(action_scale_.begin(), action_scale_.end(), [](float value) { return value <= 0.0f; }))
 143     throw std::runtime_error("WBC defaults must be finite and action scales finite and positive");
 144 
 145   clip_actions_ = robot["clip_actions"].as<float>();
 146   control_dt_ = wbc["control_dt"].as<double>();
 147   start_hold_sec_ = wbc["start_hold_sec"].as<double>();
 148   history_steps_ = wbc["history_steps"].as<std::vector<int>>();
 149   future_steps_ = wbc["future_steps"].as<std::vector<int>>();
 150   previous_action_steps_ = wbc["previous_action_steps"].as<int>();
 151   policy_input_name_ = wbc["policy_input_name"].as<std::string>();
 152   command_input_name_ = wbc["command_input_name"].as<std::string>();
 153   output_name_ = wbc["output_name"].as<std::string>();
 154   if (!std::isfinite(control_dt_) || !std::isfinite(start_hold_sec_) || !std::isfinite(clip_actions_) ||
 155       std::abs(control_dt_ - 0.02) > 1e-9 || start_hold_sec_ < 0.0 || clip_actions_ <= 0.0f ||
 156       robot["policy_frequency"].as<int>() != 50 || robot["num_actions"].as<int>() != 21 ||
 157       policy_input_name_ != "policy" || (command_input_name_ != "command_short" && command_input_name_ != "command") ||
 158       output_name_ != "action" ||
 159       history_steps_ != std::vector<int>({0, 1, 2, 3, 4, 8, 16}) ||
 160       future_steps_ != std::vector<int>({-8, -4, -2, 0, 1, 2, 3, 4}) || previous_action_steps_ != 3)
 161     throw std::runtime_error("WBC config does not match the exported 399/240 model ABI");
 162 
 163   motion_id_ = wbc["motion_id"].as<std::string>();
 164   if (motion_id_.empty() || motion_id_.size() > 32 ||
 165       !std::all_of(motion_id_.begin(), motion_id_.end(), [](char c) {
 166         return (c >= 'A' && c <= 'Z') || (c >= 'a' && c <= 'z') || (c >= '0' && c <= '9') || c == '-' || c == '_';
 167       }))
 168     throw std::runtime_error("Invalid WBC motion id");
 169   reference_num_frames_ = wbc["reference_num_frames"].as<int>();
 170   reference_root_body_index_ = wbc["reference_root_body_index"].as<int>();
 171   if (reference_num_frames_ <= 0 || reference_num_frames_ > 100000 || reference_root_body_index_ < 0 ||
 172       reference_root_body_index_ >= static_cast<int>(kReferenceBodyCount))
 173     throw std::runtime_error("Invalid WBC reference frame count or root body index");
 174 }

 176 void RLWbc::initializeModel() {
 177   model_ = std::make_unique<RknnModel>(rknnModelPath(joinPath(config_dir_, "policy.onnx")));
 178   if (model_->inputCount() != 2 || model_->outputCount() != 1)
 179     throw std::runtime_error("WBC RKNN model must have exactly two inputs and exactly one output");
 180   policy_input_index_ = model_->inputIndex(policy_input_name_);
 181   command_input_index_ = model_->inputIndex(command_input_name_);
 182   if (command_input_index_ < 0)
 183     command_input_index_ = model_->inputIndex(command_input_name_ == "command_short" ? "command" : "command_short");
 184   output_index_ = model_->outputIndex(output_name_);
 185   if (policy_input_index_ < 0 || command_input_index_ < 0 || policy_input_index_ == command_input_index_ || output_index_ < 0)
 186     throw std::runtime_error("WBC RKNN input/output names do not match policy, command_short, action");
 187   if (model_->inputElements(policy_input_index_) != kPolicyInputSize ||
 188       model_->inputElements(command_input_index_) != kCommandInputSize ||
 189       model_->outputElements(output_index_) != kJointCount)
 190     throw std::runtime_error("WBC RKNN tensor sizes do not match policy[399], command_short[240], action[21]");
 191 
 192   reference_model_ = std::make_unique<RknnModel>(rknnModelPath(joinPath(config_dir_, "reference_motion.onnx")));
 193   if (reference_model_->inputCount() != 1 || reference_model_->inputIndex("time_step") != 0 ||
 194       reference_model_->inputElements() != 1 || reference_model_->outputCount() != kReferenceOutputs.size())
 195     throw std::runtime_error("WBC reference RKNN must have time_step[1,1] and six reference outputs");
 196   for (std::size_t i = 0; i < kReferenceOutputs.size(); ++i)
 197   {
 198     const int index = reference_model_->outputIndex(kReferenceOutputs[i]);
 199     if (index < 0 || reference_model_->outputElements(index) != kReferenceOutputSizes[i])
 200       throw std::runtime_error(std::string("WBC reference RKNN output size/name mismatch: ") + kReferenceOutputs[i]);
 201     reference_output_indices_[i] = index;
 202   }
 203 }

 205 bool RLWbc::captureFrame(RobotFrame& frame, std::array<double, 4>* orientation_wxyz, std::string& error,
 206                          int64_t* observation_stamp) {
 207   luvbot_control_interfaces::msg::PolicyObservation sensor;
 208   {
 209     std::lock_guard<std::mutex> lock(observation_mutex_);
 210     const double age =
 211         std::chrono::duration<double>(std::chrono::steady_clock::now() - observation_received_at_).count();
 212     const double stamp_age = (node_->now().nanoseconds() - observation_stamp_ns_) * 1e-9;
 213     if (!observation_received_ || age > sensor_timeout_sec_ || stamp_age < 0.0 || stamp_age > sensor_timeout_sec_ ||
 214         !policy_observation_.inference_enabled)
 215     {
 216       error = "fresh, inference-enabled PolicyObservation is required";
 217       return false;
 218     }
 219     sensor = policy_observation_;
 220     if (observation_stamp) *observation_stamp = observation_stamp_ns_;
 221   }
 222   try
 223   {
 224     const std::array<double, 4> orientation = normalize(
 225         {{sensor.imu.orientation.w, sensor.imu.orientation.x, sensor.imu.orientation.y, sensor.imu.orientation.z}});
 226     const auto gravity = rotateInverse(orientation, {{0.0, 0.0, -1.0}});
 227     const double angular_velocity[] = {sensor.imu.angular_velocity.x, sensor.imu.angular_velocity.y,
 228                                        sensor.imu.angular_velocity.z};
 229     for (std::size_t i = 0; i < 3; ++i)
 230     {
 231       frame.angular_velocity[i] = static_cast<float>(angular_velocity[i]);
 232       frame.projected_gravity[i] = static_cast<float>(gravity[i]);
 233     }
 234     frame.joint_position.resize(kJointCount);
 235     frame.joint_velocity.resize(kJointCount);
 236     for (std::size_t i = 0; i < kJointCount; ++i)
 237     {
 238       frame.joint_position[i] = static_cast<float>(sensor.joints.position[i]) - default_joint_positions_[i];
 239       frame.joint_velocity[i] = static_cast<float>(sensor.joints.velocity[i]);
 240     }
 241     if (orientation_wxyz) *orientation_wxyz = orientation;
 242   } catch (const std::exception& exception)
 243   {
 244     error = exception.what();
 245     return false;
 246   }
 247   if (!finiteVector(frame.joint_position) || !finiteVector(frame.joint_velocity))
 248   {
 249     error = "non-finite WBC sensor input";
 250     return false;
 251   }
 252   return true;
 253 }

 255 RLWbc::ReferenceFrame RLWbc::inferReferenceFrame(int index) {
 256   const float time_step = static_cast<float>(std::max(0, std::min(index, reference_num_frames_ - 1)));
 257   if (!reference_model_->run(&time_step, 1))
 258     throw std::runtime_error("WBC reference RKNN inference failed: " + reference_model_->lastError());
 259   ReferenceFrame frame;
 260   frame.joint_position = reference_model_->output(reference_output_indices_[0]);
 261   const auto& positions = reference_model_->output(reference_output_indices_[2]);
 262   const auto& orientations = reference_model_->output(reference_output_indices_[3]);
 263   for (std::size_t i = 0; i < 3; ++i) frame.root_position[i] = positions[reference_root_body_index_ * 3 + i];
 264   double norm_squared = 0.0;
 265   for (std::size_t i = 0; i < 4; ++i)
 266   {
 267     frame.root_orientation[i] = orientations[reference_root_body_index_ * 4 + i];
 268     norm_squared += frame.root_orientation[i] * frame.root_orientation[i];
 269   }
 270   if (!std::isfinite(norm_squared) || std::abs(norm_squared - 1.0) > 0.1)
 271     throw std::runtime_error("WBC reference RKNN returned an invalid root quaternion");
 272   frame.root_orientation = normalize(frame.root_orientation);
 273   return frame;
 274 }

 276 std::vector<float> RLWbc::makePolicyInput() const {
 277   std::vector<float> input;
 278   input.reserve(kPolicyInputSize);
 279   for (int field = 0; field < 4; ++field)
 280   {
 281     for (int offset : history_steps_)
 282     {
 283       const RobotFrame& frame = state_history_.at(static_cast<std::size_t>(offset));
 284       if (field == 0)
 285         input.insert(input.end(), frame.angular_velocity.begin(), frame.angular_velocity.end());
 286       else if (field == 1)
 287         input.insert(input.end(), frame.projected_gravity.begin(), frame.projected_gravity.end());
 288       else if (field == 2)
 289         input.insert(input.end(), frame.joint_position.begin(), frame.joint_position.end());
 290       else
 291         input.insert(input.end(), frame.joint_velocity.begin(), frame.joint_velocity.end());
 292     }
 293   }
 294   for (const auto& action : action_history_) input.insert(input.end(), action.begin(), action.end());
 295   if (input.size() != kPolicyInputSize) throw std::runtime_error("constructed WBC policy input is not 399 elements");
 296   return input;
 297 }

 299 std::vector<float> RLWbc::makeCommandInput(const std::array<double, 4>& robot_orientation, int reference_index,
 300                                          bool reset_anchor) {
 301   std::vector<int> indices;
 302   std::vector<ReferenceFrame> references;
 303   indices.reserve(future_steps_.size());
 304   references.reserve(future_steps_.size());
 305   for (int offset : future_steps_)
 306   {
 307     const int index = std::max(0, std::min(reference_index + offset, reference_num_frames_ - 1));
 308     const auto previous = std::find(indices.begin(), indices.end(), index);
 309     // Reuse only duplicate clamped frames within this control cycle.
 310     if (previous == indices.end())
 311       references.push_back(inferReferenceFrame(index));
 312     else
 313       references.push_back(references[static_cast<std::size_t>(previous - indices.begin())]);
 314     indices.push_back(index);
 315   }
 316   std::vector<float> input;
 317   input.reserve(kCommandInputSize);
 318   const auto& current = references[3];  // future_steps[3] is the current reference frame.
 319   if (reset_anchor)
 320     yaw_anchor_wxyz_ = multiply(yawQuaternion(robot_orientation), conjugate(yawQuaternion(current.root_orientation)));
 321   const auto reference_yaw = yawQuaternion(current.root_orientation);
 322   // Signal-major layout: 8 positions, 8 rotation matrices' first two rows, 8 joint vectors.
 323   for (const auto& future : references)
 324   {
 325     const auto local = rotateInverse(reference_yaw,
 326         {{future.root_position[0] - current.root_position[0],
 327           future.root_position[1] - current.root_position[1], future.root_position[2]}});
 328     for (double value : local) input.push_back(static_cast<float>(value));
 329   }
 330   const auto robot_inverse = conjugate(normalize(robot_orientation));
 331   for (const auto& future : references)
 332   {
 333     appendRotationFirstTwoRows(multiply(multiply(robot_inverse, yaw_anchor_wxyz_), future.root_orientation), input);
 334   }
 335   for (const auto& future : references)
 336   {
 337     const auto& joints = future.joint_position;
 338     input.insert(input.end(), joints.begin(), joints.end());
 339   }
 340   if (input.size() != kCommandInputSize) throw std::runtime_error("constructed WBC command input is not 240 elements");
 341   return input;
 342 }

 344 bool RLWbc::inferAndPublish(double elapsed_sec, bool commit_action, std::string& error) {
 345   const auto inference_started = std::chrono::steady_clock::now();
 346   RobotFrame frame;
 347   std::array<double, 4> orientation;
 348   int64_t input_stamp = 0;
 349   if (!captureFrame(frame, &orientation, error, &input_stamp)) return false;
 350   state_history_.push_front(std::move(frame));
 351   while (state_history_.size() > 17) state_history_.pop_back();
 352 
 353   try
 354   {
 355     const std::vector<float> policy_input = makePolicyInput();
 356     const int reference_index = std::min(reference_num_frames_ - 1,
 357         static_cast<int>(std::max(0.0, elapsed_sec - start_hold_sec_) / control_dt_));
 358     const std::vector<float> command_input = makeCommandInput(orientation, reference_index, !commit_action);
 359     std::vector<const float*> inputs(model_->inputCount(), nullptr);
 360     std::vector<std::size_t> sizes(model_->inputCount(), 0);
 361     inputs[policy_input_index_] = policy_input.data();
 362     sizes[policy_input_index_] = policy_input.size();
 363     inputs[command_input_index_] = command_input.data();
 364     sizes[command_input_index_] = command_input.size();
 365     if (!model_->run(inputs, sizes))
 366     {
 367       error = model_->lastError();
 368       return false;
 369     }
 370     auto action = model_->output(output_index_);
 371     if (action.size() != kJointCount || !finiteVector(action))
 372     {
 373       error = "WBC policy returned an invalid action";
 374       return false;
 375     }
 376     // Reject late results and observations invalidated during inference.
 377     RobotFrame latest_frame;
 378     if (!captureFrame(latest_frame, nullptr, error)) return false;
 379     const double inference_sec =
 380         std::chrono::duration<double>(std::chrono::steady_clock::now() - inference_started).count();
 381     const double input_age = (node_->now().nanoseconds() - input_stamp) * 1e-9;
 382     if (input_age < 0.0 || input_age > sensor_timeout_sec_ ||
 383         inference_sec > std::min(control_dt_, std::min(sensor_timeout_sec_, command_validity_ms_ / 1000.0)))
 384     {
 385       error = "WBC inference exceeded command validity budget";
 386       return false;
 387     }
 388     for (float& value : action) value = std::max(-clip_actions_, std::min(value, clip_actions_));
 389     posCmd_.target_positions.resize(kJointCount);
 390     for (std::size_t i = 0; i < kJointCount; ++i)
 391       posCmd_.target_positions[i] = static_cast<double>(action[i]) * action_scale_[i] + default_joint_positions_[i];
 392     publishPolicyCommand();
 393     if (commit_action)
 394     {
 395       action_history_.push_front(action);
 396       while (action_history_.size() > static_cast<std::size_t>(previous_action_steps_)) action_history_.pop_back();
 397     }
 398     return true;
 399   } catch (const std::exception& exception)
 400   {
 401     error = exception.what();
 402     return false;
 403   }
 404 }

 406 void RLWbc::prepareCallback(const std::shared_ptr<luvbot_control_interfaces::srv::ControlWbcMotion::Request> request,
 407                             std::shared_ptr<luvbot_control_interfaces::srv::ControlWbcMotion::Response> response) {
 408   std::lock_guard<std::mutex> lock(mutex_);
 409   response->motion_id = request->motion_id;
 410   response->robot_cfg_path = robot_cfg_path_;
 411   if (request->command == luvbot_control_interfaces::srv::ControlWbcMotion::Request::CANCEL)
 412   {
 413     active_ = false;
 414     prepared_ = false;
 415     response->success = true;
 416     response->message = "WBC motion cancelled";
 417     return;
 418   }
 419   if (request->command != luvbot_control_interfaces::srv::ControlWbcMotion::Request::START)
 420   {
 421     response->message = "unsupported WBC command";
 422     return;
 423   }
 424   if (active_ || prepared_)
 425   {
 426     response->message = "a WBC motion is already prepared or active";
 427     return;
 428   }
 429   if (request->motion_id != motion_id_)
 430   {
 431     response->message = "unknown WBC motion_id";
 432     return;
 433   }
 434   RobotFrame initial_frame;
 435   std::string error;
 436   if (!captureFrame(initial_frame, nullptr, error))
 437   {
 438     response->message = "cannot prepare WBC: " + error;
 439     return;
 440   }
 441   state_history_.assign(17, initial_frame);
 442   action_history_.assign(static_cast<std::size_t>(previous_action_steps_), std::vector<float>(kJointCount, 0.0f));
 443   done_published_ = false;
 444   if (!inferAndPublish(0.0, false, error))
 445   {
 446     response->message = "WBC pre-inference failed: " + error;
 447     return;
 448   }
 449   prepared_ = true;
 450   response->success = true;
 451   response->message = "WBC motion prepared";
 452   response->motion_id = motion_id_;
 453   response->duration_sec = start_hold_sec_ + reference_num_frames_ * control_dt_;
 454 }

 456 void RLWbc::activeCallback(const std_msgs::msg::Bool::ConstSharedPtr msg) {
 457   std::lock_guard<std::mutex> lock(mutex_);
 458   if (!msg->data)
 459   {
 460     active_ = false;
 461     prepared_ = false;
 462     return;
 463   }
 464   if (!prepared_)
 465   {
 466     RCLCPP_ERROR(node_->get_logger(), "Ignoring control/policy/wbc/active=true without a prepared motion");
 467     return;
 468   }
 469   motion_start_ = std::chrono::steady_clock::now();
 470   active_ = true;
 471   prepared_ = false;
 472 }

 474 void RLWbc::stateCallback(const luvbot_control_interfaces::msg::PolicyObservation::ConstSharedPtr msg) {
 475   const int64_t stamp_ns = rclcpp::Time(msg->header.stamp).nanoseconds();
 476   const double age = (node_->now().nanoseconds() - stamp_ns) * 1e-9;
 477   const bool valid = validatePolicyObservation(*msg, joint_names_, "WBC");
 478   std::lock_guard<std::mutex> lock(observation_mutex_);
 479   if (!valid || stamp_ns <= 0 || age < 0.0 || age > sensor_timeout_sec_ ||
 480       (observation_received_ && stamp_ns <= observation_stamp_ns_))
 481   {
 482     observation_received_ = false;
 483     return;
 484   }
 485   policy_observation_ = *msg;
 486   observation_received_at_ = std::chrono::steady_clock::now();
 487   observation_stamp_ns_ = stamp_ns;
 488   observation_received_ = true;
 489 }

 500 void RLWbc::runLoop() {
 501   rclcpp::WallRate rate(50.0);
 502   while (loop_running_.load() && rclcpp::ok())
 503   {
 504     {
 505       std::lock_guard<std::mutex> lock(mutex_);
 506       if (active_)
 507       {
 508         const double elapsed = std::chrono::duration<double>(std::chrono::steady_clock::now() - motion_start_).count();
 509         const double duration = start_hold_sec_ + reference_num_frames_ * control_dt_;
 510         RobotFrame current_frame;
 511         std::string state_error;
 512         if (!captureFrame(current_frame, nullptr, state_error))
 513         {
 514           active_ = false;
 515           publishDone(false, state_error.c_str());
 516         }
 517         else if (elapsed >= duration)
 518         {
 519           active_ = false;
 520           publishDone(true, "completed");
 521         }
 522         else
 523         {
 524           std::string error;
 525           if (!inferAndPublish(elapsed, true, error))
 526           {
 527             active_ = false;
 528             RCLCPP_ERROR(node_->get_logger(), "WBC inference failed: %s", error.c_str());
 529             publishDone(false, "runtime failure");
 530           }
 531         }
 532       }
 533     }
 534     rate.sleep();
 535   }
 536 }
