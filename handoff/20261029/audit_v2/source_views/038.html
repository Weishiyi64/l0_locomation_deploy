  37 std::vector<float> readFloats(const fs::path& path, std::size_t count) {
  38   if (!fs::is_regular_file(path) || fs::file_size(path) != count * sizeof(float))
  39     throw std::runtime_error("Wrong input byte count: " + path.string());
  40   std::vector<float> values(count);
  41   std::ifstream stream(path, std::ios::binary);
  42   if (!stream.read(reinterpret_cast<char*>(values.data()), count * sizeof(float)) ||
  43       !std::all_of(values.begin(), values.end(), [](float value) { return std::isfinite(value); }))
  44     throw std::runtime_error("Invalid/non-finite input: " + path.string());
  45   return values;
  46 }

  70     // Do not overwrite a previous run, including failed or interrupted evidence.

  77       throw std::runtime_error("Probe binary format requires little-endian host");

  79     const auto request = YAML::LoadFile((cases_root / "cases.json").string());
  80     const auto cases = request["cases"];
  81     if (request["format"].as<std::string>() != "wbc_policy_io_cases_v1" ||
  82         !cases.IsSequence() || cases.size() == 0 || cases.size() > 8192)
  83       throw std::runtime_error("Invalid probe case format/count");
  84     const auto policy = readFloats(cases_root / "policy.f32", cases.size() * 399);
  85     const auto command = readFloats(cases_root / "command_short.f32", cases.size() * 240);
  86     std::vector<std::string> selected;
  87     for (const auto& item : cases) {
  88       const auto id = item["action_id"].as<std::string>();
  89       if (std::find(selected.begin(), selected.end(), id) == selected.end()) selected.push_back(id);
  90     }
  91 
  92     // These are the SAME loader, native tensor buffers and SDK calls used by RLWbc.
  93     const auto initialized = std::chrono::steady_clock::now();
  94     WbcLoraAssets assets(argv[2], selected);
  95     RknnModel model(assets.modelPath(), RknnModel::InputMode::ResidentWbc);
  96     assets.bindModel(model);
  97     std::array<WbcLoraAssets::Buffer, 2> buffers;
  98     for (auto& buffer : buffers) assets.allocate(buffer);
  99     report["initialization_ms"] = elapsedMs(initialized);
 100     report["model_path"] = assets.modelPath();
 101     report["base_checkpoint_sha256"] = assets.baseHash();
 102     report["sdk_version"] = model.runtimeVersion();
 103     report["execution_path"] = "WbcLoraAssets + RknnModel::ResidentWbc (same as RLWbc)";
 104     report["scope"] = "policy inference only; fixed policy/command_short; no reference queries, ROS or motor output";
 105     report["requested_samples"] = cases.size();
 106     report["status"] = "running";
 107     writeYaml(output / "runtime.yaml", report);
 108     saveRuntimeMaps(output);
 109     std::ofstream layout(output / "resident_layout.txt");
 110     layout << model.residentLayout() << '\n';
 111     layout.close();
 112     if (!layout) throw std::runtime_error("Cannot save resident layout");

 114     std::ofstream values(output / "actions.f32", std::ios::binary);
 115     std::ofstream frames(output / "frames.csv");
 116     frames << "case_index,action_id,slot,generation,inference_ms,runs,parameter_prepares,parameter_binds,parameter_syncs,observation_syncs,output_syncs,observation_ms_total,execution_ms_total,output_ms_total\n";
 117     frames << std::setprecision(17);
 118     int active_slot = 0;
 119     std::uint64_t generation = 0;
 120     std::string active_action;
 121     for (std::size_t i = 0; i < cases.size(); ++i) {
 122       if (stop_requested) throw std::runtime_error("Probe interrupted; completed outputs retained");
 123       const auto id = cases[i]["action_id"].as<std::string>();
 124       if (id != active_action) {
 125         const auto load_started = std::chrono::steady_clock::now();
 126         const int slot = 1 - active_slot;
 127         assets.loadAdapter(assets.action(id), buffers[slot]);
 128         model.stageResidentParameters(slot, assets.parameterInputs(buffers[slot]));
 129         if (!model.commitResidentParameters(slot)) throw std::runtime_error(model.lastError());

 139         change["adapter_manifest_sha256"] = assets.action(id).adapter_hash;
 140         report["switches"].push_back(change);

 142       }

 144       if (!model.runResident(policy.data() + i * 399, 399, command.data() + i * 240, 240))

 146       const double inference_ms = elapsedMs(started);
 147       const auto& action = model.output(0);
 148       if (action.size() != 21 || !std::all_of(action.begin(), action.end(), [](float x) { return std::isfinite(x); }))
 149         throw std::runtime_error("Invalid action output");
 150       // Save raw network output BEFORE clipping/action scaling, exactly as returned.

 156              << stats.parameter_sync_calls << ',' << stats.observation_sync_calls << ',' << stats.output_sync_calls << ','
 157              << stats.observation_ms << ',' << stats.execution_ms << ',' << stats.output_ms << '\n';
 158       frames.flush();

 164     report["parameter_prepares"] = stats.parameter_prepares;
 165     report["commits"] = stats.commits;
 166     report["rollbacks"] = stats.rollbacks;
 167     report["runs"] = stats.runs;
 168     report["allocated_bytes"] = stats.allocated_bytes;
 169     report["status"] = "completed";
 170     writeYaml(output / "runtime.yaml", report);
 171     std::cout << "Completed " << cases.size() << " samples using one resident context" << std::endl;
 172     return 0;
 173   } catch (const std::exception& error) {
 174     report["status"] = "failed";
 175     report["error"] = error.what();
 176     if (owns_output) {
 177       try {
 178         saveRuntimeMaps(output);
 179         writeYaml(output / "runtime.yaml", report);
 180       } catch (...) {}  // Preserve the original runtime failure in stderr.
 181     }

 183     return 1;
