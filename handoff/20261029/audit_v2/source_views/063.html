  25 def load_actions(bundle, selected):
  26     actions = json.loads(selected)
  27     if (not isinstance(actions, list) or not actions or
  28             any(not isinstance(a, str) or not re.fullmatch(r'[A-Za-z0-9_-]{1,32}', a) for a in actions) or
  29             len(actions) != len(set(actions))):
  30         raise ValueError('动作名单必须非空、无重复且使用合法 ID')
  31     graph = json.loads((bundle / 'base_manifest.json').read_text())
  32     entries = {a['action_id']: a for a in graph['actions']}
  33     durations = {}
  34     for action in actions:
  35         entry = entries[action]
  36         config = (bundle / entry['config']).resolve(strict=True)
  37         if not config.is_relative_to(bundle.resolve()):
  38             raise ValueError('动作配置路径越界')
  39         data = config.read_bytes()
  40         if hashlib.sha256(data).hexdigest() != entry['config_sha256']:
  41             raise ValueError('动作配置哈希不符：' + action)
  42         cfg = yaml.safe_load(data)['WbcCfg']
  43         duration = float(cfg['reference_num_frames']) * float(cfg['control_dt']) + float(cfg.get('start_hold_sec', 0))
  44         if cfg['motion_id'] != action or not math.isfinite(duration) or duration <= 0:
  45             raise ValueError('动作 ID 或时长无效：' + action)
  46         durations[action] = duration
  47     return actions, durations

  50 def resolve_action(value, actions):
  51     # Exact ID wins if a deployment happens to use a numeric ID.
  52     if value in actions:
  53         return value
  54     if re.fullmatch(r'[0-9]{1,6}', value):
  55         index = int(value) - 1
  56         if 0 <= index < len(actions):
  57             return actions[index]
  58     return None

  69 def stop_publisher(process):
  70     if process is not None and process.poll() is None:
  71         try:
  72             os.killpg(process.pid, signal.SIGINT)
  73         except ProcessLookupError:
  74             process.wait()
  75             return
  76         try:
  77             process.wait(timeout=2)
  78         except subprocess.TimeoutExpired:
  79             os.killpg(process.pid, signal.SIGKILL)
  80             process.wait()

  83 def run_console(args, actions, durations):
  84     # Keep preview/list validation independent of ROS and DDS.
  85     import rclpy
  86     from rcl_interfaces.srv import GetParameters
  87     from rclpy.executors import ExternalShutdownException
  88     from rclpy.parameter import parameter_value_to_python
  89     from rclpy.qos import QoSProfile, ReliabilityPolicy, DurabilityPolicy
  90     from std_msgs.msg import Bool, String
  91 
  92     if not sys.stdin.isatty():
  93         raise ValueError('动作控制台需要交互终端，不接受管道或批量输入')
  94     rclpy.init(args=[])
  95     node = rclpy.create_node(f'wbc_action_console_{os.getpid()}')
  96     pending = None
  97     terminal = None
  98     active = None
  99     publisher = None
 100     last_timestamp = 0
 101     feedback_path = args.run_dir / 'action_feedback.jsonl'
 102     request_path = args.run_dir / 'action_requests.jsonl'
 103 
 104     def record(path, value):
 105         with path.open('a') as stream:
 106             stream.write(json.dumps(value, ensure_ascii=False) + '\n')
 107 
 108     def feedback(message):
 109         nonlocal terminal
 110         try:
 111             value = json.loads(message.data)
 112             if not isinstance(value, dict):
 113                 return
 114         except (ValueError, TypeError):
 115             return
 116         record(feedback_path, {'received_at_ms': time.time_ns() // 1_000_000, 'feedback': value})
 117         if pending and (value.get('timestamp_ms'), value.get('action_id')) == pending:
 118             status = value.get('status')
 119             print(f'反馈：{pending[1]} → {status}', flush=True)
 120             if status in ('completed', 'rejected', 'failed'):
 121                 terminal = status
 122 
 123     def set_active(message):
 124         nonlocal active
 125         active = message.data
 126 
 127     node.create_subscription(String, FEEDBACK_TOPIC, feedback, QoSProfile(depth=100))
 128     node.create_subscription(Bool, '/control/policy/wbc/active', set_active,
 129                              QoSProfile(depth=1, reliability=ReliabilityPolicy.RELIABLE,
 130                                         durability=DurabilityPolicy.TRANSIENT_LOCAL))
 131     client = node.create_client(GetParameters, WORKER + '/get_parameters')
 132 
 133     def ready():
 134         if not client.wait_for_service(timeout_sec=2):
 135             raise ValueError('WBC worker 未就绪，未发送动作')
 136         names = [namespace.rstrip('/') + '/' + name for name, namespace in node.get_node_names_and_namespaces()]
 137         if names.count(WORKER) != 1:
 138             raise ValueError('必须只有一个 /rl_controller_wbc')
 139         request = GetParameters.Request()
 140         request.names = ['wbc_mode', 'wbc_lora_manifest', 'available_wbc_policies']
 141         future = client.call_async(request)
 142         rclpy.spin_until_future_complete(node, future, timeout_sec=3)
 143         if not future.done():
 144             future.cancel()
 145             raise ValueError('读取 worker 参数超时，未发送动作')
 146         response = future.result()
 147         if response is None or len(response.values) != 3:
 148             raise ValueError('无法读取 worker 参数')
 149         mode, manifest, loaded = [parameter_value_to_python(v) for v in response.values]
 150         if (mode != 'lora' or not isinstance(manifest, str) or
 151                 Path(manifest).resolve() != (args.bundle / 'base_manifest.json').resolve() or
 152                 not isinstance(loaded, list) or loaded != actions):
 153             raise ValueError('运行中的 worker 与本次快照不一致，请重新启动部署/控制台')
 154         services = dict(node.get_service_names_and_types())
 155         if 'luvbot_control_interfaces/srv/ControlWbcMotion' not in services.get('/control/policy/wbc/prepare', []):
 156             raise ValueError('prepare 服务未就绪，未发送动作')
 157         deadline = time.monotonic() + 2
 158         while time.monotonic() < deadline:
 159             rclpy.spin_once(node, timeout_sec=0.1)
 160             if (active is not None and node.count_publishers(FEEDBACK_TOPIC) > 0 and
 161                     node.count_subscribers(COMMAND_TOPIC) > 0):
 162                 break
 163         if active is not False:
 164             raise ValueError('WBC 正在执行，或尚未收到 active 状态；未发送动作')
 165         if node.count_publishers(FEEDBACK_TOPIC) == 0 or node.count_subscribers(COMMAND_TOPIC) == 0:
 166             raise ValueError('动作请求/反馈端点未就绪，未发送动作')
 167 
 168     try:
 169         show_menu(actions)
 170         print('执行前按原手柄流程进入 Walking 并保持零速度；控制台不会切换状态。')
 171         while rclpy.ok():
 172             try:
 173                 value = input('\n动作> ').strip()
 174             except EOFError:
 175                 break
 176             if value.lower() in ('q', 'quit', 'exit'):
 177                 break
 178             if value.lower() in ('list', 'help', '?'):
 179                 show_menu(actions)
 180                 continue
 181             if not value:
 182                 continue
 183             action = resolve_action(value, actions)
 184             if action is None:
 185                 print('未识别输入。请输入列表编号或完整动作 ID，不支持模糊匹配。')
 186                 continue
 187             try:
 188                 ready()
 189             except ValueError as error:
 190                 print(error)
 191                 termios.tcflush(sys.stdin.fileno(), termios.TCIFLUSH)
 192                 continue
 193             timestamp = max(time.time_ns() // 1_000_000, last_timestamp + 1)
 194             last_timestamp = timestamp
 195             pending, terminal = (timestamp, action), None
 196             payload = json.dumps({'data': json.dumps({'timestamp_ms': timestamp, 'action_id': action})})
 197             command = ['ros2', 'topic', 'pub', '--once', '--wait-matching-subscriptions', '1',
 198                        '--keep-alive', '0.2', '--qos-reliability', 'reliable', '--qos-durability', 'volatile',
 199                        COMMAND_TOPIC, 'std_msgs/msg/String', payload]
 200             print(shlex.join(command), flush=True)
 201             record(request_path, {'timestamp_ms': timestamp, 'action_id': action, 'command': command, 'status': 'attempted'})
 202             deadline = time.monotonic() + 3
 203             with (args.run_dir / 'action_publish.log').open('a') as log:
 204                 publisher = subprocess.Popen(command, stdin=subprocess.DEVNULL, stdout=log, stderr=log,
 205                                              start_new_session=True)
 206                 while publisher.poll() is None and time.monotonic() < deadline:
 207                     rclpy.spin_once(node, timeout_sec=0.05)
 208                 if publisher.poll() is None:
 209                     stop_publisher(publisher)
 210                     raise RuntimeError('发布超时，结果不确定；控制台退出且不重发，请检查反馈与机器人状态')
 211                 if publisher.returncode != 0:
 212                     raise RuntimeError('发布指令失败；控制台退出且不重发，请检查 action_publish.log')
 213             publisher = None
 214             deadline = time.monotonic() + durations[action] + 30
 215             while terminal is None and time.monotonic() < deadline:
 216                 rclpy.spin_once(node, timeout_sec=0.1)
 217             # Do not turn input typed during execution into a queued action.
 218             termios.tcflush(sys.stdin.fileno(), termios.TCIFLUSH)
 219             if terminal is None:
 220                 raise RuntimeError('未收到终态反馈；控制台退出且不重发，请检查部署日志')
 221             if terminal == 'failed':
 222                 raise RuntimeError('动作反馈 failed；控制台退出，请按原流程检查故障')
 223             print('本次请求结束；执行期间提前输入的内容已丢弃。')
 224             pending = None
 225     except ExternalShutdownException:
 226         print('\nROS 上下文已关闭；未发送新的动作请求。')
 227     finally:
 228         stop_publisher(publisher)
 229         node.destroy_node()
 230         if rclpy.ok():
 231             rclpy.shutdown()

 234 def main():
 235     parser = argparse.ArgumentParser(description=__doc__)
 236     parser.add_argument('--bundle', type=Path, required=True)

 248         return 0
 249     except KeyboardInterrupt:
 250         print('\n控制台已退出；机器人和正在执行的动作仍由原控制链管理。')
 251         return 130

 253         print(f'错误：{error}', file=sys.stderr)
