  34     def resolve_config_path(self, raw_path: str) -> str:
  35         if not raw_path:
  36             return ""
  37         if os.path.isabs(raw_path):
  38             return raw_path
  39         try:
  40             return os.path.join(get_package_share_directory("legged_controllers"), raw_path)
  41         except Exception:
  42             return raw_path

  44     def load_bindings(self, raw_config_path: str) -> dict:
  45         config_path = self.resolve_config_path(raw_config_path)
  46         if not config_path:
  47             self.get_logger().error("keyboard_action_trigger.py requires config_path")
  48             return {}
  49         try:
  50             with open(config_path, "r", encoding="utf-8") as stream:
  51                 root = yaml.safe_load(stream) or {}
  52         except Exception as exc:
  53             self.get_logger().error(f"Failed to load keyboard action config {config_path}: {exc}")
  54             return {}
  55 
  56         cfg = root.get("keyboard_actions", root)
  57         actions = cfg.get("actions", [])
  58         if not isinstance(actions, list):
  59             self.get_logger().error(f"keyboard action config {config_path} must contain actions: []")
  60             return {}
  61 
  62         command_to_action = {}
  63         for entry in actions:
  64             if not isinstance(entry, dict):
  65                 self.get_logger().warning("Skipping keyboard action entry that is not a map")
  66                 continue
  67             command = str(entry.get("key", "")).strip()
  68             action_name = str(entry.get("name") or command).strip()
  69             if not command or not action_name:
  70                 self.get_logger().warning("Skipping keyboard action entry without key/name")
  71                 continue
  72             command_to_action[command] = action_name
  73         return command_to_action
  74 
  75     def publish_trigger(self, action_name: str) -> None:
  76         msg = String()
  77         msg.data = action_name
  78         self.publisher.publish(msg)
  79         self.get_logger().info(f"Published keyboard action trigger: {action_name}")
