  10 def _legged_mujoco_sim_setup(context, *args, **kwargs):
  11     cfg = context.launch_configurations
  12 
  13     legged_controllers_share = get_package_share_directory('legged_controllers')
  14     command_arbiter_share = get_package_share_directory('system_command_arbiter')
  15     robot_type = cfg['robot_type']
  16     subpolicy = cfg['subpolicy']
  17     legged_deploy_share = get_package_share_directory('legged_deploy')
  18 
  19     robot_cfg_path = os.path.abspath(
  20         os.path.join(legged_deploy_share, 'policy', robot_type, subpolicy, 'rl_config.yaml')
  21     )
  22 
  23     legged_controller_config_path = os.path.abspath(
  24         os.path.join(legged_controllers_share, 'config', 'legged_controller_config.yaml')
  25     )
  26     joy_cfg_path = os.path.abspath(
  27         os.path.join(legged_deploy_share, 'policy', robot_type, subpolicy, 'joy_config.yaml')
  28     )
  29     mapping_file = os.path.abspath(os.path.join(command_arbiter_share, 'config', f'{robot_type}.yaml'))
  30 
  31     start_delay = float(cfg.get('start_delay', '1.0'))
  32 
  33     command_arbiter_node = Node(
  34         package='system_command_arbiter',
  35         executable='system_command_arbiter_node',
  36         name='system_command_arbiter_node',
  37         namespace='system/command_arbiter',
  38         output='screen',
  39         on_exit=Shutdown(reason='system command arbiter exited'),
  40         parameters=[{'mapping_file': mapping_file, 'pilot_mode': False}],
  41         remappings=[
  42             ('motor_command', '/actuator/motor/command'),
  43             ('motor_state', '/actuator/motor/state'),
  44             ('set_motor_enabled', '/actuator/motor/set_enabled'),
  45             ('emergency_stop', '/system/safety/emergency_stop'),
  46             ('hardware_fault', '/system/safety/hardware_fault'),
  47         ],
  48     )
  49 
  50     adapter_node = Node(
  51         package='legged_real',
  52         executable='topic_hardware_adapter',
  53         name='topic_hardware_adapter',
  54         output='screen',
  55         on_exit=Shutdown(reason='topic hardware adapter exited'),
  56         parameters=[
  57             {'mapping_file': mapping_file},
  58             {'robot_type': robot_type},
  59             {'robot_cfg_path': robot_cfg_path},
  60             {'legged_controller_config_path': legged_controller_config_path},
  61             {'joy_cfg_path': joy_cfg_path},
  62             {'isSim': True},
  63             {'wbc_enabled': cfg['wbc_enabled'] == 'true'},
  64             {'/RobotHw/imu_enabled': True},
  65             {'motor_lifecycle_period_ms': 1},
  66             {'rviz': cfg['rviz'] == 'true'},
  67             {'head_script_enabled': cfg['head_script_enabled'] == 'true'},
  68             {'head_script_config': cfg['head_script_config']},
  69             {'joystick_imitation_enabled': cfg.get('joystick_imitation_enabled', 'false') == 'true'},
  70             {'joystick_imitation_config_path': cfg.get('joystick_imitation_config_path', '')},
  71             {'joystick_sequence_enabled': cfg.get('joystick_sequence_enabled', 'false') == 'true'},
  72             {'joystick_sequence_config_path': cfg.get('joystick_sequence_config_path', '')},
  73             {'emergency.damping_kd': float(cfg.get('emergency_damping_kd', '5.0'))},
  74             {
  75                 'joystick_back_emergency_sequence_enabled': cfg.get('joystick_back_emergency_sequence_enabled', 'true')
  76                 == 'true'
  77             },
  78             {
  79                 'joystick_back_emergency_sequence_path': cfg.get(
  80                     'joystick_back_emergency_sequence_path', 'action_sequences/walking_back_emergency.yaml'
  81                 )
  82             },
  83             {'arm_swing.enabled': cfg.get('arm_swing_enabled', 'true') == 'true'},
  84             {'arm_swing.stop_vx_threshold': float(cfg.get('arm_swing_stop_vx_threshold', '0.05'))},
  85             {'arm_swing.high_vx_threshold': float(cfg.get('arm_swing_high_vx_threshold', '0.8'))},
  86             {'arm_swing.vx_hysteresis': float(cfg.get('arm_swing_vx_hysteresis', '0.03'))},
  87             {'arm_swing.blend_sec': float(cfg.get('arm_swing_blend_sec', '0.35'))},
  88         ],
  89         remappings=[
  90             ('emergency_stop', '/system/safety/emergency_stop'),
  91             ('hardware_fault', '/system/safety/hardware_fault'),
  92             ('motors_ready', '/hardware_status/motors_ready'),
  93         ],
  94     )
  95 
  96     return [
  97         command_arbiter_node,
  98         TimerAction(
  99             period=start_delay,
 100             actions=[adapter_node],
 101         ),
 102     ]
