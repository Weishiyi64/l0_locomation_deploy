   7 void JointCommandComposer::composeAll(const Context& ctx) const {
   8   if (ctx.robot_hw == nullptr) return;
   9 
  10   const int joint_motor_commands_size = ctx.robot_hw->getJointMotorCommandsSize();
  11   const auto& q_snapshot = ctx.policy_inputs.q_snapshot;
  12   const auto& active_gains = ctx.policy_inputs.gains;
  13 
  14   if (ctx.release_handoff != nullptr)
  15   {
  16     ctx.release_handoff->clearIfExpired(ctx.wall_sec);
  17   }
  18   const bool release_handoff_available =
  19       ctx.release_handoff != nullptr && ctx.release_handoff->active() && !ctx.action_sequence_active;
  20 
  21   for (size_t j = 0; j < static_cast<size_t>(joint_motor_commands_size); ++j)
  22   {
  23     double q_j, dq_j, kp_j, kd_j, tau_ff_j;
  24     const bool is_leg_joint = ctx.is_leg_joint != nullptr && j < ctx.is_leg_joint->size() && (*ctx.is_leg_joint)[j];
  25 
  26     const bool csv_allowed_for_joint = ctx.action_sequence_full_csv || !is_leg_joint;
  27     if (csv_allowed_for_joint && ctx.joint_csv != nullptr && ctx.joint_csv->isEnabled() &&
  28         ctx.joint_csv->getJointCmd(j, q_j, dq_j, kp_j, kd_j, tau_ff_j))
  29     {
  30       // Joint CSV timeline (auxiliary joints normally; full body only for back-emergency full_csv).
  31     }
  32     else if (ctx.arm_swing_ctrl != nullptr && ctx.arm_swing_ctrl->getJointCmd(j, q_j, dq_j, kp_j, kd_j, tau_ff_j))
  33     {
  34       // Procedural walking arm swing.
  35     }
  36     else if (ctx.allow_auxiliary_overlays && ctx.head_ctrl != nullptr &&
  37              ctx.head_ctrl->getJointCmd(j, q_j, dq_j, kp_j, kd_j, tau_ff_j))
  38     {
  39       // Walking-only head overlay.
  40     }
  41     else
  42     {
  43       q_j = (j < q_snapshot.size()) ? q_snapshot[j] : 0.0;
  44       dq_j = 0.0;
  45       kp_j = kd_j = tau_ff_j = 0.0;
  46     }
  47 
  48     if (release_handoff_available && !ctx.emergency && !ctx.blend_policy_gains)
  49     {
  50       (void)ctx.release_handoff->apply(j, ctx.wall_sec, q_j, dq_j, kp_j, kd_j, tau_ff_j);
  51     }
  52 
  53     if (ctx.walking)
  54     {
  55       kp_j = (j < active_gains.kps.size()) ? active_gains.kps[j] : 0.0;
  56       kd_j = (j < active_gains.kds.size()) ? active_gains.kds[j] : 0.0;
  57     }
  58     else if (ctx.is_sim)
  59     {
  60       kp_j = 80.0;
  61       kd_j = 1.0;
  62     }
  63     else
  64     {
  65       double kp_limit = ctx.kp_standby;
  66       if (j < ctx.rl_gains.kps.size()) kp_limit = std::min(ctx.kp_standby, ctx.rl_gains.kps[j]);
  67       kp_j = kp_limit;
  68       kd_j = 0.05;
  69     }
  70 
  71     // WBC changes gain profiles as well as position targets.
  72     if (release_handoff_available && !ctx.emergency && ctx.blend_policy_gains)
  73       (void)ctx.release_handoff->apply(j, ctx.wall_sec, q_j, dq_j, kp_j, kd_j, tau_ff_j);
  74 
  75     if (ctx.emergency)
  76     {
  77       q_j = (j < ctx.motor_states.size()) ? ctx.motor_states[j].q : q_j;
  78       dq_j = 0.0;
  79       kp_j = 0.0;
  80       kd_j = std::max(0.0, ctx.emergency_damping_kd);
  81       tau_ff_j = 0.0;
  82     }
  83 
  84     legged::JointMotorCommand cmd(q_j, dq_j, kp_j, kd_j, tau_ff_j);
  85     ctx.robot_hw->set_joint_motor_commands(static_cast<int>(j), cmd);
  86   }
  87 }
