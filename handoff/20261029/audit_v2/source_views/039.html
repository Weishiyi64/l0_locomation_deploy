  24 std::vector<std::pair<double, double>> wbcJsonJointLimits(
  25     const std::string& mapping_file, const std::vector<std::string>& joint_names) {
  26   const auto motors = YAML::LoadFile(mapping_file)["motors"];
  27   require(motors.IsSequence(), "mapping has no motors sequence");
  28   std::vector<std::pair<double, double>> result;
  29   for (const auto& name : joint_names) {
  30     unsigned count = 0;
  31     std::pair<double, double> bounds;
  32     for (const auto& motor : motors) {
  33       if (motor["joint_name"].as<std::string>() != name) continue;
  34       ++count;
  35       bounds = {finite(motor["joint_lower"]), finite(motor["joint_upper"])};
  36       require(motor["enabled"].as<bool>() && bounds.first < bounds.second, "invalid/disabled joint limit");
  37     }
  38     require(count == 1, "missing/duplicate JSON WBC mapping joint");
  39     result.push_back(bounds);
  40   }
  41   return result;
  42 }

  59       std::all_of(id_.begin(), id_.end(), [](unsigned char c) {

  62       }), "invalid text motion request ID");
  63   expires_at_ms_ = root["expires_at_ms"].as<std::int64_t>();

  66   const auto names = root["joint_names"];

  68   for (std::size_t i = 0; i < 9; ++i)

  72   const auto snapshot_joints = snapshot["joint_positions"];

  75   for (std::size_t j = 0; j < 21; ++j) {
  76     require(snapshot_names[j].as<std::string>() == joint_names[j], "snapshot joint order differs from WBC Base");
  77     const double q = finite(snapshot_joints[j]);
  78     require(q >= limits[j].first - 1e-6 && q <= limits[j].second + 1e-6, "snapshot outside robot joint limits");
  79     start_[j] = static_cast<float>(std::clamp(q, limits[j].first, limits[j].second));
  80   }
  81   const auto position = snapshot["base_position"];
  82   const auto orientation = snapshot["base_quaternion_wxyz"];
  83   require(position.IsSequence() && position.size() == 3 && orientation.IsSequence() && orientation.size() == 4,
  84           "invalid JSON root pose dimensions");
  85   for (std::size_t j = 0; j < 3; ++j) root_position_[j] = finite(position[j]);
  86   const auto reference_source = root["root_reference_source"].as<std::string>("snapshot");
  87   require(reference_source == "snapshot" || reference_source == "wbc_base_action", "unknown JSON root reference source");
  88   if (reference_source == "wbc_base_action") {

  91   }

  93   double norm = 0;

  97     require(std::isfinite(root_orientation_[j]), "non-finite WBC reference quaternion");
  98     norm += root_orientation_[j] * root_orientation_[j];
  99   }

 105   require(tokens[0] >= 0 && tokens[0] < 47 && tokens[1] >= 0 && tokens[1] < 43 && tokens[2] >= 0 && tokens[2] < 43,
 106           "invalid text model token IDs");
 107   require(tokens[0] || tokens[1] || tokens[2], "no active body part");

 113   action.reserve(count);
 114   for (std::size_t i = 0; i < count; ++i) {
 115     const auto frame = frames[i];
 116     require(frame["frame_index"].as<std::size_t>(count) == i &&
 117             std::abs(finite(frame["time"]) - i / 50.0) <= 1e-5, "invalid JSON frame timeline");
 118     const auto values = frame["joint_positions"];
 119     require(values.IsSequence() && values.size() == 9, "invalid upper-body frame size");
 120     Joints row = start_;
 121     for (std::size_t j = 0; j < 9; ++j) {
 122       const double q = finite(values[j]);
 123       const auto& bound = limits[12 + j];
 124       require(q >= bound.first - 1e-6 && q <= bound.second + 1e-6, "JSON frame exceeds robot joint limits");
 125       const bool active = tokens[j < 3 ? 1 : (j < 6 ? 2 : 0)] != 0;
 126       if (!active) require(std::abs(q - start_[12 + j]) <= 0.005, "inactive body part differs from captured pose");
 127       else row[12 + j] = static_cast<float>(std::clamp(q, bound.first, bound.second));
 128     }
 129     action.push_back(row);
 130   }
 131   // Preserve the reference packer's standing hold and smooth entry/exit. No
 132   // second arm mirror, FK pass, or per-request ONNX/RKNN conversion is needed.
 133   constexpr std::size_t hold = 10, transition = 40, settle = 10;
 134   frames_.reserve(count + hold + 2 * transition + settle);
 135   frames_.insert(frames_.end(), hold, start_);
 136   const auto appendBlend = [&](const Joints& from, const Joints& to) {
 137     for (std::size_t i = 0; i < transition; ++i) {
 138       const float u = static_cast<float>(i) / (transition - 1);
 139       const float alpha = u * u * (3 - 2 * u);
 140       Joints row;
 141       for (std::size_t j = 0; j < 21; ++j) row[j] = from[j] + alpha * (to[j] - from[j]);
 142       frames_.push_back(row);
 143     }
 144   };
 145   appendBlend(start_, action.front());
 146   frames_.insert(frames_.end(), action.begin(), action.end());
 147   appendBlend(action.back(), start_);
 148   frames_.insert(frames_.end(), settle, start_);
