  30 set(LEGGED_DEPLOY_BACKEND "AUTO" CACHE STRING "Inference backend: AUTO, OPENVINO, TENSORRT, or RKNN")
  31 set_property(CACHE LEGGED_DEPLOY_BACKEND PROPERTY STRINGS AUTO OPENVINO TENSORRT RKNN)
  32 string(TOUPPER "${LEGGED_DEPLOY_BACKEND}" LEGGED_DEPLOY_BACKEND)
  33 
  34 if(LEGGED_DEPLOY_BACKEND STREQUAL "AUTO")
  35   string(TOLOWER "${CMAKE_SYSTEM_PROCESSOR}" _processor)
  36   if(_processor MATCHES "aarch64|arm64|armv8")
  37     set(_device_tree_hex "")
  38     foreach(_compatible_file
  39         "/proc/device-tree/compatible"
  40         "/sys/firmware/devicetree/base/compatible")
  41       if(EXISTS "${_compatible_file}")
  42         file(READ "${_compatible_file}" _device_tree_hex HEX)
  43         string(TOLOWER "${_device_tree_hex}" _device_tree_hex)
  44         break()
  45       endif()
  46     endforeach()
  47 
  48     # Device-tree strings contain NUL bytes, so match their ASCII hex values.
  49     if(_device_tree_hex MATCHES "726f636b636869702c726b33353838") # rockchip,rk3588
  50       set(LEGGED_DEPLOY_BACKEND "RKNN")
  51       message(STATUS "legged_deploy: AUTO detected Rockchip RK3588")
  52     elseif(_device_tree_hex MATCHES "6e7669646961" OR EXISTS "/etc/nv_tegra_release") # nvidia
  53       set(LEGGED_DEPLOY_BACKEND "TENSORRT")
  54       message(STATUS "legged_deploy: AUTO detected NVIDIA Jetson")
  55     else()
  56       message(FATAL_ERROR
  57         "Unable to identify this ARM64 platform as Jetson or RK3588. "
  58         "Set -DLEGGED_DEPLOY_BACKEND=TENSORRT or RKNN explicitly.")
  59     endif()
  60   else()
  61     set(LEGGED_DEPLOY_BACKEND "OPENVINO")
  62     message(STATUS "legged_deploy: AUTO selected OpenVINO for ${CMAKE_SYSTEM_PROCESSOR}")
  63   endif()
  64 endif()
  65 
  66 set(LEGGED_DEPLOY_USE_TENSORRT OFF)
  67 set(LEGGED_DEPLOY_USE_RKNN OFF)
  68 if(LEGGED_DEPLOY_BACKEND STREQUAL "TENSORRT")
  69   set(LEGGED_DEPLOY_USE_TENSORRT ON)
  70   message(STATUS "legged_deploy: TensorRT backend")
  71   add_compile_definitions(USE_TENSORRT)
  72   find_package(CUDA REQUIRED)
  73   set(TENSORRT_INCLUDE_DIRS "/usr/include/aarch64-linux-gnu")
  74   set(TENSORRT_LIBRARIES "/usr/lib/aarch64-linux-gnu/libnvinfer.so")
  75   set(PLATFORM_INCLUDE_DIRS
  76     ${TENSORRT_INCLUDE_DIRS}
  77     ${CUDA_INCLUDE_DIRS}
  78     "/usr/local/cuda/include"
  79   )
  80   set(PLATFORM_LIBS ${TENSORRT_LIBRARIES} ${CUDA_LIBRARIES})
  81 elseif(LEGGED_DEPLOY_BACKEND STREQUAL "RKNN")
  82   set(LEGGED_DEPLOY_USE_RKNN ON)
  83   message(STATUS "legged_deploy: RKNN backend (RK3588)")
  84   add_compile_definitions(USE_RKNN)
  85   find_package(OpenSSL REQUIRED)
  86   set(RKNN_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/third_party/rknn_runtime"
  87       CACHE PATH "RKNN Runtime or rknpu2 root")
  88   find_path(RKNN_INCLUDE_DIR rknn_api.h
  89     HINTS ${RKNN_ROOT}
  90     PATH_SUFFIXES include runtime/Linux/librknn_api/include
  91   )
  92   find_library(RKNN_LIBRARY NAMES rknnrt
  93     HINTS ${RKNN_ROOT}
  94     PATH_SUFFIXES lib lib64 aarch64 runtime/Linux/librknn_api/aarch64
  95   )
  96   if(NOT RKNN_INCLUDE_DIR OR NOT RKNN_LIBRARY)
  97     message(FATAL_ERROR "RKNN Runtime not found. Set -DRKNN_ROOT=<rknpu2 runtime root>.")
  98   endif()
  99   set(PLATFORM_INCLUDE_DIRS ${RKNN_INCLUDE_DIR})
 100   set(PLATFORM_LIBS ${RKNN_LIBRARY})
 101 elseif(LEGGED_DEPLOY_BACKEND STREQUAL "OPENVINO")
 102   message(STATUS "legged_deploy: OpenVINO backend")
 103   add_compile_definitions(USE_OPENVINO)
 104   find_package(OpenVINO REQUIRED COMPONENTS Runtime)
 105   set(PLATFORM_INCLUDE_DIRS "")
 106   set(PLATFORM_LIBS openvino::runtime)
 107 else()
 108   message(FATAL_ERROR "Unsupported LEGGED_DEPLOY_BACKEND='${LEGGED_DEPLOY_BACKEND}'")
 109 endif()
