  26 def training_files(source, actions):
  27     """Freeze the selected inputs, including any supplied full checkpoints."""
  28     from wbc_lora_assets import ID_PATTERN, asset_path, sha256
  29 
  30     index = json.loads((source / "emotion_index.json").read_text(encoding="utf-8"))
  31     if not isinstance(index, list) or not index:
  32         raise ValueError("emotion_index.json must be a nonempty array")
  33     ids = [entry["motion"] for entry in index]
  34     if len(ids) != len(set(ids)) or any(not ID_PATTERN.fullmatch(x) for x in ids):
  35         raise ValueError("Invalid/duplicate action IDs in emotion_index.json")
  36     actions = ids if actions is None else actions
  37     if not actions or len(actions) != len(set(actions)) or not set(actions) <= set(ids):
  38         raise ValueError("--actions must select unique IDs from emotion_index.json")
  39     paths = {name: None for name in (
  40         "emotion_index.json", "base/base_ckpt.pt", "base/policy.onnx", "base/policy.yaml")}
  41     for entry in index:
  42         if entry["motion"] not in actions:
  43             continue
  44         paths[entry["adapter"]] = entry["adapter_sha256"]
  45         paths[entry["motion_onnx"]] = entry.get("motion_onnx_sha256")
  46         if entry.get("checkpoint") is not None:
  47             checkpoint = asset_path(source, entry["checkpoint"])
  48             if checkpoint.exists():
  49                 if not entry.get("checkpoint_sha256"):
  50                     raise ValueError(f"Existing checkpoint requires a checksum: {checkpoint}")
  51                 paths[entry["checkpoint"]] = entry["checkpoint_sha256"]
  52     hashes = {}
  53     for name, expected in paths.items():
  54         digest = sha256(asset_path(source, name))
  55         if expected is not None and digest != expected:
  56             raise ValueError(f"Training checksum mismatch: {name}")
  57         hashes[name] = digest
  58     return actions, hashes

  61 def verify_assets(root, expected=None):
  62     """Check the exact file set and every SHA256; never ship links or checkpoints."""
  63     from wbc_lora_assets import asset_path, sha256
  64 
  65     paths = list(root.rglob("*"))
  66     if any(p.is_symlink() for p in paths):
  67         raise ValueError(f"Symlinks are forbidden in generated bundles: {root}")
  68     actual = {p.relative_to(root).as_posix() for p in paths if p.is_file()}
  69     if any(Path(name).suffix.lower() in (".pt", ".pth", ".ckpt") for name in actual):
  70         raise ValueError(f"Training checkpoints are forbidden in generated bundles: {root}")
  71     if expected is not None and actual != expected:
  72         raise ValueError(f"Unexpected board bundle files: extra={sorted(actual - expected)}, "
  73                          f"missing={sorted(expected - actual)}")
  74     hashes = {}
  75     for line in (root / "ASSETS.sha256").read_text(encoding="utf-8").splitlines():
  76         digest, name = line.split("  ", 1)
  77         if not re.fullmatch(r"[0-9a-f]{64}", digest) or name in hashes:
  78             raise ValueError("Invalid/duplicate ASSETS.sha256 entry")
  79         if sha256(asset_path(root, name)) != digest:
  80             raise ValueError(f"Generated asset checksum mismatch: {name}")
  81         hashes[name] = digest
  82     if set(hashes) != actual - {"ASSETS.sha256"}:
  83         raise ValueError("ASSETS.sha256 must cover every generated file exactly once")
  84     return {"file_count": len(actual), "size_bytes": sum((root / n).stat().st_size for n in actual),
  85             "checkpoint_count": 0, "sha256": "passed"}

  88 def board_files(actions):
  89     # Exported paths are fixed by export_wbc_lora, not selected by a glob.
  90     files = {"base_manifest.json", "base_lora.onnx", "PREPROCESSING.json", "ASSETS.sha256"}

  92         files.update({f"adapters/{action}/manifest.json", f"adapters/{action}/weights.bin",
  93                       f"actions/{action}/rl_config.yaml", f"actions/{action}/reference_motion.onnx"})

  97 def process(args):
  98     from wbc_lora_assets import asset_path, sha256
  99 
 100     source, workspace = args.source.resolve(), args.workspace.resolve()
 101     name = args.name or datetime.datetime.now().strftime("wbc_lora_%Y%m%d_%H%M%S_%f")
 102     if not re.fullmatch(r"[A-Za-z0-9_-]{1,96}", name):
 103         raise ValueError("--name must contain 1-96 letters, digits, underscores or hyphens")
 104     if args.samples < 20:
 105         raise ValueError("At least 20 verification samples are required")
 106     archive = workspace / "model_inbox" / name
 107     board = workspace / "model_sources" / name
 108     logs = workspace / "smoke_logs" / name
 109     for path in (archive, board, logs):
 110         if path.exists() or path.is_symlink():
 111             raise ValueError(f"Output already exists; choose a new --name: {path}")
 112         resolved = path.resolve()
 113         if resolved.is_relative_to(source) or source.is_relative_to(resolved):
 114             raise ValueError("Training input and output directories must not overlap")
 115     actions, input_hashes = training_files(source, args.actions)
 116     env = dict(os.environ, PYTHONDONTWRITEBYTECODE="1", OMP_NUM_THREADS="1",
 117                MKL_NUM_THREADS="1", OPENBLAS_NUM_THREADS="1")
 118     # Check the actual subprocess environment before creating any output.
 119     subprocess.run([sys.executable, "-B", "-c", "import torch, numpy, yaml, onnx, onnxruntime"],
 120                    check=True, env=env)
 121     logs.mkdir(parents=True, exist_ok=False)
 122     report_path = logs / "PROCESSING.json"
 123     report = {"format": "wbc_lora_processing_v1", "status": "running", "source": str(source),
 124               "local_archive": str(archive), "board_bundle": str(board), "actions": actions,
 125               "samples_per_action": args.samples, "source_files": input_hashes, "steps": [],
 126               "python": sys.executable, "board_operations": "not_run", "rknn_compilation": "not_run"}
 127     write_json(report_path, report)
 128 
 129     def run_step(label, script, *arguments):
 130         command = [sys.executable, "-B", str(SCRIPT_DIR / script), *map(str, arguments)]
 131         log = logs / (label + ".log")
 132         print(f"[{label}] {log}", flush=True)
 133         with log.open("x", encoding="utf-8") as stream:
 134             result = subprocess.run(command, stdout=stream, stderr=subprocess.STDOUT, env=env, check=False)
 135         report["steps"].append({"step": label, "command": command, "log": str(log), "exit_code": result.returncode})
 136         write_json(report_path, report)
 137         if result.returncode:
 138             tail = "\n".join(log.read_text(encoding="utf-8", errors="replace").splitlines()[-20:])
 139             raise RuntimeError(f"{label} failed (exit {result.returncode}):\n{tail}")
 140 
 141     try:
 142         run_step("01_export_fp16", "prepare_wbc_deploy.py", source, archive,
 143                  "--mode", "lora", "--fp16", "--samples", args.samples, "--actions", *actions)
 144         report["archive_audit"] = verify_assets(archive)
 145         run_step("02_preprocess", "preprocess_wbc_lora_fp16.py", archive, board)
 146         report["board_audit"] = verify_assets(board, board_files(actions))
 147         # Exercise relocated converter discovery and manifest planning without RKNN.
 148         run_step("03_conversion_plan", "batch_convert_onnx_to_rknn_rk3588.py", board,
 149                  "--build-only", "--dry-run")
 150         for relative, digest in input_hashes.items():
 151             if sha256(asset_path(source, relative)) != digest:
 152                 raise ValueError(f"Training input changed during processing: {relative}")
 153         validation = json.loads((archive / "validation_report.json").read_text(encoding="utf-8"))
 154         report.update(status="passed", source_unchanged=True,
 155                       validation_report=str(archive / "validation_report.json"),
 156                       validation_report_sha256=sha256(archive / "validation_report.json"),
 157                       full_checkpoint_verification={a["action_id"]: a["full_checkpoint_verification"]
 158                                                     for a in validation["actions"]})
 159     except BaseException as error:
 160         report.update(status="failed", error=str(error))
 161         write_json(report_path, report)
 162         raise
 163     write_json(report_path, report)
 164     audit = report["board_audit"]
 165     print(f"Completed {len(actions)} LoRA actions; {audit['file_count']} board files, "
 166           f"{audit['size_bytes'] / 1024**2:.2f} MiB, 0 checkpoints", flush=True)
 167     print(f"Board bundle (transfer only this directory): {board}")
 168     print(f"Local archive: {archive}\nProcessing report: {report_path}")
 169     print("RKNN compilation and robot runtime validation remain on the board; no board operations performed.")
