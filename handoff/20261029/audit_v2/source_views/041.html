  38 std::vector<unsigned char> readFile(const std::string& path, std::size_t maximum) {
  39   std::ifstream stream(path, std::ios::binary | std::ios::ate);
  40   require(stream.good(), "cannot open " + path);

  48 std::string fileHash(const std::string& path) {
  49   std::ifstream stream(path, std::ios::binary);
  50   require(stream.good(), "cannot hash " + path);
  51   std::unique_ptr<EVP_MD_CTX, decltype(&EVP_MD_CTX_free)> ctx(EVP_MD_CTX_new(), EVP_MD_CTX_free);
  52   require(ctx && EVP_DigestInit_ex(ctx.get(), EVP_sha256(), nullptr) == 1, "SHA256 init failed");
  53   std::array<char, 65536> chunk{};
  54   while (stream.read(chunk.data(), chunk.size()) || stream.gcount())
  55     require(EVP_DigestUpdate(ctx.get(), chunk.data(), stream.gcount()) == 1, "SHA256 update failed");
  56   require(stream.eof(), "hash read failed: " + path);
  57   unsigned char out[EVP_MAX_MD_SIZE];
  58   unsigned int length = 0;
  59   require(EVP_DigestFinal_ex(ctx.get(), out, &length) == 1, "SHA256 final failed");
  60   std::ostringstream hex;
  61   for (unsigned i = 0; i < length; ++i) hex << std::hex << std::setw(2) << std::setfill('0') << unsigned(out[i]);
  62   return hex.str();
  63 }

 103 float halfToFloat(std::uint16_t bits) {
 104   const int sign = (bits & 0x8000U) ? -1 : 1;
 105   const int exponent = (bits >> 10U) & 31U;
 106   const int fraction = bits & 1023U;
 107   require(exponent != 31, "non-finite FP16 parameter");
 108   return static_cast<float>(sign * std::ldexp(exponent ? 1024 + fraction : fraction,
 109                                              exponent ? exponent - 25 : -24));
 110 }

 195 void WbcLoraAssets::bindModel(const RknnModel& model) {
 196   const auto inputs = graph_["inputs"];
 197   require(inputs.IsSequence() && inputs.size() == 33 && model.inputCount() == 33 && model.outputCount() == 1 &&
 198           model.outputIndex("action") == 0 && model.outputElements(0) == 21, "WBC shared model must have 33 inputs / action[21]");
 199   std::set<std::string> seen;
 200   for (const auto& item : inputs) {
 201     const auto name = item["name"].as<std::string>();
 202     const auto dims = shape(item["shape"]);
 203     const auto role = item["role"].as<std::string>();
 204     const int index = model.inputIndex(name);
 205     require(seen.insert(name).second && index >= 0 && item["dtype"].as<std::string>() == "float32", "invalid input: " + name);
 206     require(withoutSingletons(model.inputShape(index)) == withoutSingletons(dims) &&
 207             (model.inputType(index) == RKNN_TENSOR_FLOAT16 || model.inputType(index) == RKNN_TENSOR_FLOAT32),
 208             "RKNN input shape/type mismatch: " + name);
 209     if (name == "policy" || name == "command_short") {
 210       require(role == "observation" && dims == std::vector<std::size_t>{name == "policy" ? 399U : 240U}, "wrong WBC observation ABI");
 211     } else {
 212       auto tensor = std::find_if(tensors_.begin(), tensors_.end(), [&](const Tensor& spec) { return spec.name == name; });
 213       require(role == "adapter" && tensor != tensors_.end() && tensor->shape == dims, "unknown adapter input: " + name);
 214       tensor->input_index = index;
 215     }
 216   }
 217   require(seen.count("policy") && seen.count("command_short"), "missing WBC observations");
 218   for (const auto& tensor : tensors_) require(tensor.input_index >= 0, "unbound parameter: " + tensor.name);
 219 }

 226 void WbcLoraAssets::loadAdapter(const Action& action, Buffer& buffer) const {
 227   require(fileHash(action.adapter) == action.adapter_hash, "adapter manifest changed since startup");
 228   const auto manifest = readManifest(action.adapter);
 229   require(manifest["format"].as<std::string>() == "wbc_lora_adapter_v1" &&
 230           manifest["action_id"].as<std::string>() == action.id, "adapter identity mismatch");
 231   for (const char* key : {"base_checkpoint_sha256", "normalization_sha256", "parameter_schema_sha256", "rank", "alpha"})
 232     require(manifest[key].as<std::string>() == graph_[key].as<std::string>(), std::string("adapter mismatch: ") + key);
 233   const auto weights = manifest["weights"];
 234   const auto path = asset(fs::path(action.adapter).parent_path().string(), weights["path"].as<std::string>());
 235   const auto payload = readFile(path, 4 * 1024 * 1024);
 236   require(weights["dtype"].as<std::string>() == "float16-le" && weights["size_bytes"].as<std::size_t>() == payload.size() &&
 237           digest(payload.data(), payload.size()) == weights["sha256"].as<std::string>(), "invalid adapter payload");
 238   const auto records = manifest["tensors"];
 239   require(records.IsSequence() && records.size() == tensors_.size() && buffer.parameters.size() == tensors_.size(),
 240           "incomplete adapter buffer/schema");
 241   std::size_t end = 0;
 242   for (std::size_t i = 0; i < tensors_.size(); ++i) {
 243     const auto& spec = tensors_[i];
 244     const auto record = records[i];
 245     const auto offset = record["offset_bytes"].as<std::size_t>();
 246     const auto bytes = spec.elements * 2;
 247     require(record["name"].as<std::string>() == spec.name && record["actor_key"].as<std::string>() == spec.actor_key &&
 248             shape(record["shape"]) == spec.shape && record["size_bytes"].as<std::size_t>() == bytes &&
 249             offset >= end && offset % 64 == 0 && offset <= payload.size() && bytes <= payload.size() - offset &&
 250             buffer.parameters[i].size() == spec.elements, "invalid adapter tensor: " + spec.name);
 251     for (std::size_t j = 0; j < spec.elements; ++j)
 252       buffer.parameters[i][j] = halfToFloat(static_cast<std::uint16_t>(payload[offset + j * 2]) |
 253                                            (static_cast<std::uint16_t>(payload[offset + j * 2 + 1]) << 8U));
 254     end = offset + bytes;
 255   }
 256   require(end == payload.size(), "trailing adapter payload");
 257 }
