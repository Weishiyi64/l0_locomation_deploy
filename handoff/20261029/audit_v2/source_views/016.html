  15 def plan_configured_astar_route(
  16     root_xy: np.ndarray,
  17     goal_xy: np.ndarray,
  18     planner,
  19     waypoint_spacing_m: float,
  20     preferred_direction=None,
  21 ) -> np.ndarray:
  22     """净空简化后按固定前视距离采样；必要避障拐点与近终点除外。"""
  23     if not math.isfinite(waypoint_spacing_m) or waypoint_spacing_m <= 0:
  24         raise ValueError("A*路线点间距必须为有限正数")
  25     if not np.isfinite(root_xy).all():
  26         raise ValueError(f"起点不是有限坐标：{root_xy}")
  27     if not np.isfinite(goal_xy).all() or not planner.segment_is_free(goal_xy, goal_xy):
  28         raise ValueError(f"目标位于地图外或膨胀障碍内：{goal_xy}")
  29     if not planner.segment_is_free(root_xy, root_xy):
  30         exits = (
  31             planner.recovery_exits(
  32                 root_xy,
  33                 goal_xy,
  34                 max_distance=waypoint_spacing_m,
  35                 preferred_direction=preferred_direction,
  36             )
  37             if isinstance(planner, RecoveryAStarPlanner)
  38             else []
  39         )
  40         for endpoint in exits:
  41             try:
  42                 route = _plan_clearance_route(
  43                     endpoint,
  44                     goal_xy,
  45                     planner,
  46                     waypoint_spacing_m,
  47                     preferred_direction=preferred_direction,
  48                 )
  49             except GoalPlanningError:
  50                 continue
  51             # Keep the checked escape connector intact: never shortcut past it.
  52             print(
  53                 f"[RECOVERY] 起点净空不足：{np.asarray(root_xy).round(3)} → "
  54                 f"退出点 {endpoint.round(3)}；先退出，再接回原目标路线。"
  55             )
  56             return np.concatenate((np.asarray(root_xy, dtype=np.float32)[None], route))
  57         raise GoalPlanningError(f"起点净空不足且未找到可验证的退出路线：{root_xy}")
  58     return _plan_clearance_route(
  59         root_xy,
  60         goal_xy,
  61         planner,
  62         waypoint_spacing_m,

  67 def _plan_clearance_route(
  68     root_xy,
  69     goal_xy,
  70     planner,
  71     waypoint_spacing_m,
  72     *,
  73     exact_first_lookahead=True,
  74     preferred_direction=None,
  75 ):
  76     """Normal route; every segment must satisfy the unchanged inflated grid."""
  77     root_xy, goal_xy = np.asarray(root_xy, dtype=np.float64), np.asarray(goal_xy, dtype=np.float64)
  78     if np.array_equal(root_xy, goal_xy):
  79         if not planner.segment_is_free(root_xy, goal_xy):
  80             raise GoalPlanningError("A*路线端点或连线缺少净空；请重新选择目标")
  81         return np.stack((root_xy, goal_xy)).astype(np.float32)
  82     if np.linalg.norm(goal_xy - root_xy) <= waypoint_spacing_m + 1.0e-6 and planner.segment_is_free(root_xy, goal_xy):
  83         return np.stack((root_xy, goal_xy)).astype(np.float32)
  84     if exact_first_lookahead and isinstance(planner, RecoveryAStarPlanner):
  85         endpoint = planner.safe_lookahead_endpoint(
  86             root_xy,
  87             goal_xy,
  88             distance=waypoint_spacing_m,
  89             preferred_direction=preferred_direction,
  90         )
  91         if endpoint is not None and np.linalg.norm(endpoint - goal_xy) > 1.0e-5:
  92             tail = _plan_clearance_route(
  93                 endpoint,
  94                 goal_xy,
  95                 planner,
  96                 waypoint_spacing_m,
  97                 exact_first_lookahead=False,
  98             )
  99             result = np.concatenate((root_xy.astype(np.float32)[None], tail))
 100             if all(planner.segment_is_free(start, end) for start, end in zip(result[:-1], result[1:])):
 101                 return result
 102     dense_path = planner.plan(root_xy, goal_xy)
 103     if dense_path is None:
 104         raise GoalPlanningError("当前起点到配置目标点不存在A*路径")
 105     # Include exact endpoints in shortcutting, so grid-center connections do not
 106     # become tiny first/last tracking targets. Never relocate a requested goal.
 107     dense_path = np.concatenate((root_xy[None], dense_path, goal_xy[None]))
 108     keep = np.r_[True, np.any(np.diff(dense_path, axis=0) != 0.0, axis=1)]
 109     dense_path = dense_path[keep]
 110     if len(dense_path) == 1:
 111         return np.repeat(dense_path, 2, axis=0).astype(np.float32)
 112     # Greedy farthest-visible shortcuts remove the artificial alternating
 113     # axial/diagonal bends of an 8-connected grid, but cannot cut through walls.
 114     vertices = [dense_path[0]]
 115     current = 0
 116     while current < len(dense_path) - 1:
 117         for candidate in range(len(dense_path) - 1, current, -1):
 118             if planner.segment_is_free(dense_path[current], dense_path[candidate]):
 119                 vertices.append(dense_path[candidate])
 120                 current = candidate
 121                 break
 122         else:
 123             raise GoalPlanningError("A*路线端点或连线缺少净空；请重新选择目标")
 124     try:
 125         result = sample_guidance_path(vertices, planner, waypoint_spacing_m)
 126     except ValueError as exc:
 127         raise GoalPlanningError(str(exc)) from exc
 128     if not all(planner.segment_is_free(start, end) for start, end in zip(result[:-1], result[1:])):
 129         raise GoalPlanningError("A*路线端点或连线缺少净空；请重新选择目标")
 130     return result

 133 class FlatGroundPlanner:
 134     """Explicit obstacle-free deployment assumption; no sensor map input."""
 135 
 136     def segment_is_free(self, start, end):
 137         return bool(np.isfinite((start, end)).all())
 138 
 139     def start_is_free(self, xy):
 140         return bool(np.isfinite(xy).all())
 141 
 142     def plan(self, start, goal):
 143         points = np.asarray([start, goal], dtype=float)
 144         low = np.floor(points.min(axis=0) / 0.1) * 0.1 - 0.2
 145         high = np.ceil(points.max(axis=0) / 0.1) * 0.1 + 0.2
 146         width, height = np.ceil((high - low) / 0.1).astype(int) + 1
 147         if width * height > 2_000_000:
 148             raise ValueError('Flat-ground planning extent exceeds 2 million cells')
 149         grid = np.zeros((height, width), dtype=np.uint8)
 150         return AStarPlanner(grid, 0.1, tuple(low), prevent_corner_cutting=True).plan(start, goal)

 153 def load_house_planner(scene):
 154     if str(scene) == 'flat':
 155         return FlatGroundPlanner()
 156     obstacles = []
 157     for geom in ET.parse(scene).getroot().findall('geom'):
 158         x, y, _ = map(float, geom.attrib['pos'].split())
 159         hx, hy, _ = map(float, geom.attrib['size'].split())
 160         obstacles.append(dict(shape='box', x=x, y=y, size_x=2 * hx, size_y=2 * hy))
 161     return RecoveryAStarPlanner(obstacles, (10.0, 10.0), 0.1)

 164 class NavigationRoute:
 165     def __init__(self, planner, goal):
 166         self.planner, self.goal = planner, goal
 167         self.path = None
 168         self.index = 0
 169         self.last_plan = -math.inf
 170         self.revision = 0
 171         self.error = ''
 172         self.mode = 'Nav'
 173         self.complete = False
 174         self.hold_steps = 0
 175         self.arrived_since = None
 176         self.preview = np.zeros(4, dtype=np.float32)
 177 
 178     def replan(self, xy, now, force=False):
 179         self.last_plan = now
 180         # Finish a certified recovery connector before considering another side.
 181         if (
 182             not force
 183             and self.path is not None
 184             and 1 <= self.index < len(self.path)
 185             and not self.planner.start_is_free(xy)
 186             and self.planner.recovery_segment_is_safe(xy, self.path[self.index])
 187         ):
 188             return False
 189         preferred = None if self.path is None else self.path[self.index] - self.path[self.index - 1]
 190         try:
 191             refreshed = None
 192             if not force and self.path is not None:
 193                 refreshed = refresh_committed_path(self.path, self.index, xy, self.planner, 1.0)
 194             if refreshed is None:
 195                 refreshed = plan_configured_astar_route(
 196                     xy, np.array([self.goal.x, self.goal.y]), self.planner, 1.0, preferred
 197                 )
 198             self.path = refreshed
 199             self.index = 1
 200             self.error = ''
 201         except (GoalPlanningError, ValueError) as exc:
 202             self.path, self.index, self.error = None, 0, str(exc)
 203             self.hold_steps = 0
 204         self.revision += 1
 205         return True
 206 
 207     def update(self, pose, speed, now):
 208         xy = np.asarray(pose[:2], dtype=float)
 209 
 210         def evaluate():
 211             recovering = not self.complete and self.mode == 'Nav' and not self.planner.start_is_free(xy)
 212             return update_route_command(
 213                 self.path,
 214                 self.index,
 215                 xy,
 216                 pose[2],
 217                 self.goal.heading or 0.0,
 218                 self.goal,
 219                 heading_enabled=self.goal.heading is not None,
 220                 linear_speed=self.goal.speed,
 221                 angular_speed=self.goal.turn_speed,
 222                 recovering=recovering,
 223                 planner=self.planner,
 224                 planar_speed_mps=speed,
 225             )
 226 
 227         self.index, self.mode, command, rearmed, force = evaluate()
 228         if rearmed:
 229             self.complete, self.hold_steps = False, 0
 230         # Same order as sim2sim: arrival/waypoint advance -> replan -> recompute -> preview.
 231         if (
 232             not self.complete
 233             and (self.path is None or self.mode == 'Nav')
 234             and (force or now < self.last_plan or now - self.last_plan >= 1.0 - 1e-9)
 235         ):
 236             if self.replan(xy, now, force=force):
 237                 self.index, self.mode, command, _, _ = evaluate()
 238         # Deployment settling period: stop translation before final heading control.
 239         waiting_for_face = False
 240         if self.path is None or not self.goal.position_locked:
 241             self.arrived_since = None
 242         elif self.goal.heading is not None:
 243             if self.arrived_since is None or now < self.arrived_since:
 244                 self.arrived_since = now
 245             waiting_for_face = now - self.arrived_since < 1.0 - 1e-9
 246         if waiting_for_face:
 247             command = np.zeros(5, dtype=np.float32)
 248             self.mode = 'Hold'
 249         if self.complete:
 250             command = np.zeros(5, dtype=np.float32)
 251         self.preview = future_waypoints_in_body(
 252             self.path, self.index, xy, pose[2], 'Hold' if self.complete else self.mode
 253         )
 254         if self.path is not None and not self.complete:
 255             reached = (
 256                 not waiting_for_face
 257                 and self.index == len(self.path) - 1
 258                 and self.goal.position_locked
 259                 and (self.goal.heading is None or self.goal.heading_locked)
 260             )
 261             self.hold_steps = self.hold_steps + 1 if reached else 0
 262             required = max(1, round((1.5 if self.goal.heading is not None else 4.0) * 5))
 263             self.complete = self.hold_steps >= required
 264         return command, self.mode

 267 def update_route_command(
 268     path_xy,
 269     waypoint_index,
 270     root_xy,
 271     yaw,
 272     target_yaw,
 273     arrival,
 274     *,
 275     heading_enabled,
 276     linear_speed,
 277     angular_speed,
 278     waypoint_radius=0.2,
 279     recovering=False,
 280     planner=None,
 281     planar_speed_mps=0.0,
 282     decision_period_s=0.2,
 283 ):
 284     """模拟全局规划上游，并报告已越点但不可安全跳点的失效路线。"""
 285     if path_xy is None:
 286         # No route is a waiting state, not a zero-length completed goal.
 287         return 0, "Hold", np.zeros(5, dtype=np.float32), False, False
 288     if len(path_xy) < 2 or not 1 <= waypoint_index < len(path_xy):
 289         raise ValueError("A*路径索引无效")
 290 
 291     # A fast robot can cross a waypoint between two 5 Hz planner ticks without
 292     # ever entering its switching ball. Distance-only advancement would then
 293     # keep commanding the now-behind point until the next 1 Hz global replan.
 294     # Crossing the endpoint-normal plane of the incoming segment also means the
 295     # waypoint has been passed. The clearance check below still prevents this
 296     # shortcut from cutting through a mandatory obstacle corner.
 297     def waypoint_reached_or_passed(index):
 298         target = path_xy[index]
 299         reached = np.linalg.norm(target - root_xy) <= waypoint_radius
 300         incoming = target - path_xy[index - 1]
 301         passed = np.dot(root_xy - target, incoming) >= 0.0 and np.dot(incoming, incoming) > 1.0e-12
 302         return bool(reached or passed)
 303 
 304     # Refresh an intermediate target before the robot can cross it during the
 305     # next 5 Hz hold. This keeps the published point ahead without changing the
 306     # final goal's strict 0.2 m arrival condition.
 307     anticipation_radius = waypoint_radius + max(0.0, float(planar_speed_mps)) * max(0.0, float(decision_period_s))
 308 
 309     def intermediate_is_due(index):
 310         return bool(
 311             index < len(path_xy) - 1
 312             and (np.linalg.norm(path_xy[index] - root_xy) <= anticipation_radius or waypoint_reached_or_passed(index))
 313         )
 314 
 315     intermediate_due = intermediate_is_due(waypoint_index)
 316     # Prefer the cached forward point when it is still directly reachable. A
 317     # blocked connector is rebuilt from the live pose while the old point is
 318     # still ahead, rather than after it has become a reverse command.
 319     cached_recovery_is_safe = bool(
 320         recovering
 321         and isinstance(planner, RecoveryAStarPlanner)
 322         and planner.recovery_segment_is_safe(root_xy, path_xy[waypoint_index])
 323     )
 324     replan_required = bool(recovering and not cached_recovery_is_safe)
 325     while not recovering and intermediate_due and waypoint_index < len(path_xy) - 1:
 326         if planner is not None and not planner.segment_is_free(root_xy, path_xy[waypoint_index + 1]):
 327             # This is a clearance-critical bend, not a redundant point.  The
 328             # speed anticipation radius must not repeatedly rebuild the route
 329             # before the robot reaches it. Replan only after the point is
 330             # actually reached/passed and the outgoing connector is still
 331             # unavailable from the live pose.
 332             replan_required = waypoint_reached_or_passed(waypoint_index)
 333             break
 334         waypoint_index += 1
 335         intermediate_due = intermediate_is_due(waypoint_index)
 336     error = math.atan2(math.sin(target_yaw - yaw), math.cos(target_yaw - yaw))
 337     distance = float(np.linalg.norm(path_xy[-1] - root_xy))
 338     if waypoint_index == len(path_xy) - 1 and not recovering:
 339         previous = (arrival.position_locked, arrival.heading_locked)
 340         arrival.update(float(root_xy[0]), float(root_xy[1]), yaw)
 341         rearmed = (previous[0] and not arrival.position_locked) or (previous[1] and not arrival.heading_locked)
 342     else:
 343         # Passing near the final point cannot skip unfinished waypoints.
 344         arrival.position_locked = arrival.heading_locked = False
 345         rearmed = False
 346     command = np.zeros(5, dtype=np.float32)
 347     if not arrival.position_locked:
 348         phase = "Nav"
 349         delta = path_xy[waypoint_index] - root_xy
 350         c, s = math.cos(yaw), math.sin(yaw)
 351         command[:3] = (c * delta[0] + s * delta[1], -s * delta[0] + c * delta[1], np.clip(linear_speed, 0.3, 1.2))
 352     elif heading_enabled and not arrival.heading_locked:
 353         phase = "Face"
 354         command[3:] = (error, np.clip(abs(angular_speed), 1.57, 3.14))
 355     else:
 356         phase = "Hold"
 357     return waypoint_index, phase, command, rearmed, replan_required

 360 def future_waypoints_in_body(path_xy, waypoint_index, root_xy, yaw, phase):
 361     """Return the next two A* targets in one base frame; pad with the endpoint."""
 362     if path_xy is None or phase != "Nav":
 363         return np.zeros(4, dtype=np.float32)
 364     if not 0 <= waypoint_index < len(path_xy):
 365         raise ValueError("Cannot preview an invalid A* waypoint index")
 366     indices = np.minimum(waypoint_index + np.array([1, 2]), len(path_xy) - 1)
 367     relative = path_xy[indices] - root_xy
 368     c, s = math.cos(yaw), math.sin(yaw)
 369     return (
 370         np.stack((c * relative[:, 0] + s * relative[:, 1], -s * relative[:, 0] + c * relative[:, 1]), axis=-1)
 371         .astype(np.float32)
 372         .reshape(4)
 373     )
