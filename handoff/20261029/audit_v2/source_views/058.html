   1 #!/usr/bin/env bash
   2 # One optional rosbag group per terminal, independent of deployment/actions.
   3 set -eo pipefail
   4 if [[ ${1:-} == --help || ${1:-} == -h || $# == 0 ]]; then
   5   echo '用法：bash scripts/record_wbc_rosbag.sh A|B|C|D [--workspace 工作空间] [--dry-run]'
   6   echo 'A 策略输入输出；B 电机与 IMU；C 动作与安全事件；D ROS 日志。'
   7   echo '每组独立终端；Ctrl+C 停止并写盘。不启动 sim2real/HPM，不发送动作。'
   8   exit 0
   9 fi
  10 group=$1
  11 shift
  12 source "$(dirname -- "${BASH_SOURCE[0]}")/wbc_runtime_common.bash"
  13 wbc_options "$@"
  14 case "$group" in
  15   A|a) label=policy; topics=(/control/policy/observation /control/policy/command/rl /control/policy/command/wbc) ;;
  16   B|b) label=actuator; topics=(/actuator/motor/state /actuator/motor/command /imu/data) ;;
  17   C|c) label=events; topics=(/algorithm/brain/wbc_action_cmd /system/state_machine/wbc_action_fdb
  18     /control/policy/wbc/active /control/policy/wbc/done /hardware_status/motors_ready
  19     /system/safety/emergency_stop /system/safety/hardware_fault /actuator/motor/enable /joy) ;;
  20   D|d) label=diagnostics; topics=(/rosout) ;;
  21   *) wbc_die '录包组必须是 A、B、C 或 D' ;;
  22 esac
  23 wbc_current_run
  24 qos="$wbc_workspace/deploy_config/wbc-record-qos.yaml"
  25 [[ -f "$qos" ]] || qos="$wbc_script_dir/wbc-record-qos.yaml"
  26 stamp=$(date +%Y%m%d_%H%M%S_%N)
  27 bag_path="$wbc_run_dir/bags/${label}_$stamp"
  28 cmd=(ros2 bag record --storage sqlite3 --node-name "record_wbc_$label"
  29   --output "$bag_path" --max-cache-size 16777216 --max-bag-size 268435456
  30   --max-bag-duration 300 --qos-profile-overrides-path "$qos" "${topics[@]}")
  31 wbc_print_command "${cmd[@]}"
  32 $wbc_dry_run && exit 0
  33 exec 9>"$wbc_run_dir/.record_$label.lock"
  34 flock -n 9 || wbc_die "本次运行的 $label 组已在录制"
  35 mkdir -p -- "$wbc_run_dir/bags"
  36 cp -- "$qos" "$wbc_run_dir/record_${label}_$stamp.qos.yaml"
  37 printf '录包目录：%s\n按 Ctrl+C 停止并等待 metadata 写入。\n' "$bag_path"
  38 if wbc_logged_command "$wbc_run_dir/record_${label}_$stamp.log" "${cmd[@]}"; then result=0; else result=$?; fi
  39 printf '%s\n' "$result" > "$wbc_run_dir/record_${label}_$stamp.exitcode"
  40 [[ -f "$bag_path/metadata.yaml" ]] || wbc_die "录包未正常封存，请检查日志：$bag_path"
  41 printf '录包已停止：%s\n' "$bag_path"
  42 exit "$result"
