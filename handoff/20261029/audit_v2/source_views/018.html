  36 function process_group_alive() {
  37     local pid="$1"
  38     [[ -n "$pid" ]] && kill -0 -- "-$pid" 2>/dev/null
  39 }
  40 
  41 function signal_process_group() {
  42     local pid="$1"
  43     local signal="$2"
  44     kill "-$signal" -- "-$pid" 2>/dev/null || true
  45 }
  46 
  47 function wait_for_process_group_exit() {
  48     local pid="$1"
  49     local timeout_sec="$2"
  50     local deadline=$((SECONDS + timeout_sec))
  51 
  52     while process_group_alive "$pid" && (( SECONDS < deadline )); do
  53         sleep 0.2
  54     done
  55 
  56     ! process_group_alive "$pid"
  57 }

  59 function start_managed_process() {
  60     local name="$1"
  61     shift
  62 
  63     setsid "$@" &
  64     STARTED_PID=$!

  69 function cleanup() {
  70     local exit_code=$?
  71 
  72     if [[ "$CLEANUP_STARTED" == "true" ]]; then
  73         return "$exit_code"
  74     fi
  75     CLEANUP_STARTED=true
  76     trap '' INT TERM
  77     trap - EXIT
  78 
  79     if [[ ${#MANAGED_PIDS[@]} -gt 0 ]]; then
  80         echo ""
  81         echo "Cleaning up launched ROS process groups..."
  82         echo "Repeated Ctrl+C/SIGTERM will be ignored until cleanup completes."
  83     fi
  84 
  85     for ((idx=${#MANAGED_PIDS[@]}-1; idx>=0; idx--)); do
  86         local pid="${MANAGED_PIDS[$idx]}"
  87         local name="${MANAGED_NAMES[$idx]}"
  88 
  89         if process_group_alive "$pid"; then
  90             echo "Stopping $name (PGID: $pid) with SIGINT..."
  91             signal_process_group "$pid" INT
  92             if ! wait_for_process_group_exit "$pid" "$SIGINT_GRACE_SEC"; then
  93                 echo "Stopping $name (PGID: $pid) with SIGTERM..."
  94                 signal_process_group "$pid" TERM
  95                 if ! wait_for_process_group_exit "$pid" "$SIGTERM_GRACE_SEC"; then
  96                     echo "Force stopping $name (PGID: $pid) with SIGKILL..."
  97                     signal_process_group "$pid" KILL
  98                     wait_for_process_group_exit "$pid" 2 || true
  99                 fi
 100             fi
 101         fi
 102 
 103         wait "$pid" 2>/dev/null || true
 104     done
 105 
 106     return "$exit_code"
 107 }
