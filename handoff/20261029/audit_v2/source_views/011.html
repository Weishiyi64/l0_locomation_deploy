   5 """Simple 2D A* global planner utilities for Isaac Lab terrain navigation.
   6 
   7 This module intentionally has no Isaac Sim / Isaac Lab dependency.
   8 It can be reused by visualization scripts, training commands, and env logic.
   9 
  10 Coordinate convention:
  11     - world xy: meters, [x, y]
  12     - grid rc : row/col, grid[row, col]
  13     - grid value: 0 = free, 1 = occupied
  14 """
  15 
  16 from __future__ import annotations
  17 
  18 import heapq
  19 import math
  20 from typing import Iterable
  21 
  22 import numpy as np
  23 
  24 
  25 class AStarPlanner:
  26     """2D occupancy-grid A* planner.
  27 
  28     Args:
  29         grid: ``np.ndarray`` with shape ``[height, width]``. 0 means free, 1 means occupied.
  30         resolution: meters per grid cell.
  31         origin_xy: world coordinate of grid cell ``grid[0, 0]`` lower-left corner.
  32     """
  33 
  34     def __init__(
  35         self,
  36         grid: np.ndarray,
  37         resolution: float,
  38         origin_xy: tuple[float, float],
  39         prevent_corner_cutting: bool = False,
  40     ):
  41         if grid.ndim != 2:
  42             raise ValueError(f"grid must be 2-D, got shape={grid.shape}")
  43         if resolution <= 0.0:
  44             raise ValueError(f"resolution must be positive, got {resolution}")
  45 
  46         self.grid = grid.astype(np.uint8)
  47         self.resolution = float(resolution)
  48         self.origin_xy = np.array(origin_xy, dtype=np.float32)
  49         self.prevent_corner_cutting = bool(prevent_corner_cutting)
  50         self.h, self.w = self.grid.shape
  51 
  52         self.neighbors = [
  53             (-1, 0, 1.0),
  54             (1, 0, 1.0),
  55             (0, -1, 1.0),
  56             (0, 1, 1.0),
  57             (-1, -1, math.sqrt(2.0)),
  58             (-1, 1, math.sqrt(2.0)),
  59             (1, -1, math.sqrt(2.0)),
  60             (1, 1, math.sqrt(2.0)),
  61         ]
  62 
  63     def world_to_grid(self, xy: Iterable[float]) -> tuple[int, int]:
  64         """Convert world xy coordinate to grid row/col."""
  65         xy = np.asarray(xy, dtype=np.float32)
  66         col = int(math.floor((xy[0] - self.origin_xy[0]) / self.resolution))
  67         row = int(math.floor((xy[1] - self.origin_xy[1]) / self.resolution))
  68         return row, col
  69 
  70     def grid_to_world(self, rc: tuple[int, int]) -> np.ndarray:
  71         """Convert grid row/col to the center of the cell in world xy."""
  72         row, col = rc
  73         x = self.origin_xy[0] + (col + 0.5) * self.resolution
  74         y = self.origin_xy[1] + (row + 0.5) * self.resolution
  75         return np.array([x, y], dtype=np.float32)
  76 
  77     def valid(self, rc: tuple[int, int]) -> bool:
  78         """Return whether a grid cell is inside bounds and free."""
  79         row, col = rc
  80         return 0 <= row < self.h and 0 <= col < self.w and self.grid[row, col] == 0
  81 
  82     def segment_is_free(self, start_xy: Iterable[float], end_xy: Iterable[float]) -> bool:
  83         """Check a whole segment against closed occupied cells, including corners.
  84 
  85         Used for visibility shortcuts on the already robot-inflated map. Exact
  86         segment/AABB intersections avoid missing a short corner crossing between
  87         discrete samples. Touching an occupied cell or the map boundary is unsafe.
  88         """
  89         endpoints = np.asarray((start_xy, end_xy), dtype=np.float64)
  90         if endpoints.shape != (2, 2) or not np.isfinite(endpoints).all():
  91             return False
  92         endpoints = (endpoints - self.origin_xy) / self.resolution
  93         epsilon = 1e-6  # Grid units; conservatively include float32 boundary ties.
  94         lower = np.floor(endpoints.min(0) - epsilon).astype(np.int64)
  95         upper = np.floor(endpoints.max(0) + epsilon).astype(np.int64)
  96         if (lower < 0).any() or upper[0] >= self.w or upper[1] >= self.h:
  97             return False
  98         occupied = np.argwhere(self.grid[lower[1] : upper[1] + 1, lower[0] : upper[0] + 1])
  99         if len(occupied) == 0:
 100             return True
 101         cell = occupied[:, ::-1] + lower
 102         box_min, box_max = cell - epsilon, cell + 1.0 + epsilon
 103         start, end = endpoints
 104         delta = end - start
 105         entry, leave = np.zeros(len(cell)), np.ones(len(cell))
 106         possible = np.ones(len(cell), dtype=bool)
 107         for axis in range(2):
 108             if delta[axis] == 0.0:
 109                 possible &= (start[axis] >= box_min[:, axis]) & (start[axis] <= box_max[:, axis])
 110             else:
 111                 t0 = (box_min[:, axis] - start[axis]) / delta[axis]
 112                 t1 = (box_max[:, axis] - start[axis]) / delta[axis]
 113                 entry = np.maximum(entry, np.minimum(t0, t1))
 114                 leave = np.minimum(leave, np.maximum(t0, t1))
 115         return not np.any(possible & (entry <= leave))
 116 
 117     @staticmethod
 118     def heuristic(a: tuple[int, int], b: tuple[int, int]) -> float:
 119         """Euclidean grid-distance heuristic."""
 120         return math.hypot(a[0] - b[0], a[1] - b[1])
 121 
 122     def plan(self, start_xy: Iterable[float], goal_xy: Iterable[float]) -> np.ndarray | None:
 123         """Plan a dense path in world xy coordinates.
 124 
 125         Returns:
 126             ``np.ndarray`` with shape ``[N, 2]`` if path is found, otherwise ``None``.
 127         """
 128         start = self.world_to_grid(start_xy)
 129         goal = self.world_to_grid(goal_xy)
 130 
 131         if not self.valid(start):
 132             raise ValueError(f"start is invalid: rc={start}, xy={start_xy}")
 133         if not self.valid(goal):
 134             raise ValueError(f"goal is invalid: rc={goal}, xy={goal_xy}")
 135 
 136         open_set: list[tuple[float, tuple[int, int]]] = []
 137         heapq.heappush(open_set, (0.0, start))
 138         came_from: dict[tuple[int, int], tuple[int, int]] = {}
 139         g_score = {start: 0.0}
 140         closed = set()
 141 
 142         while open_set:
 143             _, current = heapq.heappop(open_set)
 144             if current in closed:
 145                 continue
 146             closed.add(current)
 147 
 148             if current == goal:
 149                 return self._reconstruct(came_from, current)
 150 
 151             for dr, dc, move_cost in self.neighbors:
 152                 nxt = (current[0] + dr, current[1] + dc)
 153                 if not self.valid(nxt):
 154                     continue
 155                 # Do not allow diagonal corner-cutting through walls.  A diagonal
 156                 # move is only valid when both adjacent cardinal cells are free.
 157                 if (
 158                     self.prevent_corner_cutting
 159                     and dr != 0
 160                     and dc != 0
 161                     and (not self.valid((current[0] + dr, current[1])) or not self.valid((current[0], current[1] + dc)))
 162                 ):
 163                     continue
 164 
 165                 tentative_g = g_score[current] + move_cost
 166                 if nxt not in g_score or tentative_g < g_score[nxt]:
 167                     came_from[nxt] = current
 168                     g_score[nxt] = tentative_g
 169                     f_score = tentative_g + self.heuristic(nxt, goal)
 170                     heapq.heappush(open_set, (f_score, nxt))
 171 
 172         return None
 173 
 174     def _reconstruct(self, came_from: dict, current: tuple[int, int]) -> np.ndarray:
 175         cells = [current]
 176         while current in came_from:
 177             current = came_from[current]
 178             cells.append(current)
 179         cells.reverse()
 180         return np.array([self.grid_to_world(cell) for cell in cells], dtype=np.float32)
