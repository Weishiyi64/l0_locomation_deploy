  56 # Function to wait for a topic to publish with timeout (ROS2)
  57 # Prefer Python rclpy - ros2 topic echo often fails when check runs after nodes start
  58 wait_for_topic() {
  59     local topic_name="$1"
  60     local timeout="$2"
  61     local description="$3"
  62 
  63     log_info "Waiting for $description ($topic_name)..."
  64 
  65     # Method 1: Python rclpy subscriber (reliable when launched as Node child)
  66     local python_script="${SCRIPT_DIR}/wait_for_hardware_ready.py"
  67     if [[ -x "$python_script" || -f "$python_script" ]]; then
  68         if python3 "$python_script" "$topic_name" "$timeout" "$description" 2>/dev/null; then
  69             log_success "$description is ready!"
  70             return 0
  71         else
  72             log_error "$description timeout after ${timeout}s"
  73             return 1
  74         fi
  75     fi
  76 
  77     # Method 2: ros2 topic echo fallback (may miss messages if published before subscribe)
  78     local start_time=$(date +%s)
  79     local elapsed=0
  80     while [[ $elapsed -lt $timeout ]]; do
  81         local msg
  82         msg=$(timeout 5 ros2 topic echo "$topic_name" std_msgs/msg/Bool -n 1 2>/dev/null) || true
  83         if [[ -n "$msg" ]] && echo "$msg" | grep -qE "data:\s*(true|True)"; then
  84             log_success "$description is ready!"
  85             return 0
  86         fi
  87         sleep 0.5
  88         elapsed=$(($(date +%s) - start_time))
  89         if [[ $((elapsed % 2)) -eq 0 && $elapsed -gt 0 ]]; then
  90             log_info "Still waiting for $description... (${elapsed}s/${timeout}s)"
  91         fi
  92     done
  93     log_error "$description timeout after ${timeout}s"
  94     return 1
  95 }
