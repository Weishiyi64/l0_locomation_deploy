  15 def verify_cases(runtime, root, graph, max_abs_error):
  16     """Use one caller-owned initialized RKNN context; never reload it between adapters."""
  17     arrays = {}
  18     for key in ("verification_data", "verification_expected"):
  19         path = asset_path(root, graph[key])
  20         if sha256(path) != graph[key + "_sha256"]:
  21             raise ValueError(f"Verification data checksum mismatch: {key}")
  22         with np.load(path, allow_pickle=False) as source:
  23             arrays[key] = {name: source[name].copy() for name in source.files}
  24     observations = arrays["verification_data"]
  25     expected_outputs = arrays["verification_expected"]
  26     for item in graph["inputs"]:
  27         if item["role"] == "observation":
  28             value = observations[item["name"]]
  29             if value.dtype != np.float32 or list(value.shape[1:]) != item["shape"] or not np.isfinite(value).all():
  30                 raise ValueError(f"Invalid recorded observation: {item['name']}")
  31     counts = {len(observations[x["name"]]) for x in graph["inputs"] if x["role"] == "observation"}
  32     if len(counts) != 1 or min(counts) < 20:
  33         raise ValueError("At least 20 aligned observations are required")
  34     count = min(counts)
  35     first_feed, first_output, reports = None, None, []
  36     for action in graph["actions"]:
  37         parameters = load_adapter(asset_path(root, action["adapter_manifest"]), graph, action["action_id"])
  38         expected = expected_outputs[action["action_id"]]
  39         if expected.dtype != np.float32 or list(expected.shape) != [count] + graph["outputs"][0]["shape"]:
  40             raise ValueError("Invalid expected output shape/dtype")
  41         if not np.isfinite(expected).all():
  42             raise ValueError("Non-finite expected output")
  43         maximum, total, worst_ms = 0.0, 0.0, 0.0
  44         for i in range(count):
  45             feed = graph_feeds(graph, {k: v[i] for k, v in observations.items()}, parameters)
  46             started = time.perf_counter()
  47             outputs = runtime.inference(inputs=[feed[x["name"]] for x in graph["inputs"]])
  48             elapsed = (time.perf_counter() - started) * 1000
  49             if outputs is None or len(outputs) != 1:
  50                 raise RuntimeError("RKNN inference did not return one output")
  51             actual = np.asarray(outputs[0], np.float32).reshape(-1)
  52             reference = expected[i].reshape(-1)
  53             if actual.shape != reference.shape or not np.isfinite(actual).all():
  54                 raise RuntimeError("Invalid/non-finite RKNN output")
  55             error = np.abs(actual - reference)
  56             maximum, total, worst_ms = max(maximum, float(error.max())), total + float(error.mean()), max(worst_ms, elapsed)
  57             if error.max() > max_abs_error:
  58                 raise RuntimeError(f"{action['action_id']}: sample {i} max_abs={error.max()} > {max_abs_error}")
  59             if first_feed is None:
  60                 first_feed, first_output = feed, actual.copy()
  61         reports.append({"action_id": action["action_id"], "samples": count, "max_abs": maximum,
  62                         "mean_abs": total / count, "worst_ms": worst_ms})
  63     if first_feed is None:
  64         raise ValueError("No adapters selected")
  65     repeated = runtime.inference(inputs=[first_feed[x["name"]] for x in graph["inputs"]])
  66     if repeated is None or len(repeated) != 1 or not np.isfinite(repeated[0]).all() or \
  67             not np.allclose(np.asarray(repeated[0]).reshape(-1), first_output, atol=max_abs_error, rtol=0):
  68         raise RuntimeError("A-to-B-to-A output mismatch")
  69     return {"actions": reports, "a_to_b_to_a": "passed" if len(reports) > 1 else "one_adapter_only",
  70             "runtime_initializations_in_this_run": 1, "max_abs_error": max_abs_error}

  73 def main():
  74     parser = argparse.ArgumentParser(description=__doc__)
  75     parser.add_argument("--manifest", required=True, type=Path)
  76     parser.add_argument("--rknn", type=Path, help="defaults to the manifest RKNN path")
  77     parser.add_argument("--max-abs-error", type=float, default=0.05)
  78     parser.add_argument("--report", type=Path, help="optional JSON report path")
  79     args = parser.parse_args()
  80     if not math.isfinite(args.max_abs_error) or args.max_abs_error <= 0:
  81         raise ValueError("Invalid precision threshold")
  82     path = args.manifest.resolve()
  83     root, graph = path.parent, load_graph_manifest(path)
  84     model_path = (args.rknn or asset_path(root, graph["model"]["rknn"])).resolve()
  85     record = read_json(Path(str(model_path) + ".json"))
  86     if record.get("source_sha256") != graph["model"]["sha256"] or \
  87             record.get("rknn_sha256") != sha256(model_path) or record.get("precision") != "float16":
  88         raise ValueError("RKNN conversion record does not match this FP16 graph/model")
  89     from rknnlite.api import RKNNLite
  90     runtime = RKNNLite()
  91     try:
  92         ret = runtime.load_rknn(str(model_path))
  93         if ret != 0:
  94             raise RuntimeError(f"load_rknn failed: {ret}")
  95         ret = runtime.init_runtime()
  96         if ret != 0:
  97             raise RuntimeError(f"init_runtime failed: {ret}")
  98         report = verify_cases(runtime, root, graph, args.max_abs_error)
  99         report.update(backend="RKNNLite_on_target", manifest_sha256=sha256(path),
 100                       rknn_sha256=sha256(model_path), precision="float16",
 101                       operator_placement="inspect compiler/runtime profiling separately")
 102         print(json.dumps(report, indent=2, allow_nan=False))
 103         if args.report:
 104             args.report.parent.mkdir(parents=True, exist_ok=True)
 105             args.report.write_text(json.dumps(report, indent=2, allow_nan=False) + "\n")
 106     finally:
 107         runtime.release()
