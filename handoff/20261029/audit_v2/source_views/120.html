 130   {
 131     sequenceRuntime_.jointCsv().step(period.seconds(), time.seconds());

 155 void LeggedController::update(const rclcpp::Time& time, const rclcpp::Duration& period) {
 156   if (!fsm_ || !robot_hw_)
 157   {
 158     RCLCPP_ERROR_THROTTLE(nh_->get_logger(), *nh_->get_clock(), 1000,
 159                           "LeggedController::update called before init completed (fsm/robot_hw not ready)");
 160     return;
 161   }
 162 
 163   last_controller_wall_sec_ = time.seconds();
 164   prepareControllerClock(time);
 165   handleActionSequenceUpdate(time);
 166   RuntimeControlState runtime = computeRuntimeControlState();
 167   if (!runtime.policy_drives_legs && !sequenceRuntime_.isActive() && policyMux_.source() != PolicyCmdSource::kRl)
 168   {
 169     abortWbcImmediately("[WBC] Leaving Walking mode");
 170     setPolicySource(PolicyCmdSource::kRl, "[PolicyMux] Leaving walk -> default RL source");
 171   }
 172   policy_observation_.inference_enabled = runtime.rl_inference_enabled;
 173   updatePolicyObservationFromHardware();
 174   runSafetyAndEmergencyFeedback(time, runtime.policy_drives_legs);
 175   runtime = computeRuntimeControlState();
 176   if (runtime.emergency_control)
 177   {
 178     policy_observation_.inference_enabled = false;
 179   }
 180   publishPolicyObservationWithJoystick(time.seconds());
 181   updateWbcControlState();
 182   runtime = computeRuntimeControlState();
 183   syncFaceFollowState(runtime.allow_auxiliary_overlays && !sequenceRuntime_.releaseHandoffActive(time.seconds()),
 184                       time.seconds());
 185   stepAuxiliaryMotionSources(runtime.allow_auxiliary_overlays, period, time);
 186   applyJointCommands(runtime.policy_drives_legs, runtime.allow_auxiliary_overlays, runtime.emergency_control);
 187   maybePublishVisualization(time);
 188 }

 190 void LeggedController::applyJointCommands(bool walking, bool allow_auxiliary_overlays, bool emergency_control) {
 191   const bool action_sequence_active = sequenceRuntime_.isActive();
 192   const PolicyInputs policy_inputs = policyMux_.selectInputs(walking);
 193   const bool policy_expired = walking && !policy_inputs.valid;
 194   const bool effective_emergency = emergency_control || policy_expired;
 195   if (policy_expired)
 196   {
 197     if (policyMux_.source() == PolicyCmdSource::kWbc) triggerEmergencyStop();
 198     RCLCPP_ERROR_THROTTLE(nh_->get_logger(), *nh_->get_clock(), 1000,
 199                           "Policy command expired or unavailable; applying measured-position damping hold");
 200   }
 201   const bool need_motor_state = effective_emergency;
 202   const auto motor_states =
 203       need_motor_state ? robot_hw_->getHyridJointMotorState() : std::vector<legged::JointMotorState>{};
 204   const bool arm_swing_allowed = allow_auxiliary_overlays && !action_sequence_active && !effective_emergency;
 205   armSwingCtrl_.update(last_controller_wall_sec_, policy_observation_.velocity_command.linear.x, arm_swing_allowed,
 206                        policy_inputs.q_snapshot);
 207 
 208   JointCommandComposer::Context ctx;
 209   ctx.robot_hw = robot_hw_.get();
 210   ctx.head_ctrl = &headCtrl_;
 211   ctx.joint_csv = &sequenceRuntime_.jointCsv();
 212   ctx.arm_swing_ctrl = &armSwingCtrl_;
 213   ctx.release_handoff = &sequenceRuntime_.releaseHandoff();
 214   ctx.policy_inputs = policy_inputs;
 215   ctx.blend_policy_gains = wbc_handoff_in_progress_.load();
 216   ctx.rl_gains = policyMux_.rlGainsSnapshot();
 217   ctx.motor_states = motor_states;
 218   ctx.is_leg_joint = &is_leg_joint_;
 219   // Active CSV/action phases use the AMP profile too; only a selected mimic policy
 220   // changes PolicyMux::gains away from the AMP profile.
 221   ctx.walking = walking || action_sequence_active;
 222   ctx.allow_auxiliary_overlays = allow_auxiliary_overlays;
 223   ctx.emergency = effective_emergency;
 224   ctx.is_sim = isSim_;
 225   ctx.action_sequence_full_csv = isActionSequenceFullCsv();
 226   ctx.action_sequence_active = action_sequence_active;
 227   ctx.kp_standby = kp_standby_;
 228   ctx.emergency_damping_kd = emergency_damping_kd_;
 229   ctx.wall_sec = last_controller_wall_sec_;
 230   jointCommandComposer_.composeAll(ctx);
 231 }

 233 void LeggedController::checkSafety() {
 234   if (isSim_ || !safetyChecker_)
 235   {
 236     orientation_violation_count_ = 0;
 237     return;
 238   }
 239 
 240   const auto& q = policy_observation_.imu.orientation;
 241   const double roll = std::atan2(2.0 * (q.w * q.x + q.y * q.z), 1.0 - 2.0 * (q.x * q.x + q.y * q.y));
 242   const double pitch = std::asin(std::max(-1.0, std::min(1.0, 2.0 * (q.w * q.y - q.z * q.x))));
 243 
 244   SafetyChecker::OrientationResult orientationResult = safetyChecker_->checkOrientation(roll, pitch);
 245   if (orientationResult.ok)
 246   {
 247     orientation_violation_count_ = 0;
 248     return;
 249   }
 250 
 251   ++orientation_violation_count_;
 252   RCLCPP_WARN_THROTTLE(nh_->get_logger(), *nh_->get_clock(), 500,
 253                        "[SafetyChecker] Orientation violation %d/%d: roll=%.1f deg, pitch=%.1f deg",
 254                        orientation_violation_count_, safetyChecker_->requiredConsecutiveOrientationViolations(),
 255                        radiansToDegrees(roll), radiansToDegrees(pitch));
 256 
 257   if (orientation_violation_count_ >= safetyChecker_->requiredConsecutiveOrientationViolations() &&
 258       !fsm_->isEmergency())
 259   {
 260     RCLCPP_ERROR(nh_->get_logger(),
 261                  "[SafetyChecker] Orientation violation persisted for %d cycles (roll=%.1f deg, pitch=%.1f deg)",
 262                  orientation_violation_count_, radiansToDegrees(roll), radiansToDegrees(pitch));
 263     triggerEmergencyStop();
 264   }
 265 }

 267 void LeggedController::updatePolicyObservationFromHardware() {
 268   std::vector<double> bodyPosition;
 269   std::vector<double> bodyOrientation;
 270   bodyPosition.resize(3);
 271   bodyOrientation.resize(4);
 272   bodyPosition = robot_hw_->getBodyPosition();
 273   bodyOrientation = robot_hw_->getBodyOrientation();
 274 
 275   if (bodyPosition.size() >= 3)
 276   {
 277     policy_observation_.base_pose.position.x = bodyPosition[0];
 278     policy_observation_.base_pose.position.y = bodyPosition[1];
 279     policy_observation_.base_pose.position.z = bodyPosition[2];
 280   }
 281   if (bodyOrientation.size() >= 4)
 282   {
 283     policy_observation_.base_pose.orientation.x = bodyOrientation[0];
 284     policy_observation_.base_pose.orientation.y = bodyOrientation[1];
 285     policy_observation_.base_pose.orientation.z = bodyOrientation[2];
 286     policy_observation_.base_pose.orientation.w = bodyOrientation[3];
 287   }
 288 
 289   std::vector<legged::JointMotorState> motor_states = robot_hw_->getHyridJointMotorState();
 290   const size_t state_count = std::min(motor_states.size(), policy_observation_.joints.position.size());
 291   for (size_t i = 0; i < state_count; ++i)
 292   {
 293     policy_observation_.joints.position[i] = motor_states[i].q;
 294     policy_observation_.joints.velocity[i] = motor_states[i].dq;
 295     policy_observation_.joints.effort[i] = motor_states[i].tau;
 296   }
 297 
 298   std::vector<double> imu_quaternion = robot_hw_->getImuQuaternion();
 299   std::vector<double> imu_angular_velocity = robot_hw_->getImuAngularVelocity();
 300 
 301   if (imu_quaternion.size() >= 4)
 302   {
 303     policy_observation_.imu.orientation.x = imu_quaternion[0];
 304     policy_observation_.imu.orientation.y = imu_quaternion[1];
 305     policy_observation_.imu.orientation.z = imu_quaternion[2];
 306     policy_observation_.imu.orientation.w = imu_quaternion[3];
 307   }
 308   if (imu_angular_velocity.size() >= 3)
 309   {
 310     policy_observation_.imu.angular_velocity.x = imu_angular_velocity[0];
 311     policy_observation_.imu.angular_velocity.y = imu_angular_velocity[1];
 312     policy_observation_.imu.angular_velocity.z = imu_angular_velocity[2];
 313   }
 314   policy_observation_.motion_time = shifted_time;
 315 }

 193   const bool action_sequence_active = sequenceRuntime_.isActive();
 194   const PolicyInputs policy_inputs = policyMux_.selectInputs(walking);
 195   const bool policy_expired = walking && !policy_inputs.valid;
 196   const bool effective_emergency = emergency_control || policy_expired;
 197   const bool text_motion_allowed = allow_auxiliary_overlays && !effective_emergency &&
 198                                    policyMux_.source() == PolicyCmdSource::kRl &&

 201   text_motion_start_allowed_.store(text_motion_enabled_ && text_motion_allowed);
 202   textMotionCtrl_.setAllowed(text_motion_enabled_ && text_motion_allowed);
 203   if (text_motion_enabled_ && textMotionCtrl_.busy())
 204     publishTextMotionEvent(textMotionCtrl_.step(policy_inputs.q_snapshot, robot_hw_->getJointMotorCommands(),
 205                                                text_motion_allowed));
 206   if (policy_expired)
 207   {
 208     if (policyMux_.source() == PolicyCmdSource::kWbc) triggerEmergencyStop();
 209     RCLCPP_ERROR_THROTTLE(nh_->get_logger(), *nh_->get_clock(), 1000,
 210                           "Policy command expired or unavailable; applying measured-position damping hold");
 211   }
 212   const bool need_motor_state = effective_emergency;
 213   const auto motor_states =
 214       need_motor_state ? robot_hw_->getHyridJointMotorState() : std::vector<legged::JointMotorState>{};
 215   const bool arm_swing_allowed = allow_auxiliary_overlays && !action_sequence_active && !effective_emergency;
 216   armSwingCtrl_.update(last_controller_wall_sec_, policy_observation_.velocity_command.linear.x, arm_swing_allowed,
 217                        policy_inputs.q_snapshot);
 218 
 219   JointCommandComposer::Context ctx;
 220   ctx.robot_hw = robot_hw_.get();
 221   ctx.head_ctrl = &headCtrl_;
 222   ctx.joint_csv = &sequenceRuntime_.jointCsv();
 223   ctx.arm_swing_ctrl = &armSwingCtrl_;
 224   ctx.text_motion = text_motion_enabled_ ? &textMotionCtrl_ : nullptr;
 225   ctx.release_handoff = &sequenceRuntime_.releaseHandoff();
 226   ctx.policy_inputs = policy_inputs;
 227   ctx.blend_policy_gains = wbc_handoff_in_progress_.load();
 228   ctx.rl_gains = policyMux_.rlGainsSnapshot();
 229   ctx.motor_states = motor_states;
 230   ctx.is_leg_joint = &is_leg_joint_;
 231   // Active CSV/action phases use the AMP profile too; only a selected mimic policy
 232   // changes PolicyMux::gains away from the AMP profile.
 233   ctx.walking = walking || action_sequence_active;
 234   ctx.allow_auxiliary_overlays = allow_auxiliary_overlays;
 235   ctx.emergency = effective_emergency;
 236   ctx.is_sim = isSim_;
 237   ctx.action_sequence_full_csv = isActionSequenceFullCsv();
 238   ctx.action_sequence_active = action_sequence_active;
 239   ctx.kp_standby = kp_standby_;
 240   ctx.emergency_damping_kd = emergency_damping_kd_;
 241   ctx.wall_sec = last_controller_wall_sec_;
 242   jointCommandComposer_.composeAll(ctx);
 243 }

 245 void LeggedController::checkSafety() {
 246   if (isSim_ || !safetyChecker_)
 247   {
 248     orientation_violation_count_ = 0;
 249     return;
 250   }
 251 
 252   const auto& q = policy_observation_.imu.orientation;
 253   const double roll = std::atan2(2.0 * (q.w * q.x + q.y * q.z), 1.0 - 2.0 * (q.x * q.x + q.y * q.y));
 254   const double pitch = std::asin(std::max(-1.0, std::min(1.0, 2.0 * (q.w * q.y - q.z * q.x))));
 255 
 256   SafetyChecker::OrientationResult orientationResult = safetyChecker_->checkOrientation(roll, pitch);
 257   if (orientationResult.ok)
 258   {
 259     orientation_violation_count_ = 0;
 260     return;
 261   }
 262 
 263   ++orientation_violation_count_;
 264   RCLCPP_WARN_THROTTLE(nh_->get_logger(), *nh_->get_clock(), 500,
 265                        "[SafetyChecker] Orientation violation %d/%d: roll=%.1f deg, pitch=%.1f deg",
 266                        orientation_violation_count_, safetyChecker_->requiredConsecutiveOrientationViolations(),
 267                        radiansToDegrees(roll), radiansToDegrees(pitch));
 268 
 269   if (orientation_violation_count_ >= safetyChecker_->requiredConsecutiveOrientationViolations() &&
 270       !fsm_->isEmergency())
 271   {
 272     RCLCPP_ERROR(nh_->get_logger(),
 273                  "[SafetyChecker] Orientation violation persisted for %d cycles (roll=%.1f deg, pitch=%.1f deg)",
 274                  orientation_violation_count_, radiansToDegrees(roll), radiansToDegrees(pitch));
 275     triggerEmergencyStop();
 276   }
 277 }

 279 void LeggedController::updatePolicyObservationFromHardware() {
 280   std::vector<double> bodyPosition;
 281   std::vector<double> bodyOrientation;
 282   bodyPosition.resize(3);
 283   bodyOrientation.resize(4);
 284   bodyPosition = robot_hw_->getBodyPosition();
 285   bodyOrientation = robot_hw_->getBodyOrientation();
 286 
 287   if (bodyPosition.size() >= 3)
 288   {
 289     policy_observation_.base_pose.position.x = bodyPosition[0];
 290     policy_observation_.base_pose.position.y = bodyPosition[1];
 291     policy_observation_.base_pose.position.z = bodyPosition[2];
 292   }
 293   if (bodyOrientation.size() >= 4)
 294   {
 295     policy_observation_.base_pose.orientation.x = bodyOrientation[0];
 296     policy_observation_.base_pose.orientation.y = bodyOrientation[1];
 297     policy_observation_.base_pose.orientation.z = bodyOrientation[2];
 298     policy_observation_.base_pose.orientation.w = bodyOrientation[3];
 299   }
 300 
 301   std::vector<legged::JointMotorState> motor_states = robot_hw_->getHyridJointMotorState();
 302   const size_t state_count = std::min(motor_states.size(), policy_observation_.joints.position.size());
 303   for (size_t i = 0; i < state_count; ++i)
 304   {
 305     policy_observation_.joints.position[i] = motor_states[i].q;
 306     policy_observation_.joints.velocity[i] = motor_states[i].dq;
 307     policy_observation_.joints.effort[i] = motor_states[i].tau;
 308   }
 309 
 310   std::vector<double> imu_quaternion = robot_hw_->getImuQuaternion();
 311   std::vector<double> imu_angular_velocity = robot_hw_->getImuAngularVelocity();
 312 
 313   if (imu_quaternion.size() >= 4)
 314   {
 315     policy_observation_.imu.orientation.x = imu_quaternion[0];
 316     policy_observation_.imu.orientation.y = imu_quaternion[1];
 317     policy_observation_.imu.orientation.z = imu_quaternion[2];
 318     policy_observation_.imu.orientation.w = imu_quaternion[3];
 319   }
 320   if (imu_angular_velocity.size() >= 3)
 321   {
 322     policy_observation_.imu.angular_velocity.x = imu_angular_velocity[0];
 323     policy_observation_.imu.angular_velocity.y = imu_angular_velocity[1];
 324     policy_observation_.imu.angular_velocity.z = imu_angular_velocity[2];
 325   }
 326   policy_observation_.motion_time = shifted_time;
 327 }

 329 void LeggedController::rvizVisualization() {
 330   if (!enableRviz_)
 331   {
 332     return;
 333   }
 334 
 335   rvizVisualPos_.header.stamp = rclcpp::Clock().now();
 336   std::vector<legged::JointMotorState> motor_states = robot_hw_->getHyridJointMotorState();
 337   for (size_t i = 0; i < motor_states.size(); ++i)
 338   {
 339     rvizVisualPos_.position[i] = motor_states[i].q;
 340     rvizVisualPos_.velocity[i] = motor_states[i].dq;
 341     rvizVisualPos_.effort[i] = motor_states[i].tau;
 342   }
 343   vizPub_->publish(rvizVisualPos_);
 344 }
