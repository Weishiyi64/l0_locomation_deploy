cmake_minimum_required(VERSION 3.8)
project(luvbot_control_interfaces)

find_package(ament_cmake REQUIRED)
find_package(action_msgs REQUIRED)
find_package(builtin_interfaces REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(rosidl_default_generators REQUIRED)
find_package(std_msgs REQUIRED)

ament_package_xml()

rosidl_generate_interfaces(${PROJECT_NAME}
  "action/FollowNavigationPath.action"
  "msg/WbcActorSample.msg"
  "msg/PathReference.msg"
  "msg/NavigationGoal.msg"
  "msg/NavigationPlannerCommand.msg"
  "msg/ExploreStatus.msg"
  "srv/MapControl.srv"
  "srv/GetVolumeState.srv"
  "srv/SetCustomWakeWord.srv"
  "srv/SetVolume.srv"
  "srv/UpdateUsers.srv"
  DEPENDENCIES action_msgs builtin_interfaces geometry_msgs std_msgs
)

ament_export_dependencies(rosidl_default_runtime)

if(BUILD_TESTING)
  find_package(ament_lint_auto REQUIRED)
  ament_lint_auto_find_test_dependencies()
endif()

file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/version.py"
  "# Generated by CMake. Do not edit.\n"
  "APP_PACKAGE_NAME = \"${PROJECT_NAME}\"\n"
  "APP_VERSION = \"${${PROJECT_NAME}_VERSION}\"\n"
)
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/version.py" DESTINATION "lib/${PROJECT_NAME}")

ament_package()
