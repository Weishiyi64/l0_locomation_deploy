  "msg/DisplayBatteryState.msg"
  "msg/DisplayWifiState.msg"
  "msg/DisplayProvisionState.msg"
  "msg/DisplayFault.msg"
  "msg/DisplayFaultState.msg"
  "msg/DisplayUpgradeState.msg"
  "msg/DisplayIdentityState.msg"
  "msg/DisplayExpression.msg"
  "msg/DisplayVoiceState.msg"
  "msg/DisplayIndicator.msg"

  "msg/DisplayResult.msg"
  "msg/DisplayDeviceState.msg"
  "msg/BatteryState.msg"

  "msg/MotorTarget.msg"
  "msg/TactileAdc.msg"
  "srv/MotorSetZero.srv"
  "srv/MotorCalibrateId.srv"
  "srv/MotorReadLsError.srv"
