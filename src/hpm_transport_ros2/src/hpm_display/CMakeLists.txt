cmake_minimum_required(VERSION 3.8)
project(hpm_display)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(std_msgs REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(std_srvs REQUIRED)
find_package(luvbot_hardware_interfaces REQUIRED)
find_package(Threads REQUIRED)
add_library(hpm_display SHARED src/protocol.cpp src/display.cpp)
target_include_directories(hpm_display PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
target_compile_options(hpm_display PRIVATE -Wall -Wextra -Wpedantic)
ament_target_dependencies(hpm_display rclcpp std_msgs sensor_msgs std_srvs luvbot_hardware_interfaces)
target_link_libraries(hpm_display Threads::Threads)
install(TARGETS hpm_display EXPORT hpm_displayTargets
  ARCHIVE DESTINATION lib LIBRARY DESTINATION lib RUNTIME DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY include/ DESTINATION include)
install(DIRECTORY config DESTINATION share/${PROJECT_NAME})
install(FILES README.md DESTINATION share/${PROJECT_NAME} OPTIONAL)
install(FILES "docs/hpm_display_ROS2对外接口说明_V2.0.md" DESTINATION share/${PROJECT_NAME}/docs OPTIONAL)
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(test_display_protocol test/test_display_protocol.cpp src/protocol.cpp)
  target_include_directories(test_display_protocol PRIVATE include)
endif()
ament_export_targets(hpm_displayTargets HAS_LIBRARY_TARGET)
ament_export_dependencies(rclcpp std_msgs sensor_msgs std_srvs luvbot_hardware_interfaces)
ament_package()
