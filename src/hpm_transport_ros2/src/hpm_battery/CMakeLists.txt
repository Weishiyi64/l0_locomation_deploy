cmake_minimum_required(VERSION 3.8)
project(hpm_battery)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
find_package(ament_cmake REQUIRED)
include("${CMAKE_CURRENT_SOURCE_DIR}/../../cmake/workspace_version.cmake")
find_package(rclcpp REQUIRED)
find_package(hpm_transport REQUIRED)
find_package(luvbot_hardware_interfaces REQUIRED)
add_library(${PROJECT_NAME} SHARED src/decode.cpp src/battery.cpp)
target_include_directories(${PROJECT_NAME} PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
target_compile_features(${PROJECT_NAME} PUBLIC cxx_std_17)
target_compile_options(${PROJECT_NAME} PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(${PROJECT_NAME} PUBLIC hpm_transport::hpm_transport)
ament_target_dependencies(${PROJECT_NAME} PUBLIC rclcpp luvbot_hardware_interfaces)
install(TARGETS ${PROJECT_NAME} EXPORT export_${PROJECT_NAME}
  ARCHIVE DESTINATION lib LIBRARY DESTINATION lib RUNTIME DESTINATION bin)
install(DIRECTORY include/ DESTINATION include)
ament_export_targets(export_${PROJECT_NAME} HAS_LIBRARY_TARGET)
ament_export_dependencies(rclcpp hpm_transport luvbot_hardware_interfaces)
if(BUILD_TESTING)
  find_package(ament_cmake_test REQUIRED)
  add_executable(test_battery test/test_battery.cpp)
  target_link_libraries(test_battery ${PROJECT_NAME})
  ament_add_test(test_battery COMMAND $<TARGET_FILE:test_battery>
    GENERATE_RESULT_FOR_RETURN_CODE_ZERO)
endif()
ament_package()
